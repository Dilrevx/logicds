package main;

import commons.*;

public class X509CrtParser {

        if ((ret = Helper.x509GetAlg(buf, p, sig_oid2, sig_params2)) != 0) {
            Err.x509crtFree(crt);
            return ret;
        }
        
        /* 
         * Added check: Ensure that the signature algorithm (OID) and its parameters
         * as parsed from the TBSCertificate structure match those from the outer signature.
         * Without this check an attacker could supply a certificate in which the two signature
         * algorithm fields differ, possibly leading to a misinterpretation of the signature.
         */
        if (!Helper.bufEqual(crt.sigOid, sig_oid2) || !Helper.bufEqual(sig_params1, sig_params2)) {
            Err.x509crtFree(crt);
            return Constants.MBEDTLS_ERR_X509_SIG_MISMATCH;
        }
        
        // Continue processing any remaining certificate fields...
        return 0;
    }

    private static boolean equalOids(byte[] a, byte[] b, int len) {
        for (int i = 0; i < len; i++)
            if (a[i] != b[i]) return false;
        return true;
    }
    private static boolean equalBytes(byte[] a, byte[] b) {
        if (a.length != b.length) return false;
        for (int i = 0; i < a.length; i++)
            if (a[i] != b[i]) return false;
        return true;
    }
}
