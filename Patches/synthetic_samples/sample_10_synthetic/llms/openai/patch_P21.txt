        // First, check that the algorithm identifiers (OIDs) match.
        if (crt.sigOid.len != sig_oid2.len

            return Constants.MBEDTLS_ERR_X509_SIG_MISMATCH;
        }
        // Now enforce that both algorithm parameters are either both absent or, if present, identical.
        if ((sig_params1.len == 0 && sig_params2.len != 0)
                || (sig_params1.len != 0 && sig_params2.len == 0)
                || (sig_params1.len != 0 && !Helper.equalBytes(sig_params1.p, sig_params2.p))) {
            Err.x509crtFree(crt);
            return Constants.MBEDTLS_ERR_X509_SIG_MISMATCH;
        }
        // Proceed to extract the signature.
        if ((ret = Helper.x509GetSig(buf, p, new byte[][]{ crt.sig })) != 0) {
            Err.x509crtFree(crt);
            return ret;
        }
        return ret;
    }

    private static boolean equalOids(byte[] a, byte[] b, int len) {
        for (int i = 0; i < len; i++)
            if (a[i] != b[i]) return false;
        return true;
    }
    private static boolean equalBytes(byte[] a, byte[] b) {
        if (a.length != b.length) return false;
        for (int i = 0; i < a.length; i++)
            if (a[i] != b[i]) return false;
        return true;
    }
