package main;

import commons.*;

public class X509CrtParser {

    // Compare signature algorithm OIDs to ensure they match
    if (crt.sigOid.len != sig_oid2.len) {
        Err.x509crtFree(crt);
        return Constants.MBEDTLS_ERR_X509_SIG_MISMATCH;
    }
    for (int i = 0; i < crt.sigOid.len; i++) {
        if (crt.sigOid.p[i] != sig_oid2.p[i]) {
            Err.x509crtFree(crt);
            return Constants.MBEDTLS_ERR_X509_SIG_MISMATCH;
        }
    }
    return 0;

    private static boolean equalOids(byte[] a, byte[] b, int len) {
        for (int i = 0; i < len; i++)
            if (a[i] != b[i]) return false;
        return true;
    }
    private static boolean equalBytes(byte[] a, byte[] b) {
        if (a.length != b.length) return false;
        for (int i = 0; i < a.length; i++)
            if (a[i] != b[i]) return false;
        return true;
    }
}
