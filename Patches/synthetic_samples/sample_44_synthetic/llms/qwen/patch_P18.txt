            }

            if (targisdir) {
                np = targ + "/" + filename;
            } else {
                np = targ;

            exists = ctx.getFs().stat(np, stb) == 0;
            omode = stb.getMode();

            if (cp.charAt(0) == 'D') {
                if (!exists || !FileSystem.sIsDir(omode)) {
                    if (ctx.getFs().mkdir(np, mode) != 0) {
                        ScpLog.runErr("cannot make directory %s", np);
                        return 1;
                    }
                }
                if (ctx.isRecursive()) {
                    if (sink(np, reader, ctx) != 0) {
                        return 1;
                    }
                }
                if (setimes > 0) {
                    setimes--;
                    if (setimes == 0) {
                        if (ctx.getFs().utimes(np, stb.getAtime(), stb.getMtime()) != 0) {
                            ScpLog.runErr("cannot set times on directory %s", np);
                            return 1;
                        }
                    }
                }
            } else {
                if (exists && FileSystem.sIsDir(omode)) {
                    ScpLog.runErr("%s is a directory", np);
                    return 1;
                }
                ofd = ctx.getFs().open(np, FileSystem.O_WRONLY | FileSystem.O_CREAT | FileSystem.O_TRUNC, mode);
                if (ofd < 0) {
                    ScpLog.runErr("cannot open %s", np);
                    return 1;
                }
                if (copy(ofd, reader, size, ctx) != 0) {
                    ctx.getFs().close(ofd);
                    return 1;
                }
                ctx.getFs().close(ofd);
                if (setimes > 0) {
                    setimes--;
                    if (setimes == 0) {
                        if (ctx.getFs().utimes(np, stb.getAtime(), stb.getMtime()) != 0) {
                            ScpLog.runErr("cannot set times on file %s", np);
                            return 1;
                        }
                    }
                }
            }
        }
        return 0;


    private static void verifyDir(FileSystem fs, String targ) {
        ScpStat stb = new ScpStat();
        fs.stat(targ, stb);
    }
