
    public int sink(String targ, ScpReader reader, ScpContext ctx) {
        ScpStat stb = new ScpStat();
        int mode, omode, mask;
        long size;
        int setimes = 0;
        boolean targisdir = false;
        boolean exists;
        String cp, np;
        int ofd;
        boolean first;

        mask = ctx.getUmask();
        if (!ctx.isPflag()) {
            ctx.setUmask(mask);
        }
        if (ctx.isTargetshouldbedirectory()) {
            verifyDir(ctx.getFs(), targ);
        }

        if (ctx.getFs().stat(targ, stb) == 0 && FileSystem.sIsDir(stb.getMode())) {
            targisdir = true;
        }

        first = true;
        while (reader.hasMore()) {
            String line = reader.readLine();
            if (line == null) {
                return 0;
            }
            if (ctx.isVerboseMode()) {
                ScpLog.verbose("Sink: " + line);
            }

            if (line.length() >= 1 && (line.charAt(0) == '\01' || line.charAt(0) == '\02')) {
                if (!ctx.isIamremote()) {
                    ScpLog.runErr("%s", line.substring(1));
                }
                if (line.charAt(0) == '\02') {
                    return 1;
                }
                ctx.incErrs();
                continue;
            }
            if (line.length() >= 1 && line.charAt(0) == 'E') {
                return 0;
            }

            cp = line;
            if (cp.length() >= 1 && cp.charAt(0) == 'T') {
                setimes++;
                first = false;
                continue;
            }

            if (cp.length() < 1 || (cp.charAt(0) != 'C' && cp.charAt(0) != 'D')) {
                if (first) {
                    ScpLog.runErr("%s", cp);
                    return 1;
                }
                ScpLog.screwup("expected control record");
                return 1;
            }

            mode = 0;
            int idx = 1;
            for (; idx < 5; idx++) {
                char c = cp.charAt(idx);
                if (c < '0' || c > '7') {
                    ScpLog.screwup("bad mode");
                    return 1;
                }
                mode = (mode << 3) | (c - '0');
            }
            if (!ctx.isPflag()) {
                mode &= ~mask;
            }
            if (cp.charAt(idx) != ' ') {
                ScpLog.screwup("mode not delimited");
                return 1;
            }
            idx++;

            int sizeStart = idx;
            while (idx < cp.length() && Character.isDigit(cp.charAt(idx))) {
                idx++;
            }
            if (idx == sizeStart) {
                ScpLog.screwup("size not present");
                return 1;
            }
            size = Long.parseLong(cp.substring(sizeStart, idx));
            if (idx >= cp.length() || cp.charAt(idx) != ' ') {
                ScpLog.screwup("size not delimited");
                return 1;
            }
            idx++;

            String filename = cp.substring(idx);

            if (filename.indexOf('/') >= 0 || filename.equals("..")) {
// start of generated patch
if(filename.indexOf('/')>=0||newSrc.equals("..")){
ScpLog.runErr("error: unexpected filename: %s",filename);
return 1;
}
// end of generated patch
/* start of original code
                ScpLog.runErr("error: unexpected filename: %s", filename);
                return 1;
            }

 end of original code*/
            if (targisdir) {
                np = targ + (targ.equals("/") ? "" : "/") + filename;
            } else {
                np = targ;
            }
            exists = ctx.getFs().stat(np, stb) == 0;

            if (cp.charAt(0) == 'D') {
                if (!ctx.isIamrecursive()) {
                    ScpLog.screwup("received directory without -r");
                    return 1;
                }
                if (exists) {
                    if (!FileSystem.sIsDir(stb.getMode())) {
                        ScpLog.runErr("%s: Not a directory", np);
                        first = false;
                        continue;
                    }
                    if (ctx.isPflag()) {
                        ctx.getFs().chmod(np, mode);
                    }
                } else {
                    if (ctx.getFs().mkdir(np, mode | ScpFlags.S_IRWXU) < 0) {
                        ScpLog.runErr("%s: mkdir failed", np);
                        first = false;
                        continue;
                    }
                }
                int inner = sink(np, reader, ctx);
                if (inner != 0) {
                    return inner;
                }
                if (ctx.isPflag() || !exists) {
                    ctx.getFs().chmod(np, mode);
                }
                first = false;
                continue;
            }

            omode = mode;
            mode |= ScpFlags.S_IWUSR;
            ofd = ctx.getFs().open(np, ScpFlags.O_WRONLY | ScpFlags.O_CREAT, mode);
            if (ofd < 0) {
                ScpLog.runErr("%s: open failed", np);
                first = false;
                continue;
            }

            byte[] payload = reader.readPayload((int) size);
            ctx.getFs().write(ofd, payload);

            if (ctx.isPflag() && (exists || omode != mode)) {
                ctx.getFs().chmod(np, omode);
            }
            ctx.getFs().close(ofd);
            first = false;
        }

        return 0;
    }

    private static void verifyDir(FileSystem fs, String targ) {
        ScpStat stb = new ScpStat();
        fs.stat(targ, stb);
    }
