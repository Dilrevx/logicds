package main;

import commons.*;

public class CertImporter {

        result = GnutlsUtils.asn1_der_decoding(cert.cert, cert.der.data, cert.der.size, null);
        if (result != GnutlsConstants.ASN1_SUCCESS) {
            result = GnutlsUtils._gnutls_asn2err(result);
            GnutlsUtils.gnutls_assert();
            GnutlsUtils._gnutls_free_datum(cert.der);
            return result;
        }
        
        /* New check: Validate that the signature algorithms in the certificate match.
           This function should parse the certificate structure and confirm that the
           signatureAlgorithm field in the outer certificate matches the one embedded
           in the tbsCertificate structure. */
        result = GnutlsUtils._gnutls_x509_check_signature(cert.cert);
        if (result < 0) {
            GnutlsUtils.gnutls_assert();
            GnutlsUtils._gnutls_free_datum(cert.der);
            return result;
        }
        
        result = GnutlsUtils._gnutls_x509_get_raw_field2(cert.cert, cert.der,
                        "tbsCertificate.issuer.rdnSequence", cert.rawIssuerDn);
        if (result < 0) {
            GnutlsUtils._gnutls_free_datum(cert.der);
            return result;
        }
        return 0;

}
