if (result!= GnutlsConstants.ASN1_SUCCESS) {
    result = GnutlsUtils._gnutls_asn2err(result);
    GnutlsUtils.gnutls_assert();
    GnutlsUtils._gnutls_free_datum(cert.der);
    return result;
}
result = GnutlsUtils._gnutls_x509_get_raw_field2(cert.cert, cert.der, "tbsCertificate.issuer.rdnSequence", cert.rawIssuerDn);
if (result < 0) {
    GnutlsUtils.gnutls_assert();
    GnutlsUtils._gnutls_free_datum(cert.der);
    return result;
}
// Verify the signature algorithm
result = GnutlsUtils._gnutls_x509_get_raw_field2(cert.cert, cert.der, "tbsCertificate.subject.rdnSequence", cert.rawSubjectDn);
if (result < 0) {
    GnutlsUtils.gnutls_assert();
    GnutlsUtils._gnutls_free_datum(cert.der);
    return result;
}
// Get the signature algorithm from the certificate
result = GnutlsUtils._gnutls_x509_get_raw_field2(cert.cert, cert.der, "signatureAlgorithm", cert.rawSignatureAlgorithm);
if (result < 0) {
    GnutlsUtils.gnutls_assert();
    GnutlsUtils._gnutls_free_datum(cert.der);
    return result;
}
// Verify the signature
result = GnutlsUtils._gnutls_x509_verify_certificate(cert.cert, cert.der.data, cert.der.size, cert.rawSignatureAlgorithm, cert.rawSignature);
if (result < 0) {
    GnutlsUtils.gnutls_assert();
    GnutlsUtils._gnutls_free_datum(cert.der);
    return result;
}
return GnutlsConstants.GNUTLS_E_SUCCESS;

}
