package SD1_fixed.src;

import SD1_fixed.src.commons.Pjmedia_tp_cb_param;
import SD1_fixed.src.commons.SrtpCrypto;
import SD1_fixed.src.commons.TransportSrtp;

public class SrtpTransportCallback {
    public static final int PJ_EIGNORED = -1;
    public static final int PJ_SUCCESS = 0;
    public static final int SRTP_ERR_REPLAY_OLD = 1;
    public static final int SRTP_ERR_REPLAY_FAIL = 2;
    public static int unprotectCallCount = 0;


public static void srtp_rtp_cb(Pjmedia_tp_cb_param param) {
    // Validate pointer and length before processing
    if (param == null || param.user_data == null || param.pkt == null || param.size <= 0) {
        // Invalid parameters: log error and discard packet
        System.err.println("srtp_rtp_cb: Invalid parameters; dropping packet.");
        return;
    }

    TransportSrtp srtp = (TransportSrtp) param.user_data;
    // Assuming the packet is a byte array; adjust cast as needed.
    byte[] pkt;
    try {
        pkt = (byte[]) param.pkt;
    } catch (ClassCastException ex) {
        System.err.println("srtp_rtp_cb: Packet data is of invalid type.");
        return;
    }
    
    int size = param.size;
    
    // Call srtp_unprotect and check the result.
    int err = srtp_unprotect(srtp.srtp_rx_ctx, pkt, size);
    if (err != 0) {
        // If unprotection fails, report error and discard packet.
        System.err.println("srtp_unprotect failed with error code: " + err);
        return;
    }
    
    // If srtp_unprotect modifies the length of the valid payload,
    // and if it returns the new length, update the size accordingly.
    // (If it returns 0 on success, then the incoming size is still valid.)
    // For example, if err represented new length:
    // int newSize = err;
    // Otherwise, newSize remains size.
    
    // Continue processing the now-protected (decrypted and authenticated) packet.
    // ... rest of the function logic ...
}


    static int pjmedia_transport_send_rtp(SD1_fixed.src.commons.Keying key, Object pkt, int size) { return PJ_EIGNORED; }
    static void pjmedia_transport_srtp_stop(TransportSrtp srtp) {}
    static int pjmedia_transport_srtp_start(TransportSrtp srtp, SrtpCrypto tx, SrtpCrypto rx) { return PJ_SUCCESS; }
    static int srtp_unprotect(Object ctx, Object pkt, int len) {
        unprotectCallCount++;
        return unprotectCallCount == 1 ? SRTP_ERR_REPLAY_OLD : PJ_SUCCESS;
    }
}
