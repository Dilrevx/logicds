            case commons.Nas5gsHdr.security_header_type_opts.plain_5gs_nas_message:
                // If the security context is activated, plain NAS messages (like identity_req_plain) must not be accepted.
                if (nas_msg.is_security_context_activated()) {
                    logger.error("Plain NAS message received after NAS security activation");
                    return SRSRAN_ERROR;
                }
                break;
                
            case commons.Nas5gsHdr.security_header_type_opts.integrity_protected:
                if (!integrity_check(pdu.get())) {
                    logger.error("Not handling NAS message with integrity check error");
                    return SRSRAN_ERROR;
                }
                break;
                
            case commons.Nas5gsHdr.security_header_type_opts.integrity_protected_and_ciphered:
                if (!integrity_check(pdu.get())) {
                    logger.error("Not handling NAS message with integrity check error");
                    return SRSRAN_ERROR;
                } else {
                    cipher_decrypt(pdu.get());
                }
                break;
                
            case commons.Nas5gsHdr.security_header_type_opts.integrity_protected_with_new_5G_nas_context:

            case commons.Nas5gsHdr.security_header_type_opts.integrity_protected_and_ciphered_with_new_5G_nas_context:
                logger.error("Not handling NAS message with unsupported security header (new 5G NAS context with ciphering and integrity)");
                return SRSRAN_ERROR;
                
            default:
                logger.error("Not handling NAS message with unknown security header");
                return SRSRAN_ERROR;
