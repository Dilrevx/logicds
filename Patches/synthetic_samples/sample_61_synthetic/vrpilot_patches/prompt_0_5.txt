public class SaslService {

    public static final String SASLSVS_NICK = "SaslServ";
    public static final String MOWGLI_FUNC_NAME = "sasl_handle_login";

    public boolean saslHandleLogin(SaslSession p, User u, MyUser mu) {
        boolean wasKilled = false;

        // If no MyUser was supplied, then we are in the handshake phase.
        if (mu == null) {
            // A complete SASL handshake must supply a non‑empty authzid.
            String authzid = p.getAuthzid();
            if (authzid == null || authzid.isEmpty()) {
                IrcUtil.slog(IrcUtil.LG_INFO,
                        "%s: session for '%s' without an authzid (BUG)",
                        MOWGLI_FUNC_NAME, u.getNick());
                IrcUtil.notice(SASLSVS_NICK, u.getNick(), IrcUtil.LOGIN_CANCELLED_STR);
                return false;
            }
            // Handshake is complete: look up the MyUser using the provided authzid.
            mu = UserRegistry.myuserFindUid(authzid);
            if (mu == null) {
                // If the account corresponding to the authzid cannot be found, notify the user.
                IrcUtil.notice(SASLSVS_NICK, u.getNick(),
                        "Account %s dropped; login cancelled", authzid);
                return false;
            }
        }

        // If the user is already logged in as a different account,
        // log out the current session.
        if (u.getMyuser() != null && !u.getMyuser().equals(mu)) {
            IrcUtil.logcommandUser(SASLSVS_NICK, u, IrcUtil.CMDLOG_LOGIN, "LOGOUT");
        }

        // Proceed with setting the new MyUser and continue with further login logic.
        u.setMyuser(mu);
        // ... (other login code) ...

        return true;
    }
}
