public class SaslService {

    public static final String SASLSVS_NICK = "SaslServ";
    public static final String MOWGLI_FUNC_NAME = "sasl_handle_login";

    public boolean saslHandleLogin(SaslSession p, User u, MyUser mu) {
        boolean wasKilled = false;


        if (mu == null) {
            /* Ensure that both the authoritative identity (authzeid) and the provided identity (authzid)
             * are present and consistent. Without a valid, complete exchange, we must refuse to authenticate.
             */
            if (p.getAuthzeid().isEmpty() || p.getAuthzid().isEmpty()) {
                IrcUtil.slog(IrcUtil.LG_INFO,
                        "%s: session for '%s' missing required authentication identities; login cancelled",
                        MOWGLI_FUNC_NAME, u.getNick());
                IrcUtil.notice(SASLSVS_NICK, u.getNick(), IrcUtil.LOGIN_CANCELLED_STR);
                return false;
            }

            if (!p.getAuthzeid().equals(p.getAuthzid())) {
                IrcUtil.slog(IrcUtil.LG_INFO,
                        "%s: session for '%s' provided mismatched authzeid (%s) and authzid (%s); login cancelled",
                        MOWGLI_FUNC_NAME, u.getNick(), p.getAuthzeid(), p.getAuthzid());
                IrcUtil.notice(SASLSVS_NICK, u.getNick(), IrcUtil.LOGIN_CANCELLED_STR);
                return false;
            }

            mu = UserRegistry.myuserFindUid(p.getAuthzeid());
            if (mu == null) {
                IrcUtil.notice(SASLSVS_NICK, u.getNick(),
                        "Account %s dropped; login cancelled", p.getAuthzeid());
                return false;
            }
        }


        if (u.getMyuser() != null && u.getMyuser() != mu) {
            if (IrcUtil.isSoper(u.getMyuser())) {
                IrcUtil.logcommandUser(SASLSVS_NICK, u, IrcUtil.CMDLOG_ADMIN,
                        "DESOPER: %s as %s", u.getNick(), u.getMyuser().getName());
            }

            IrcUtil.logcommandUser(SASLSVS_NICK, u, IrcUtil.CMDLOG_LOGIN, "LOGOUT");

            wasKilled = IrcUtil.ircdOnLogout(u, u.getMyuser().getName());
            if (!wasKilled) {
                u.getMyuser().getLogins().remove(u);
                u.setMyuser(null);
            }
        }

        if (!wasKilled) {
            if (u.getMyuser() != mu) {
                IrcUtil.myuserLogin(SASLSVS_NICK, u, mu, false);
                IrcUtil.logcommandUser(SASLSVS_NICK, u, IrcUtil.CMDLOG_LOGIN,
                        "LOGIN (%s)", p.getMechptr().getName());
            } else {
                mu.setLastLogin(System.currentTimeMillis());
                IrcUtil.logcommandUser(SASLSVS_NICK, u, IrcUtil.CMDLOG_LOGIN,
                        "REAUTHENTICATE (%s)", p.getMechptr().getName());
            }
        }

        return true;
    }
}
