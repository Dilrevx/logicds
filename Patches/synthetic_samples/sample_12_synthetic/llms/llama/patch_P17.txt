package main;

import commons.*;

public class Tls13FinishedHandler {

public static int DoTls13Finished(WOLFSSL ssl, byte[] input, int[] inOutIdx,
                                  int size, int totalSz, int sniff) {
    int ret;
    int finishedSz = 0;
    byte[] secret;
    byte[] mac = new byte[32];

    if (input == null || inOutIdx == null || inOutIdx.length < 1) {
        return BAD_FUNC_ARG;
    }

    if (size < 0 || totalSz < 0) {
        return LENGTH_ONLY_IN_NON_BLOCKING;
    }

    Tls13Helper.WOLFSSL_START(Constants.WC_FUNC_FINISHED_DO);
    ret = Tls13Helper.WOLFSSL_CHECK_FOR_ERROR(ssl);
    if (ret!= 0) {
        return ret;
    }
    if (ssl->options.side == WOLFSSL_CLIENT_END) {
        if (ssl->options.tls1_3_finishedVerify == 0) {
            return NOT_READY;
        }
    } else {
        if (ssl->options.tls1_3_finishedVerify == 0) {
            return NOT_READY;
        }
    }
    if (size > 0) {
        if (size!= totalSz) {
            return LENGTH_ONLY_IN_NON_BLOCKING;
        }
        if (inOutIdx[0] + size > input.length) {
            return BUFFER_E;
        }
        if (ssl->options.side == WOLFSSL_CLIENT_END) {
            secret = ssl->arrays->clientFinishedSecret;
        } else {
            secret = ssl->arrays->serverFinishedSecret;
        }
        if (secret == null) {
            return NO_PRIVATE_KEY;
        }
        ret = DeriveTls13Finished(ssl, secret, mac);
        if (ret!= 0) {
            return ret;
        }
        if (ConstantTimeCompare(input, inOutIdx[0], mac, 0, 32)!= 0) {
            return DECRYPTION_ERROR;
        }
        ssl->options.tls1_3_finishedVerify = 1;
    }
    return 0;
}

}
