package main;

import commons.*;

public class Tls13FinishedHandler {

    if (ssl == null || ssl.tls13 == null) {
        return BAD_FUNC_ARG;
    }

    if (ssl.tls13.handshakeHash == null) {
        return SSL_FATAL_ERROR;
    }

    secret = ssl.tls13.handshakeHash;
    finishedSz = ssl.tls13.handshakeHashSz;

    if (finishedSz > size) {
        return BUFFER_E;
    }

    ret = DeriveTls13Secret(ssl, Constants.TLS13_FINISHED_SECRET_LABEL,
            ssl.tls13.handshakeHash, ssl.tls13.handshakeHashSz, mac, 32);
    if (ret!= 0) {
        return ret;
    }

    ret = Hmac(ssl.tls13.macHash, mac, 32, input, inOutIdx[0], size, mac, 0);
    if (ret!= 0) {
        return ret;
    }

    if (CmpConstTime(mac, 0, input, inOutIdx[0], 32)!= 0) {
        return SSL_FATAL_ERROR;
    }

    inOutIdx[0] += 32;
    return 0;

}
