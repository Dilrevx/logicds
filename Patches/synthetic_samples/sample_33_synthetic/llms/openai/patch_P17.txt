package master;
import common.*;

public class Security_Default{


        if (securityOpts == null) {
            return MosqErr.MOSQ_ERR_ACL_DENIED;
        }

        /*
         * Instead of immediately granting access (or skipping further ACL checks)
         * ensure that the default ACL rules are evaluated. Assume the security
         * options structure now includes a collection of default ACL entries.
         * If no default ACL is configured then the request should be denied.
         */
        if (securityOpts.defaultACLs == null || securityOpts.defaultACLs.isEmpty()) {
            return MosqErr.MOSQ_ERR_ACL_DENIED;
        }

        boolean accessAllowed = false;
        // Iterate over all default ACL entries to find a matching rule.
        for (AclEntry entry : securityOpts.defaultACLs) {
            // The applies method should check if this ACL entry allows the given
            // topic and access type (e.g. read, write). It should perform safe validations.
            if (entry.applies(topic, access)) {
                accessAllowed = true;
                break;
            }
        }

        return accessAllowed ? MosqErr.MOSQ_ERR_SUCCESS : MosqErr.MOSQ_ERR_ACL_DENIED;
    }


    public static void main(String[] args) {
        MosquittoDb db = new MosquittoDb();
        MosquittoContext context = new MosquittoContext();

        db.config.perListenerSettings = false;
        db.config.securityOptions.aclFile = null;
        db.config.securityOptions.aclList = null;
        db.config.securityOptions.aclPatterns = null;

        MosqErr result = mosquittoAclCheckDefault(db, context, "someTopic", Access.MOSQ_ACL_READ);
        System.out.println("Case1 => " + result);

        db.config.securityOptions.aclFile = "/path/to/acl";

        result = mosquittoAclCheckDefault(db, context, "someTopic", Access.MOSQ_ACL_READ);
        System.out.println("Case2 => " + result);

    }
}
