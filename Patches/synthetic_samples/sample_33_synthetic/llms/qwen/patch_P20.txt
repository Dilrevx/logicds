        if (securityOpts.aclList != null) {
            for (MosquittoAclEntry entry : securityOpts.aclList) {
                if (entry.matches(topic, access)) {
                    aclMatched = true;
                    break;
                }
            }
        }

        if (!aclMatched && securityOpts.aclPatterns != null) {
            for (MosquittoAclPattern pattern : securityOpts.aclPatterns) {
                if (pattern.matches(topic, access)) {
                    aclMatched = true;
                    break;
                }
            }
        }

        return aclMatched ? MosqErr.MOSQ_ERR_SUCCESS : MosqErr.MOSQ_ERR_ACL_DENIED;
    }
