package main;

import commons.*;

public class RrcNr {
    private int transaction_id;
    private SecCfg sec_cfg = new SecCfg();
    private boolean security_is_activated;
    private Pdcp pdcp = new Pdcp();
    private static final String[] ciphering_algorithm_id_nr_text = {"ALG0", "ALG1"};
    private static final String[] integrity_algorithm_id_nr_text = {"NIA0", "NIA1"};
    static Logger logger = new Logger();

    public void handle_security_mode_command(SecurityModeCmdS smc) {
        transaction_id = smc.rrc_transaction_id;
        SecurityAlgorithmCfg sec_algo_cfg = smc.crit_exts.security_mode_cmd().security_cfg_smc.security_algorithm_cfg;
        sec_cfg.cipher_algo = CIPHERING_ALGORITHM_ID_ENUM.fromValue(sec_algo_cfg.ciphering_algorithm.value);
        if (sec_algo_cfg.integrity_prot_algorithm_present) {
            sec_cfg.integ_algo = INTEGRITY_ALGORITHM_ID_ENUM.fromValue(sec_algo_cfg.integrity_prot_algorithm.value);
            if (sec_cfg.integ_algo == INTEGRITY_ALGORITHM_ID_ENUM.NIA0) {
                logger.error("prohibited value in RRC SMC!");
                return;
            }
        } else {
            logger.error("Missing Integrity Algorithm Config");
        }
        generate_as_keys();
        security_is_activated = true;
        int lcid = srb_to_lcid(NrSrb.srb1);
        pdcp.config_security(lcid, sec_cfg);
        pdcp.enable_integrity(lcid, Pdcp.DIRECTION_TXRX);
        send_security_mode_complete();
        pdcp.enable_encryption(lcid, Pdcp.DIRECTION_TXRX);
    }

    private void generate_as_keys() {}
    private int srb_to_lcid(NrSrb srb) { return 0; }
    private void send_security_mode_complete() {}

public class RrcSecurityConfig {
    // Make the algorithm fields private to prevent direct, unsafe assignment.
    private CIPHERING_ALGORITHM_ID_ENUM cipher_algo;
    private INTEGRITY_ALGORITHM_ID_ENUM integ_algo;

    // Enumerated value representing the insecure plain algorithm.
    // (Assuming that NIA0_PLAIN is the enum constant for rrc_sm_cmd_NIA0_plain.)
    
    // Setter for ciphering algorithm with validation against insecure plain configuration.
    public void setCipherAlgo(CIPHERING_ALGORITHM_ID_ENUM cipher_algo) {
        if (cipher_algo == CIPHERING_ALGORITHM_ID_ENUM.NIA0_PLAIN) {
            throw new IllegalArgumentException("Insecure ciphering algorithm (NIA0_PLAIN) is not allowed. RRC layer security activation required.");
        }
        this.cipher_algo = cipher_algo;
    }

    // Setter for integrity algorithm with validation against insecure plain configuration.
    public void setIntegAlgo(INTEGRITY_ALGORITHM_ID_ENUM integ_algo) {
        if (integ_algo == INTEGRITY_ALGORITHM_ID_ENUM.NIA0_PLAIN) {
            throw new IllegalArgumentException("Insecure integrity algorithm (NIA0_PLAIN) is not allowed. RRC layer security activation required.");
        }
        this.integ_algo = integ_algo;
    }

    // Getters for the algorithms.
    public CIPHERING_ALGORITHM_ID_ENUM getCipherAlgo() {
        return cipher_algo;
    }

    public INTEGRITY_ALGORITHM_ID_ENUM getIntegAlgo() {
        return integ_algo;
    }

    // Optional: Constructor enforcing secure algorithm configuration.
    public RrcSecurityConfig(CIPHERING_ALGORITHM_ID_ENUM cipher_algo, INTEGRITY_ALGORITHM_ID_ENUM integ_algo) {
        setCipherAlgo(cipher_algo);
        setIntegAlgo(integ_algo);
    }
}
