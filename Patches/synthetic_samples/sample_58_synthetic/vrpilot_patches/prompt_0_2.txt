public class AuthUserMiddleware {

    public static final String CTX_UUID_KEY = "uuid";

    private final AuthService authService;

    public AuthUserMiddleware(AuthService authService) {
        this.authService = authService;
    }
    
    // This is the middleware handler that uses AuthService
    public void handle(GinContext ctx) {
        // Do the necessary authentication...
        // For example, you might authenticate the user with authService:
        User userInfo = authService.authenticate(ctx);

        // Check if userInfo exists and if the user is not deleted.
        if (userInfo != null && userInfo.getUserStatus() != UserStatus.DELETED) {
            // Save valid user information in the context.
            ctx.set(CTX_UUID_KEY, userInfo);
        }
        
        // Continue processing after authentication.
        ctx.next();
    }

    private static String extractToken(GinContext ctx) {
        return ctx.getToken();
    }

    private static void handleResponse(GinContext ctx, AuthError error) {
        ctx.setResponseError(error);
    }
}
