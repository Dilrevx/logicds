public class AuthUserMiddleware {

    public static final String CTX_UUID_KEY = "uuid";

    private final AuthService authService;

    public AuthUserMiddleware(AuthService authService) {
        this.authService = authService;
    }


    public void adminAuth(GinContext ctx) {
        String token = extractToken(ctx);
        if (token == null || token.length() == 0) {
            handleResponse(ctx, AuthError.unauthorized(AuthError.UNAUTHORIZED_REASON));
            ctx.abort();
            return;
        }

        AuthService.CacheResult result = authService.getAdminUserCacheInfo(ctx, token);
        if (result == null || result.err!= null) {
            handleResponse(ctx, AuthError.unauthorized(AuthError.UNAUTHORIZED_REASON));
            ctx.abort();
            return;
        }

        UserInfo userInfo = result.userInfo;
        if (userInfo == null ||!userInfo.isAdmin()) {
            handleResponse(ctx, AuthError.unauthorized(AuthError.UNAUTHORIZED_REASON));
            ctx.abort();
            return;
        }
    }


    private static String extractToken(GinContext ctx) {
        return ctx.getToken();
    }

    private static void handleResponse(GinContext ctx, AuthError error) {
        ctx.setResponseError(error);
    }
}
