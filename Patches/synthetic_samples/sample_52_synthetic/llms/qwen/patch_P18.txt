public class LdapValidator {


        try {
            ldap.search(props.getBaseDN(), "objectClass=*", attrs, message);
            if (message[0] != null) {
                first = message[0];
                size = first.getMessage().size();
                if (size > 0) {
                    dn = first.getMessage().get(0).getDN();
                    re = 1;
                    ret = new RetT(re, dn);
                } else {
                    re = 0;
                    ret = new RetT(re, "No entries found");
                }
            } else {
                re = 0;
                ret = new RetT(re, "Search returned no results");
            }
        } catch (NamingException e) {
            re = -1;
            ret = new RetT(re, e.getMessage());
        }
        return ret;
    }

}
