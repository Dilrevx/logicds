import java.util.Map;

public class ShelfController {

    private final User currentUser;

    public ShelfController(User currentUser) {
        this.currentUser = currentUser;
    }

    public ShelfResponse createEditShelf(Shelf shelf, String pageTitle, int page, Integer shelfId, Request request) {
        boolean syncOnlySelectedShelves = currentUser.isKoboOnlyShelvesSync();
  
        // New: Access control and owner check
        if (shelfId != null) {
            // Editing an existing shelf. Retrieve the current shelf and check that it belongs to the current user.
            Shelf existingShelf = shelfService.getShelfById(shelfId);
            if (existingShelf == null || !existingShelf.getOwner().equals(currentUser)) {
                throw new AccessControlException("Unauthorized attempt to modify a shelf");
            }
        } else {
            // Creating a new shelf. Check if currentUser is allowed to create one.
            if (!currentUser.canCreateShelf()) {
                throw new AccessControlException("User not permitted to create a shelf");
            }
            // Set the current user as the owner of the new shelf.
            shelf.setOwner(currentUser);
        }
  
        // Replace the previous handling of the "is_public" flag.
        // Using the Request parameter as the source for the is_public value.
        String isPublicRaw = request.getParameter("is_public");
        // Assuming the parameter "on" means true.
        int isPublic = (isPublicRaw != null && !isPublicRaw.isEmpty() && "on".equals(isPublicRaw)) ? 1 : 0;
  
        // Retrieve the shelf title from the request parameters.
        String shelfTitle = request.getParameter("title") != null ? request.getParameter("title") : "";
  
        // Validate the shelf's uniqueness.
        if (checkShelfIsUnique(shelf, shelfTitle, isPublic, shelfId)) {
            shelf.setName(shelfTitle);
            shelf.setIsPublic(isPublic);
            String shelfAction;
            String flashText;
  
            if (shelfId == null) {
                // For a new shelf, assign the current user's id and add the shelf.
                shelf.setUserId(currentUser.getId());
                Session.add(shelf);
                shelfAction = "created";
                flashText = "Shelf " + shelfTitle + " created";
            } else {
                shelfAction = "changed";
                flashText = "Shelf " + shelfTitle + " changed";
            }
  
            try {
                Session.commit();
                Logger.info("Shelf " + shelfTitle + " " + shelfAction);
                return ShelfResponse.redirect("/shelf/show_shelf?id=" + shelf.getId(),
                                                flashText, "success");
            } catch (CalibreErrors.OperationalError ex) {
                Session.rollback();
                Logger.error("Settings DB is not Writeable");
            } catch (CalibreErrors.InvalidRequestError ex) {
                Session.rollback();
                Logger.error("Settings DB is not Writeable");
            } catch (Exception ex) {
                Session.rollback();
                Logger.error("There was an error");
                }
            }
        }
        return ShelfResponse.template("shelf_edit.html");
    }

    private static boolean checkShelfIsUnique(Shelf shelf, String title, int isPublic, Integer shelfId) {
        return title != null && title.length() > 0;
    }

    static class Logger {
        static void info(String msg) {}
        static void error(String msg) {}
    }
}
