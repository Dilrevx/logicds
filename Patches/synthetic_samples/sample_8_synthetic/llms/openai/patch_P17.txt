import common.*;

public class SSHKeyExchange{
    private SSHSession session;
    private SSHCrypto crypto;
    private SSHKex server;
    private SSHKex client;

    public SSHKeyExchange(SSHSession session) {
        this.session = session;
        this.crypto = session.getNextCrypto();
        this.server = crypto.getServerKex();
        this.client = crypto.getClientKex();
    }


    public int selectMethods() {
        // Retrieve methods from client and server, and check for null values.
        String[] clientMethods = client.getMethods();
        String[] serverMethods = server.getMethods();
        if (clientMethods == null || serverMethods == null) {
            // Return error code or throw an exception as appropriate.
            return -1;
        }
        
        // Initialize extStart safely.
        String extStart = null;
        // Iterate over both arrays to select a mutually supported method.
        for (String clientMethod : clientMethods) {
            // Ignore invalid client method entries.
            if (clientMethod == null || clientMethod.isEmpty()) {
                continue;
            }
            for (String serverMethod : serverMethods) {
                // Ignore invalid server method entries.
                if (serverMethod == null || serverMethod.isEmpty()) {
                    continue;
                }
                // Use exact match; if cryptographic methods are compared,
                // consider constant-time comparison if needed.
                if (clientMethod.equals(serverMethod)) {
                    extStart = clientMethod;
                    break;
                }
            }
            if (extStart != null) {
                break;
            }
        }
        
        // Ensure a mutually supported method has been found.
        if (extStart == null) {
            // No matching method found. Handle this error condition appropriately.
            return -1;
        }
        
        // Proceed with selecting the method.
        // Assume selectMethodByName performs further validation and returns an identifier.
        int selectedMethodId = selectMethodByName(extStart);
        return selectedMethodId;
    }


    private String findSubstring(String source, String target) {
        int index = source.indexOf(target);
        return index == -1 ? null : source.substring(index);
    }

    private String truncateString(String source, String toRemove) {
        return source.replace(toRemove, "");
    }

    private String findMatching(String serverMethod, String clientMethod) {
        String[] serverTokens = serverMethod.split(",");
        String[] clientTokens = clientMethod.split(",");

        for (String clientToken : clientTokens) {
            for (String serverToken : serverTokens) {
                if (serverToken.equals(clientToken)) {
                    return clientToken;
                }
            }
        }
        return null;
    }

    private String findAeadHmac(String method) {
        return method != null && method.contains("AEAD") ? "HMAC" : null;
    }

    public SSHCrypto getSSHCrypto() {
        return crypto;
    }
}
