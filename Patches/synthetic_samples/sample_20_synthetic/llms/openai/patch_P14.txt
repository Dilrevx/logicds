package main;

import commons.*;

public class X509Verifier {
    public static void x509_crt_verify_name(MbedtlsX509Crt crt, String cn, int[] flags) {
        MbedtlsX509Name     name;
        MbedtlsX509Sequence cur;
        int                 cn_len = cn.length();
        if ((crt.extTypes & MbedtlsConstants.MBEDTLS_X509_EXT_SUBJECT_ALT_NAME) != 0) {
            for (cur = crt.subjectAltNames; cur != null; cur = cur.next) {

/* 
 * Updated fix: When verifying the certificate's common name, restrict the comparison 
 * to subjectAltName entries of type dNSName. This avoids matching certificate names of other
 * types (such as those corresponding to IP addresses) which could be misused.
 *
 * Note: This repair assumes that the function x509_crt_check_cn() has been modified to accept
 * an additional parameter that specifies the allowed SAN type (e.g. MBEDTLS_X509_SAN_DNS_NAME)
 * and only compare against entries of that type.
 */
if (CommonUtils.x509_crt_check_cn(cur.buf, cn, cn_len, MBEDTLS_X509_SAN_DNS_NAME) == 0)
    break;

                    break;

            }
            if (cur == null)
                flags[0] |= MbedtlsConstants.MBEDTLS_X509_BADCERT_CN_MISMATCH;
        } else {
            for (name = crt.subject; name != null; name = name.next) {
                if (CommonUtils.MBEDTLS_OID_CMP(MbedtlsConstants.MBEDTLS_OID_AT_CN, name.oid) == 0
                        && CommonUtils.x509_crt_check_cn(name.val, cn, cn_len) == 0) {
                    break;
                }
            }
            if (name == null)
                flags[0] |= MbedtlsConstants.MBEDTLS_X509_BADCERT_CN_MISMATCH;
        }
    }
}
