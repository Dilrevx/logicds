package main;

import commons.*;

public class X509Verifier {

    public static void x509_crt_verify_name(MbedtlsX509Crt crt, String cn, int[] flags) {
        MbedtlsX509Name name;
        MbedtlsX509Sequence cur;
        int cn_len = cn.length();

        // When the certificate includes SubjectAltName, only DNS names should be used to verify the expected hostname.
        if ((crt.extTypes & MbedtlsConstants.MBEDTLS_X509_EXT_SUBJECT_ALT_NAME) != 0) {
            for (cur = crt.subjectAltNames; cur != null; cur = cur.next) {
                // Only consider entries that are DNS names.
                // It is assumed that cur.tag indicates the type of the alternative name,
                // and MBEDTLS_X509_SAN_DNS corresponds to DNS entries.
                if (cur.tag == MbedtlsConstants.MBEDTLS_X509_SAN_DNS) {
                    if (CommonUtils.x509_crt_check_cn(cur.buf, cn, cn_len) == 0) {
                        // Found a match in the valid dNSName field; verification success.
                        return;
                    }
                }
            }
            // No valid DNS name in the SubjectAltName extension matched: mark as CN mismatch.
            flags[0] |= MbedtlsConstants.MBEDTLS_X509_BADCERT_CN_MISMATCH;
        } else {
            // Fallback: iterate over the subject names for a commonName match.
            for (name = crt.subject; name != null; name = name.next) {
                if (CommonUtils.x509_crt_check_cn(name.buf, cn, cn_len) == 0) {
                    return;
                }
            }
            // If no match found in the subject's common name, mark as CN mismatch.
            flags[0] |= MbedtlsConstants.MBEDTLS_X509_BADCERT_CN_MISMATCH;
        }
    }

        }
    }
}
