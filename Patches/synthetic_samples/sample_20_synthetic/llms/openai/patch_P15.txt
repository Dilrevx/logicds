package main;

import commons.*;

public class X509Verifier {
    public static void x509_crt_verify_name(MbedtlsX509Crt crt, String cn, int[] flags) {
        MbedtlsX509Name     name;
        MbedtlsX509Sequence cur;
        int                 cn_len = cn.length();
        if ((crt.extTypes & MbedtlsConstants.MBEDTLS_X509_EXT_SUBJECT_ALT_NAME) != 0) {
            for (cur = crt.subjectAltNames; cur != null; cur = cur.next) {

                /* Repaired call: verify the certificate’s subject name with proper SAN extension handling.
                 * The new function inspects the subjectAltName extension, only matching names of the correct type
                 * (e.g. matching a DNS name to an expected domain name). It falls back to the common name check only if
                 * no SAN entries exist. This change prevents mismatches where an IP address in SAN is incorrectly
                 * accepted as a domain name.
                 */
                if (CommonUtils.x509_crt_verify_subject_name(cur.buf, cn, cn_len) == 0)
                    break;

                    break;

            }
            if (cur == null)
                flags[0] |= MbedtlsConstants.MBEDTLS_X509_BADCERT_CN_MISMATCH;
        } else {
            for (name = crt.subject; name != null; name = name.next) {
                if (CommonUtils.MBEDTLS_OID_CMP(MbedtlsConstants.MBEDTLS_OID_AT_CN, name.oid) == 0
                        && CommonUtils.x509_crt_check_cn(name.val, cn, cn_len) == 0) {
                    break;
                }
            }
            if (name == null)
                flags[0] |= MbedtlsConstants.MBEDTLS_X509_BADCERT_CN_MISMATCH;
        }
    }
}
