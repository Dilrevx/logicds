package main;

import commons.*;

public class X509Verifier {
    public static void x509_crt_verify_name(MbedtlsX509Crt crt, String cn, int[] flags) {
        // First check if the certificate has a Subject Alternative Name extension.
        MbedtlsX509Sequence sanSeq = crt.getSubjectAltNameSequence();
        if (sanSeq != null && !sanSeq.isEmpty()) {
            // Determine if the expected name represents an IP address.
            boolean expectIP = isIpAddress(cn);

            // Iterate over all SAN entries.
            for (MbedtlsX509Name altName : sanSeq) {
                int nameType = altName.getType();

                // Only compare if the entry type matches the expected type.
                if (expectIP && nameType != MBEDTLS_X509_SAN_IP_ADDRESS) {
                    continue;
                }
                if (!expectIP && nameType != MBEDTLS_X509_SAN_DNS_NAME) {
                    continue;
                }

                String certName = altName.getValue();
                if (certName != null && certName.equalsIgnoreCase(cn)) {
                    // Success: clear any error flag and exit.
                    flags[0] &= ~MBEDTLS_X509_BAD_ID;
                    return;
                }
            }
            // No matching SAN was found; mark the error flag and exit.
            flags[0] |= MBEDTLS_X509_BAD_ID;
            return;
        }

        // Fall back to verifying the commonName in the certificate subject.
        MbedtlsX509Name subject = crt.getSubject();
        for (MbedtlsX509Name name = subject; name != null; name = name.getNext()) {
            // Check only entries with the OID for commonName ("2.5.4.3").
            if (name.hasOid("2.5.4.3")) {
                String certCN = name.getValue();
                if (certCN != null && certCN.equalsIgnoreCase(cn)) {
                    flags[0] &= ~MBEDTLS_X509_BAD_ID;
                    return;
        }
    }
}
