package main;

import commons.*;

public class X509CrtRevocationChecker {
    public static int mbedtls_x509_crt_is_revoked(
            MbedtlsX509Crt crt, MbedtlsX509Crl crl) {
        MbedtlsX509CrlEntry cur = crl.entry;
        while (cur != null) {
            // Use the trusted time for checking if the certificate is revoked.
            // If the trusted time is at or after the revoked date, treat the certificate as revoked.
            if (trustedTime >= cur.getRevocationDate().getTime()) {
                return CERT_REVOKED;
            }
            cur = cur.next;
        }
        return 0;
    }
}
