package main;

import commons.*;

public class X509CrtRevocationChecker {

    // Helper method to convert a custom X509 time (MbedtlsX509Time) into epoch milliseconds.
    private static long convertMbedtlsX509TimeToMillis(MbedtlsX509Time x509Time) {
        Calendar cal = Calendar.getInstance();
        // Assuming MbedtlsX509Time has fields for year, mon, day, hour, min, and sec.
        cal.set(Calendar.YEAR, x509Time.year);
        // Calendar.MONTH is 0-based. Assuming 'mon' is 1-based.
        cal.set(Calendar.MONTH, x509Time.mon - 1);
        cal.set(Calendar.DAY_OF_MONTH, x509Time.day);
        cal.set(Calendar.HOUR_OF_DAY, x509Time.hour);
        cal.set(Calendar.MINUTE, x509Time.min);
        cal.set(Calendar.SECOND, x509Time.sec);
        cal.set(Calendar.MILLISECOND, 0);
        return cal.getTimeInMillis();
    }

    public static int mbedtls_x509_crt_is_revoked(MbedtlsX509Crt crt, MbedtlsX509Crl crl) {
        MbedtlsX509CrlEntry cur = crl.entry;

        // Iterate through all entries in the CRL.
        while (cur != null) {
            // Verify that the certificate's serial number matches the CRL entry's serial number.
            if (crt.serial.len == cur.serial.len &&
                MemUtil.memcmp(crt.serial.p, cur.serial.p, crt.serial.len) == 0) {
                // If the current system time is at or after the revocation date,
                // consider the certificate revoked.
                if (System.currentTimeMillis() >= convertMbedtlsX509TimeToMillis(cur.revocationDate)) {
                    return 1;  // Certificate is revoked.
                }
            }
            // Move on to the next CRL entry.
            cur = cur.next;
        }

        // No matching CRL entry with a past revocation date was found.
        return 0;
    }
}
