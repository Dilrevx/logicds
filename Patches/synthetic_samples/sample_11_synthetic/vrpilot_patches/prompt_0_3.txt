package main;

import commons.*;

public class X509CrtRevocationChecker {
    public static int mbedtls_x509_crt_is_revoked(
            MbedtlsX509Crt crt, MbedtlsX509Crl crl) {
        MbedtlsX509CrlEntry cur = crl.entry;
        
        // Iterate through all entries in the CRL.
        while (cur != null) {
            // Verify that the certificate's serial number matches the CRL entry's serial number.
            if (crt.serial.len == cur.serial.len &&
                    MemUtil.memcmp(crt.serial.p, cur.serial.p, crt.serial.len) == 0) {
                // If the current system time is at or after the revocation date,
                // consider the certificate revoked.
                if (System.currentTimeMillis() >= cur.getRevocationDate().getTime()) {
                    return 1;  // Certificate is revoked.
                }
            }
            // Move on to the next CRL entry.
            cur = cur.next;
        }
        
        // No matching CRL entry with a past revocation date was found.
        return 0;
    }
}
