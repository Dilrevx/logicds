package main;

import commons.*;

public class X509CrtRevocationChecker {
    public static int mbedtls_x509_crt_is_revoked(
            MbedtlsX509Crt crt, MbedtlsX509Crl crl) {
        MbedtlsX509CrlEntry cur = crl.entry;
        while (cur != null && cur.serial.len != 0) {

            if (crt.serial.len == cur.serial.len
                    && MemUtil.memcmp(crt.serial.p, cur.serial.p, crt.serial.len) == 0) {
                if (cur.revocationDate.type == MBEDTLS_X509_TIME_DATE && Helper.mbedtls_x509_time_is_past(cur.revocationDate))
                    return 1;
                else if (cur.revocationDate.type == MBEDTLS_X509_TIME_GENERALIZED_TIME) {
                    time_t revocationTime = mbedtls_x509_get_time_t(&cur.revocationDate);
                    time_t currentTime = time(NULL);
                    if (revocationTime <= currentTime)
                        return 1;
                }
            }

            cur = cur.next;
        }
        return 0;
    }
}
