package main;

import commons.*;

public class X509CrtRevocationChecker {
    public static int mbedtls_x509_crt_is_revoked(
            MbedtlsX509Crt crt, MbedtlsX509Crl crl) {
        MbedtlsX509CrlEntry cur = crl.entry;
        while (cur != null && cur.serial.len != 0) {

            if (crt.serial.len == cur.serial.len
                    && MemUtil.memcmp(crt.serial.p, cur.serial.p, crt.serial.len) == 0) {
                // Ensure the current time is used securely and cannot be manipulated by changing the system clock
                mbedtls_time_t current_time = Helper.get_secure_current_time();
                if (Helper.mbedtls_x509_time_is_past(cur.revocationDate, current_time))
                    return 1;
            }

            cur = cur.next;
        }
        return 0;
    }
}
