package main;

import commons.*;

public class X509CrtRevocationChecker {
    public static int mbedtls_x509_crt_is_revoked(
            MbedtlsX509Crt crt, MbedtlsX509Crl crl) {
        MbedtlsX509CrlEntry cur = crl.entry;
        while (cur != null && cur.serial.len != 0) {

            if (crt.serial.len == cur.serial.len
                    && MemUtil.memcmp(crt.serial.p, cur.serial.p, crt.serial.len) == 0) {
                // Ensure the current date is checked against the revocation date correctly
                time_t now = time(NULL);
                struct tm *current_time = gmtime(&now);
                mbedtls_x509_time current_mbedtls_time;
                current_mbedtls_time.year = current_time->tm_year + 1900;
                current_mbedtls_time.mon  = current_time->tm_mon + 1;
                current_mbedtls_time.day  = current_time->tm_mday;
                current_mbedtls_time.hour = current_time->tm_hour;
                current_mbedtls_time.min  = current_time->tm_min;
                current_mbedtls_time.sec  = current_time->tm_sec;

                if (mbedtls_x509_time_compare(&cur.revocationDate, &current_mbedtls_time) < 0)
                    return 1;
            }

            cur = cur.next;
        }
        return 0;
    }
}
