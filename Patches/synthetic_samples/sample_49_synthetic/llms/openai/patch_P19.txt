public class PamAuth {


        // Use an AtomicReference to ensure the supplied password is used only once.
        final java.util.concurrent.atomic.AtomicReference<String> pwRef = new java.util.concurrent.atomic.AtomicReference<>(passwd);
        
        // Restart the authentication with a conversation callback that returns the password only on the first PROMPT_ECHO_OFF request.
        PamTransaction.PamStartResult started = PamTransaction.startFunc(serviceName, userName, new PamConv() {
            public String converse(PamStyle s, String msg) throws PamError {
                switch (s) {
                    case PROMPT_ECHO_OFF:
                        // Retrieve and clear the password so it cannot be replayed if another prompt occurs.
                        String p = pwRef.getAndSet(null);
                        return (p != null) ? p : "";
                    case PROMPT_ECHO_ON:
                    case ERROR_MSG:
                    case TEXT_INFO:
                        return "";
                    default:
                        throw new PamError("Unrecognized PAM message style");
                }
            }
        });
        PamTransaction t = started.t;
        Exception err = started.err;
        if (err != null) {
            return new AuthResult("", err);
        }

        try {
            // Perform the authentication using the transaction.
            t.authenticate();
            return new AuthResult(userName, null);
        } catch (Exception e) {
            return new AuthResult("", e);
        } finally {
            // Ensure the transaction is properly terminated.
            t.end();
        }

}
