public class PamAuth {


    public AuthResult auth(String serviceName, String userName, final String passwd) {
        // Added authorization check before proceeding with authentication.
        if (!isAuthorizedService(serviceName, userName)) {
            return new AuthResult("", new PamError("User not authorized for service: " + serviceName));
        }
        
        PamTransaction.PamStartResult started = PamTransaction.startFunc(serviceName, userName, new PamConv() {
            public String converse(PamStyle s, String msg) throws PamError {
                switch (s) {
                    case PROMPT_ECHO_OFF:
                        return passwd;
                    case PROMPT_ECHO_ON:
                    case ERROR_MSG:
                    case TEXT_INFO:
                        return "";
                    default:
                        throw new PamError("Unrecognized PAM message style");
                }
            }
        });
        PamTransaction t = started.t;
        Exception err = started.err;
        if (err != null) {
            return new AuthResult("", err);
        }
        
        err = t.authenticate(0);
        if (err != null) {
            return new AuthResult("", err);
        }
        
        PamTransaction.ItemResult item = t.getItem(PamItem.USER);
        return new AuthResult(item.value, item.err);
    }
    
    // Helper method to check if the service and user are authorized.
    private boolean isAuthorizedService(String serviceName, String userName) {
        // Example authorization logic:
        // Only allow authentication for the specific PAM service designed for this application.
        // You can customize the logic as per your authorization requirements.
        if (serviceName == null || userName == null || userName.isEmpty()) {
            return false;
        }
        // In this example, we allow only a specific service name. Adjust as needed.
        if ("gitea-pam-service".equals(serviceName)) {
            return true;
        }
        return false;
    }

}
