public class PamAuth {

    public AuthResult auth(String serviceName, String userName, final String passwd) {
        PamTransaction.PamStartResult started = PamTransaction.startFunc(serviceName, userName, new PamConv() {
            public String converse(PamStyle s, String msg) throws PamError {
                switch (s) {
                    case PROMPT_ECHO_OFF:
                        return passwd;
                    case PROMPT_ECHO_ON:
                    case ERROR_MSG:
                    case TEXT_INFO:
                        return "";
                    default:
                        throw new PamError("Unrecognized PAM message style");
                }
            }
        });
        PamTransaction t = started.t;
        Exception err = started.err;
        if (err != null) {
            return new AuthResult("", err);
        }

        err = t.authenticate(0);
        if (err != null) {
            return new AuthResult("", err);
        }


        err = t.authenticate(0);
        if (err != null) {
            return new AuthResult("", err);
        }
        
        // Added authorization check to ensure that the user has permission to use the service.
        if (!t.isAuthorized(userName, serviceName)) {
            return new AuthResult("", "Authorization failed: User '" + userName + "' is not permitted to access service '" + serviceName + "'.");
        }
        
        PamTransaction.ItemResult item = t.getItem(PamItem.USER);


        PamTransaction.ItemResult item = t.getItem(PamItem.USER);
        return new AuthResult(item.value, item.err);
    }
}
