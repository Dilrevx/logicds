public class PamAuth {

    public AuthResult auth(String serviceName, String userName, final String passwd) {
        // Create a PamTransaction using the provided credentials.
        PamTransaction t = new PamTransaction(serviceName, userName, passwd);

        // Attempt to authenticate. The authenticate() method returns an error code.
        int err = t.authenticate(0);
        if (err != 0) {
            // If an error occurred, return an AuthResult with an empty user value.
            return new AuthResult("", err);
        } else {
            // Otherwise, retrieve the PamItem.USER and return its value.
            PamItem userItem = t.getItem(PamItem.USER);
            return new AuthResult(userItem.getValue(), 0);
    }
}
