public class PamAuth {

    public AuthResult auth(String serviceName, String userName, final String passwd) {
        // Obtain or create the PamTransaction using provided credentials.
        PamTransaction t = new PamTransaction(serviceName, userName, passwd);

        // Authenticate and check if there's an error.
        int err = t.authenticate(0);
        if (err != 0) {
            // If an error occurred, return early with an empty user value.
            return new AuthResult("", err);
        } else {
            // Otherwise, retrieve the PamItem.USER and return its result.
        PamTransaction.ItemResult item = t.getItem(PamItem.USER);
        return new AuthResult(item.value, item.err);
    }
}
