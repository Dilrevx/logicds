public class CaptchaService {

    private final CaptchaRepo captchaRepo;

    public CaptchaService(CaptchaRepo captchaRepo) {
        this.captchaRepo = captchaRepo;
    }


    CaptchaRepo.CaptchaResult fetched = captchaRepo.getCaptcha(ctx, key);
    if (fetched.err != null || fetched.value == null) {
        return new VerifyResult(false, "Captcha retrieval error");
    }
    String realCaptcha = fetched.value;
    
    // Invalidate the captcha to prevent replay attacks.
    // Remove or mark the captcha as used so that it cannot be reused.
    captchaRepo.deleteCaptcha(ctx, key);
    
    boolean verified = trimSpace(captcha).equals(realCaptcha);
    return new VerifyResult(verified, verified ? "Captcha verified" : "Captcha mismatch");


    private static String trimSpace(String s) {
        return s == null ? "" : s.trim();
    }
}
