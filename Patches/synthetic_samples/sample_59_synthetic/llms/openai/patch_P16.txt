public class CaptchaService {

    private final CaptchaRepo captchaRepo;

    public CaptchaService(CaptchaRepo captchaRepo) {
        this.captchaRepo = captchaRepo;
    }

    public VerifyResult verifyCaptcha(Context ctx, String key, String captcha) {

        // Fetch the captcha using the provided key.
        CaptchaRepo.CaptchaResult fetched = captchaRepo.getCaptcha(ctx, key);
        if (fetched.err != null) {
            // Log the error for auditing purposes.
            logger.error("Error fetching captcha for key: " + key, fetched.err);
            return new VerifyResult(false, null);
        }

        // Attempt to delete the captcha to prevent replay attacks.
        try {
            captchaRepo.deleteCaptcha(ctx, key);
        } catch (Exception deletionException) {
            // Log the deletion error.
            logger.error("Error deleting captcha for key: " + key, deletionException);
            return new VerifyResult(false, null);
        }

        // Verify the trimmed user input against the fetched captcha value.
        return new VerifyResult(trimSpace(captcha).equals(fetched.value), null);

    }

    private static String trimSpace(String s) {
        return s == null ? "" : s.trim();
    }
}
