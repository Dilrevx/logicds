public class CaptchaService {

    private final CaptchaRepo captchaRepo;

    public CaptchaService(CaptchaRepo captchaRepo) {
        this.captchaRepo = captchaRepo;
    }

    public VerifyResult verifyCaptcha(Context ctx, String key, String captcha) {

import java.nio.charset.StandardCharsets;
import java.security.MessageDigest;

// Retrieve the captcha result.
CaptchaRepo.CaptchaResult fetched = captchaRepo.getCaptcha(ctx, key);
if (fetched.err != null || fetched.value == null) {
    // Remove any stored captcha to prevent replay attacks.
    captchaRepo.deleteCaptcha(ctx, key);
    return new VerifyResult(false, null);
}
// Remove the captcha immediately to prevent reuse.
captchaRepo.deleteCaptcha(ctx, key);

// Compare using a constant‐time algorithm to prevent timing attacks.
boolean valid = MessageDigest.isEqual(
    trimSpace(captcha).getBytes(StandardCharsets.UTF_8),
    fetched.value.getBytes(StandardCharsets.UTF_8)
);
return new VerifyResult(valid, null);

    }

    private static String trimSpace(String s) {
        return s == null ? "" : s.trim();
    }
}
