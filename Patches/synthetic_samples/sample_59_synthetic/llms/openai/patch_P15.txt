public class CaptchaService {

    private final CaptchaRepo captchaRepo;

    public CaptchaService(CaptchaRepo captchaRepo) {
        this.captchaRepo = captchaRepo;
    }

    public VerifyResult verifyCaptcha(Context ctx, String key, String captcha) {

        // Fetch the captcha from the repository
        CaptchaRepo.CaptchaResult fetched = captchaRepo.getCaptcha(ctx, key);
        if (fetched.err != null) {
            // Log the error for audit purposes
            System.err.println("Error fetching captcha for key " + key + ": " + fetched.err.getMessage());
            return new VerifyResult(false, null);
        }
        
        // Delete the captcha to prevent replay attacks
        Exception deletionErr = captchaRepo.deleteCaptcha(ctx, key);
        if (deletionErr != null) {
            // Log the deletion error and avoid proceeding with verification
            System.err.println("Error deleting captcha for key " + key + ": " + deletionErr.getMessage());
            return new VerifyResult(false, null);
        }
        
        // Compare the trimmed user input with the stored captcha value
        return new VerifyResult(trimSpace(captcha).equals(fetched.value), null);

    }

    private static String trimSpace(String s) {
        return s == null ? "" : s.trim();
    }
}
