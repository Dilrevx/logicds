public class CaptchaService {

    private final CaptchaRepo captchaRepo;

    public CaptchaService(CaptchaRepo captchaRepo) {
        this.captchaRepo = captchaRepo;
    }

    /**
     * Verifies the provided captcha token.
     *
     * @param captchaId    the unique identifier of the captcha
     * @param userSolution the captcha answer provided by the user
     * @return true if the captcha is valid, false otherwise.
     */
    public boolean verifyCaptcha(String captchaId, String userSolution) {
        // Retrieve the captcha record using the identifier.
        CaptchaRecord record = captchaRepo.findById(captchaId);
        if (record == null) {
            return false;
    }

    private static String trimSpace(String s) {
        return s == null ? "" : s.trim();
    }
}
