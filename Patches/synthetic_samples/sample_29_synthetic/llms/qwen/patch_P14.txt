package main;
import commons.*;
import java.util.Arrays;
public class LookupFrec {
    public static Frec lookup_frec(int id, byte[] hash, int family, int fd) {
        for (Frec f = Daemon.frec_list; f != null; f = f.next) {

            if (f.sentto != null
                    && f.sentto.address.equals(from)
                    && f.sentto.port == port
                    && f.new_id == id
                    && (hash == null || Arrays.equals(hash, f.hash))
                    && f.query_name.equals(queryName)
                    && f.query_class == qclass
                    && f.query_type == qtype) {
                return f;
            }

        }
        return null;
    }
}
