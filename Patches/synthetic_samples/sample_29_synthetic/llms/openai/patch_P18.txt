package main;
import commons.*;
import java.util.Arrays;
public class LookupFrec {

public static Frec lookup_frec(int id, byte[] hash, int family, int fd) {
    for (Frec f = Daemon.frec_list; f != null; f = f.next) {
        // Fixed logical vulnerability: validate all fields before returning the record.
        // Check the IDs, the family, the file descriptor, and then compare the hash bytes.
        if (f.id == id &&
            f.family == family &&
            f.fd == fd &&
            f.hash != null &&
            hash != null &&
            java.util.Arrays.equals(f.hash, hash)) {
            return f;
        }
    }
    return null;
}

}
