package main;
import commons.*;
import java.util.Arrays;
public class LookupFrec {
    public static Frec lookup_frec(int id, byte[] hash, int family, int fd) {
        for (Frec f = Daemon.frec_list; f != null; f = f.next) {

            if (f.sentto != null
                    && f.new_id == id
                    && (hash == null || Arrays.equals(hash, f.hash))) {

                // Begin additional network channel validation.
                // Assume that 'replySocket' is an object representing the socket on which the reply
                // was received and that it provides a method getFileDescriptor() to retrieve its FD.
                boolean validReply = false;

                // Check if the query was sent over IPv4.
                if (f.sentto.getAddressFamily() == AddressFamily.IPv4 && f.ipv4Socket != null) {
                    if (f.ipv4Socket.getFileDescriptor() == replySocket.getFileDescriptor()) {
                        validReply = true;
                    }
                }
                // Check if the query was sent over IPv6.
                else if (f.sentto.getAddressFamily() == AddressFamily.IPv6 && f.ipv6Socket != null) {
                    if (f.ipv6Socket.getFileDescriptor() == replySocket.getFileDescriptor()) {
                        validReply = true;
                    }
                }
                // Fallback: if the sent-to structure holds a specific socket, validate its FD.
                else if (f.sentto.getSocket() != null) {
                    if (f.sentto.getSocket().getFileDescriptor() == replySocket.getFileDescriptor()) {
                        validReply = true;
                    }
                }

                if (validReply) {
                    return f;
                }
            }

        }
        return null;
    }
}
