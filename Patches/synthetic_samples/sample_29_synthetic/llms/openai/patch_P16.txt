package main;
import commons.*;
import java.util.Arrays;
public class LookupFrec {
    public static Frec lookup_frec(int id, byte[] hash, int family, int fd) {
        for (Frec f = Daemon.frec_list; f != null; f = f.next) {

            if (f.sentto != null
                    && f.new_id == id
                    && (hash == null || Arrays.equals(hash, f.hash))) {

                // Additional network channel validation:
                // 'fd' is assumed to be the file descriptor (an int) on which the reply was received.
                // Depending on the address family used for the original query, check the matching file descriptor.
                boolean validFd = false;

                // For IPv4: verify that the IPv4-specific file descriptor matches.
                if (f.sentto.getAddress() instanceof Inet4Address) {
                    if (f.ipv4Fd != null && f.ipv4Fd.intValue() == fd) {
                        validFd = true;
                    }
                }
                // For IPv6: verify that the IPv6-specific file descriptor matches.
                else if (f.sentto.getAddress() instanceof Inet6Address) {
                    if (f.ipv6Fd != null && f.ipv6Fd.intValue() == fd) {
                        validFd = true;
                    }
                }

                // Fallback check: if a dedicated socket descriptor exists in the sent-to structure, validate it.
                if (!validFd && f.sentto.getSocketFd() != null) {
                    if (f.sentto.getSocketFd().intValue() == fd) {
                        validFd = true;
                    }
                }

                if (validFd) {
                    return f;
                }
            }

        }
        return null;
    }
}
