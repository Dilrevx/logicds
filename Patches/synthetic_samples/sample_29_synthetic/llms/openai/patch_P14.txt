package main;
import commons.*;
import java.util.Arrays;
public class LookupFrec {
    public static Frec lookup_frec(int id, byte[] hash, int family, int fd) {
        for (Frec f = Daemon.frec_list; f != null; f = f.next) {

            if (f.sentto != null
                    && f.new_id == id
                    && (hash == null || Arrays.equals(hash, f.hash))
                    && reply_src.equals(f.sentto.getAddress())       // reply's source must match query destination
                    && reply_dst.equals(f.local_ip)                    // reply's destination must match our local address
                    && reply_dst_port == f.local_port) {               // reply's destination port must match our query port
                return f;
            }

        }
        return null;
    }
}
