package main;
import commons.*;
import java.util.Arrays;
public class LookupFrec {
    public static Frec lookup_frec(int id, byte[] hash, int family, int fd) {
        for (Frec f = Daemon.frec_list; f != null; f = f.next) {

            // Require non-null hash values so that an omitted hash cannot bypass verification.
            if (f.sentto != null
                    && f.new_id == id
                    && hash != null
                    && f.hash != null
                    && MessageDigest.isEqual(hash, f.hash)) {
                return f;
            }

        }
        return null;
    }
}
