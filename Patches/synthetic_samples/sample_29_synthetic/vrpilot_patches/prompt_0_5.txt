package main;
import commons.*;
import java.util.Arrays;
public class LookupFrec {
public static Frec lookup_frec(int id, byte[] hash, int family, int fd) {
    // To prevent concurrent modifications, synchronize on the global list.
    synchronized (Daemon.frec_list) {
        Frec prev = null;
        // Traverse the linked list.
        for (Frec curr = Daemon.frec_list; curr != null; ) {
            // Check that the node is valid (sentto not null), has the matching id,
            // and that both hashes match (handling possible nulls).
            if (curr.new_id == id &&
                curr.sentto != null &&
                ((hash == null && curr.hash == null) ||
                 (hash != null && Arrays.equals(hash, curr.hash))))
            {
                // Remove the matching node from the list.
                if (prev == null) {
                    Daemon.frec_list = curr.next;
                } else {
                    prev.next = curr.next;
                }
                return curr;
            } else {
                prev = curr;
                curr = curr.next;
            }
            }
        }
        return null;
    }
}
