package main;
import commons.*;
import java.util.Arrays;
public class LookupFrec {
    public static Frec lookup_frec(int id, byte[] hash, int family, int fd) {
        synchronized(pendingForwardQueries) {
            Iterator<Frec> iterator = pendingForwardQueries.iterator();
            while (iterator.hasNext()) {
                Frec frec = iterator.next();
                // Fix: require an exact match for ID, family, hash, and destination fd.
                if (frec.getId() == id &&
                    frec.getFamily() == family &&
                    Arrays.equals(frec.getHash(), hash) &&
                    frec.getDestinationFD() == fd) {
                    // Remove it so that the same pending query cannot be reused.
                    iterator.remove();
                    return frec;
                }
            }
        }
        return null;
    }
}
