package main;
import commons.*;
import java.util.Arrays;
public class LookupFrec {
    public static Frec lookup_frec(int id, byte[] hash, int family, int fd) {
        // Synchronize on the global linked list of pending queries.
        synchronized (Daemon.frec_list) {
            // Walk the linked list while keeping track of the previous node.
            Frec prev = null;
            for (Frec curr = Daemon.frec_list; curr != null; ) {
                if (curr.new_id == id &&
                    curr.family == family &&
                    // For the hash, require both are null or that both exist and are equal.
                    ((hash == null && curr.hash == null) ||
                     (hash != null && Arrays.equals(hash, curr.hash))) &&
                    // Check that the destination file descriptor matches.
                    curr.destfd == fd)
                {
                    // Remove the matching entry from the linked list.
                    if (prev == null) { // removing the head.
                        Daemon.frec_list = curr.next;
                    } else {
                        prev.next = curr.next;
                    }
                    return curr;
                } else {
                    // Advance in the list.
                    prev = curr;
                    curr = curr.next;
                }
            }
        }
        return null;
    }
}
