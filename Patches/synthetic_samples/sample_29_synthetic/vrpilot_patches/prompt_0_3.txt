package main;
import commons.*;
import java.util.Arrays;
public class LookupFrec {
    public static Frec lookup_frec(int id, byte[] hash, int family, int fd) {
        synchronized(Daemon.pendingForwardQueries) {
            Iterator<Frec> iter = Daemon.pendingForwardQueries.iterator();
            while (iter.hasNext()) {
                Frec pending = iter.next();
                // Require an exact match for id, family, hash and destination fd.
                // For the hash, we check both are null or both non-null and equal.
                if (pending.new_id == id &&
                    pending.family == family &&
                    ((hash == null && pending.hash == null) ||
                     (hash != null && Arrays.equals(pending.hash, hash))) &&
                    pending.fd == fd)
                {
                    // Remove the pending query so it cannot be reused.
                    iter.remove();
                    return pending;
                }
            }
        }
        return null;
    }
}
