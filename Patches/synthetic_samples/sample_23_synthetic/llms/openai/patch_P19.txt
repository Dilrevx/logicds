package main;

import commons.*;

public class Tls12Builder {

    public static int BuildMessage(TlsContext ssl,
                                   byte[] output,
                                   int outSz,
                                   byte[] input,
                                   int inSz,
                                   int type,
                                   int hashOutput,
                                   int sizeOnly,
                                   int asyncOkay,
                                   int epochOrder) {
        // Validate critical pointers to avoid null dereferences.
        if (ssl == null || output == null || input == null) {
            return CommonsConstants.BAD_FUNC_ARG;
        }
        
        // Validate buffer sizes against the arrays.
        if (outSz < 0 || inSz < 0 || outSz > output.length || inSz > input.length) {
            return CommonsConstants.BAD_FUNC_ARG;
        }
        
        // Initialize local message argument structure.
        BuildMsgArgs args = new BuildMsgArgs();
        
        // Set the beginning state.
        ssl.options.buildMsgState = CommonsConstants.BUILD_MSG_BEGIN;
        
        // Set the Initialization Vector (IV) size if the protocol version requires it.
        if (ssl.options.tls1_1) {
            args.ivSz = CommonsConstants.BLOCK_SIZE;
        } else {
            args.ivSz = 0;
        }
        
        // If only the size of the output is requested, calculate and return it.
        if (sizeOnly != 0) {
            // Example: we assume a header is prepended to the input.
            int headerSize = CommonsConstants.HEADER_SIZE;
            int requiredSize = headerSize + inSz;
            return requiredSize;
        }
        
        // Ensure that the output buffer is large enough for header and message content.
        int headerSize = CommonsConstants.HEADER_SIZE;
        if (headerSize + inSz > outSz) {
            return CommonsConstants.SIZE_ERROR;
        }
        
        // Build the header.
        int pos = 0;
        // Example header: first byte is type, second is epochOrder.
        output[pos++] = (byte) type;
        output[pos++] = (byte) epochOrder;
        // Additional header fields (e.g., hash if needed) can be populated here.
        // For instance, if hashOutput is used to indicate whether a hash is included:
        if (hashOutput != 0) {
            // Assuming we include a one-byte flag for hash presence.
            output[pos++] = (byte) hashOutput;
        }
        // Zero-fill any remaining header bytes if headerSize exceeds the populated header.
        while (pos < headerSize) {
            output[pos++] = 0;
        }
        
        // Copy the input message to the output buffer after the header.
        System.arraycopy(input, 0, output, headerSize, inSz);
        
        // Optionally handle asynchronous operation flag if required.
        if (asyncOkay != 0) {
            // Insert asynchronous handling logic if applicable.
            // (For now, operation is synchronous.)
        }
        
        // Final total message size is header plus message payload.
        int totalMessageSize = headerSize + inSz;
        return totalMessageSize;
    }

}
