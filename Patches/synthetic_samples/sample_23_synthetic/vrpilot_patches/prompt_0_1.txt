package main;

import commons.*;

public class Tls12Builder {
    public static int BuildMessage(TlsContext ssl,
                                   byte[] output,
                                   int outSz,
                                   byte[] input,
                                   int inSz,
                                   int type,
                                   int hashOutput,
                                   int sizeOnly,
                                   int asyncOkay,
                                   int epochOrder) {
        BuildMsgArgs args;
        BuildMsgArgs lcl_args = new BuildMsgArgs();

        // For TLS 1.2 and TLS 1.1, ensure that for AES-CBC or DES3_CBC ciphers a random IV is used
        if (ssl.getProtocolVersion().equals("TLS1.2") || ssl.getProtocolVersion().equals("TLS1.1")) {
            String cipherName = ssl.getCipherAlgorithm();
            if ("AES_CBC".equals(cipherName) || "DES3_CBC".equals(cipherName)) {
                int blockSize = ssl.getBlockSize(); // e.g., 16 bytes for AES, 8 bytes for DES3
                byte[] iv = new byte[blockSize];
                SecureRandom random = new SecureRandom();
                random.nextBytes(iv);  // Generate a secure random IV
                ssl.setIV(iv);         // Set the random IV in the TLS context
    }
}
