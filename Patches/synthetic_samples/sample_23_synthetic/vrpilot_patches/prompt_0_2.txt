package main;

import commons.*;

public class Tls12Builder {
    public static int BuildMessage(TlsContext ssl,
                                   byte[] output,
                                   int outSz,
                                   byte[] input,
                                   int inSz,
                                   int type,
                                   int hashOutput,
                                   int sizeOnly,
                                   int asyncOkay,
                                   int epochOrder) {
        // Check for null TLS context
        if (ssl == null) {
            return CommonsConstants.BAD_FUNC_ARG;
        }
        
        // Create and initialize message arguments
        BuildMsgArgs args = new BuildMsgArgs();
        ssl.options.buildMsgState = CommonsConstants.BUILD_MSG_BEGIN;

        // For TLS 1.2 and TLS 1.1, ensure that for AES-CBC or DES3_CBC ciphers a random IV is used
        if ("TLS1.2".equals(ssl.getProtocolVersion()) || "TLS1.1".equals(ssl.getProtocolVersion())) {
            String cipherName = ssl.getCipherAlgorithm();
            if ("AES_CBC".equals(cipherName) || "DES3_CBC".equals(cipherName)) {
                int blockSize = ssl.getBlockSize(); // e.g., 16 bytes for AES, 8 bytes for DES3
                byte[] iv = new byte[blockSize];
                SecureRandom random = new SecureRandom();
                random.nextBytes(iv);  // Generate a secure random IV
                ssl.setIV(iv);         // Set the generated IV into the TLS context
                
                // Set the IV on the args structure as well
                args.ivSz = blockSize;
                args.iv = iv;
            }
        }
        
        // If no IV was set, ensure ivSz is initialized to 0
        if (args.iv == null) {
            args.ivSz = 0;
        }
        
        // Calculate the overall message size using input size and the IV size
        args.sz = inSz + args.ivSz;
        return args.sz;
    }
}
