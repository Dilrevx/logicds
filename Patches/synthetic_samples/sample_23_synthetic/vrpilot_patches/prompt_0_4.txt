package main;

import commons.*;

public class Tls12Builder {
    public static int BuildMessage(TlsContext ssl,
                                   byte[] output,
                                   int outSz,
                                   byte[] input,
                                   int inSz,
                                   int type,
                                   int hashOutput,
                                   int sizeOnly,
                                   int asyncOkay,
                                   int epochOrder) {
        // 1. Ensure that the TLS context is not null.
        if (ssl == null) {
            // It is advisable to return an error code or throw an exception.
            return -1;
        }

        // 2. Create and initialize the message arguments.
        BuildMsgArgs args = new BuildMsgArgs();

        // 3. For TLS 1.2 and TLS 1.1, when using AES_CBC or DES3_CBC ciphers,
        //    generate a cryptographically strong random IV.
        if ("TLS1.2".equals(ssl.protocolVersion) || "TLS1.1".equals(ssl.protocolVersion)) {
            String cipherName = ssl.cipherAlgorithm;
            if ("AES_CBC".equals(cipherName) || "DES3_CBC".equals(cipherName)) {
                // Use the blockSize defined in the TLS context (e.g., 16 for AES, 8 for DES3).
                int blockSize = ssl.blockSize;
                byte[] iv = new byte[blockSize];

                // Use SecureRandom to fill the IV with secure random bytes.
                SecureRandom random = new SecureRandom();
                random.nextBytes(iv);

                // Set the generated IV in the TLS context.
                ssl.lastIv = iv;

                // Also configure the args structure with the IV information.
                args.ivSz = blockSize;
                args.iv = iv;
            }
        }

        // Configure additional state if needed. For example, compute the final message size.
        args.sz = inSz + args.ivSz;

        // Continue additional processing if needed.
        // For instance:
        // ssl.options.buildMsgState = CommonsConstants.BUILD_MSG_BEGIN;
        // (other processing steps)

        return args.sz;
    }
}
