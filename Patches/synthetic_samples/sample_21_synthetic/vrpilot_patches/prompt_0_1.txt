        // For TLS 1.3 the handshake requires an Initial Key Material (IKM) for deriving the master secret.
        // In the original version, if no PSK or Key Share extension was present, a default predictable IKM was used.
        // This vulnerability is fixed by generating a secure random IKM instead.

        // Check for PSK and Key Share extensions (assumed functions within ssl or helper utilities).
        boolean hasPskExtension = ssl.hasExtension(ExtensionType.PSK);
        boolean hasKeyShareExtension = ssl.hasExtension(ExtensionType.KEY_SHARE);

        byte[] ikm; // initial key material for master secret generation

        if (!hasPskExtension && !hasKeyShareExtension) {
            // Instead of using a default predictable buffer, securely generate the IKM.
            // Here a 32-byte IKM is generated; adjust the size if your key derivation requires a different length.
            ikm = new byte[32];
            SecureRandom random = new SecureRandom();
            random.nextBytes(ikm);
        } else {
            // Use the normally derived IKM available from the SSL context.
            ikm = ssl.getIkm();
        }

        // Continue with the master secret computation using either the secure random ikm or the derived one.
        ret = computeMasterSecret(ssl, suite, ikm);

        return ret;
    }
