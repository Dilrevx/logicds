package main;

import commons.*;

public class Tls13ServerHello {
    public static int DoTls13ServerHello(TlsContext ssl, byte[] input, int[] inOutIdx, int helloSz, byte[] extMsgType) {
        int ret;
        byte[] suite = new byte[2];
        byte tls12minor = CommonsConstants.TLSv1_2_MINOR;
        if (ssl == null || ssl.arrays == null)
            return CommonsConstants.BAD_FUNC_ARG;
        ret = resetState(ssl);
        int idx = inOutIdx[0];
        if (helloSz < CommonsConstants.OPAQUE16_LEN)
            return CommonsConstants.BUFFER_ERROR;
        idx += CommonsConstants.OPAQUE16_LEN;
        if (idx > helloSz)
            return CommonsConstants.BUFFER_ERROR;

        // Process the Server Hello extensions to decide on the Initial Key Material (IKM)
        // for TLS 1.3 master secret derivation.
        // We check for the presence of the PSK and Key Share extensions using the TLSX.find utility.
        boolean hasPskExtension = (TLSX.find(ssl.extensions, CommonsConstants.TLSX_PRE_SHARED_KEY) != null);
        boolean hasKeyShareExtension = (TLSX.find(ssl.extensions, CommonsConstants.TLSX_KEY_SHARE) != null);

        byte[] ikm; // the IKM for master secret generation

        // If no PSK or Key Share extension is provided, generate a secure random IKM.
        if (!hasPskExtension && !hasKeyShareExtension) {
            ikm = new byte[32]; // adjust the length according to key derivation requirements
            java.security.SecureRandom random = new java.security.SecureRandom();
            random.nextBytes(ikm);
        } else {
            // Otherwise, use the normally derived IKM provided by TlsContext.
            // Ensure that TlsContext now exposes getDerivedIkm() that returns a byte[]
            if (ssl.getDerivedIkm() == null || ssl.getDerivedIkm().length == 0)
                return CommonsConstants.BUFFER_ERROR;  // or an appropriate error code

            ikm = ssl.getDerivedIkm();
        }
        
        // (Optionally process or store ikm for further cryptographic derivation...)

        // Set encryption flag
        ssl.keys.encryptionOn = true;

        // Set server state based on the extMsgType value.
        if (extMsgType[0] == CommonsConstants.server_hello) {
            ssl.options.serverState = CommonsConstants.SERVER_HELLO_COMPLETE;
        } else {
            ssl.options.tls1_3 = true;
            ssl.options.serverState = CommonsConstants.SERVER_HELLO_RETRY_REQUEST_COMPLETE;
        }
        return 0;
    }

    private static int resetState(TlsContext ssl) {
        ssl.options.asyncState = CommonsConstants.TLS_ASYNC_BEGIN;
        return 0;
    }

    private static int SetupPskKey(TlsContext ssl, PreSharedKey psk, int zero) {
        return 0;
    }
}
