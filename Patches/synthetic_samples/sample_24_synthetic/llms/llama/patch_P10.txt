package main;

import commons.*;

public class CertParser {

    public static int ParseCertRelative(DecodedCert cert, int type, int verify, Object cm) {
        int ret = 0;
        int confirmOID = 0;
        if (cert == null) {
            return Commons.BAD_FUNC_ARG;
        }
        if (type == Commons.CERTREQ_TYPE) {
            cert.isCSR = true;
        }
        if (cert.sigCtx.state == Commons.SIG_STATE_BEGIN) {
            cert.badDate = 0;
            cert.criticalExt = 0;
            ret = Commons.DecodeToKey(cert, verify);
            if (ret < 0) {
                if (ret == Commons.ASN_BEFORE_DATE_E || ret == Commons.ASN_AFTER_DATE_E) {
                    cert.badDate = ret;
                }
                else {
                    return ret;
                }
            }

            if (cert.srcIdx < cert.sigIndex) {
                if (!Commons.ALLOW_V1_EXTENSIONS && cert.version < 2) {
                    return Commons.ASN_VERSION_E;
                }
                cert.extensions     = cert.source;
                cert.extensionsSz   = cert.sigIndex - cert.srcIdx;
                cert.extensionsIdx  = cert.srcIdx;
                ret = Commons.DecodeCertExtensions(cert);
                if (ret < 0) {
                    if (ret == Commons.ASN_CRIT_EXT_E) {
                        cert.criticalExt = ret;
                    }
                    else {
                        return ret;
                    }
                }
                if (cert.ocspNoCheckSet && verify == Commons.VERIFY_OCSP) {
                    // Fix: Check if the OCSP response is for the same certificate or its issuer
                    if (!isOCSPResponseForCertOrIssuer(cert, cert.ocspResponse)) {
                        return Commons.OCSP_INVALID_RESPONSE_E;
                    }
                    verify = Commons.NO_VERIFY;
                }
                cert.srcIdx = cert.sigIndex;
            }

            confirmOID = Commons.GetAlgoId(cert);
            if (confirmOID < 0) {
                return -1;
            }
            ret = Commons.GetSignature(cert);
            if (ret < 0) {
                return ret;
            }
            if (confirmOID!= cert.signatureOID) {
                return Commons.ASN_SIG_OID_E;
            }
        }


        if (verify!= Commons.NO_VERIFY) {
            if (verify == Commons.VERIFY_OCSP) {
                return Commons.OCSP_ONLY;
            }
            else {
                return Commons.FULL_VERIFY;
            }
        }

        return ret;
    }

    private static boolean isOCSPResponseForCertOrIssuer(DecodedCert cert, byte[] ocspResponse) {
        // Implement logic to check if the OCSP response is for the same certificate or its issuer
        // This can be done by comparing the certificate's subject and issuer with the OCSP response's subject and issuer
        // For simplicity, this example assumes that the OCSP response is a DER-encoded OCSPResponse structure
        // In a real-world implementation, you would need to parse the OCSP response and compare the relevant fields
        return Arrays.equals(cert.subject, getOCSPResponseSubject(ocspResponse)) || Arrays.equals(cert.issuer, getOCSPResponseIssuer(ocspResponse));
    }

    private static byte[] getOCSPResponseSubject(byte[] ocspResponse) {
        // Implement logic to extract the subject from the OCSP response
        // This can be done by parsing the OCSP response and extracting the subject field
        // For simplicity, this example assumes that the subject is the first 10 bytes of the OCSP response
        return Arrays.copyOfRange(ocspResponse, 0, 10);
    }

    private static byte[] getOCSPResponseIssuer(byte[] ocspResponse) {
        // Implement logic to extract the issuer from the OCSP response
        // This can be done by parsing the OCSP response and extracting the issuer field
        // For simplicity, this example assumes that the issuer is the next 10 bytes of the OCSP response
        return Arrays.copyOfRange(ocspResponse, 10, 20);
    }

}
