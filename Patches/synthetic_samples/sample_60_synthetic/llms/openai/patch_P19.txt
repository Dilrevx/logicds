    private static byte[] concat(String a, String b, String c) {
        byte[] ab = a.getBytes();
        byte[] bb = b.getBytes();
        byte[] cb = c.getBytes();
        byte[] out = new byte[ab.length + bb.length + cb.length];
        System.arraycopy(ab, 0, out, 0, ab.length);
        System.arraycopy(bb, 0, out, ab.length, bb.length);
        System.arraycopy(cb, 0, out, ab.length + bb.length, cb.length);
        return out;
    }

    public static String hexEncode(byte[] data) {
        StringBuilder sb = new StringBuilder(data.length * 2);
        for (byte b : data) {
            sb.append(String.format("%02x", b & 0xff));
        }
        return sb.toString();
    }
