import java.util.Map;

public class WebHandler {

    public static final String FAKE_HASH =
            "$2y$10$QOauhQNbBCuQDKes6eFzPeMqBSjb7Mr5DUmpZ/VcEd00UAV/LDeSi";

    private final String tlsConfigPath;
    private final InnerHandler handler;
    private final Logger logger = new Logger();
    private final AuthCache cache = new AuthCache();
    private final Object bcryptMtx = new Object();

    public WebHandler(String tlsConfigPath, InnerHandler handler) {
        this.tlsConfigPath = tlsConfigPath;
        this.handler = handler;
    }

    public AuthCache getCache() { return cache; }

    public void serveHttp(HttpResponse w, HttpRequest r) {
        ConfigLoader.LoadResult cLoad = ConfigLoader.getConfig(tlsConfigPath);
        WebConfig c = cLoad.config;
        Exception err = cLoad.err;
        if (err != null) {
            logger.log("msg", "Unable to parse configuration", "err", err);
            httpError(w, HttpStatus.statusText(HttpStatus.INTERNAL_SERVER_ERROR),
                    HttpStatus.INTERNAL_SERVER_ERROR);
            return;
        }

        for (Map.Entry<String, String> e : c.getHttpConfig().header.entrySet()) {
            w.getHeaders().put(e.getKey(), e.getValue());
        }

        if (c.getUsers().isEmpty()) {
            handler.serveHttp(w, r);
            return;
        }

        HttpRequest.BasicAuthResult ba = r.basicAuth();
        String user = ba.user;
        String pass = ba.pass;
        boolean auth = ba.present;
        if (auth) {
            String hashedPassword = c.getUsers().get(user);
            boolean validUser = hashedPassword != null;

            if (!validUser) {
                hashedPassword = FAKE_HASH;
            }

            String cacheKey = hexEncode(concat(user, hashedPassword, pass));
            AuthCache.GetResult cached = cache.get(cacheKey);
            boolean authOk = cached.value;
            boolean ok = cached.present;

            if (!ok) {
                Exception bcryptErr;

synchronized (bcryptMtx) {
    bcryptErr = Bcrypt.compareHashAndPassword(
            hashedPassword.getBytes(), pass.getBytes());
    authOk = bcryptErr == null;
    cache.set(cacheKey, authOk);
}


            if (authOk && validUser) {
                handler.serveHttp(w, r);
                return;
            }
        }

        w.getHeaders().put("WWW-Authenticate", "Basic");
        httpError(w, HttpStatus.statusText(HttpStatus.UNAUTHORIZED),
                HttpStatus.UNAUTHORIZED);
    }

    private static void httpError(HttpResponse w, String msg, int status) {
        w.setStatus(status);
        w.setBody(msg);
    }

    private static byte[] concat(String a, String b, String c) {
        byte[] ab = a.getBytes();
        byte[] bb = b.getBytes();
        byte[] cb = c.getBytes();
        byte[] out = new byte[ab.length + bb.length + cb.length];
        System.arraycopy(ab, 0, out, 0, ab.length);
        System.arraycopy(bb, 0, out, ab.length, bb.length);
        System.arraycopy(cb, 0, out, ab.length + bb.length, cb.length);
        return out;
    }

    public static String hexEncode(byte[] data) {
        StringBuilder sb = new StringBuilder(data.length * 2);
        for (byte b : data) {
            sb.append(String.format("%02x", b & 0xff));
        }
        return sb.toString();
    }
}
