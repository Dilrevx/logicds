public class MandrilPlugin {

    private static final String REFERER = "Referer";

    private final MkApi mkApi;

    public MandrilPlugin(MkApi mkApi) {
        this.mkApi = mkApi;
    }

    public int mkpStage30(Plugin p, ClientSession cs, SessionRequest sr) {
        MkPtr referer;

        PluginTrace.trace("[FD %d] Mandril validating URL", cs.getSocket());

    /* Decode the incoming URI before checking to normalize any encoded characters */
    char *decoded_uri = mk_utils_uri_decode(sr->uri);
    if (!decoded_uri) {
        PluginTrace.trace("[FD %d] URI decoding failed", cs->socket);
        mkApi.headerSetHttpStatus(sr, PluginConst.MK_INTERNAL_ERROR);
        return PluginConst.MK_PLUGIN_RET_CLOSE_CONX;
    }

    if (MkSecurity.checkUrl(decoded_uri) < 0) {
        PluginTrace.trace("[FD %d] Close connection, blocked URL", cs->socket);
        mkApi.headerSetHttpStatus(sr, PluginConst.MK_CLIENT_FORBIDDEN);
        free(decoded_uri);
        return PluginConst.MK_PLUGIN_RET_CLOSE_CONX;
    }

    free(decoded_uri);
    /* Continue processing the request normally */


        PluginTrace.trace("[FD %d] Mandril validating hotlinking", cs.getSocket());
        referer = mkApi.headerGet(sr.getHeadersToc(), REFERER, REFERER.length());
        if (MkSecurity.checkHotlink(sr.getUriProcessed(), sr.getHost(), referer) < 0) {
            PluginTrace.trace("[FD %d] Close connection, deny hotlinking.", cs.getSocket());
            mkApi.headerSetHttpStatus(sr, PluginConst.MK_CLIENT_FORBIDDEN);
            return PluginConst.MK_PLUGIN_RET_CLOSE_CONX;
        }

        return PluginConst.MK_PLUGIN_RET_NOT_ME;
    }
}
