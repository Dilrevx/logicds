        for (i = 0; i < ctx.chain.size() - 1; i++) {
            // If the same certificate appears twice in the chain, abort to prevent a loop.
            if (ctx.chain.subList(i + 1, ctx.chain.size()).contains(ctx.chain.get(i))) {
                Helper.raiseError("X509_R_CERT_CHAIN_LOOP");
                return -1;
            }
        }
        // Repair: If extra untrusted certificates were appended beyond a trust anchor, trim them.
        // (This protects against a chain extension attack where additional untrusted certs are appended.)
        while (ctx.chain.size() > j) {
            ctx.chain.remove(ctx.chain.size() - 1);
        }
        // Final verification step: verify the assembled chain.
        // (Assuming Helper.verifyChain performs signature/path validation using ctx.param)
        ok = Helper.verifyChain(ctx.chain, param);
