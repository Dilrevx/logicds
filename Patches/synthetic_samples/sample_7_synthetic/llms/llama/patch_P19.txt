package main;

import commons.X509;
import commons.X509_VERIFY_PARAM;
import commons.X509_STORE_CTX;
import commons.VerifyCallback;
import commons.Helper;
import commons.Constants;
import java.util.List;
import java.util.ArrayList;

public class X509CertVerifier {

        for (i = 0; i < num; i++) {
            xtmp = ctx.chain.get(i);
            if (xtmp == null) continue;
            if (Helper.cert_self_signed(xtmp)) {
                if (i == 0) {
                    ok = 1;
                    break;
                }
                if (ctx.last_untrusted > 0) {
                    ctx.last_untrusted--;
                    ctx.chain.remove(i);
                    num--;
                    i--;
                    continue;
                }
                bad_chain = 1;
                break;
            }
            if (i == 0) continue;
            xtmp2 = ctx.chain.get(i - 1);
            if (xtmp2 == null) continue;
            if (!Helper.check_issued(xtmp, xtmp2)) {
                bad_chain = 1;
                break;
            }
        }
        if (bad_chain) {
            Helper.raiseError("X509_R_CERTIFICATE_CHAIN_TOO_LONG");
            return -1;
        }
        if (ok == 0) {
            Helper.raiseError("X509_R_NO_TRUSTED_CERT_FOUND");
            return -1;
        }
        retry = 0;
        while (ctx.chain.size() > 0) {
            x = ctx.chain.remove(0);
            if (x!= null) {
                if (ctx.verify_cb!= null) {
                    if (ctx.verify_cb(x, ctx.error_depth, ctx.error) <= 0) {
                        Helper.free(x);
                        continue;
                    }
                }
                if (ctx.error_depth > 0) {
                    ctx.error_depth--;
                }
                if (ctx.error!= null) {
                    ctx.error = null;
                }
                Helper.free(x);
            }
        }
        return 1;
    }

}
