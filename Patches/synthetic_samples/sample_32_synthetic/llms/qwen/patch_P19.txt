            if (x == null) {
                ret = 0;

            }
            ret = Util.verifyCA(x);
            if (ret != 1) {
                break;
            }
            boolean strict = (ctx.param.flags & X509Flags.X509_V_FLAG_X509_STRICT) != 0;

            // Check if the certificate is a CA
            if (!x.isCA()) {
                if (i != num - 1) { // Not the last certificate in the chain
                    ret = 0;
                    break;
                }
            } else {
                if (must_be_ca >= 0 && must_be_ca != x.getBasicConstraints()) {
                    ret = 0;
                    break;
                }
                must_be_ca = x.getBasicConstraints();
            }

            // Check path length constraint
            if (proxy_path_length > 0) {
                if (x.getPathLenConstraint() < proxy_path_length - 1) {
                    ret = 0;
                    break;
                }
            }
            proxy_path_length++;

            // Check policy constraints
            if (strict) {
                plen = x.getPolicyConstraints();
                if (plen > 0 && plen < num - i - 1) {
                    ret = 0;
                    break;
                }
            }
        }
        return ret;
    }

}
