package master;

import common.*;

public class CheckChainExtensions {


    public static int checkChainExtensions(X509StoreCtx ctx) {
        int num = ctx.chain.size();
        int plen = 0;
        int proxy_path_length = 0;
        int must_be_ca = -1;
        int ret = 1;

        for (int i = 0; i < num; i++) {
            X509 x = ctx.chain.get(i);
            // Determine strict mode before CA verification so that the correct logic is applied.
            boolean strict = (ctx.param.flags & X509Flags.X509_V_FLAG_X509_STRICT) != 0;

            // Pass the certificate (and optionally the strict flag) to the CA verifier.
            // This fixes the logical vulnerability by ensuring that the strict parameter affects the verification.
            ret = Util.verifyCA(x, strict);
            if (ret != 1) {
                return ret;
            }

            // If the certificate is not self-issued then update the path length.
            if (!Util.isSelfIssued(x)) {
                // If a maximum path length is specified, enforce it.
                if (must_be_ca >= 0 && plen > must_be_ca) {
                    return X509VErr.INVALID_CA; // or the appropriate error constant
                }
                plen++;
            }

            // Additional chain extension validations can be performed here.
            // For example, handling proxy certificate path lengths if needed:
            // if (Util.isProxy(x)) { proxy_path_length = ...; }
        }
        return ret;
    }

}
