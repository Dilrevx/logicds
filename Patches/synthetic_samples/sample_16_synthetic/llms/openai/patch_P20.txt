public class Ssl3GetCertVerify {

            if (peer != null && (type & Constants.EVP_PKT_SIGN) != 0) {
                // A certificate with a signing key MUST be accompanied by a certificate verify message.
                al = Constants.SSL_AD_UNEXPECTED_MESSAGE;
                Helper.SSLerr(Constants.SSL_F_SSL3_GET_CERT_VERIFY, Constants.SSL_R_MISSING_VERIFY_MESSAGE);
                Helper.ssl3_send_alert(s, Constants.SSL3_AL_FATAL, al);
                return ret;
            } else {
                // If no certificate was provided or the key type doesn't require signing,
                // then we set the flag to reuse the current message.
                s.s3.tmp.reuse_message = 1;
                return 1;
            }
        }


    private static byte[] slice(byte[] array, int offset) {
        byte[] result = new byte[array.length - offset];
        System.arraycopy(array, offset, result, 0, result.length);
        return result;
    }
}
