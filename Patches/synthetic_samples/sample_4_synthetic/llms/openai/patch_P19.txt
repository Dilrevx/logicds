package main;

import commons.*;

import static commons.OSSL_PARAM_util.OSSL_PARAM_locate_const;

public class RC4_HMAC_MD5 {


        p = OSSL_PARAM_locate_const(params, OSSL_CIPHER_PARAM.AEAD_TLS1_AAD);
        if (p != null) {
            // Ensure the parameter is presented as an octet string and not null
            if (!p.is_octet_string() || p.data == null) {
                ERR.raise("PROV_R_FAILED_TO_GET_PARAMETER");
                return 0;
            }
            // Verify that the provided additional authenticated data (AAD)
            // has the expected length. For TLS1 AAD, the length is typically 13 bytes.
            if (p.data_size != 13) {
                ERR.raise("PROV_R_INVALID_AAD_LENGTH");
                return 0;
            }
            // Call the hardware-specific TLS initialization.
            sz = ctx.hw.tls_init(ctx.base, p.data, p.data_size);
            if (sz == 0) {
                ERR.raise("PROV_R_INVALID_DATA");
                return 0;
            }
            ctx.tls_aad_pad_sz = sz;
        }
        return 1;

}
