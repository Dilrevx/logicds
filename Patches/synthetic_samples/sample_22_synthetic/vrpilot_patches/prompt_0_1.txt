package main;

import commons.*;

public class Tls13CertificateVerify {
    public static int DoTls13CertificateVerify(TlsContext ssl, byte[] input, int[] inOutIdx, int totalSz) {
        int ret = 0;
        Buffer sig = ssl.buffers.sig;
        Dcv13Args args = new Dcv13Args();
        ssl.options.asyncState = CommonsConstants.TLS_ASYNC_BEGIN;
        
        // Set the hash algorithm as before.
        args.hashAlgo = CommonsConstants.sha_mac;
        
        // FIX: Instead of using a hard-coded anonymous algorithm,
        // use the peer certificate’s signature algorithm (obtained earlier in the handshake).
        args.sigAlgo = ssl.handshakeContext.peerCertSignatureAlgorithm;
        
        // Perform a validity check: if the signature algorithm is still anonymous,
        // this indicates a problem and we return an error.
        if (args.sigAlgo == CommonsConstants.anonymous_sa_algo) {
            // Log the error or handle it appropriately.
            return CommonsConstants.ALERT_ILLEGAL_PARAMETER;  // Return an appropriate error code.
        }
        
        args.idx = inOutIdx[0];
        args.begin = inOutIdx[0];
        
        // ... (rest of code that verifies the signature using args.sigAlgo and args.hashAlgo)
        
        return ret;
        }
    }
}
