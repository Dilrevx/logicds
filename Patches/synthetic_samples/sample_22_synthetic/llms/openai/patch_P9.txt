package main;

import commons.*;

public class Tls13CertificateVerify {
    public static int DoTls13CertificateVerify(TlsContext ssl, byte[] input, int[] inOutIdx, int totalSz) {
        int ret = 0;
        Buffer sig = ssl.buffers.sig;
        Dcv13Args args = new Dcv13Args();
        ssl.options.asyncState = CommonsConstants.TLS_ASYNC_BEGIN;
        args.hashAlgo = CommonsConstants.sha_mac;
        args.sigAlgo = CommonsConstants.anonymous_sa_algo;
        args.idx = inOutIdx[0];
        args.begin = inOutIdx[0];
        switch (ssl.options.asyncState) {
            case CommonsConstants.TLS_ASYNC_BEGIN:
                ssl.options.asyncState = CommonsConstants.TLS_ASYNC_BUILD;
            case CommonsConstants.TLS_ASYNC_BUILD:
                if ((args.idx - args.begin) + CommonsConstants.ENUM_LEN + CommonsConstants.ENUM_LEN > totalSz)
                    return CommonsConstants.BUFFER_ERROR;
                ret = CommonUtils.DecodeTls13SigAlg(input, args);
                if (ret < 0)
                    return ret;
                args.idx += CommonsConstants.OPAQUE16_LEN;
                if ((args.idx - args.begin) + CommonsConstants.OPAQUE16_LEN > totalSz)
                    return CommonsConstants.BUFFER_ERROR;
                args.sz = CommonUtils.ato16(input, args.idx);
                args.idx += CommonsConstants.OPAQUE16_LEN;
                if ((args.idx - args.begin) + args.sz > totalSz || args.sz > CommonsConstants.ENCRYPT_LEN)
                    return CommonsConstants.BUFFER_ERROR;

                /* 
                 * In TLS 1.3 the certificate_verify message’s signature algorithm must match
                 * the algorithm of the key in the certificate. This patch ensures that if the
                 * certificate does not include the correct key type for the used signature 
                 * algorithm, that the handshake is aborted.
                 */
                if (args.sigAlgo == CommonsConstants.ed25519_sa_algo) {
                    if (!ssl.peerEd25519KeyPresent) {
                        Wslib.WOLFSSL_MSG("Peer sent ED25519 signature but certificate does not contain ED25519 key");
                        return CommonsConstants.SIG_VERIFY_E;
                    }
                }
                else if (args.sigAlgo == CommonsConstants.ed448_sa_algo) {
                    if (!ssl.peerEd448KeyPresent) {
                        Wslib.WOLFSSL_MSG("Peer sent ED448 signature but certificate does not contain ED448 key");
                        return CommonsConstants.SIG_VERIFY_E;
                    }
                }
                else if (args.sigAlgo == CommonsConstants.ecc_dsa_sa_algo) {
                    if (!ssl.peerEccDsaKeyPresent) {
                        Wslib.WOLFSSL_MSG("Peer sent ECC signature but certificate does not contain ECC key");
                        return CommonsConstants.SIG_VERIFY_E;
                    }
                }
                else if (args.sigAlgo == CommonsConstants.falcon_level1_sa_algo ||
                         args.sigAlgo == CommonsConstants.falcon_level5_sa_algo) {
                    if (!ssl.peerFalconKeyPresent) {
                        Wslib.WOLFSSL_MSG("Peer sent Falcon signature but certificate does not contain Falcon key");
                        return CommonsConstants.SIG_VERIFY_E;
                    }
                }
                else if (args.sigAlgo == CommonsConstants.rsa_pss_sa_algo) {
                    if (ssl.peerRsaKey == null || !ssl.peerRsaKeyPresent) {
                        Wslib.WOLFSSL_MSG("Peer sent RSA-PSS signature but certificate does not contain RSA key");
                        return CommonsConstants.SIG_VERIFY_E;
                    }
                }
                else if (args.sigAlgo == CommonsConstants.rsa_sa_algo) {
                    /* 
                     * TLS 1.3 does not allow RSA PKCS#1.5 (rsa_sa_algo) for certificate_verify.
                     * Reject any attempt to use it. 
                     */
                    Wslib.WOLFSSL_MSG("Peer sent unsupported RSA PKCS#1.5 signature algorithm");
                    return CommonsConstants.SIG_VERIFY_E;
                }
                else {
                    Wslib.WOLFSSL_MSG("Peer sent unknown signature algorithm");
                    return CommonsConstants.SIG_VERIFY_E;
                }

                return 0;
            default:
                return 0;
        }
    }
}
