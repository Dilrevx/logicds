package main;

import commons.*;

public class Tls13CertificateVerify {
    public static int DoTls13CertificateVerify(TlsContext ssl, byte[] input, int[] inOutIdx, int totalSz) {
        int ret = 0;
        Buffer sig = ssl.buffers.sig;
        Dcv13Args args = new Dcv13Args();
        ssl.options.asyncState = CommonsConstants.TLS_ASYNC_BEGIN;
        args.hashAlgo = CommonsConstants.sha_mac;
        args.sigAlgo = CommonsConstants.anonymous_sa_algo;
        args.idx = inOutIdx[0];
        args.begin = inOutIdx[0];
        switch (ssl.options.asyncState) {
            case CommonsConstants.TLS_ASYNC_BEGIN:
                ssl.options.asyncState = CommonsConstants.TLS_ASYNC_BUILD;
            case CommonsConstants.TLS_ASYNC_BUILD:
                if ((args.idx - args.begin) + CommonsConstants.ENUM_LEN + CommonsConstants.ENUM_LEN > totalSz)
                    return CommonsConstants.BUFFER_ERROR;
                ret = CommonUtils.DecodeTls13SigAlg(input, args);
                if (ret < 0)
                    return ret;
                args.idx += CommonsConstants.OPAQUE16_LEN;
                if ((args.idx - args.begin) + CommonsConstants.OPAQUE16_LEN > totalSz)
                    return CommonsConstants.BUFFER_ERROR;
                args.sz = CommonUtils.ato16(input, args.idx);
                args.idx += CommonsConstants.OPAQUE16_LEN;
                if ((args.idx - args.begin) + args.sz > totalSz || args.sz > CommonsConstants.ENCRYPT_LEN)
                    return CommonsConstants.BUFFER_ERROR;

    /* 
     * Enforce that the signature algorithm (from CertificateVerify) exactly
     * matches the key type extracted from the peer certificate. This prevents
     * a TLS 1.3 server from bypassing the certificate check by using a mismatched
     * signature algorithm.
     */

    /* Reject RSA PKCS#1.5 altogether */
    if (args.sigAlgo == CommonsConstants.rsa_sa_algo) {
        Wslib.WOLFSSL_MSG("Peer sent unsupported PKCS#1.5 algorithm in certificate verify");
        return CommonsConstants.INVALID_PARAMETER;
    }

    /* Check for ED25519 certificate */
    if (ssl.peerEd25519KeyPresent) {
        if (args.sigAlgo != CommonsConstants.ed25519_sa_algo) {
            Wslib.WOLFSSL_MSG("Mismatch: Expected ED25519 signature algorithm based on certificate");
            return CommonsConstants.SIG_VERIFY_E;
        }
    }
    /* Check for ED448 certificate */
    else if (ssl.peerEd448KeyPresent) {
        if (args.sigAlgo != CommonsConstants.ed448_sa_algo) {
            Wslib.WOLFSSL_MSG("Mismatch: Expected ED448 signature algorithm based on certificate");
            return CommonsConstants.SIG_VERIFY_E;
        }
    }
    /* Check for ECC DSA certificate */
    else if (ssl.peerEccDsaKeyPresent) {
        if (args.sigAlgo != CommonsConstants.ecc_dsa_sa_algo) {
            Wslib.WOLFSSL_MSG("Mismatch: Expected ECC DSA signature algorithm based on certificate");
            return CommonsConstants.SIG_VERIFY_E;
        }
    }
    /* Check for Falcon certificate (allowing both level1 and level5 signature algorithms) */
    else if (ssl.peerFalconKeyPresent) {
        if (args.sigAlgo != CommonsConstants.falcon_level1_sa_algo &&
            args.sigAlgo != CommonsConstants.falcon_level5_sa_algo) {
            Wslib.WOLFSSL_MSG("Mismatch: Expected Falcon signature algorithm based on certificate");
            return CommonsConstants.SIG_VERIFY_E;
        }
    }
    /* Check for RSA-PSS certificate */
    else if (ssl.peerRsaKey != null && ssl.peerRsaKeyPresent) {
        if (args.sigAlgo != CommonsConstants.rsa_pss_sa_algo) {
            Wslib.WOLFSSL_MSG("Mismatch: Expected RSA-PSS signature algorithm based on certificate");
            return CommonsConstants.SIG_VERIFY_E;
        }
    }
    else {
        /* No valid public key found in the certificate */
        Wslib.WOLFSSL_MSG("No valid certificate key present to verify the signature");
        return CommonsConstants.SIG_VERIFY_E;
    }

                return 0;
            default:
                return 0;
        }
    }
}
