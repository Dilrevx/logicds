        // Check for SPNEGO token in Authorization header
        String authHeader = r.getHeadersIn().getAuthorization();
        if (authHeader != null && authHeader.startsWith("Negotiate ")) {
            SpnegoDebug.d0("SPNEGO token found in Authorization header");
            ctx.token.data = authHeader.substring(10); // Remove "Negotiate " prefix
            ctx.token.len = ctx.token.data.length();
            ctx.head = 1; // Indicate that we have a token

            // Process the SPNEGO token
            ret = AuthSpnego.processSpnegoToken(r, ctx, alcf);
            if (ret == NgxConst.NGX_OK) {
                SpnegoDebug.d0("SPNEGO authentication successful");
                return NgxConst.NGX_OK;
            } else {
                SpnegoDebug.d0("SPNEGO authentication failed");
                ctx.ret = NgxConst.NGX_HTTP_UNAUTHORIZED;
            }
        } else {
            SpnegoDebug.d0("No SPNEGO token found in Authorization header");
            ctx.ret = NgxConst.NGX_HTTP_UNAUTHORIZED;
        }

        SpnegoDebug.d1("Returning %d", ctx.ret);
        return ctx.ret;
    }
