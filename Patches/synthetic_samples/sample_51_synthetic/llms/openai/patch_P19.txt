            if (alcf.isAllowBasic()) {
                SpnegoDebug.d0("Detect basic auth");
                ret = AuthSpnego.ngxHttpAuthBasicUser(r);
                if (NgxConst.NGX_OK == ret) {
                    SpnegoDebug.d0("Basic auth credentials supplied by client");
                    return ret; // Fixed: return immediately when valid basic auth credentials are supplied.
                }
            }

            /*
             * Proceed with SPNEGO negotiation if no basic authentication credentials
             * were supplied (or the basic auth check did not return OK).
             * For example, attempt to process an Authorization header containing a Negotiate token.
             */
            ret = AuthSpnego.ngxHttpAuthNegotiate(r, ctx);
            if (NgxConst.NGX_OK == ret) {
                SpnegoDebug.d0("SPNEGO authentication successful");
            } else {
                SpnegoDebug.d0("SPNEGO authentication failed, returning code: %d", ret);
            }
            return ret;
    }

}
