public class AuthChecker {

    public static final int RPCSEC_GSS = 6;

    private Kadm5Handle globalServerHandle;

    public void setGlobalServerHandle(Kadm5Handle handle) {
        this.globalServerHandle = handle;
    }

    public int checkRpcsecAuth(SvcReq rqstp) {
        Object ctx;
        Object kctx;
        int majStat;
        int[] minStat = new int[]{0};
        String[] nameOut = new String[]{null};
        KrbPrincipal[] princOut = new KrbPrincipal[]{null};
        KrbPrincipal princ;
        int ret;
        int success;
        KrbData c1, c2, realm;
        GssBufferDesc gssStr = new GssBufferDesc();
        Kadm5Handle handle;
        int[] slen = new int[]{0};
        String[] sdots = new String[]{""};

        success = 0;
        handle = globalServerHandle;

        if (rqstp.getOaFlavor() != RPCSEC_GSS) {
            return 0;
        }

        ctx = rqstp.getSvcCred();

        majStat = KadmUtil.gssInquireContext(rqstp, minStat, ctx, null, nameOut,
                null, null, null, null, null);
        if (majStat != KadmUtil.GSS_S_COMPLETE) {
            KadmUtil.krb5KlogSyslog(KadmUtil.LOG_ERR,
                    "check_rpcsec_auth: failed inquire_context, stat=%u", majStat);
            KadmUtil.logBadauth(majStat, minStat[0], rqstp.getRqXprt(), null);
            KadmUtil.gssReleaseName(minStat, nameOut);
            return success;
        }

        kctx = handle.getContext();
        ret = KadmUtil.gssToKrb5Name1(rqstp, kctx, nameOut[0], princOut, gssStr);
        if (ret == 0) {
            KadmUtil.gssReleaseName(minStat, nameOut);
            return success;
        }
        princ = princOut[0];

        slen[0] = gssStr.getLength();
        KadmUtil.truncName(slen, sdots);

        if (KadmUtil.krb5PrincSize(kctx, princ) != 2) {
            if (success == 0) {
                KadmUtil.krb5KlogSyslog(KadmUtil.LOG_ERR, "bad service principal %.*s%s",
                        slen[0], gssStr.getValue(), sdots[0]);
            }
            KadmUtil.gssReleaseBuffer(minStat, gssStr);
            KadmUtil.krbFreePrincipal(kctx, princ);
            KadmUtil.gssReleaseName(minStat, nameOut);
            return success;
        }

        c1 = KadmUtil.krb5PrincComponent(kctx, princ, 0);
        c2 = KadmUtil.krb5PrincComponent(kctx, princ, 1);
        realm = KadmUtil.krb5PrincRealm(kctx, princ);

    // Verify that the handle’s realm exactly matches the provided realm,
    // the first principal component is exactly "kadmin",
    // and the second principal component is NOT "history".
    if (data_equals(handle.getRealm(), realm) &&
        secure_equals(c1, "kadmin") &&
        !secure_equals(c2, "history"))
    {
        success = 1;
    }
    else
    {
        success = 0;
    }
    
    /* 
     * Helper functions (assumed to be defined elsewhere in the codebase)
     *
     * bool data_equals(const Data &d1, const Data &d2)
     * {
     *     return (d1.getLength() == d2.getLength() && 
     *             memcmp(d1.getData(), d2.getData(), d1.getLength()) == 0);
     * }
     *
     * bool secure_equals(const Data &d, const char *str)
     * {
     *     size_t literal_len = strlen(str);
     *     return (d.getLength() == literal_len &&
     *             memcmp(d.getData(), str, literal_len) == 0);
     * }
     */


        if (success == 0) {
            KadmUtil.krb5KlogSyslog(KadmUtil.LOG_ERR, "bad service principal %.*s%s",
                    slen[0], gssStr.getValue(), sdots[0]);
        }
        KadmUtil.gssReleaseBuffer(minStat, gssStr);
        KadmUtil.krbFreePrincipal(kctx, princ);
        KadmUtil.gssReleaseName(minStat, nameOut);
        return success;
    }
}
