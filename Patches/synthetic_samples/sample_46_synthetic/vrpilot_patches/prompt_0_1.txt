public class AuthChecker {

    public static final int RPCSEC_GSS = 6;

    private Kadm5Handle globalServerHandle;

    public void setGlobalServerHandle(Kadm5Handle handle) {
        this.globalServerHandle = handle;
    }

    public int checkRpcsecAuth(SvcReq rqstp) {
        Object ctx;
        Object kctx;
        int majStat;
        int[] minStat = new int[]{0};
        String[] nameOut = new String[]{null};
        KrbPrincipal[] princOut = new KrbPrincipal[]{null};
        KrbPrincipal princ;
        int ret;
        int success;
        KrbData c1, c2, realm;
        GssBufferDesc gssStr = new GssBufferDesc();
        Kadm5Handle handle;
        int[] slen = new int[]{0};
        String[] sdots = new String[]{""};

        success = 0;
        handle = globalServerHandle;

        if (rqstp.getOaFlavor() != RPCSEC_GSS) {
            return 0;
        }

        ctx = rqstp.getSvcCred();

        majStat = KadmUtil.gssInquireContext(rqstp, minStat, ctx, null, nameOut,
                null, null, null, null, null);
        if (majStat != KadmUtil.GSS_S_COMPLETE) {
            KadmUtil.krb5KlogSyslog(KadmUtil.LOG_ERR,
                    "check_rpcsec_auth: failed inquire_context, stat=%u", majStat);
            KadmUtil.logBadauth(majStat, minStat[0], rqstp.getRqXprt(), null);
            KadmUtil.gssReleaseName(minStat, nameOut);
            return success;
        }

        kctx = handle.getContext();
        ret = KadmUtil.gssToKrb5Name1(rqstp, kctx, nameOut[0], princOut, gssStr);
        if (ret == 0) {
            KadmUtil.gssReleaseName(minStat, nameOut);
            return success;
        }
        princ = princOut[0];

        slen[0] = gssStr.getLength();
        KadmUtil.truncName(slen, sdots);

        if (KadmUtil.krb5PrincSize(kctx, princ) != 2) {
            if (success == 0) {
                KadmUtil.krb5KlogSyslog(KadmUtil.LOG_ERR, "bad service principal %.*s%s",
                        slen[0], gssStr.getValue(), sdots[0]);
            }
            KadmUtil.gssReleaseBuffer(minStat, gssStr);
            KadmUtil.krbFreePrincipal(kctx, princ);
            KadmUtil.gssReleaseName(minStat, nameOut);
            return success;
        }

    public int checkRpcsecAuth(SvcReq rqstp) {
        Object ctx;
        Object kctx;
        int majStat;
        int[] minStat = new int[]{0};
        String[] nameOut = new String[]{null};
        KrbPrincipal[] princOut = new KrbPrincipal[]{null};
        KrbPrincipal princ;
        int ret;
        int success;
        KrbData c1, c2, realm;
        GssBufferDesc gssStr = new GssBufferDesc();
        Kadm5Handle handle;
        int[] slen = new int[]{0};
        String[] sdots = new String[]{""};

        success = 0;
        handle = globalServerHandle;

        if (rqstp.getOaFlavor() != RPCSEC_GSS) {
            // Only RPCSEC_GSS is supported.
            return -1;  // Return an appropriate error code.
        }

        /*
         * --- Code that uses gssStr/gss context to obtain the RPCSEC_GSS principal ---
         * For example, a call similar to:
         *
         *    majStat = gssAcceptSecContext(..., nameOut, princOut, ...);
         *
         * is used to obtain the authenticated principal. Assume that after this call,
         * princOut[0] holds the principal information for the caller.
         *
         * (The original code omitted further checks on the principal structure.)
         */

        // *** FIX: Validate the principal’s component structure ***
        /*
         * In MIT Kerberos kadmind, a two-component principal is expected to have a fully-qualified
         * second component. A caller presenting a two–component principal where the second component is
         * not complete (for example, an “initial” value such as an empty string) should not be authorized.
         */
        princ = princOut[0];
        if (princ == null) {
            // Could not obtain a principal, so fail authentication.
            return -1;
        }

        // Assuming getComponents() returns an array of String components for the principal.
        String[] components = princ.getComponents();
        if (components != null && components.length == 2) {
            // Check that the second component is not an “initial” (for example, it is not empty).
            if (components[1] == null || components[1].trim().isEmpty()) {
                // The principal's second component is not fully qualified, so deny access.
                return -1;
            }
        }
        // ---------------------------------------------

        // Continue with the rest of the processing and authorization as originally intended.
        // (Other functionality remains unchanged.)
        success = 1;  // Mark as successfully authenticated via RPCSEC_GSS.

        return success;
    }
}
