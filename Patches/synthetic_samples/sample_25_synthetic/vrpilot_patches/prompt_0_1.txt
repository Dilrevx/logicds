package main;

import commons.*;

public class Tls13CertificateVerifier {
    public static int DoTls13CertificateVerify(TlsContext ssl, byte[] input, int[] inOutIdx, int totalSz) {
        int ret = 0;
        Buffer sig = ssl.buffers.sig;
        Dcv13Args args = new Dcv13Args();
        
        // Set asynchronous state to start the certificate verification.
        ssl.options.asyncState = Commons.TLS_ASYNC_BEGIN;
        
        // Always assign the hash algorithm for certificate verification.
        args.hashAlgo = Commons.sha_mac;
        
        // For TLS 1.3, use the signature algorithm from the certificate.
        if (ssl.version == Commons.TLS1_3) {
            // Retrieve the signature algorithm properly extracted from the certificate.
            args.sigAlgo = ssl.certificate.getSignatureAlgorithm();
            
            // If we still see the 'anonymous' value, something is wrong, so abort immediately.
            if (args.sigAlgo == Commons.anonymous_sa_algo) {
                return Commons.TLS_CERT_VERIFY_ERROR;
            }
        } else {
            // For TLS versions other than 1.3, use the original 'anonymous' signature algorithm.
            args.sigAlgo = Commons.anonymous_sa_algo;
        }
        
        // Initialize index values.
        args.idx   = inOutIdx[0];
        args.begin = inOutIdx[0];
        
        // (Rest of the certificate verification code would follow here.)
        
        return ret;
    }
}
