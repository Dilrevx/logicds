public class NamespacesSetup {

    public void setupNamespaces(LoData lo, FuseSession se) {
        int child;

        if (OsApi.unshare(NamespaceFlags.CLONE_NEWPID
                | NamespaceFlags.CLONE_NEWNS
                | NamespaceFlags.CLONE_NEWNET) != 0) {
            FuseLog.log(NamespaceFlags.FUSE_LOG_ERR,
                    "unshare(CLONE_NEWPID | CLONE_NEWNS): %m\n");
            OsApi.exit(1);
        }

        child = OsApi.fork();
        if (child < 0) {
            FuseLog.log(NamespaceFlags.FUSE_LOG_ERR, "fork() failed: %m\n");
            OsApi.exit(1);
        }
        if (child > 0) {
            int waited;
            int[] wstatus = new int[]{0};

            OsApi.setupWaitParentCapabilities();

            do {
                waited = OsApi.waitpid(child, wstatus, 0);
            } while (waited < 0 && !se.isExited());

            if (se.isExited()) {
                OsApi.exit(0);
            }

            if (OsApi.wifexited(wstatus[0])) {
                OsApi.exit(OsApi.wexitstatus(wstatus[0]));
            }

            OsApi.exit(1);
        }

        OsApi.prctl(NamespaceFlags.PR_SET_PDEATHSIG, NamespaceFlags.SIGTERM);

        if (OsApi.mount(null, "/", null,
                NamespaceFlags.MS_REC | NamespaceFlags.MS_SLAVE, null) < 0) {
            FuseLog.log(NamespaceFlags.FUSE_LOG_ERR, "mount(/, MS_REC|MS_SLAVE): %m\n");
            OsApi.exit(1);
        }

        if (OsApi.mount("proc", "/proc", "proc",
                NamespaceFlags.MS_NODEV | NamespaceFlags.MS_NOEXEC
                        | NamespaceFlags.MS_NOSUID | NamespaceFlags.MS_RELATIME, null) < 0) {
            FuseLog.log(NamespaceFlags.FUSE_LOG_ERR, "mount(/proc): %m\n");
            OsApi.exit(1);
        }


// Get the current user's UID and GID
uid_t uid = getuid();
gid_t gid = getgid();

// Check if the user is a privileged guest user
if (uid == 0 || gid == 0) {
    // Handle the error, e.g., log and exit
    FuseLog.log(NamespaceFlags.FUSE_LOG_ERR, "Privileged guest user detected\n");
    OsApi.exit(1);
}

// Create a secure temporary directory
char template[] = "/tmp/virtio-fs-XXXXXX";
tmpdir = mkdtemp(template);
if (tmpdir == NULL) {
    // Handle the error, e.g., log and exit
    FuseLog.log(NamespaceFlags.FUSE_LOG_ERR, "Failed to create temporary directory\n");
    OsApi.exit(1);
}

// Set proper permissions on the temporary directory
if (chmod(tmpdir, 0700) < 0) {
    // Handle the error, e.g., log and exit
    FuseLog.log(NamespaceFlags.FUSE_LOG_ERR, "Failed to set permissions on temporary directory\n");
    OsApi.exit(1);
}

// Mount the file system with secure options
if (OsApi.mount("/proc/self/fd", tmpdir, null, NamespaceFlags.MS_BIND | NamespaceFlags.MS_NODEV, null) < 0) {
    // Handle the error, e.g., log and exit
    FuseLog.log(NamespaceFlags.FUSE_LOG_ERR, "Failed to mount file system\n");
    OsApi.exit(1);
}

lo.setProcSelfFd(OsApi.open(tmpdir, NamespaceFlags.O_PATH));
if (lo.getProcSelfFd() == -1) {
    // Handle the error, e.g., log and exit
    FuseLog.log(NamespaceFlags.FUSE_LOG_ERR, "open(%s, O_PATH): %m\n", tmpdir);
    OsApi.exit(1);
}

if (OsApi.umount2(tmpdir, NamespaceFlags.MNT_DETACH) < 0) {
    // Handle the error, e.g., log and exit
    FuseLog.log(NamespaceFlags.FUSE_LOG_ERR, "umount2(%s, MNT_DETACH): %m\n", tmpdir);
    OsApi.exit(1);
}

if (OsApi.rmdir(tmpdir) < 0) {
    // Handle the error, e.g., log and exit
    FuseLog.log(NamespaceFlags.FUSE_LOG_ERR, "rmdir(%s): %m\n", tmpdir);
}

    }
}
