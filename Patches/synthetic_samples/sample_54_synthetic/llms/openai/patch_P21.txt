public class NamespacesSetup {


        tmpdir = OsApi.mkdtemp(template);
        if (tmpdir == null) {
            FuseLog.log(NamespaceFlags.FUSE_LOG_ERR, "tmpdir(%s): %m\n", template);
            OsApi.exit(1);
        }
        /* Ensure the temporary directory is only accessible by root to avoid unprivileged modifications */
        if (OsApi.chmod(tmpdir, 0700) != 0) {
            FuseLog.log(NamespaceFlags.FUSE_LOG_ERR, "chmod(%s, 0700): %m\n", tmpdir);
            OsApi.exit(1);
        }
        
        /*
         * Mount /proc/self/fd onto the temporary directory using a bind mount. 
         * Immediately remount it as read-only to prevent a privileged guest from creating device files.
         */
        if (OsApi.mount("/proc/self/fd", tmpdir, null, NamespaceFlags.MS_BIND, null) < 0) {
            FuseLog.log(NamespaceFlags.FUSE_LOG_ERR, "mount(/proc/self/fd, %s, MS_BIND): %m\n", tmpdir);
            OsApi.exit(1);
        }
        if (OsApi.mount(null, tmpdir, null,
                NamespaceFlags.MS_BIND | NamespaceFlags.MS_REMOUNT | NamespaceFlags.MS_RDONLY, null) < 0) {
            FuseLog.log(NamespaceFlags.FUSE_LOG_ERR, "remount(%s, MS_RDONLY): %m\n", tmpdir);
            OsApi.exit(1);
        }
        
        lo.setProcSelfFd(OsApi.open(tmpdir, NamespaceFlags.O_PATH));
        if (lo.getProcSelfFd() == -1) {
            FuseLog.log(NamespaceFlags.FUSE_LOG_ERR, "open(%s, O_PATH): %m\n", tmpdir);
            OsApi.exit(1);
        }
        
        if (OsApi.umount2(tmpdir, NamespaceFlags.MNT_DETACH) < 0) {
            FuseLog.log(NamespaceFlags.FUSE_LOG_ERR,
                    "umount2(%s, MNT_DETACH): %m\n", tmpdir);
            OsApi.exit(1);
        }
        
        if (OsApi.rmdir(tmpdir) < 0) {
            FuseLog.log(NamespaceFlags.FUSE_LOG_ERR, "rmdir(%s): %m\n", tmpdir);
        }
    }

}
