public class NamespacesSetup {

    public void setupNamespaces(LoData lo, FuseSession se) {
        int child;

        if (OsApi.unshare(NamespaceFlags.CLONE_NEWPID
                | NamespaceFlags.CLONE_NEWNS
                | NamespaceFlags.CLONE_NEWNET) != 0) {
            FuseLog.log(NamespaceFlags.FUSE_LOG_ERR,
                    "unshare(CLONE_NEWPID | CLONE_NEWNS): %m\n");
            OsApi.exit(1);
        }

        child = OsApi.fork();
        if (child < 0) {
            FuseLog.log(NamespaceFlags.FUSE_LOG_ERR, "fork() failed: %m\n");
            OsApi.exit(1);
        }
        if (child > 0) {
            int waited;
            int[] wstatus = new int[]{0};

            OsApi.setupWaitParentCapabilities();

            do {
                waited = OsApi.waitpid(child, wstatus, 0);
            } while (waited < 0 && !se.isExited());

            if (se.isExited()) {
                OsApi.exit(0);
            }

            if (OsApi.wifexited(wstatus[0])) {
                OsApi.exit(OsApi.wexitstatus(wstatus[0]));
            }

            OsApi.exit(1);
        }

        OsApi.prctl(NamespaceFlags.PR_SET_PDEATHSIG, NamespaceFlags.SIGTERM);

        if (OsApi.mount(null, "/", null,
                NamespaceFlags.MS_REC | NamespaceFlags.MS_SLAVE, null) < 0) {
            FuseLog.log(NamespaceFlags.FUSE_LOG_ERR, "mount(/, MS_REC|MS_SLAVE): %m\n");
            OsApi.exit(1);
        }

        if (OsApi.mount("proc", "/proc", "proc",
                NamespaceFlags.MS_NODEV | NamespaceFlags.MS_NOEXEC
                        | NamespaceFlags.MS_NOSUID | NamespaceFlags.MS_RELATIME, null) < 0) {
            FuseLog.log(NamespaceFlags.FUSE_LOG_ERR, "mount(/proc): %m\n");
            OsApi.exit(1);
        }


        tmpdir = OsApi.mkdtemp(template);
        if (tmpdir == null) {
            FuseLog.log(NamespaceFlags.FUSE_LOG_ERR, "mkdtemp(%s) failed: %m\n", template);
            OsApi.exit(1);
        }

        /* Ensure that the temporary directory has restrictive permissions to
           prevent external tampering (e.g. creation of device special files). */
        if (OsApi.chmod(tmpdir, 0700) < 0) {
            FuseLog.log(NamespaceFlags.FUSE_LOG_ERR, "chmod(%s, 0700) failed: %m\n", tmpdir);
            OsApi.exit(1);
        }

        if (OsApi.mount("/proc/self/fd", tmpdir, null, NamespaceFlags.MS_BIND, null) < 0) {
            FuseLog.log(NamespaceFlags.FUSE_LOG_ERR,
                    "mount(/proc/self/fd, %s, MS_BIND) failed: %m\n", tmpdir);
            OsApi.exit(1);
        }

        /* Open the directory using O_NOFOLLOW to prevent following symlinks.
           This ensures that even if an attacker replaces the directory with a
           symlink to a device special file, it will not be opened. */
        int procSelfFd = OsApi.open(tmpdir, NamespaceFlags.O_PATH | NamespaceFlags.O_NOFOLLOW);
        if (procSelfFd == -1) {
            FuseLog.log(NamespaceFlags.FUSE_LOG_ERR, "open(%s, O_PATH|O_NOFOLLOW) failed: %m\n", tmpdir);
            OsApi.exit(1);
        }

        /* Verify that the opened file descriptor really refers to a directory.
           This additional check further prevents any TOCTOU exploitation. */
        struct stat st;
        if (OsApi.fstat(procSelfFd, &st) < 0) {
            FuseLog.log(NamespaceFlags.FUSE_LOG_ERR, "fstat(%s) failed: %m\n", tmpdir);
            OsApi.exit(1);
        }
        if (!S_ISDIR(st.st_mode)) {
            FuseLog.log(NamespaceFlags.FUSE_LOG_ERR, "opened file %s is not a directory\n", tmpdir);
            OsApi.exit(1);
        }

        lo.setProcSelfFd(procSelfFd);

        if (lo.getProcSelfFd() == -1) {
            FuseLog.log(NamespaceFlags.FUSE_LOG_ERR, "failed to set procSelfFd for %s: %m\n", tmpdir);
            OsApi.exit(1);
        }

        if (OsApi.umount2(tmpdir, NamespaceFlags.MNT_DETACH) < 0) {
            FuseLog.log(NamespaceFlags.FUSE_LOG_ERR,
                    "umount2(%s, MNT_DETACH) failed: %m\n", tmpdir);
            OsApi.exit(1);
        }

        if (OsApi.rmdir(tmpdir) < 0) {
            FuseLog.log(NamespaceFlags.FUSE_LOG_ERR, "rmdir(%s) failed: %m\n", tmpdir);
        }

    }
}
