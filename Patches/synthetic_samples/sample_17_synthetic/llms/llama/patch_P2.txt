package main;

import commons.Logger;
import commons.Session;
import commons.ClientSession;
import commons.CliOptions;

public class AuthHandler {
    public static Session ses = new Session();
    public static ClientSession cli_ses = new ClientSession();
    public static CliOptions cli_opts = new CliOptions();

    public static void recv_msg_userauth_success() {

        Logger.trace("received msg_userauth_success");
        if (cli_ses.authstate.auth_methods.contains(ClientSession.AUTH_TYPE_PUBKEY)) {
            cli_ses.state = ClientSession.USERAUTH_SUCCESS_RCVD;
            cli_ses.lastauthtype = ClientSession.AUTH_TYPE_PUBKEY;
        } else if (cli_ses.authstate.auth_methods.contains(ClientSession.AUTH_TYPE_PASSWORD)) {
            cli_ses.state = ClientSession.USERAUTH_SUCCESS_RCVD;
            cli_ses.lastauthtype = ClientSession.AUTH_TYPE_PASSWORD;
        } else if (cli_ses.authstate.auth_methods.contains(ClientSession.AUTH_TYPE_KEYBOARD_INTERACTIVE)) {
            cli_ses.state = ClientSession.USERAUTH_SUCCESS_RCVD;
            cli_ses.lastauthtype = ClientSession.AUTH_TYPE_KEYBOARD_INTERACTIVE;
        } else {
            Logger.error("Unknown authentication type received");
            cli_ses.disconnect("Unknown authentication type received");
            return;
        }
        ClientSession.cli_auth_pubkey_cleanup();

    }
}
