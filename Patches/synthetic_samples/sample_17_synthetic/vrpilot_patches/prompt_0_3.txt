package main;

import commons.Logger;
import commons.Session;
import commons.ClientSession;
import commons.CliOptions;

public class AuthHandler {
    public static Session ses = new Session();
    public static ClientSession cli_ses = new ClientSession();
    public static CliOptions cli_opts = new CliOptions();

    public static void recv_msg_userauth_success() {
        Logger.trace("received msg_userauth_success");
        
        // Retrieve the authentication method used from the Session instance using the getter.
        String authMethodUsed = ses.getLastAuthMethod();
        
        // Validate that the authentication method is one of the RFC-approved methods.
        if (authMethodUsed == null ||
            (!"publickey".equals(authMethodUsed) &&
             !"password".equals(authMethodUsed) &&
             !"keyboard-interactive".equals(authMethodUsed))) {
            // If the authentication method does not match, terminate the session.
            throw new SecurityException("Detected non-RFC-compliant authentication method: " + authMethodUsed);
        }
        
        // Proceed with the rest of the authentication success logic.
        ses.authstate.authdone = true;
        cli_ses.state = ClientSession.USERAUTH_SUCCESS_RCVD;
        cli_ses.lastauthtype = ClientSession.AUTH_TYPE_NONE;
    }
}
