        Object armorKey = null;

        if (pa == null || pa.getPaDataValue() == null) {
            respond(context, new KrbError("Missing pre-authentication (OTP) data"), arg);
            return;
        }
        
        // 2. Decode the OTP request from the provided PA data.
        try {
            req = decodeKrbPaOtpReq(pa.getPaDataValue());
        } catch (Exception e) {
            respond(context, new KrbError("OTP request decoding failed: " + e.getMessage()), arg);
            return;
        }
        
        if (req == null) {
            respond(context, new KrbError("Invalid OTP request"), arg);
            return;
        }
        
        // 3. Additional validation of the OTP request payload.
        if (!req.isValid()) {
            respond(context, new KrbError("OTP request validation failed"), arg);
            return;
        }
        
        // 4. If the callback requires an armor key, retrieve and validate it.
        if (cb.requiresArmorKey(context)) {
            armorKey = cb.getArmorKey(context);
            if (armorKey == null) {
                respond(context, new KrbError("Armor key is required but missing"), arg);
                return;
            }
        }
        
        // 5. Prepare the internal state for OTP verification.
        rs = new RequestState();
        rs.setOtpRequest(req);
        rs.setArmorKey(armorKey);
        rs.setUserRequest(request);
        
        // 6. Perform OTP verification. This usually means that a computed OTP value,
        //    obtained from the supplied request, is compared against a locally computed or stored value.
        boolean otpValid = verifyOtp(rs);
        if (!otpValid) {
            respond(context, new KrbError("OTP verification failed"), arg);
            return;
        }
        
        // 7. Update the ticket reply on successful OTP validation.
        processSuccessfulVerification(rs, encTktReply);
        
        // 8. Report success (null error indicates success).
        respond(context, null, arg);
    }
