    private static String formatCollection(String reCollection, Matcher userMatch) {
        String out = reCollection;
        int groupCount = userMatch.groupCount();
        for (int i = 0; i < groupCount; i++) {
            String g = userMatch.group(i + 1);
            if (g == null) g = "";
            out = out.replace("{" + i + "}", g);
        }
        return out;
    }
