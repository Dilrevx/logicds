import java.util.HashMap;
import java.util.Map;
import java.util.regex.Matcher;
import java.util.regex.Pattern;
import java.util.regex.PatternSyntaxException;

public class RegexRights {


    public boolean readFromSections(String user, String collectionUrl, String permission) {
        // Validate input parameters to avoid null values or unexpected input.
        if (user == null || collectionUrl == null || permission == null) {
            return false;
        }
        
        // Validate the collectionUrl to guard against directory traversal or injection attacks.
        if (collectionUrl.contains("..") || collectionUrl.contains("~") || collectionUrl.contains("//")) {
            return false;
        }
        
        // Retrieve the file configuration and normalize rights type.
        String filename = Config.get("rights", "file");
        String rightsType = Config.get("rights", "type");
        if (rightsType != null) {
            rightsType = rightsType.toLowerCase();
        } else {
            rightsType = "";
        }
        
        // Ensure the filename is a safe file by canonicalizing and checking it lies within an allowed directory.
        File rightsFile = new File(filename);
        try {
            String canonicalPath = rightsFile.getCanonicalPath();
            // Define an allowed directory for rights files (update this path as appropriate).
            File allowedDirectory = new File("/opt/rights").getCanonicalFile();
            if (!canonicalPath.startsWith(allowedDirectory.getPath())) {
                // If the file does not reside in the approved directory, abort.
                return false;
            }
        } catch (IOException e) {
            // On any error resolving the file path, do not proceed.
            return false;
        }
        
        // Read the rights file and check whether the specified user, URL, and permission match an entry.
        try {
            // Use the default charset or specify one (e.g., StandardCharsets.UTF_8)
            List<String> lines = java.nio.file.Files.readAllLines(rightsFile.toPath(), java.nio.charset.StandardCharsets.UTF_8);
            for (String line : lines) {
                // Assume each line has three comma-separated values: user, collectionUrl, and permission.
                String[] parts = line.split(",");
                if (parts.length != 3) {
                    continue;
                }
                
                // Trim possible extra spaces in each field.
                String fileUser = parts[0].trim();
                String fileCollectionUrl = parts[1].trim();
                String filePermission = parts[2].trim().toLowerCase();
                
                // Compare the values. Permission comparison is done case-insensitively.
                if (fileUser.equals(user) && fileCollectionUrl.equals(collectionUrl)
                        && filePermission.equals(permission.toLowerCase())) {
                    return true;
                }
            }
        } catch (IOException e) {
            // If the file could not be read, return false to disallow unauthorized access.
            return false;
        }
        
        return false;
    }


    private static String formatCollection(String reCollection, Matcher userMatch) {
        String out = reCollection;
        int groupCount = userMatch.groupCount();
        for (int i = 0; i < groupCount; i++) {
            String g = userMatch.group(i + 1);
            if (g == null) g = "";
            out = out.replace("{" + i + "}", g);
        }
        return out;
    }
}
