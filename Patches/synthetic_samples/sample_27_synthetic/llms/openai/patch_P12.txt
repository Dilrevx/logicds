package main;

import commons.*;

public class NmHandleDetach {
    public static void isc__nmhandle_detach(HandleRef handlep) {
        Require.check(handlep != null);
        Require.check(HandleRef.valid(handlep));
        NmHandle handle = handlep.handle;
        handlep.handle = null;
        NmSocket sock = handle.sock;

if (sock == null || sock.mgr == null || sock.mgr.workers == null ||
    sock.tid < 0 || sock.tid >= sock.mgr.workers.length) {
    // Invalid sock state – reject the operation or log an error.
    Logger.error("Invalid sock or thread id detected in detach operation.");
    return;
}

if (sock.tid == NodeManager.nmTid()) {
    // Directly handle the detach on the current thread.
    NodeManager.nmhandle_detach_cb(handle);
} else {
    NetiEventDetach event = NodeManager.getNetiEventDetach(sock.mgr, sock);
    if (event == null) {
        Logger.error("Failed to create detach event.");
        return;
    }
    // Validate the handle before assignment if needed.
    if (handle == null) {
        Logger.error("Invalid handle passed to detach event.");
        return;
    }
    event.handle = handle;
    
    // Ensure safe passing of event to any other processing
    if (!EventPasser.passEvent(event)) {
        Logger.error("Event passing failed in detach event.");
        return;
    }
    
    NodeManager.enqueueEvent(sock.mgr.workers[sock.tid], event);
}

    }
}
