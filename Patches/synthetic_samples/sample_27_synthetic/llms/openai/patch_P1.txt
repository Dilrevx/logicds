package main;

import commons.*;

public class NmHandleDetach {
    public static void isc__nmhandle_detach(HandleRef handlep) {
        Require.check(handlep != null);
        Require.check(HandleRef.valid(handlep));
        NmHandle handle = handlep.handle;
        handlep.handle = null;
        NmSocket sock = handle.sock;

        if (sock.tid == NodeManager.nmTid()) {
            // If we’re in the same thread, immediately perform the detach and close.
            NodeManager.nmhandle_detach_cb(handle);
        } else {
            // For off‐thread sockets, create the detach event as before…
            NetiEventDetach event = NodeManager.getNetiEventDetach(sock.mgr, sock);
            event.handle = handle;
            EventPasser.passEvent(event);
            // Attempt to enqueue the event; if that fails then fall back to synchronous detach.
            if (!NodeManager.enqueueEvent(sock.mgr.workers[sock.tid], event)) {
                NodeManager.nmhandle_detach_cb(handle);
            }
            // In addition, schedule a forced detachment after a fixed timeout so that even if
            // a specially crafted TCP stream delays the normal handling the connection is finally closed.
            NodeManager.scheduleForcedDetach(handle, /* timeout in ms */ 5000);
        }

    }
}
