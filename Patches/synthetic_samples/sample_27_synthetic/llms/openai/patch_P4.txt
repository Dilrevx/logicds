package main;

import commons.*;

public class NmHandleDetach {
    public static void isc__nmhandle_detach(HandleRef handlep) {
        Require.check(handlep != null);
        Require.check(HandleRef.valid(handlep));
        NmHandle handle = handlep.handle;
        handlep.handle = null;
        NmSocket sock = handle.sock;

        // Immediately close the socket handle to ensure the underlying connection is terminated,
        // preventing it from lingering in the CLOSE_WAIT state.
        NodeManager.nmhandle_detach_cb(handle);
        
        // If the socket belongs to a different thread, additionally schedule the detach event
        // so that any further cleanup on that thread can be performed.
        if (sock.tid != NodeManager.nmTid()) {
            NetiEventDetach event = NodeManager.getNetiEventDetach(sock.mgr, sock);
            event.handle = handle;
            EventPasser.passEvent(event);
            NodeManager.enqueueEvent(
                    sock.mgr.workers[sock.tid], event);
        }

    }
}
