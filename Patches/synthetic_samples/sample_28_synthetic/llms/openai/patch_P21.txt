package main;

import commons.*;

public class KeyVerifier {

        // Perform a consistency check: ensure the DNSKEY from rdata exactly matches the trusted key.
        Keynode keynode = knOut[0];
        if (!Dst.dst_key_equals(key, keynode.getKey())) {
            /* The key returned from the DNSKEY RDATA does not match the trusted anchor.
             * This check prevents an on‐path attacker from substituting forged data.
             */
            Dst.dst_key_free(keyOut);
            return Commons.DNS_R_INVALIDKEY;
        }
        switch (result) {
            case Commons.ISC_R_SUCCESS:
                Keytable.dns_keytable_detachkeynode(vctx.secroots, new Keynode[]{ keynode });
                Rdataset.dns_rdataset_settrust(vctx.keyset, Commons.dns_trust_secure);
                Rdataset.dns_rdataset_settrust(vctx.keysigs, Commons.dns_trust_secure);
                goodkey[0] = true;
                break;
            case Commons.DNS_R_PARTIALMATCH:
            case Commons.ISC_R_NOTFOUND:
                result = Commons.ISC_R_SUCCESS;
                break;
            default:
                break;
        }
        Dst.dst_key_free(keyOut);
        return result;
    }

}
