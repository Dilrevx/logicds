package main;

import commons.*;

public class OCSPVerifier {

    public static int ocsp_verify_signer(X509 signer, int response,
                                         X509Store store, long flags,
                                         X509Stack untrusted, X509Stack[] chain) {
        X509StoreContext ctx = X509StoreContext.newContext();
        X509VerifyParam vp;
        int ret = -1;

        if (ctx == null) {
            ERR.raise("ERR_R_MALLOC_FAILURE");
            return ret;
        }

        if (!ctx.init(store, signer, untrusted)) {
            ERR.raise("ERR_R_X509_LIB");
            ctx.free();
            return ret;
        }

        vp = ctx.getParam();
        if (vp == null) {
            ctx.free();
            return ret;
        }

        // If a chain array is supplied with at least one element,
        // build the certificate chain from the context.
        if (chain != null && chain.length > 0) {
            // Use the accessor provided by X509StoreContext that returns the verified certificate chain.
            chain[0] = ctx.getVerifiedChain();
        }

        ctx.free();
        return ret;
    }
}
