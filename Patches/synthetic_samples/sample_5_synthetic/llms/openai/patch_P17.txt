package main;

import commons.*;

public class OCSPVerifier {


    ret = ctx.verify();

    /* If verification succeeded and a chain pointer was provided, capture the resulting chain */
    if (chain != null && ret == 1) {
        X509Stack resultChain = ctx.getChain();
        if (resultChain == null) {
            ERR.raise("ERR_R_X509_LIB");
            ret = -1;
        } else {
            chain[0] = resultChain;
        }
    }

    ctx.free();
    return ret;

}
