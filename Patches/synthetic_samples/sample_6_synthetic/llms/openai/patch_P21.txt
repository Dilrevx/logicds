package main;


import static commons.ChaCha20Poly1305Commons.*;
import commons.ChaCha20Poly1305Commons.EvpCipherCtx;
import commons.ChaCha20Poly1305Commons.EvpChaChaAeadCtx;
import commons.ChaCha20Poly1305Commons.Len;

public class ChaCha20Poly1305Ctrl {

            case EVP_CTRL_AEAD_SET_IVLEN:
                /* Enforce that nonce length must be exactly 12 bytes (96 bits) as specified by RFC 7539.
                   Allowing a nonce length different than 12 would risk nonce reuse vulnerabilities. */
                if (arg != 12) {
                    return 0;
                }
                actx.nonce_len = arg;
                return 1;

            default:
                return -1;
        }
    }

}
