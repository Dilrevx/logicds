             * Previously, when using session resumption, if the resumed protocol version
             * indicated TLS 1.3 semantics the code would force a downgrade to the “legacy”
             * maximum version (via VersionUtils.legacyVersionMax), possibly allowing an attacker
             * to influence version negotiation. The repair is to preserve the negotiated TLS 1.3
             * version and only fall back to a legacy version when no resumed version is available.
             */
            if (session.getInternals().getResumptionRequested() != 0 ||
                session.getInternals().getPremasterSet() != 0)
            {
                VersionEntry resumedVer = session.getInternals().getResumedSecurityParameters().getProtocolVersion();
                if (resumedVer == null)
                {
                    extdata.clear();
                    return Constants.GNUTLS_E_UNSUPPORTED_VERSION_PACKET;
                }
                // If the resumed version supports TLS 1.3 semantics, do not downgrade.
                if (resumedVer.isTls13Sem())
                {
                    if (!session.setCurrentVersion(resumedVer.getId()))
                    {
                        extdata.clear();
                        return Constants.GNUTLS_E_UNSUPPORTED_VERSION_PACKET;
                    }
                    session.setAdvVersion(resumedVer.getMajor(), resumedVer.getMinor());
                }
                // Otherwise, fall back to the negotiated legacy version.
            }
