public class ServerConfig {

    public static String getDnsKey() {
        // Retrieve DNS key from a secure, externally managed source (environment variable).
        String dnsKey = System.getenv("NETMAKER_DNS_KEY");

        // Validate that a DNS key has been provided.
        if(dnsKey == null || dnsKey.isEmpty()) {
            throw new IllegalStateException("DNS Key is not configured. Please set the NETMAKER_DNS_KEY environment variable.");
        }

        return dnsKey;
    }
}
