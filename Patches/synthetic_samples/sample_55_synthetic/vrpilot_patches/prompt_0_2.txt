public class ServerConfig {

    public static String getDnsKey() {
        // First, check if the DNS key is supplied as an environment variable.
        String dnsKey = System.getenv("DNS_KEY");
        if (dnsKey != null && !dnsKey.isEmpty()) {
            return dnsKey;
        }

        // Fallback: check the DNS key defined in configuration.
        dnsKey = Config.Server.getDnsKey();
        if (dnsKey != null && !dnsKey.isEmpty()) {
            return dnsKey;
        }

        // Neither the environment nor the config provided a non-empty key.
        throw new IllegalStateException(
            "DNS Key is not configured. Please set the DNS_KEY environment variable or provide it in the server configuration."
        );
    }
}
