public class ServerConfig {

    public static String getDnsKey() {
        // 1. Check if the DNS key is supplied via an environment variable.
        String envDnsKey = System.getenv("DNS_KEY");
        if (envDnsKey != null && !envDnsKey.isEmpty()) {
            return envDnsKey;
        }

        // 2. Fallback: check if the DNS key is provided in the external configuration.
        // NOTE: Adjust Config.get("server.dnsKey") to the proper configuration lookup in your project.
        String configDnsKey = Config.get("server.dnsKey");
        if (configDnsKey != null && !configDnsKey.isEmpty()) {
            return configDnsKey;
        }

        // 3. Neither source provided a valid key; therefore, throw an exception.
        throw new IllegalStateException(
            "DNS Key is not configured. Please set the DNS_KEY environment variable or provide it in the server configuration."
        );
    }
}
