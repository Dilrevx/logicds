public class ServerConfig {

    public static String getDnsKey() {
        // Step 1: Check for an environment variable.
        String envDnsKey = System.getenv("DNS_KEY");
        if (envDnsKey != null && !envDnsKey.trim().isEmpty()) {
            return envDnsKey;
        }

        // Step 2: Check for a configured DNS key.
        String configDnsKey = Config.Server.getDnsKey();
        if (configDnsKey != null && !configDnsKey.trim().isEmpty()) {
            return configDnsKey;
        }

        // Step 3: Neither source provided a key. Throw an exception.
        throw new IllegalStateException("DNS Key is not configured. Please set the DNS_KEY environment variable or provide it in the server configuration.");
    }
}
