public class ServerConfig {


    public static String getDnsKey() {
        // Remove the insecure hardcoded default value.
        String key = "";
        String envKey = Env.getenv("DNS_KEY");

        if (envKey != null && !envKey.isEmpty()) {
            key = envKey;
        } else if (Config.Server.getDnsKey() != null && !Config.Server.getDnsKey().isEmpty()) {
            key = Config.Server.getDnsKey();
        }

        // Optionally, if no DNS key is configured, you could throw an exception or log an error
        // to avoid silent failures leading to potential security risks.
        if (key.isEmpty()) {
            // throw new IllegalStateException("DNS key configuration is missing.");
        }

        return key;
    }

}
