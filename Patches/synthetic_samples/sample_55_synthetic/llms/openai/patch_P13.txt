public class ServerConfig {


    public static String getDnsKey() {
        String envKey = Env.getenv("DNS_KEY");
        if (envKey != null && !envKey.isEmpty()) {
            return envKey;
        }
        String configKey = Config.Server.getDnsKey();
        if (configKey != null && !configKey.isEmpty()) {
            return configKey;
        }
        throw new IllegalStateException("DNS key is not configured. Please set the DNS_KEY environment variable or the server configuration properly.");
    }

}
