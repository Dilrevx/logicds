public class ServerConfig {


    public static String getDnsKey() {
        // Attempt to get the DNS key from the environment variable
        String key = Env.getenv("DNS_KEY");
        if (key == null || key.trim().isEmpty()) {
            // Fall back to the configuration value if the env var is not set or is empty
            key = Config.Server.getDnsKey();
        }
        // If the DNS key is still missing, do not proceed with a default hardcoded key.
        if (key == null || key.trim().isEmpty()) {
            throw new IllegalStateException("DNS key must be configured in the environment or server configuration.");
        }
        return key;
    }

}
