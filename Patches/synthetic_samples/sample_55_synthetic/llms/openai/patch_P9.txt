public class ServerConfig {


public static String getDnsKey() {
    // First, try to get the DNS key from environment variables.
    String envKey = Env.getenv("DNS_KEY");
    if (envKey != null && !envKey.isEmpty()) {
        return envKey;
    }
    
    // Next, try to get the DNS key from configuration settings.
    String configKey = Config.Server.getDnsKey();
    if (configKey != null && !configKey.isEmpty()) {
        return configKey;
    }
    
    // Neither source provided a key.
    // Instead of falling back to a hardcoded default, we throw an exception to prevent unauthenticated access.
    throw new IllegalStateException("DNS Key is not properly configured.");
}

}
