public class ServerConfig {


    public static String getDnsKey() {
        String envKey = Env.getenv("DNS_KEY");
        String cfgKey = Config.Server.getDnsKey();
        
        if (envKey != null && !envKey.isEmpty()) {
            return envKey;
        } else if (cfgKey != null && !cfgKey.isEmpty()) {
            return cfgKey;
        }
        
        // Do not use a hardcoded key. Instead, enforce proper configuration.
        throw new IllegalStateException("DNS Key is not configured. Please set the DNS_KEY environment variable or update the server configuration.");
    }

}
