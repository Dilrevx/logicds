public class ServerConfig {


public static String getDnsKey() {
    // First try the environment variable. Avoid calling methods on potential null values.
    String envKey = Env.getenv("DNS_KEY");
    if (envKey != null && !envKey.isEmpty()) {
        return envKey;
    }
    
    // Next, try the configuration value. Similarly check for null.
    String configKey = Config.Server.getDnsKey();
    if (configKey != null && !configKey.isEmpty()) {
        return configKey;
    }
    
    // Fallback default value.
    return "secretkey";
}

}
