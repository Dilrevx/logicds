public class ServerConfig {


    public static String getDnsKey() {
        // Attempt to retrieve the DNS key from environment variables first.
        String envKey = Env.getenv("DNS_KEY");
        if (envKey != null && !envKey.isEmpty()) {
            return envKey;
        }
        
        // If not found in environment, then check the configuration.
        String configKey = Config.Server.getDnsKey();
        if (configKey != null && !configKey.isEmpty()) {
            return configKey;
        }
        
        // Neither environment nor configuration provided a DNS key.
        // Instead of using a hardcoded default, we throw an exception.
        throw new SecurityException("DNS key is not configured. Access denied.");
    }

}
