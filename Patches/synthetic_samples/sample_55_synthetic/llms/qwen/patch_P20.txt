public class ServerConfig {


        String key = "secretkey";
        String envKey = Env.getenv("DNS_KEY");
        if (envKey != null && !envKey.equals("")) {
            key = envKey;
        } else {
            String configKey = Config.Server.getDnsKey();
            if (configKey != null && !configKey.equals("")) {
                key = configKey;
            }
        }
        return key;
    }

}
