package master;

import common.*;

public class EapPeap {
    public static final int EAP_TLV_RESULT_SUCCESS = 1;
    public static final int EAP_TLV_RESULT_FAILURE = 0;

    /**
     * Converts the given 16-bit value from big-endian to the host’s byte order.
     * In other words, swaps the two bytes.
     *
     * @param x the 16-bit integer in big-endian order.
     * @return the integer in the host's byte order.
     */
    public static int WPA_GET_BE16(int x) {
        return ((x & 0xff) << 8) | ((x >> 8) & 0xff);
    }

    /**
     * Processes the EAP-TLV part of the message.
     *
     * @param sm            The state machine.
     * @param data          The EAP-PEAP data.
     * @param ret           The structure carrying method return information.
     * @param req           The WPABuf request.
     * @param force_failure If set to true, forces a failure outcome.
     * @return              The decision value from ret.
     */
    public static int eap_tlv_process(EapSm sm, EapPeapData data,
                                      EapMethodRet ret, WPABuf req,
                                      boolean force_failure) {
        int resp_status;
        // Convert the TLV status from big-endian to the host order.
        int tlvStatus = WPA_GET_BE16(EAP_TLV_RESULT_SUCCESS);

        if (tlvStatus == EAP_TLV_RESULT_SUCCESS) {
            if (force_failure) {
                resp_status = EAP_TLV_RESULT_FAILURE;
                ret.decision = EapMethodRet.DECISION_FAIL;
            } else {
                resp_status = EAP_TLV_RESULT_SUCCESS;
                ret.decision = EapMethodRet.DECISION_UNCOND_SUCC;
            }
        } else if (tlvStatus == EAP_TLV_RESULT_FAILURE) {
            resp_status = EAP_TLV_RESULT_FAILURE;
            ret.decision = EapMethodRet.DECISION_FAIL;
        } else {
            resp_status = EAP_TLV_RESULT_FAILURE;
            ret.decision = EapMethodRet.DECISION_FAIL;
        }
        
        // Setting additional state after processing.
        ret.methodState = 1;
        return ret.decision;
    }

    public static boolean peapPhase2Sufficient(EapSm sm, EapPeapData data) {
        if ((data.phase2Auth == EapPeapData.ALWAYS ||
                (data.phase2Auth == EapPeapData.FOR_INITIAL &&
                        !sm.tlsConnectionResumed() &&
                        !data.ssl.clientCertConf) ||
                data.phase2EapStarted) &&
                !data.phase2EapSuccess) {
            return false;
        }
        return true;
    }
}
