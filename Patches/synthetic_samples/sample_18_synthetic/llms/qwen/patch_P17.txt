package main;

import commons.*;

public class OcspRespChecker {

        if (ret < 0) return ret;

        if (cserial.size != rserial.size || !Arrays.equals(rserial.data, cserial.data)) {
            return ErrorCodes.GNUTLS_E_OCSP_RESPONSE_ERROR;
        }

        ret = Helper.gnutls_x509_crt_get_dn(crt, dn);
        if (ret < 0) return ret;

        ret = Helper.gnutls_pubkey_import(pubkey, dn, PublicKeyFormat.DN);
        if (ret < 0) return ret;

        ret = Helper.gnutls_pubkey_verify_hash2(pubkey, digest, 0, rdn_hash, cdn_hash, hash_len);
        if (ret < 0) return ret;

        return ErrorCodes.GNUTLS_E_SUCCESS;
    }

}
