package main;

import commons.*;

public class OcspRespChecker {

        ret = Helper.gnutls_x509_crt_get_serial(crt, cserial, tHolder);
        if (ret < 0) {
            return ret;
        }

        /* Ensure that the certificate serial and the OCSP response serial
           have the same length before comparing their contents. */
        if (cserial.size != rserial.size) {
            return ErrorCodes.GNUTLS_E_OCSP_RESPONSE_ERROR;
        }

        /* Compare the certificate serial with the OCSP response serial
           in a constant‐time manner to avoid timing attacks. */
        boolean match = true;
        for (int i = 0; i < cserial.size; i++) {
            if (rserial.data[i] != cserial.data[i]) {
                match = false;
            }
        }
        if (!match) {
            return ErrorCodes.GNUTLS_E_OCSP_RESPONSE_ERROR;
        }

        return 0;
    }

}
