package main;

import commons.*;

public class OcspRespChecker {
    public static int gnutls_ocsp_resp_check_crt(Object resp, int indx, Object crt) {
        int ret;
        DigestAlgorithm digest;
        Datum rdn_hash = new Datum();
        Datum rserial = new Datum();
        Datum cserial = new Datum();
        Datum dn = new Datum();
        byte[] cdn_hash = new byte[Helper.MAX_HASH_SIZE];
        long[] tHolder;
        int hash_len;
        DigestAlgorithm[] digestHolder = new DigestAlgorithm[1];

        // ... [earlier code that initializes rserial and cserial from resp and crt] ...

        // FIX: Verify that the OCSP response serial number length matches the certificate serial number length.
        // This check protects against trailing bytes extra data that might be present due to
        // gnutls_malloc behavior.
        if (rserial.getLength() != cserial.getLength()) {
            // The lengths do not match. This can indicate that the OCSP response carries extra trailing data.
            return GNUTLS_E_CERTIFICATE_ERROR;
        }

        ret = Helper.gnutls_x509_crt_get_raw_issuer_dn(crt, dn);
        if (ret < 0) {
            return ret;
        }

        ret = Helper.gnutls_hash_fast(digest, dn.data, dn.size, cdn_hash);
        if (ret < 0) {
            return ret;
        }

        if (!Helper.memcmp(cdn_hash, rdn_hash.data, hash_len)) {
            return ErrorCodes.GNUTLS_E_OCSP_RESPONSE_ERROR;
        }

        return 0;
    }
}
