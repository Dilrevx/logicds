public class Config {

    private static final int MOSQ_LOG_ERR     = 1;
    private static final int MOSQ_LOG_WARNING = 2;
    private static final int INT_MAX = Integer.MAX_VALUE;


// Ensure default listener is assigned
MosquittoListener def = config.getDefaultListener();
// FIX: Avoid potential NullPointerException by initializing security options if not set.
if (def.getSecurityOptions() == null) {
    def.setSecurityOptions(new SecurityOptions());
}

if (config.getListenerCount() == 0
        || def.getCafile() != null
        || def.getCapath() != null
        || def.getCertfile() != null
        || def.getKeyfile() != null
        || def.getCiphers() != null
        || def.getPskHint() != null
        || def.isRequireCertificate()
        || def.getCrlfile() != null
        || def.isUseIdentityAsUsername()
        || def.isUseSubjectAsUsername()
        || def.isUseUsernameAsClientid()
        || def.getHost() != null
        || def.getPort() != 0
        || def.getMaxConnections() != -1
        || def.getMountPoint() != null
        || def.getProtocol() != MosqProtocol.MQTT
        || def.getSocketDomain() != 0
        || def.getSecurityOptions().getPasswordFile() != null
        || def.getSecurityOptions().getPskFile() != null
        || def.getSecurityOptions().getAuthPluginConfigCount() != 0
        || def.getSecurityOptions().getAllowAnonymous() != -1) {

    config.incListenerCount();
    MosquittoListener promoted = new MosquittoListener();
    config.getListeners().add(promoted);

    // Set port: if none provided, default to 1883.
    if (def.getPort() != 0) {
        promoted.setPort(def.getPort());
    } else {
        promoted.setPort(1883);
    }

    // Propagate host and mount point if defined.
    promoted.setHost(def.getHost());
    promoted.setMountPoint(def.getMountPoint());
    
    // Copy remaining configuration settings.
    promoted.setMaxConnections(def.getMaxConnections());
    promoted.setProtocol(def.getProtocol());
    promoted.setSocketDomain(def.getSocketDomain());
    promoted.setClientCount(0);
    promoted.setSocks(null);
    promoted.setSockCount(0);
    promoted.setUseUsernameAsClientid(def.isUseUsernameAsClientid());
    promoted.setTlsVersion(def.getTlsVersion());
    promoted.setCafile(def.getCafile());
    promoted.setCapath(def.getCapath());
    promoted.setCertfile(def.getCertfile());
    promoted.setKeyfile(def.getKeyfile());
    promoted.setCiphers(def.getCiphers());
    promoted.setPskHint(def.getPskHint());
    promoted.setRequireCertificate(def.isRequireCertificate());
    promoted.setSslCtx(null);
    promoted.setCrlfile(def.getCrlfile());
    promoted.setUseIdentityAsUsername(def.isUseIdentityAsUsername());
    promoted.setUseSubjectAsUsername(def.isUseSubjectAsUsername());
}

return MosqErr.SUCCESS;
} 

}
