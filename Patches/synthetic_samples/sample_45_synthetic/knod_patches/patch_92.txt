public class Config {

    private static final int MOSQ_LOG_ERR     = 1;
    private static final int MOSQ_LOG_WARNING = 2;
    private static final int INT_MAX = Integer.MAX_VALUE;

    public int configParseArgs(MosquittoDb db, MosquittoConfig config, String[] argv) {
        int i;
        int portTmp;

        int argc = argv.length;
        for (i = 0; i < argc; i++) {
            if ("-c".equals(argv[i]) || "--config-file".equals(argv[i])) {
                if (i < argc - 1) {
                    db.setConfigFile(argv[i + 1]);
                    if (ConfUtil.configRead(db, config, false) != 0) {
                        ConfUtil.logPrintf(null, MOSQ_LOG_ERR,
                                "Error: Unable to open configuration file.");
                        return MosqErr.INVAL;
                    }
                } else {
                    ConfUtil.logPrintf(null, MOSQ_LOG_ERR,
                            "Error: -c argument given, but no config file specified.");
                    return MosqErr.INVAL;
                }
                i++;
            } else if ("-d".equals(argv[i]) || "--daemon".equals(argv[i])) {
                config.setDaemon(true);
            } else if ("-h".equals(argv[i]) || "--help".equals(argv[i])) {
                ConfUtil.printUsage();
                return MosqErr.INVAL;
            } else if ("-p".equals(argv[i]) || "--port".equals(argv[i])) {
                if (i < argc - 1) {
                    try {
                        portTmp = Integer.parseInt(argv[i + 1]);
                    } catch (NumberFormatException e) {
                        portTmp = 0;
                    }
                    if (portTmp < 1 || portTmp > 65535) {
                        ConfUtil.logPrintf(null, MOSQ_LOG_ERR,
                                "Error: Invalid port specified (%d).", portTmp);
                        return MosqErr.INVAL;
                    } else {
                        if (config.getDefaultListener().getPort() != 0) {
                            ConfUtil.logPrintf(null, MOSQ_LOG_WARNING,
                                    "Warning: Default listener port specified multiple times.");
                        }
                        config.getDefaultListener().setPort(portTmp);
                    }
                } else {
                    ConfUtil.logPrintf(null, MOSQ_LOG_ERR,
                            "Error: -p argument given, but no port specified.");
                    return MosqErr.INVAL;
                }
                i++;
            } else if ("-v".equals(argv[i]) || "--verbose".equals(argv[i])) {
                db.setVerbose(true);
            } else {
                ConfUtil.logPrintf(null, MOSQ_LOG_ERR,
                        "Error: Unknown option '%s'.", argv[i]);
                ConfUtil.printUsage();
                return MosqErr.INVAL;
            }
        }

        MosquittoListener def = config.getDefaultListener();
        if (config.getListenerCount() == 0
                || def.getCafile() != null
                || def.getCapath() != null
                || def.getCertfile() != null
                || def.getKeyfile() != null
                || def.getCiphers() != null
                || def.getPskHint() != null
                || def.isRequireCertificate()
                || def.getCrlfile() != null
                || def.isUseIdentityAsUsername()
                || def.isUseSubjectAsUsername()
                || def.isUseUsernameAsClientid()
                || def.getHost() != null
                || def.getPort() != 0
                || def.getMaxConnections() != -1
                || def.getMountPoint() != null
                || def.getProtocol() != MosqProtocol.MQTT
                || def.getSocketDomain() != 0
                || def.getSecurityOptions().getPasswordFile() != null
                || def.getSecurityOptions().getPskFile() != null
                || def.getSecurityOptions().getAuthPluginConfigCount() != 0
                || def.getSecurityOptions().getAllowAnonymous() != -1) {

            config.incListenerCount();
            MosquittoListener promoted = new MosquittoListener();
            config.getListeners().add(promoted);

            if (def.getPort() != 0) {
                promoted.setPort(def.getPort());
            } else {
                promoted.setPort(1883);
            }
            if (def.getHost() != null) {
                promoted.setHost(def.getHost());
            } else {
                promoted.setHost(null);
            }
            if (def.getMountPoint() != null) {
                promoted.setMountPoint(def.getMountPoint());
            } else {
                promoted.setMountPoint(null);
            }
            promoted.setMaxConnections(def.getMaxConnections());
            promoted.setProtocol(def.getProtocol());
            promoted.setSocketDomain(def.getSocketDomain());
            promoted.setClientCount(0);
            promoted.setSocks(null);
            promoted.setSockCount(0);
            promoted.setClientCount(0);
            promoted.setUseUsernameAsClientid(def.isUseUsernameAsClientid());

            promoted.setTlsVersion(def.getTlsVersion());
            promoted.setCafile(def.getCafile());
            promoted.setCapath(def.getCapath());
            promoted.setCertfile(def.getCertfile());
            promoted.setKeyfile(def.getKeyfile());
            promoted.setCiphers(def.getCiphers());
            promoted.setPskHint(def.getPskHint());
            promoted.setRequireCertificate(def.isRequireCertificate());
            promoted.setSslCtx(null);
            promoted.setCrlfile(def.getCrlfile());
            promoted.setUseIdentityAsUsername(def.isUseIdentityAsUsername());
            promoted.setUseSubjectAsUsername(def.isUseSubjectAsUsername());

            
//start of generated patch
promoted.getSecurityOptions(promoted);
//end of generated patch

            promoted.getSecurityOptions().setPskFile(def.getSecurityOptions().getPskFile());
            promoted.getSecurityOptions().setAuthPluginConfigs(def.getSecurityOptions().getAuthPluginConfigs());
            promoted.getSecurityOptions().setAuthPluginConfigCount(def.getSecurityOptions().getAuthPluginConfigCount());
            promoted.getSecurityOptions().setAllowAnonymous(def.getSecurityOptions().getAllowAnonymous());
        }

        if (config.getUser() == null) {
            config.setUser("mosquitto");
        }
        if (db.isVerbose()) {
            config.setLogType(INT_MAX);
        }
        return ConfUtil.configCheck(config);
    }
}
