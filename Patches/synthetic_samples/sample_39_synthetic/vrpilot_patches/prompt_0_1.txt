package main;

import commons.*;

public class Nas5g {
    public static final int SRSRAN_SUCCESS = 0;
    public static final int SRSRAN_ERROR = -1;
    static Logger logger = new Logger();
    private CtxtBase ctxt_base = new CtxtBase();
    private Ctxt5g ctxt_5g = new Ctxt5g();
    private Usim usim = new Usim();
    private boolean initial_sec_command = true;

    public int handle_security_mode_command(SecurityModeCommand security_mode_command, UniqueByteBuffer pdu) {

        // Fix: Reject the plain NAS security mode command during initial security activation.
        // Assuming SecurityModeCommand has a method getAlgorithm() returning an identifier
        // for the algorithm, and a constant NAS_SM_CMD_NIA0_plain to represent the plain command.
        if (security_mode_command.getAlgorithm() == SecurityModeCommand.NAS_SM_CMD_NIA0_plain &&
            initial_sec_command) {
            logger.error("Received plain NAS_SM_CMD_NIA0_plain command during initial security activation. Rejecting command.");
            return SRSRAN_ERROR;
        }
        send_security_mode_complete(security_mode_command);
        ctxt_base.rx_count++;
        return SRSRAN_SUCCESS;
    }

    }

    private void send_security_mode_reject(int cause) {}

    private void send_security_mode_complete(SecurityModeCommand cmd) {}
}
