"""
Rights management.

Rights are based on a regex-based file whose name is specified in the config
(section "right", key "file").

Authentication login is matched against the "user" key, and collection's path
is matched against the "collection" key. You can use Python's ConfigParser
interpolation values %(login)s and %(path)s. You can also get groups from the
user regex in the collection with {0}, {1}, etc.

For example, for the "user" key, ".+" means "authenticated user" and ".*"
means "anybody" (including anonymous users).

Section names are only used for naming the rule.

Leading or ending slashes are trimmed from collection's path.

"""

import re
import sys
import os.path

from .. import config, log

# Manage Python2/3 different modules
if sys.version_info[0] == 2:
    from ConfigParser import ConfigParser
    from StringIO import StringIO
else:
    from configparser import ConfigParser
    from io import StringIO


DEFINED_RIGHTS = {
    "authenticated": "[rw]\nuser:.+\ncollection:.*\npermission:rw",
    "owner_write": "[r]\nuser:.+\ncollection:.*\npermission:r\n"
                   "[w]\nuser:.+\ncollection:^%(login)s(/.*)?$\npermission:w",
    "owner_only": "[rw]\nuser:.+\ncollection:^%(login)s(/.*)?$\npermission:rw",
}

def _read_from_sections(user, collection_url, permission, rights_type, filename=None):
    """
    Check if the given user is allowed the required permission on the
    collection_url by matching against rights defined in sections.

    This function loads rights from a predefined dictionary (or file)
    then iterates over all sections. It uses full matching (with re.fullmatch)
    for both the user and collection regex. Before reading the rights
    definitions, any '%(login)s' placeholder is safely replaced with the
    escaped username to prevent regex injection.
    """
    rights_config = ConfigParser()

    # If a filename is provided, try to read rights definitions from file.
    if filename:
        log.debug("Reading rights from file: %s", filename)
        try:
            with open(filename, "r") as f:
                rights_config.read_file(f)
        except FileNotFoundError:
            log.error("File '%s' not found for rights", filename)
            return False
        except Exception as exc:
            log.error("Error reading file '%s': %s", filename, exc)
            return False
    else:
        # Use in-code rights definitions based on rights_type.
        if rights_type not in DEFINED_RIGHTS:
            log.error("Unknown rights type '%s'", rights_type)
            return False
        log.debug("Using in-code rights type '%s'", rights_type)
        # Safely replace any '%(login)s' placeholder using re.escape on the username.
        rights_text = DEFINED_RIGHTS[rights_type].replace("%(login)s", re.escape(user))
        try:
            rights_config.read_string(rights_text)
        except Exception as exc:
            log.error("Error reading rights definitions: %s", exc)
    return False


def authorized(user, collection, permission):
    """Check if the user is allowed to read or write the collection.

       If the user is empty it checks for anonymous rights
    """
    collection_url = collection.url.rstrip("/") or "/"
    if collection_url in (".well-known/carddav", ".well-known/caldav"):
        return permission == "r"
    rights_type = config.get("rights", "type").lower()
    return (
        rights_type == "none" or
        _read_from_sections(user or "", collection_url, permission))
