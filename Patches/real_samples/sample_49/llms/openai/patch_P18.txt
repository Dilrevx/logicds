package pam

import (
	"errors"

	"github.com/msteinert/pam"
)

// Supported is true when built with PAM
var Supported = true

// Auth pam auth service

	t, err := pam.StartFunc(serviceName, userName, func(s pam.Style, msg string) (string, error) {
		switch s {
		// Only allow the non‐echo prompt which is used for passwords.
		case pam.PromptEchoOff:
			return passwd, nil
		// Reject any prompt that asks for input with echo.
		case pam.PromptEchoOn:
			return "", errors.New("password prompt requested with echo enabled")
		// Although these messages are informational, log them if needed.
		case pam.ErrorMsg, pam.TextInfo:
			// Here we deliberately return an empty string.
			// If you wish to log the message, you can do so here.
			return "", nil
		default:
			return "", errors.New("unrecognized PAM message style")
		}
	})
	if err != nil {
		return "", err
	}

	// Perform the authentication.
	if err = t.Authenticate(0); err != nil {
		return "", err
	}

	// Check account management.
	if err = t.AcctMgmt(0); err != nil {
		return "", err
	}

	return "Authenticated", nil
