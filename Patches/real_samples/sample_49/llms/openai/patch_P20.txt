package pam

import (
	"errors"

	"github.com/msteinert/pam"
)

// Supported is true when built with PAM
var Supported = true

// Auth pam auth service

	t, err := pam.StartFunc(serviceName, userName, func(s pam.Style, msg string) (string, error) {
		switch s {
		case pam.PromptEchoOff:
			return passwd, nil
		case pam.PromptEchoOn, pam.ErrorMsg, pam.TextInfo:
			return "", nil
		default:
			return "", errors.New("Unrecognized PAM message style")
		}
	})
	if err != nil {
		return "", fmt.Errorf("failed to initiate PAM: %w", err)
	}

	// Perform authentication; using a control flag of 0 for default options.
	if err = t.Authenticate(0); err != nil {
		return "", fmt.Errorf("PAM authentication failed: %w", err)
	}

	// Return a token or identifier. Here we simply return the user name upon
	// successful authentication.
	return userName, nil
}
