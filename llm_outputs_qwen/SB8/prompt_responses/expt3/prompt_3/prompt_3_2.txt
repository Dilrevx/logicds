if (extMsgType[0] == CommonsConstants.server_hello) {
    PreSharedKey psk = null;
    KeyShare kse = null;

    // Check for PSK extension
    TlsExtension ext = TLSX.find(ssl.extensions, CommonsConstants.TLSX_PRE_SHARED_KEY);
    if (ext != null)
        psk = (PreSharedKey) ext.data;
    while (psk != null && !psk.chosen)
        psk = psk.next;

    // Check for KSE extension
    ext = TLSX.find(ssl.extensions, CommonsConstants.TLSX_KEY_SHARE);
    if (ext != null)
        kse = (KeyShare) ext.data;

    // If neither PSK nor KSE is provided, abort the handshake
    if (psk == null && kse == null) {
        System.out.println("Error: Server did not provide PSK or KSE extensions.");
        return -1; // Or some other error code
    } else {
        if (psk != null) {
            int ret = SetupPskKey(ssl, psk, 0);
            if (ret != 0)
                return ret;
            ssl.options.pskNegotiated = true;
        }
        // If KSE was negotiated, additional handling would be required here
    }

    ssl.keys.encryptionOn = true;
    ssl.options.serverState = CommonsConstants.SERVER_HELLO_COMPLETE;
}
