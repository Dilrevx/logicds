#include <gnutls/x509.h>
#include <assert.h>
#include <stdio.h>

int import_certificate(gnutls_datum_t *der_data, gnutls_x509_crt_t *cert) {
    int ret;
    gnutls_datum_t cert_der = *der_data;

    // Initialize the certificate structure
    ret = gnutls_x509_crt_init(cert);
    if (ret < 0) {
        fprintf(stderr, "Failed to initialize certificate: %s\n", gnutls_strerror(ret));
        return ret;
    }

    // Import the certificate
    ret = gnutls_x509_crt_import(*cert, &cert_der, GNUTLS_X509_FMT_DER);
    if (ret < 0) {
        fprintf(stderr, "Failed to import certificate: %s\n", gnutls_strerror(ret));
        gnutls_x509_crt_deinit(*cert);
        return ret;
    }

    // Retrieve the signature algorithm from the certificate
    gnutls_sign_algorithm_t cert_sign_algo;
    ret = _gnutls_x509_get_signature_algorithm(*cert, "signatureAlgorithm.algorithm", &cert_sign_algo);
    if (ret < 0) {
        fprintf(stderr, "Failed to get certificate signature algorithm: %s\n", gnutls_strerror(ret));
        gnutls_x509_crt_deinit(*cert);
        return GNUTLS_E_X509_CERTIFICATE_ERROR;
    }

    // Retrieve the signature algorithm from the TBSCertificate
    gnutls_sign_algorithm_t tbs_sign_algo;
    ret = _gnutls_x509_get_signature_algorithm(*cert, "tbsCertificate.signature.algorithm", &tbs_sign_algo);
    if (ret < 0) {
        fprintf(stderr, "Failed to get TBSCertificate signature algorithm: %s\n", gnutls_strerror(ret));
        gnutls_x509_crt_deinit(*cert);
        return GNUTLS_E_X509_CERTIFICATE_ERROR;
    }

    // Compare the signature algorithms
    if (cert_sign_algo != tbs_sign_algo) {
        fprintf(stderr, "Signature algorithm mismatch: Certificate uses %s, but TBSCertificate uses %s\n",
                gnutls_sign_get_name(cert_sign_algo), gnutls_sign_get_name(tbs_sign_algo));
        gnutls_x509_crt_deinit(*cert);
        return GNUTLS_E_X509_CERTIFICATE_ERROR;
    }

    // If algorithms match, proceed with further operations
    // For demonstration, let's assume we want to retrieve the issuer DN
    char issuer_dn[256];
    size_t issuer_dn_size = sizeof(issuer_dn);
    ret = gnutls_x509_crt_get_issuer_dn(*cert, issuer_dn, &issuer_dn_size);
    if (ret < 0) {
        fprintf(stderr, "Failed to get issuer DN: %s\n", gnutls_strerror(ret));
        gnutls_x509_crt_deinit(*cert);
        return ret;
    }
    printf("Issuer DN: %s\n", issuer_dn);

    return 0; // Success
}
