int mbedtls_x509_crt_is_revoked( const mbedtls_x509_crt *crt, const mbedtls_x509_crl *crl )
{
    const mbedtls_x509_crl_entry *cur = crl->entry;

    while( cur != NULL && cur->serial.len != 0 )
    {
        if( crt->serial.len == cur->serial.len &&
            memcmp( crt->serial.p, cur->serial.p, crt->serial.len ) == 0 )
        {
            // Check if the revocation date is before the current time
            int ret = mbedtls_x509_time_is_past( &cur->revocation_date );
            if( ret == 0 ) // Time is not past, so certificate is still considered valid
                return( 0 );
            else if( ret == 1 ) // Time is past, so certificate is revoked
                return( 1 );
            else // Error in time comparison
                return( ret ); // Propagate error
        }

        cur = cur->next;
    }

    // No matching entry found or no entries in CRL
    return( 0 );
}
