The current code does not check the `emm_cause` value for `Attach/Service Reject` messages, which can lead to unauthorized processing of these messages.

```c++
if (sec_hdr_type == LIBLTE_MME_SECURITY_HDR_TYPE_PLAIN_NAS) {
    switch (msg_type) {
        case LIBLTE_MME_MSG_TYPE_EMM_INFORMATION:
        case LIBLTE_MME_MSG_TYPE_EMM_STATUS:
        case LIBLTE_MME_MSG_TYPE_DETACH_REQUEST:
            // These messages are allowed in plaintext.
            break;
        case LIBLTE_MME_MSG_TYPE_ATTACH_REJECT:
        case LIBLTE_MME_MSG_TYPE_SERVICE_REJECT:
            // Check if emm_cause is 25 before allowing these messages in plaintext.
            if (emm_cause == 25) {
                break;
            } else {
                logger.error(
                        "Not handling NAS message MSG_TYPE=%02X with SEC_HDR_TYPE=%02X and invalid EMM_CAUSE=%02X without integrity protection!",
                        msg_type,
                        sec_hdr_type,
                        emm_cause
                );
                return SRSRAN_ERROR;
            }
        default:
            logger.error(
                    "Not handling NAS message MSG_TYPE=%02X with SEC_HDR_TYPE=%02X without integrity protection!",
                    msg_type,
                    sec_hdr_type
            );
            return SRSRAN_ERROR;
    }
}
```
