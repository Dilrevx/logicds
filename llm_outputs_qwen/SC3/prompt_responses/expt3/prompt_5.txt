if (securityOpts.aclList == NULL && securityOpts.aclPatterns == NULL) {
    // Check if an ACL file is specified
    if (securityOpts.aclFile != NULL) {
        // Attempt to read and parse the ACL file
        FILE *aclStream = fopen(securityOpts.aclFile, "r");
        if (aclStream != NULL) {
            bool isEmptyOrCommentsOnly = true;
            char line[1024];
            while (fgets(line, sizeof(line), aclStream)) {
                // Trim whitespace and check if the line is not a comment or empty
                char *trimmedLine = line;
                while (*trimmedLine && isspace((unsigned char)*trimmedLine)) trimmedLine++;
                size_t len = strlen(trimmedLine);
                if (len > 0 && trimmedLine[len - 1] == '\n') trimmedLine[len - 1] = '\0';
                if (len > 0 && trimmedLine[len - 1] == '\r') trimmedLine[len - 1] = '\0';
                if (strlen(trimmedLine) > 0 && trimmedLine[0] != '#') {
                    isEmptyOrCommentsOnly = false;
                    break;
                }
            }
            fclose(aclStream);

            // If the ACL file is empty or contains only comments, deny all access
            if (isEmptyOrCommentsOnly) {
                return MosqErr.MOSQ_ERR_ACL_DENIED;
            }
        } else {
            // If we cannot open the ACL file, log an error and deny all access
            fprintf(stderr, "Failed to open ACL file: %s\n", securityOpts.aclFile);
            return MosqErr.MOSQ_ERR_ACL_DENIED;
        }
    }
    // If no ACL file is specified, defer the decision
    return MosqErr.MOSQ_ERR_PLUGIN_DEFER;
}
