  bool security_active = is_security_context_active(); // Assume this function checks if NAS security context is active

      if (!security_active) {

      }
      if (integrity_check(pdu.get()) == false) {
        logger.error("Not handling NAS message with integrity check error");
        return SRSRAN_ERROR;
      }
      break;
    case nas_5gs_hdr::security_header_type_opts::integrity_protected_and_ciphered:
      if (!security_active) {
        logger.error("Received integrity protected and ciphered NAS message before security activation");
        return SRSRAN_ERROR;
      }
      if (integrity_check(pdu.get()) == false) {
        logger.error("Not handling NAS message with integrity check error");
        return SRSRAN_ERROR;

        cipher_decrypt(pdu.get());
      }
      break;

      if (!security_active) {
