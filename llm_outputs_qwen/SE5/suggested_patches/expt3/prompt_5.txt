package main;

import commons.*;

public class RrcNr {
    private int transaction_id;
    private SecCfg sec_cfg = new SecCfg();
    private boolean security_is_activated;
    private Pdcp pdcp = new Pdcp();
    private static final String[] ciphering_algorithm_id_nr_text = {"ALG0", "ALG1"};
    private static final String[] integrity_algorithm_id_nr_text = {"NIA0", "NIA1"};
    static Logger logger = new Logger();

    public void handle_security_mode_command(SecurityModeCmdS smc) {
        transaction_id = smc.rrc_transaction_id;
        SecurityAlgorithmCfg sec_algo_cfg = smc.crit_exts.security_mode_cmd().security_cfg_smc.security_algorithm_cfg;
        sec_cfg.cipher_algo = CIPHERING_ALGORITHM_ID_ENUM.fromValue(sec_algo_cfg.ciphering_algorithm.value);
        if (sec_algo_cfg.integrity_prot_algorithm_present) {
            sec_cfg.integ_algo = INTEGRITY_ALGORITHM_ID_ENUM.fromValue(sec_algo_cfg.integrity_prot_algorithm.value);
            if (sec_cfg.integ_algo == INTEGRITY_ALGORITHM_ID_ENUM.NIA0) {
                logger.error("prohibited value in RRC SMC!");
                return;
            }
        } else {
            logger.error("Missing Integrity Algorithm Config");
        }
        generate_as_keys();
        security_is_activated = true;
        int lcid = srb_to_lcid(NrSrb.srb1);
        pdcp.config_security(lcid, sec_cfg);
        pdcp.enable_integrity(lcid, Pdcp.DIRECTION_TXRX);
        send_security_mode_complete();
        pdcp.enable_encryption(lcid, Pdcp.DIRECTION_TXRX);
    }

    private void generate_as_keys() {}
    private int srb_to_lcid(NrSrb srb) { return 0; }
    private void send_security_mode_complete() {}

    public enum CIPHERING_ALGORITHM_ID_ENUM {
        AES_128_EEA2,
        // Add other algorithms as needed
    }

    public enum INTEGRITY_ALGORITHM_ID_ENUM {
        AES_128_CMAC,
        // Add other algorithms as needed
    }

    public class SecCfg {
        public CIPHERING_ALGORITHM_ID_ENUM cipher_algo;
        public INTEGRITY_ALGORITHM_ID_ENUM integ_algo;

        public SecCfg(CIPHERING_ALGORITHM_ID_ENUM cipher_algo, INTEGRITY_ALGORITHM_ID_ENUM integ_algo) {
            this.cipher_algo = cipher_algo;
            this.integ_algo = integ_algo;
        }
    }

    public enum NrSrb {
        srb1
        // Add other SRBs if needed
    }

    public class RrcLayer {
        private boolean isSecurityActivated = false; // Simulated flag for security activation

        public void activateSecurity() {
            isSecurityActivated = true;
        }

        public void applySecurityConfig(SecCfg secCfg, NrSrb srb) {
            if (!isSecurityActivated) {
                throw new SecurityException("RRC layer security must be activated before applying security configurations.");
            }
            // Apply security configuration here
            System.out.println("Applying security configuration for " + srb + ": Cipher Algo = " + secCfg.cipher_algo + ", Integrity Algo = " + secCfg.integ_algo);
        }
    }

    public class Main {
        public static void main(String[] args) {
            RrcLayer rrcLayer = new RrcLayer();
            SecCfg secCfg = new SecCfg(CIPHERING_ALGORITHM_ID_ENUM.AES_128_EEA2, INTEGRITY_ALGORITHM_ID_ENUM.AES_128_CMAC);

            try {
                // Attempt to apply security config without activating security
                rrcLayer.applySecurityConfig(secCfg, NrSrb.srb1);
            } catch (SecurityException e) {
                System.out.println(e.getMessage()); // Expected output: RRC layer security must be activated before applying security configurations.
            }

            // Activate security
            rrcLayer.activateSecurity();

            // Now apply security config
            rrcLayer.applySecurityConfig(secCfg, NrSrb.srb1); // Expected output: Applying security configuration for srb1: Cipher Algo = AES_128_EEA2, Integrity Algo = AES_128_CMAC
        }
    }
