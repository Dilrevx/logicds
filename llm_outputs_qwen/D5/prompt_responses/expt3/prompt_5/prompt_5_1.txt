```java
import org.jivesoftware.openfire.event.UserEventListener;
import org.jivesoftware.openfire.event.UserEventDispatcher;
import org.jivesoftware.openfire.user.User;
import org.jivesoftware.openfire.user.UserManager;

public class AdminManager {

    private AdminManager() {
        // Load an admin provider.
        initProvider(ADMIN_PROVIDER.getValue());

        // Register a UserEventListener to handle user events.
        UserEventDispatcher.addListener(new AdminUserEventListener());
    }

    private void initProvider(String providerClassName) {
        // Existing code to initialize the admin provider.
    }

    // Inner class implementing UserEventListener
    private static class AdminUserEventListener implements UserEventListener {

        @Override
        public void userDeleting(User user, Map<String, Object> params) {
            // Remove any admin privileges associated with the deleted user.
            removeAdminPrivileges(user.getUsername());
        }

        @Override
        public void userCreated(User user, Map<String, Object> params) {
            // No action required on user creation.
        }

        @Override
        public void userModified(User user, Map<String, Object> params) {
            // No action required on user modification.
        }

        private void removeAdminPrivileges(String username) {
            // Logic to remove admin privileges for the given username.
            // This might involve removing the username from the admin.authorizedJIDs list.
            String authorizedJIDs = System.getProperty("admin.authorizedJIDs");
            if (authorizedJIDs != null) {
                String[] jids = authorizedJIDs.split(",");
                StringBuilder newJIDs = new StringBuilder();
                for (String jid : jids) {
                    if (!jid.trim().equals(username)) {
                        if (newJIDs.length() > 0) {
                            newJIDs.append(",");
                        }
                        newJIDs.append(jid);
                    }
                }
                System.setProperty("admin.authorizedJIDs", newJIDs.toString());
            }
        }
    }
}
```

### Explanation

- **UserEventListener**: We've defined an inner class `AdminUserEventListener` that implements `UserEventListener`. This class handles user events.
- **userDeleting Method**: This method is called when a user is about to be deleted. It calls `removeAdminPrivileges` to ensure that the user's admin privileges are revoked.
- **removeAdminPrivileges Method**: This method removes the username from the `admin.authorizedJIDs` system property, effectively revoking their admin rights.
- **UserEventDispatcher.addListener**: We register our listener during the initialization of `AdminManager`.

This approach ensures that any attempt to delete and recreate a user with admin privileges will fail, thus mitigating the risk of privilege escalation through manipulation of the `admin.authorizedJIDs` property.
