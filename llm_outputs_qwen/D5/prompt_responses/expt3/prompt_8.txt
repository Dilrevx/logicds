    private AdminManager() {
        // Load an admin provider.
        initProvider(ADMIN_PROVIDER.getValue());

        // Add a listener to handle user events
        UserEventDispatcher.addListener(new UserEventListener() {
            @Override
            public void userDeleting(String username, Map<String, Object> params) throws UserAlreadyExistsException, UserNotFoundException {
                // Remove admin privileges when a user is deleted
                String adminJIDs = System.getProperty("admin.authorizedJIDs");
                if (adminJIDs != null && adminJIDs.contains(username)) {
                    List<String> jidList = Arrays.asList(adminJIDs.split(","));
                    jidList.remove(username);
                    System.setProperty("admin.authorizedJIDs", String.join(",", jidList));
                }
            }

            @Override
            public void userCreated(String username, Map<String, Object> params) throws UserAlreadyExistsException {
                // No action needed on user creation
            }

            @Override
            public void userModified(String username, Map<String, Object> params) throws UserNotFoundException {
                // No action needed on user modification
            }
        });
    }
