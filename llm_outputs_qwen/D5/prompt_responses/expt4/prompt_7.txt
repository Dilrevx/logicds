    private AdminManager() {
        // Load an admin provider, ensuring it's within an allowed list to prevent privilege escalation.
        Class<?> adminProviderClass = ADMIN_PROVIDER.getValue();
        if (isAllowedAdminProvider(adminProviderClass)) {
            initProvider(adminProviderClass);
        } else {
            Log.error("Attempt to set unauthorized admin provider: " + adminProviderClass.getName());
            throw new SecurityException("Unauthorized admin provider specified.");
        }
    }

    private boolean isAllowedAdminProvider(Class<?> clazz) {
        // Define a list of allowed classes here.
        List<Class<?>> allowedProviders = Arrays.asList(DefaultAdminProvider.class, AnotherAllowedProvider.class);
        return allowedProviders.contains(clazz);
    }
