		dns_keytable_detachkeynode(vctx->secroots, &keynode);
		dns_rdataset_settrust(&vctx->keyset, dns_trust_secure);
		dns_rdataset_settrust(&vctx->keysigs, dns_trust_secure);
		*goodkey = true;
	} else if (result == DNS_R_PARTIALMATCH || result == ISC_R_NOTFOUND) {
		result = ISC_R_SUCCESS; // Not an error, continue processing
	} else {
		goto cleanup; // Error occurred, clean up and return
	}

	// Iterate over all key nodes to ensure thorough validation
	while (result == ISC_R_SUCCESS) {
		result = dns_keytable_findnextkeynode(vctx->secroots, &keynode);
		if (result != ISC_R_SUCCESS) {
			break;
		}
		if (dst_key_compare(key, keynode->key) == 0) {
			dns_keytable_detachkeynode(vctx->secroots, &keynode);
			dns_rdataset_settrust(&vctx->keyset, dns_trust_secure);
			dns_rdataset_settrust(&vctx->keysigs, dns_trust_secure);
			*goodkey = true;
			break;
		}
		dns_keytable_detachkeynode(vctx->secroots, &keynode);
	}

	dst_key_free(&key);
