    if (srtp->probation_cnt > 0 &&

        pjmedia_srtp_setting tx_policy, rx_policy;
        pjmedia_transport_srtp_get_setting(srtp, PJMEDIA_DIR_ENCODING, &tx_policy);

        // Stop the current SRTP session
        pjmedia_transport_srtp_stop(srtp);

        err = pjmedia_transport_srtp_start(srtp, &tx_policy, &rx_policy);
        if (err != PJ_SUCCESS) {

            return err; // Return error if restart fails

        // Reattempt packet protection
        if (!*bypass_srtp) {
            err = srtp_unprotect(srtp->srtp_ctx, pkt, &pkt_len);

}
