package main;
import commons.*;
import java.util.Arrays;
public class LookupFrec {
    public static Frec lookup_frec(int id, byte[] hash, int family, int fd) {
        for (Frec f = Daemon.frec_list; f != null; f = f.next) {

if (f.sentto != null
        && f.new_id == id
        && (hash == null || Arrays.equals(hash, f.hash))) {

    int af = f.sentto.getFamily();
    if (af == Commons.AF_INET) {
        if (f.rfd4 != null && f.rfd4.getFD() == fd) {
            return f;
        }
    } else if (af == Commons.AF_INET6) {
        if (f.rfd6 != null && f.rfd6.getFD() == fd) {
            return f;
        }
    } else {
        if (f.sentto.sfd != null && f.sentto.sfd.getFD() == fd) {
            return f;
        }
    }
}
return null;

        }
        return null;
    }
}
