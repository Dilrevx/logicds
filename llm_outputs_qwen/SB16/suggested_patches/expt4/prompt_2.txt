package main;
import commons.*;
import java.util.Arrays;
public class LookupFrec {
    public static Frec lookup_frec(int id, byte[] hash, int family, int fd) {
        for (Frec f = Daemon.frec_list; f != null; f = f.next) {

            if (f.sentto != null
                    && f.new_id == id
                    && (hash == null || Arrays.equals(hash, f.hash))
                    && ((f.sentto.family == Commons.AF_INET && f.sentto.rfd4 != null && fd == f.sentto.rfd4)
                        || (f.sentto.family == Commons.AF_INET6 && f.sentto.rfd6 != null && fd == f.sentto.rfd6)
                        || (f.sentto.sfd != null && fd == f.sentto.sfd))) {
                return f;
            }

        }
        return null;
    }
}
