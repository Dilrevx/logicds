        int resp_status;
        int tlvStatus = WPA_GET_BE16(EAP_TLV_RESULT_SUCCESS);

        if (req == null || req.length() == 0) {
            return EAP_FAIL;
        }

        // Process TLVs
        while (req.has_remaining()) {
            int type = WPA_GET_BE16(req.get_data());
            int length = WPA_GET_BE16(req.get_data() + 2);
            byte[] value = new byte[length];
            System.arraycopy(req.get_data(), 4, value, 0, length);
            req.skip(4 + length);

            if (type == EAP_TLV_RESULT_STATUS) {
                tlvStatus = WPA_GET_BE16(value);
            }
        }

        if (force_failure) {
            resp_status = EAP_FAIL;
        } else {
            resp_status = tlvStatus;
        }

        return resp_status;
