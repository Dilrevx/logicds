static int x509_crt_check_san( const mbedtls_x509_buf *san, const char *cn, size_t cn_len )
{
    // Only check DNS names (type 2)
    if( san->tag != MBEDTLS_ASN1_UTF8_STRING )
        return -1;

    if( san->len != cn_len || memcmp( san->p, cn, cn_len ) != 0 )
        return -1;

    return 0;
}

if( crt->ext_types & MBEDTLS_X509_EXT_SUBJECT_ALT_NAME )
{
    for( cur = &crt->subject_alt_names; cur != NULL; cur = cur->next )
    {
        if( x509_crt_check_san( &cur->buf, cn, cn_len ) == 0 )
            return 0; // Match found, verification successful
    }
    *flags |= MBEDTLS_X509_BADCERT_CN_MISMATCH;
}
else
{
    for( name = &crt->subject; name != NULL; name = name->next )
    {
        if( MBEDTLS_OID_CMP( MBEDTLS_OID_AT_CN, &name->oid ) == 0 &&
            x509_crt_check_cn( &name->val, cn, cn_len ) == 0 )
        {
            return 0; // Match found, verification successful
        }
    }
    *flags |= MBEDTLS_X509_BADCERT_CN_MISMATCH;
}
