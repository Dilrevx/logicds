#include "mbedtls/x509_crt.h"
#include "mbedtls/asn1.h"

// Function to check SAN for DNS name match
int x509_crt_check_san(mbedtls_asn1_sequence *san_list, const char *cn, size_t cn_len)
{
    for (mbedtls_asn1_sequence *cur = san_list; cur != NULL; cur = cur->next)
    {
        // Check if the current entry is a DNS name (type 2)
        if (cur->tag == MBEDTLS_ASN1_CONTEXT_SPECIFIC | MBEDTLS_ASN1_CONSTRUCTED | 2)
        {
            // Compare the DNS name with the expected CN
            if (x509_crt_check_cn(&cur->buf, cn, cn_len) == 0)
            {
                return 0; // Match found
            }
        }
    }
    return -1; // No match found
}

if (crt->ext_types & MBEDTLS_X509_EXT_SUBJECT_ALT_NAME)
{
    if (x509_crt_check_san(&crt->subject_alt_names, cn, cn_len) == 0)
    {
        return; // Match found, exit early
    }
    *flags |= MBEDTLS_X509_BADCERT_CN_MISMATCH; // No match found in SAN
}
else
{
    for (name = &crt->subject; name != NULL; name = name->next)
    {
        if (MBEDTLS_OID_CMP(MBEDTLS_OID_AT_CN, &name->oid) == 0 &&
            x509_crt_check_cn(&name->val, cn, cn_len) == 0)
        {
            return; // Match found, exit early
        }
    }
    *flags |= MBEDTLS_X509_BADCERT_CN_MISMATCH; // No match found in CN
}
