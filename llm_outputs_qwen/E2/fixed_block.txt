  int current_message_header;
  current_message_header = 99;  // initialization to an invalid value


  if (nas_msg.unpack_outer_hdr(pdu) != SRSRAN_SUCCESS) {
    logger.error("Unable to unpack outer NAS header");
    return SRSRAN_ERROR;
  }

    // patch for SL4
  if(nas_msg.hdr.sequence_number != 255 && nas_msg.hdr.sequence_number < ctxt_base.rx_count - 1) {
    srsran::console("replay detected. \n");
    srsran::console("Sequence number mismatch. Expected %d, got %d", ctxt_base.rx_count++, nas_msg.hdr.sequence_number);
    return SRSRAN_ERROR;
  }

  switch (nas_msg.hdr.security_header_type) {
    case nas_5gs_hdr::security_header_type_opts::plain_5gs_nas_message:
      current_message_header = 0;
      if (has_sec_ctxt) { // patch for SL2, patch for SL3
        logger.error("Not handle the plain NAS message after security context establishment.");
        return SRSRAN_ERROR;
      }
      break;
    case nas_5gs_hdr::security_header_type_opts::integrity_protected:
      current_message_header = 1;
    // SL4, missing sequence number check
      if (integrity_check(pdu.get()) == false) {
        logger.error("Not handling NAS message with integrity check error");
        return SRSRAN_ERROR;
      }
      break;
    case nas_5gs_hdr::security_header_type_opts::integrity_protected_and_ciphered:
      current_message_header = 2;
    // SL4, missing sequence number check
      if (integrity_check(pdu.get()) == false) {
        logger.error("Not handling NAS message with integrity check error");
        return SRSRAN_ERROR;
      } else {
        cipher_decrypt(pdu.get());
      }
      break;
    case nas_5gs_hdr::security_header_type_opts::integrity_protected_with_new_5G_nas_context:
      current_message_header = 3;
      // SL4, missing sequence number check
      break;
    case nas_5gs_hdr::security_header_type_opts::integrity_protected_and_ciphered_with_new_5G_nas_context:
      current_message_header = 4;
      return SRSRAN_ERROR;
    default:
      logger.error("Not handling NAS message with unkown security header");
