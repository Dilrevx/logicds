  uint8_t current_message_header = 99; // Initialize to an invalid value

  if (nas_msg.unpack_outer_hdr(pdu) != SRSRAN_SUCCESS) {
    logger.error("Unable to unpack outer NAS header");
    return SRSRAN_ERROR;
  }

  current_message_header = nas_msg.hdr.message_type;

  switch (nas_msg.hdr.security_header_type) {
    case nas_5gs_hdr::security_header_type_opts::plain_5gs_nas_message:
      if (has_sec_ctxt) {
        logger.error("Received plain NAS message after security context establishment");
        return SRSRAN_ERROR;
      }
      break;
    case nas_5gs_hdr::security_header_type_opts::integrity_protected:
      if (nas_msg.hdr.sequence_number != ctxt_base.rx_count || nas_msg.hdr.sequence_number < ctxt_base.rx_count - 1) {
        logger.error("Replay detected: sequence number mismatch");
        return SRSRAN_ERROR;
      }
      if (integrity_check(pdu.get()) == false) {
        logger.error("Not handling NAS message with integrity check error");
        return SRSRAN_ERROR;
      }
      break;
    case nas_5gs_hdr::security_header_type_opts::integrity_protected_and_ciphered:
      if (nas_msg.hdr.sequence_number != ctxt_base.rx_count || nas_msg.hdr.sequence_number < ctxt_base.rx_count - 1) {
        logger.error("Replay detected: sequence number mismatch");
        return SRSRAN_ERROR;
      }
      if (integrity_check(pdu.get()) == false) {
        logger.error("Not handling NAS message with integrity check error");
        return SRSRAN_ERROR;
      } else {
        cipher_decrypt(pdu.get());
      }
      break;
    case nas_5gs_hdr::security_header_type_opts::integrity_protected_with_new_5G_nas_context:
      if (nas_msg.hdr.sequence_number != ctxt_base.rx_count || nas_msg.hdr.sequence_number < ctxt_base.rx_count - 1) {
        logger.error("Replay detected: sequence number mismatch");
        return SRSRAN_ERROR;
      }
      break;
    case nas_5gs_hdr::security_header_type_opts::integrity_protected_and_ciphered_with_new_5G_nas_context:
      logger.error("Not handling NAS message with new 5G NAS context");
      return SRSRAN_ERROR;
    default:
      logger.error("Not handling NAS message with unknown security header");
      break;
  }

  // Update the rx_count after successful message processing
  if (current_message_header != 99) {
    ctxt_base.rx_count++;
