int process_mirror_zone_data(dns_validator_t *vctx, dns_name_t *name, dns_rdataset_t *rdataset) {
    dns_keytable_t *secure_roots = vctx->secure_roots;
    dns_keynode_t *keynode = NULL;
    dns_key_t *keyOut = NULL;
    isc_result_t result;

    // Assume some initial setup and loading of keyOut here

    result = dns_keytable_find(secure_roots, name, &keynode);
    if (result == DNS_R_PARTIALMATCH || result == DNS_R_NOTFOUND) {
        result = ISC_R_SUCCESS; // Treat partial matches or not found as success
    }
    if (result != ISC_R_SUCCESS) {
        dns_key_free(&keyOut); // Free keyOut on failure
        return result;
    }

    while (result == ISC_R_SUCCESS && keynode != NULL) {
        if (!dst_key_compare(keyOut, keynode->key)) {
            dns_keytable_detachkeynode(&keynode);
            result = dns_keytable_findnextkeynode(secure_roots, name, &keynode);

    dns_key_free(&keyOut); // Properly free keyOut

    return result;
}
