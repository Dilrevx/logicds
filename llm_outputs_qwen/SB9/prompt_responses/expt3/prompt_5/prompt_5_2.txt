boolean validSigAlgo = false;

if (args.sigAlgo == CommonsConstants.ed25519_sa_algo && ssl.peerEd25519KeyPresent) {
    Wslib.WOLFSSL_MSG("Peer sent ED25519 sig with ED25519 cert");
    validSigAlgo = true;
} else if (args.sigAlgo == CommonsConstants.ed448_sa_algo && ssl.peerEd448KeyPresent) {
    Wslib.WOLFSSL_MSG("Peer sent ED448 sig with ED448 cert");
    validSigAlgo = true;
} else if (args.sigAlgo == CommonsConstants.ecc_dsa_sa_algo && ssl.peerEccDsaKeyPresent) {
    Wslib.WOLFSSL_MSG("Peer sent ECC sig with ECC cert");
    validSigAlgo = true;
} else if ((args.sigAlgo == CommonsConstants.falcon_level1_sa_algo || args.sigAlgo == CommonsConstants.falcon_level5_sa_algo) && ssl.peerFalconKeyPresent) {
    Wslib.WOLFSSL_MSG("Peer sent Falcon sig with Falcon cert");
    validSigAlgo = true;
} else if (args.sigAlgo == CommonsConstants.rsa_pss_sa_algo && ssl.peerRsaKeyPresent) {
    Wslib.WOLFSSL_MSG("Peer sent RSA PSS sig with RSA cert");
    validSigAlgo = true;
} else if (args.sigAlgo == CommonsConstants.rsa_sa_algo) {
    Wslib.WOLFSSL_MSG("Peer sent PKCS#1.5 algo but not in certificate");
    return CommonsConstants.INVALID_PARAMETER;
}

if (!validSigAlgo) {
    Wslib.WOLFSSL_MSG("Peer sent sig with mismatched cert");
    return CommonsConstants.SIG_VERIFY_E;
}
