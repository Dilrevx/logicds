    private AdminManager() {
        Class<?> adminProviderClass = ADMIN_PROVIDER.getValue();
        if (adminProviderClass != null && AdminProvider.class.isAssignableFrom(adminProviderClass)) {
            try {
                provider = (AdminProvider) adminProviderClass.getDeclaredConstructor().newInstance();
            } catch (InstantiationException | IllegalAccessException | NoSuchMethodException | InvocationTargetException e) {
                Log.error("Failed to instantiate AdminProvider: {}", adminProviderClass.getName(), e);
                provider = new DefaultAdminProvider();
            }
        } else {
            Log.error("Invalid AdminProvider class specified: {}", adminProviderClass);
            provider = new DefaultAdminProvider();
        }
        initProvider(provider);
    }
