void recv_msg_userauth_success() {
	/* This function can validly get called multiple times
	if DROPBEAR_CLI_IMMEDIATE_AUTH is set */

	TRACE(("received msg_userauth_success"))
	/* Note: in delayed-zlib mode, setting authdone here 
	 * will enable compression in the transport layer */
	
	/* Ensure that all required authentication methods have been successfully completed */
	if (cli_ses.required_auth_methods == cli_ses.completed_auth_methods) {
		ses.authstate.authdone = 1;
		cli_ses.state = USERAUTH_SUCCESS_RCVD;
		cli_ses.lastauthtype = AUTH_TYPE_NONE;

#if DROPBEAR_CLI_PUBKEY_AUTH
		cli_auth_pubkey_cleanup();
#endif
	} else {
		dropbear_log(LOG_WARNING, "Not all required authentication methods have been completed.");
	}
}
</repair>
