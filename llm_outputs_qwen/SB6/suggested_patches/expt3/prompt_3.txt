package main;

import commons.*;

public class X509Verifier {
    public static void x509_crt_verify_name(MbedtlsX509Crt crt, String cn, int[] flags) {
        MbedtlsX509Name     name;
        MbedtlsX509Sequence cur;
        int                 cn_len = cn.length();
        if ((crt.extTypes & MbedtlsConstants.MBEDTLS_X509_EXT_SUBJECT_ALT_NAME) != 0) {
            for (cur = crt.subjectAltNames; cur != null; cur = cur.next) {

                int ret;
                mbedtls_x509_sequence san_list;
                mbedtls_x509_sequence_init(&san_list);

                ret = mbedtls_x509_crt_get_subject_alt_names(cur, &san_list);
                if (ret == 0) {
                    for (mbedtls_x509_sequence *cur_san = &san_list; cur_san != NULL; cur_san = cur_san->next) {
                        if (cur_san->tag == MBEDTLS_X509_SAN_DNS_NAME &&
                            mbedtls_x509_string_cmp(&cur_san->buf, cn, cn_len) == 0) {
                            break;
                        }
                    }
                    mbedtls_x509_sequence_free(&san_list);
                }

                if (ret != 0 || san_list.next == NULL) { // No SANs found or parsing failed
                    if (CommonUtils.x509_crt_check_cn(cur->raw.p, cn, cn_len) == 0)
                        break;
                }

                    break;

            }
            if (cur == null)
                flags[0] |= MbedtlsConstants.MBEDTLS_X509_BADCERT_CN_MISMATCH;
        } else {
            for (name = crt.subject; name != null; name = name.next) {
                if (CommonUtils.MBEDTLS_OID_CMP(MbedtlsConstants.MBEDTLS_OID_AT_CN, name.oid) == 0
                        && CommonUtils.x509_crt_check_cn(name.val, cn, cn_len) == 0) {
                    break;
                }
            }
            if (name == null)
                flags[0] |= MbedtlsConstants.MBEDTLS_X509_BADCERT_CN_MISMATCH;
        }
    }
}
