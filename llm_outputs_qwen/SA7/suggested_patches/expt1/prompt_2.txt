        while (j >= 0 && j < ctx.chain.size()) {
            x = ctx.chain.get(j);
            if (Helper.cert_self_signed(x)) {
                chain_ss = x;
                break;
            }
            j--;
        }
        if (chain_ss == null && (param.flags & Constants.X509_V_FLAG_PARTIAL_CHAIN) == 0) {
            Helper.raiseError("X509_R_UNABLE_TO_GET_ISSUER_CERT_LOCALLY");
            return -1;
        }
        retry = 1;
        while (retry) {
            retry = 0;
            num = ctx.chain.size();
            for (i = num - 1; i >= 0; i--) {
                x = ctx.chain.get(i);
                if (Helper.cert_self_signed(x)) {
                    if (i != num - 1) {
                        ctx.chain.subList(i + 1, num).clear();
                        retry = 1;
                        break;
                    }
                } else {
                    ok = ctx.check_issued(chain_ss, x);
                    if (ok <= 0) {
                        ctx.chain.subList(i, num).clear();
                        retry = 1;
                        break;
                    }
                }
            }
        }
        if (cb != null) {
            ok = cb.verify(ctx);
            if (ok <= 0) return ok;
        }
        return 1;
    }
