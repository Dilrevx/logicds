package main;

import commons.*;

public class Ssl3GetCertVerify {

        if (pkey == null) {
            SSLerr(SSL_F_SSL3_GET_CERT_VERIFY, SSL_R_NO_PUBLICKEY);
            al = SSL_AD_HANDSHAKE_FAILURE;
            goto err;
        }

        p = s.init_buf.data;
        if (n < 2) {
            SSLerr(SSL_F_SSL3_GET_CERT_VERIFY, SSL_R_LENGTH_TOO_SHORT);
            al = SSL_AD_DECODE_ERROR;
            goto err;
        }
        i = ((p[0] & 0xff) << 8) | (p[1] & 0xff);
        if (i > n - 2) {
            SSLerr(SSL_F_SSL3_GET_CERT_VERIFY, SSL_R_LENGTH_MISMATCH);
            al = SSL_AD_DECODE_ERROR;
            goto err;
        }

        if (type == EVP_PKEY_RSA && i != 0) {
            SSLerr(SSL_F_SSL3_GET_CERT_VERIFY, SSL_R_WRONG_SIGNATURE_TYPE);
            al = SSL_AD_ILLEGAL_PARAMETER;
            goto err;
        }

        if (!Helper.EVP_VerifyInit_ex(mctx, Helper.ssl_md(type), null)) {
            SSLerr(SSL_F_SSL3_GET_CERT_VERIFY, ERR_LIB_EVP);
            al = SSL_AD_INTERNAL_ERROR;
            goto err;
        }

        if (!Helper.EVP_VerifyUpdate(mctx, s.s3.handshake_buffer.data, s.s3.handshake_buffer.length)) {
            SSLerr(SSL_F_SSL3_GET_CERT_VERIFY, ERR_LIB_EVP);
            al = SSL_AD_INTERNAL_ERROR;
            goto err;
        }

        if (!Helper.EVP_VerifyFinal(mctx, p, i, pkey)) {
            SSLerr(SSL_F_SSL3_GET_CERT_VERIFY, SSL_R_BAD_SIGNATURE);
            al = SSL_AD_DECRYPT_ERROR;
            goto err;
        }

        ret = 1;
        goto end;

        err:
        s.state = Constants.SSL_ST_ERR;
        end:
        Helper.EVP_PKEY_free(pkey);
        Helper.EVP_MD_CTX_cleanup(mctx);
        return ret;
    }


    private static byte[] slice(byte[] array, int offset) {
        byte[] result = new byte[array.length - offset];
        System.arraycopy(array, offset, result, 0, result.length);
        return result;
    }
}
