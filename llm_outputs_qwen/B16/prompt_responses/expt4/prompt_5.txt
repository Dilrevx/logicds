static struct frec *lookup_frec(unsigned short id, void *hash, struct sockaddr *addr, int port)
{
  struct frec *f;

  for(f = daemon->frec_list; f; f = f->next)
    if (f->sentto && f->new_id == id && 
        (!hash || memcmp(hash, f->hash, HASH_SIZE) == 0) &&
        (addr == NULL || (memcmp(addr, &f->sentto->sa, sizeof(struct sockaddr)) == 0 && port == f->sentto_port)))
      return f;
      
  return NULL;
}
