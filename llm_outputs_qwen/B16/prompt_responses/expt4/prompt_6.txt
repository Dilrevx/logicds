static struct frec *lookup_frec(unsigned short id, void *hash, int fd)
{
  struct frec *f;

  for(f = daemon->frec_list; f; f = f->next)
    if (f->sentto && f->new_id == id && 
        (!hash || memcmp(hash, f->hash, HASH_SIZE) == 0)) {
      if ((f->sentto->sa.sa_family == AF_INET && f->rfd4 && f->rfd4->fd == fd) ||
          (f->sentto->sa.sa_family == AF_INET6 && f->rfd6 && f->rfd6->fd == fd) ||
          (f->sentto->sfd && f->sentto->sfd->fd == fd)) {
        return f;
      }
    }
      
  return NULL;
}
