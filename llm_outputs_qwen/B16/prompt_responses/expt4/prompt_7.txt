static struct frec *lookup_frec(unsigned short id, void *hash, struct sockaddr_storage *addr, int port)
{
  struct frec *f;

  for(f = daemon->frec_list; f; f = f->next)
    if (f->sentto && f->new_id == id && 
        (!hash || memcmp(hash, f->hash, HASH_SIZE) == 0) &&
        (memcmp(&f->sentto->ss, addr, sizeof(struct sockaddr_storage)) == 0) &&
        (f->sentto->port == port))
      return f;
      
  return NULL;
}
