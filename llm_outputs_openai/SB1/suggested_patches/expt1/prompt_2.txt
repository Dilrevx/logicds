package main;

import commons.*;

public class Ssl3GetCertVerify {

        // Ensure that a certificate was provided and a public key is available.
        if (peer == null || pkey == null) {
            Helper.EVP_MD_CTX_cleanup(mctx);
            // Set appropriate alert to indicate handshake failure.
            s.alert = Constants.SSL_AD_HANDSHAKE_FAILURE;
            return -1;
        }
        
        /* 
         * Continue processing the CertificateVerify message.
         * At this point, the message contents (retrieved via ssl_get_message)
         * should include a signature that must be verified using the peer's public key.
         */

        // For illustration, assume the message buffer and its length have been set somehow.
        // Typically, the certificate verify message contains:
        //   - A signature algorithm (if version >= TLS 1.2)
        //   - A signature length field
        //   - The actual signature bytes
        // The following code uses helper methods to extract these parts.
        //
        // NOTE: In real implementations, message parsing must strictly validate all
        // boundaries to avoid buffer overruns or logic flaws.

        // Extract signature length and signature data from the message.
        // (The extraction methods below need to verify that the message data is as expected.)
        int signatureLength = Helper.extractSignatureLength(s.message);
        byte[] signature = Helper.extractSignature(s.message, signatureLength);
        if (signature == null || signature.length != signatureLength) {
            Helper.EVP_MD_CTX_cleanup(mctx);
            s.alert = Constants.SSL_AD_DECODE_ERROR;
            return -1;
        }

        // Compute the expected digest that should have been signed.
        // Often this digest is computed over a transcript of previous handshake messages.
        byte[] expectedDigest = Helper.computeCertVerifyDigest(s);
        if (expectedDigest == null) {
            Helper.EVP_MD_CTX_cleanup(mctx);
            s.alert = Constants.SSL_AD_INTERNAL_ERROR;
            return -1;
        }

        // Verify the signature using the public key and the expected digest.
        if (!Helper.verifySignature(pkey, type, expectedDigest, signature)) {
            Helper.EVP_MD_CTX_cleanup(mctx);
            s.alert = Constants.SSL_AD_DECRYPT_ERROR;
            return -1;
        }

        // Cleanup the message digest context.
        Helper.EVP_MD_CTX_cleanup(mctx);

        // If everything succeeded, set ret to success (commonly 1) and return.
        ret = 1;
        return ret;
    }


    private static byte[] slice(byte[] array, int offset) {
        byte[] result = new byte[array.length - offset];
        System.arraycopy(array, offset, result, 0, result.length);
        return result;
    }
}
