    /* Check if the current cipher is RC4-MD5.
       Since RC4-MD5 incorrectly uses the AAD data as the MAC key,
       we must avoid using the AAD parameter for this ciphersuite.
       In this case, we skip looking up the AEAD_TLS1_AAD parameter.
       Otherwise, proceed normally. */
    if (ctx->cipher != NULL && ctx->cipher->nid == NID_rc4_md5) {
        p = NULL;
    } else {
        p = OSSL_PARAM_locate_const(params, OSSL_CIPHER_PARAM.AEAD_TLS1_AAD);
    }
