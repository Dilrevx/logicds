{
    int valid = 0;

    switch (args->sigAlgo) {
    #ifdef HAVE_ED25519
        case ed25519_sa_algo:
            valid = ssl->peerEd25519KeyPresent;
            if (!valid)
                WOLFSSL_MSG("Peer sent ED25519 sig but not ED25519 cert");
            break;
    #endif

    #ifdef HAVE_ED448
        case ed448_sa_algo:
            valid = ssl->peerEd448KeyPresent;
            if (!valid)
                WOLFSSL_MSG("Peer sent ED448 sig but not ED448 cert");
            break;
    #endif

    #ifdef HAVE_ECC
        case ecc_dsa_sa_algo:
            valid = ssl->peerEccDsaKeyPresent;
            if (!valid)
                WOLFSSL_MSG("Peer sent ECC sig but not ECC cert");
            break;
    #endif

    #ifdef HAVE_PQC
        case falcon_level1_sa_algo:
            valid = ssl->peerFalconKeyPresent;
            if (!valid)
                WOLFSSL_MSG("Peer sent Falcon Level 1 sig but different cert");
            break;
        case falcon_level5_sa_algo:
            valid = ssl->peerFalconKeyPresent;
            if (!valid)
                WOLFSSL_MSG("Peer sent Falcon Level 5 sig but different cert");
            break;
    #endif

    #ifndef NO_RSA
        case rsa_sa_algo:
            WOLFSSL_MSG("Peer sent PKCS#1.5 algo but not in certificate");
            ERROR_OUT(INVALID_PARAMETER, exit_dcv);
            break;
        case rsa_pss_sa_algo:
            valid = ((ssl->peerRsaKey != NULL) && ssl->peerRsaKeyPresent);
            if (!valid)
                WOLFSSL_MSG("Peer sent RSA sig but not RSA cert");
            break;
    #endif

        default:
            WOLFSSL_MSG("Peer sent unknown signature algorithm");
            valid = 0;
            break;
    }

    /* In the aggregated check, if the certificate is not consistent with the signature algorithm, abort */
    if (!valid) {
        ret = SIG_VERIFY_E;
        goto exit_dcv;
    }
}
