                goto exit_dcv;
            }
            break;
    #endif

        case falcon_level1_sa_algo:

                ret = SIG_VERIFY_E;
                goto exit_dcv;
            }
            break;
    #endif

    #ifndef NO_RSA
        case rsa_sa_algo:
            /* The PKCS#1.5 algorithm should not be negotiated in TLS v1.3.
             * This block explicitly fails if the client attempts to use it.
             */
            WOLFSSL_MSG("Peer sent PKCS#1.5 RSA sig (unsupported in TLS v1.3)");
            ERROR_OUT(INVALID_PARAMETER, exit_dcv);
            break;
        case rsa_pss_sa_algo:
            if (ssl->peerRsaKey == NULL || !ssl->peerRsaKeyPresent) {
                WOLFSSL_MSG("Peer sent RSA-PSS sig but certificate does not include an RSA key");
                ret = SIG_VERIFY_E;
                goto exit_dcv;
            }
            break;

        default:
            /* Any signature algorithm not explicitly supported here must be rejected.
             * This prevents an attacker from bypassing certificate checking by using a mismatched algorithm.

            WOLFSSL_MSG("Invalid or unsupported signature algorithm");
            ret = SIG_VERIFY_E;
            goto exit_dcv;
