    int validKeyFound = 0;

#ifdef HAVE_ED25519
    if (args->sigAlgo == ed25519_sa_algo) {
        if (!ssl->peerEd25519KeyPresent) {
            WOLFSSL_MSG("Peer sent ED25519 sig but no matching ED25519 cert");
            ret = SIG_VERIFY_E;
            goto exit_dcv;
        }
        validKeyFound = 1;
    }
#endif

#ifdef HAVE_ED448
    if (args->sigAlgo == ed448_sa_algo) {
        if (!ssl->peerEd448KeyPresent) {
            WOLFSSL_MSG("Peer sent ED448 sig but no matching ED448 cert");
            ret = SIG_VERIFY_E;
            goto exit_dcv;
        }
        validKeyFound = 1;
    }
#endif

#ifdef HAVE_ECC
    if (args->sigAlgo == ecc_dsa_sa_algo) {
        if (!ssl->peerEccDsaKeyPresent) {
            WOLFSSL_MSG("Peer sent ECC sig but no matching ECC cert");
            ret = SIG_VERIFY_E;
            goto exit_dcv;
        }
        validKeyFound = 1;
    }
#endif

#ifdef HAVE_PQC
    if (args->sigAlgo == falcon_level1_sa_algo) {
        if (!ssl->peerFalconKeyPresent) {
            WOLFSSL_MSG("Peer sent Falcon Level 1 sig but no matching Falcon cert");
            ret = SIG_VERIFY_E;
            goto exit_dcv;
        }
        validKeyFound = 1;
    }
    if (args->sigAlgo == falcon_level5_sa_algo) {
        if (!ssl->peerFalconKeyPresent) {
            WOLFSSL_MSG("Peer sent Falcon Level 5 sig but no matching Falcon cert");
            ret = SIG_VERIFY_E;
            goto exit_dcv;
        }
        validKeyFound = 1;
    }
#endif

#ifndef NO_RSA
    if (args->sigAlgo == rsa_sa_algo) {
        WOLFSSL_MSG("Peer sent PKCS#1.5 sig algorithm which is not permitted in TLS v1.3");
        ret = SIG_VERIFY_E;
        goto exit_dcv;
    }
    if (args->sigAlgo == rsa_pss_sa_algo) {
        if (ssl->peerRsaKey == NULL || !ssl->peerRsaKeyPresent) {
            WOLFSSL_MSG("Peer sent RSA-PSS sig but no matching RSA cert");
            ret = SIG_VERIFY_E;
            goto exit_dcv;
        }
        validKeyFound = 1;
    }
#endif

    /* If none of the above blocks recognized the sigAlgo,
       then the certificate does not match the expected key type. */
    if (!validKeyFound) {
        WOLFSSL_MSG("Peer signature algorithm does not match any certificate key type");
        ret = SIG_VERIFY_E;
        goto exit_dcv;
    }
