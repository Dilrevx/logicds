        }
    }

        if (!valid) {
            WOLFSSL_MSG("Peer sent ECC sig but no matching ECC cert");
        }
    }
#endif

    if (args->sigAlgo == falcon_level1_sa_algo) {
        valid = ssl->peerFalconKeyPresent;

        }
    }
    else if (args->sigAlgo == falcon_level5_sa_algo) {
        valid = ssl->peerFalconKeyPresent;
        if (!valid) {
            WOLFSSL_MSG("Peer sent Falcon Level 5 sig but no matching cert");
        }
    }
#endif

    if (args->sigAlgo == rsa_sa_algo) {
        WOLFSSL_MSG("Peer sent PKCS#1.5 algo which is not supported in TLS 1.3");
        ERROR_OUT(INVALID_PARAMETER, exit_dcv);
    }
    if (args->sigAlgo == rsa_pss_sa_algo) {
        valid = (ssl->peerRsaKey != NULL && ssl->peerRsaKeyPresent);
        if (!valid) {
            WOLFSSL_MSG("Peer sent RSA PSS sig but no matching RSA cert");
        }
    }
