    /* Initialize flag to indicate that the certificate matches the signature algorithm */
    int validSigAndCert = 0;

    #ifdef HAVE_ED25519
        if (args->sigAlgo == ed25519_sa_algo) {
            validSigAndCert = ssl->peerEd25519KeyPresent;
            if (!validSigAndCert) {
                WOLFSSL_MSG("Peer sent ED25519 sig but no corresponding ED25519 cert");
            }
        }
    #endif

    #ifdef HAVE_ED448
        if (args->sigAlgo == ed448_sa_algo) {
            validSigAndCert = ssl->peerEd448KeyPresent;
            if (!validSigAndCert) {
                WOLFSSL_MSG("Peer sent ED448 sig but no corresponding ED448 cert");
            }
        }
    #endif

    #ifdef HAVE_ECC
        if (args->sigAlgo == ecc_dsa_sa_algo) {
            validSigAndCert = ssl->peerEccDsaKeyPresent;
            if (!validSigAndCert) {
                WOLFSSL_MSG("Peer sent ECC sig but no corresponding ECC cert");
            }
        }
    #endif

    #ifdef HAVE_PQC
        if (args->sigAlgo == falcon_level1_sa_algo ||
            args->sigAlgo == falcon_level5_sa_algo) {
            validSigAndCert = ssl->peerFalconKeyPresent;
            if (!validSigAndCert) {
                WOLFSSL_MSG("Peer sent Falcon sig but no corresponding Falcon cert");
            }
        }
    #endif

    #ifndef NO_RSA
        if (args->sigAlgo == rsa_sa_algo) {
            /* PKCS#1.5 signature method is disallowed in TLS 1.3 */
            WOLFSSL_MSG("Peer sent PKCS#1.5 RSA sig algorithm which is not allowed in TLS 1.3");
            ERROR_OUT(INVALID_PARAMETER, exit_dcv);
        }
        if (args->sigAlgo == rsa_pss_sa_algo) {
            validSigAndCert = (ssl->peerRsaKey != NULL && ssl->peerRsaKeyPresent);
            if (!validSigAndCert) {
                WOLFSSL_MSG("Peer sent RSA-PSS sig but no corresponding RSA cert");
            }
        }
    #endif

    /* Final check: if the signature algorithm does not match the provided certificate, reject */
    if (!validSigAndCert) {
        ret = SIG_VERIFY_E;
        goto exit_dcv;
    }
