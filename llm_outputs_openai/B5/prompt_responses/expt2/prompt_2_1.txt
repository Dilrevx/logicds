	result = _gnutls_x509_get_raw_field2(cert->cert, &cert->der,
	                                      "tbsCertificate.signatureAlgorithm.algorithm",
	                                      &cert->tbs_sig_algo);

		goto cleanup;

	/* Validate that the signature algorithms in the outer and TBS parts match.

	if ((cert->outer_sig_algo.size != cert->tbs_sig_algo.size) ||
