		result = _gnutls_x509_get_raw_field2(cert->cert, &cert->der,
						      "tbsCertificate.signature.algorithm",
						      &tbs_sig_alg);

			gnutls_assert();
			goto cleanup;
		}

		/* Compare both signature algorithm fields for exact match */
		if (outer_sig_alg.size != tbs_sig_alg.size ||
