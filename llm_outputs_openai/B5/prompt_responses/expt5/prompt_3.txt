	{
		asn1_node tbs_sig_alg = NULL;
		asn1_node cert_sig_alg = NULL;
		char *tbs_oid = NULL, *cert_oid = NULL;

		/* Retrieve the signatureAlgorithm from within the tbsCertificate */
		result = _gnutls_x509_get_asn1_node(cert->cert,
						       "tbsCertificate.signatureAlgorithm",
						       &tbs_sig_alg);
		if (result < 0) {
			gnutls_assert();
			goto cleanup;
		}

		/* Retrieve the certificate's outer signatureAlgorithm */
		result = _gnutls_x509_get_asn1_node(cert->cert,
						       "signatureAlgorithm",
						       &cert_sig_alg);
		if (result < 0) {
			gnutls_assert();
			goto cleanup;
		}

		/* Extract the OID from each AlgorithmIdentifier */
		result = _gnutls_x509_get_oid(tbs_sig_alg, &tbs_oid);
		if (result < 0) {
			gnutls_assert();
			goto cleanup;
		}

		result = _gnutls_x509_get_oid(cert_sig_alg, &cert_oid);
		if (result < 0) {
			gnutls_free(tbs_oid);
			gnutls_assert();
			goto cleanup;
		}

		/* Compare the OIDs; if they differ, the certificate is invalid */
		if (strcmp(tbs_oid, cert_oid) != 0) {
			gnutls_assert();
			result = GNUTLS_E_INVALID_REQUEST;  /* or an appropriate error code */
			gnutls_free(tbs_oid);
			gnutls_free(cert_oid);
			goto cleanup;
		}

		gnutls_free(tbs_oid);
		gnutls_free(cert_oid);
	}

		gnutls_assert();
