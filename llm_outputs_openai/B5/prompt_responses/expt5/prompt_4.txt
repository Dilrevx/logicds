		gnutls_datum_t overall_sig_alg = { NULL, 0 };
		gnutls_datum_t tbs_sig_alg = { NULL, 0 };

		result = asn1_der_decoding(&cert->cert, cert->der.data,
		                           cert->der.size, NULL);
		if (result != ASN1_SUCCESS) {
			result = _gnutls_asn2err(result);
			gnutls_assert();
			goto cleanup;
		}

		/* Extract the overall certificate signature algorithm */
		result = _gnutls_x509_get_raw_field2(cert->cert, &cert->der,
		                                      "certificate.signatureAlgorithm",
		                                      &overall_sig_alg);
		if (result < 0) {
			gnutls_assert();
			goto cleanup;
		}

		/* Extract the TBS (to-be-signed) signature algorithm */
		result = _gnutls_x509_get_raw_field2(cert->cert, &cert->der,
		                                      "tbsCertificate.signature",
		                                      &tbs_sig_alg);
		if (result < 0) {
			gnutls_assert();
			goto cleanup;
		}

		/* Verify that both signature algorithms match */
		if (!_gnutls_datum_equal(&overall_sig_alg, &tbs_sig_alg)) {
			gnutls_assert_log("Signature algorithm mismatch: "
			                    "overall: (size=%zu), tbs: (size=%zu)",
			                    overall_sig_alg.size, tbs_sig_alg.size);
			result = GNUTLS_E_CERTIFICATE_ERROR;
			goto cleanup;
		}

		/* Continue with processing the issuer field if signature algorithms match */
		result = _gnutls_x509_get_raw_field2(cert->cert, &cert->der,
		                                      "tbsCertificate.issuer.rdnSequence",
		                                      &cert->raw_issuer_dn);
		if (result < 0) {
			gnutls_assert();
			goto cleanup;
		}
	}
