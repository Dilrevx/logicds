		result = _gnutls_x509_get_raw_field2(cert->cert, &cert->der,

		if (result < 0) {

		/* Extract the TBS (to-be-signed) signature algorithm field */
		result = _gnutls_x509_get_raw_field2(cert->cert, &cert->der,
						      "tbsCertificate.signature.algorithm",
						      &sig_alg_tbs);
		if (result < 0) {
			gnutls_assert();
			goto cleanup;
		}

		/* Compare the two signature algorithms */
		if (sig_alg_overall.size != sig_alg_tbs.size ||
		    memcmp(sig_alg_overall.data, sig_alg_tbs.data, sig_alg_overall.size) != 0) {

			            (int)sig_alg_tbs.size, sig_alg_tbs.data);
