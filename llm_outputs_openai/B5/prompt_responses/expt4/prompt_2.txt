	result = _gnutls_x509_get_raw_field2(cert->cert, &cert->der,
					      "signatureAlgorithm.algorithm",
					      &overall_sig_alg);
	if (result < 0) {
		gnutls_assert();
		goto cleanup;
	}

	/* Extract TBS signature algorithm from the certificate */
	unsigned char *tbs_sig_alg = NULL;
	result = _gnutls_x509_get_raw_field2(cert->cert, &cert->der,
					      "tbsCertificate.signature.algorithm",
					      &tbs_sig_alg);
	if (result < 0) {
		gnutls_assert();
		goto cleanup;
	}

	/* Compare the two signature algorithms.
	   The lengths of the extracted fields are assumed to be encoded within the ASN.1
	   structure; here we do a simple binary comparison.
	   If they differ, log an error and return a certificate error code. */
	if (memcmp(overall_sig_alg, tbs_sig_alg, result) != 0) {
		gnutls_debug_log("Signature algorithm mismatch: overall signature and TBS signature differ\n");
		gnutls_assert();
		result = GNUTLS_E_CERTIFICATE_ERROR;
		goto cleanup;
	}

					      &cert->raw_issuer_dn);
