	result = _gnutls_x509_get_raw_field2(cert->cert, &cert->der,
					      "signatureAlgorithm",
					      &cert->raw_sig_algo);
	if (result < 0) {
		gnutls_assert();
		goto cleanup;
	}

	/* Extract the TBS (to-be-signed) signature algorithm of the certificate */
	result = _gnutls_x509_get_raw_field2(cert->cert, &cert->der,
					      "tbsCertificate.signatureAlgorithm",
					      &cert->raw_tbs_sig_algo);
	if (result < 0) {
		gnutls_assert();
		goto cleanup;
	}

	/* Compare the two signature algorithms to ensure they match */
	if ((cert->raw_sig_algo.size != cert->raw_tbs_sig_algo.size) ||
	    (memcmp(cert->raw_sig_algo.data, cert->raw_tbs_sig_algo.data,
	            cert->raw_sig_algo.size) != 0)) {
		gnutls_debug_log("Signature algorithm mismatch: overall signature algorithm "
				  "does not match TBS signature algorithm\n");
		gnutls_assert();

		goto cleanup;
