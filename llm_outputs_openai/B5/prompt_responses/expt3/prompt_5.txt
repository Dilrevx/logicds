/* First, decode the certificate as before */
result = asn1_der_decoding(&cert->cert, cert->der.data, cert->der.size, NULL);
if (result != ASN1_SUCCESS) {
	result = _gnutls_asn2err(result);
	gnutls_assert();
	goto cleanup;
}

/* Extract the overall signature algorithm */
gnutls_datum_t raw_overall_sigalg;
result = _gnutls_x509_get_raw_field2(cert->cert, &cert->der,
	"signatureAlgorithm.algorithm", &raw_overall_sigalg);
if (result < 0) {
	gnutls_assert();
	goto cleanup;
}

/* Extract the TBS (to be signed) signature algorithm */
gnutls_datum_t raw_tbs_sigalg;
result = _gnutls_x509_get_raw_field2(cert->cert, &cert->der,
	"tbsCertificate.signatureAlgorithm.algorithm", &raw_tbs_sigalg);
if (result < 0) {
	gnutls_assert();
	goto cleanup;
}

/* Compare the two extracted signature algorithm fields */
if ((raw_overall_sigalg.size != raw_tbs_sigalg.size) ||
    (memcmp(raw_overall_sigalg.data, raw_tbs_sigalg.data, raw_overall_sigalg.size) != 0))
{
	/* Mismatch found between signature algorithms */
	gnutls_assert();
	result = GNUTLS_E_CERTIFICATE_ERROR; /* Use an appropriate error code */
	goto cleanup;
}

/* Proceed with further processing only if the signature algorithms match */
/* Extract the issuer field */
result = _gnutls_x509_get_raw_field2(cert->cert, &cert->der,
	"tbsCertificate.issuer.rdnSequence", &cert->raw_issuer_dn);
if (result < 0) {
	gnutls_assert();
	goto cleanup;
}
