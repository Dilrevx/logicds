	/* Decode the certificate in DER format */
	result = asn1_der_decoding(&cert->cert, cert->der.data, cert->der.size, NULL);
	if (result != ASN1_SUCCESS) {
		result = _gnutls_asn2err(result);
		gnutls_assert();
		goto cleanup;
	}

	/* Extract the outer signature algorithm */
	result = _gnutls_x509_get_raw_field2(cert->cert, &cert->der,
			"signatureAlgorithm.algorithm",
			&cert->raw_signature_alg);
	if (result < 0) {
		gnutls_assert();
		goto cleanup;
	}

	/* Extract the signature algorithm from the TBS (To Be Signed) part */
	result = _gnutls_x509_get_raw_field2(cert->cert, &cert->der,
			"tbsCertificate.signature.algorithm",
			&cert->raw_tbs_signature_alg);
	if (result < 0) {
		gnutls_assert();
		goto cleanup;
	}

	/* Compare the two signature algorithms to ensure they match */
	if (!_gnutls_oid_equal(cert->raw_signature_alg, cert->raw_tbs_signature_alg)) {
		result = GNUTLS_E_INVALID_CERTIFICATE;
		gnutls_assert();
		goto cleanup;
	}

	/* Continue processing by extracting issuer DN */
	result = _gnutls_x509_get_raw_field2(cert->cert, &cert->der,
			"tbsCertificate.issuer.rdnSequence",
			&cert->raw_issuer_dn);
	if (result < 0) {
		gnutls_assert();
		goto cleanup;
