Below is one example of a repair. In the fixed code the certificate’s outer signature algorithm (the one carried in the Certificate structure) is compared with the “to‐be‑signed” signature algorithm extracted from the TBSCertificate portion. If the two do not match then the certificate is rejected. (Depending on your full code base you may already have helper routines to extract or compare algorithm OIDs. In that case, substitute those functions as needed.) 

────────────────────────────
/* After decoding the ASN.1 structure, add a new check that validates the
 * signature algorithm in the Certificate header against the one specified in
 * the TBSCertificate part.
 */
 
result = asn1_der_decoding(&cert->cert, cert->der.data, cert->der.size, NULL);
if (result != ASN1_SUCCESS) {
	result = _gnutls_asn2err(result);
	gnutls_assert();
	goto cleanup;
}

/* Validate that the signature algorithms match.
 * Note: In this example we assume that:
 *   - cert->cert.signature_algorithm_oid points to the outer signature algorithm.
 *   - cert->cert.tbs_signature_algorithm_oid points to the signature algorithm
 *     from the TBSCertificate block.
 * The helper function gnutls_oid_equal compares these two OIDs.
 */
if (!gnutls_oid_equal(cert->cert.signature_algorithm_oid,
                      cert->cert.tbs_signature_algorithm_oid)) {
	result = GNUTLS_E_INVALID_REQUEST;  /* or an appropriate error code */
	gnutls_assert();
	goto cleanup;
}

result = _gnutls_x509_get_raw_field2(cert->cert, &cert->der,
				     "tbsCertificate.issuer.rdnSequence",
				     &cert->raw_issuer_dn);
if (result < 0) {
	gnutls_assert();
	goto cleanup;
}

static int validate_signature_algorithms(const your_cert_type *cert)

	if (!gnutls_oid_equal(cert->signature_algorithm_oid,
