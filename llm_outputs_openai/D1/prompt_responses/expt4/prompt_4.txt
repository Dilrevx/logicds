        /* Preserve the current SRTP policies prior to stopping the transport
         * in order to retain all necessary crypto state, including roll-over counters.
         */
        pjmedia_srtp_crypto tx = srtp->tx_policy;
        pjmedia_srtp_crypto rx = srtp->rx_policy;
        pj_status_t status;

        /* Stop the SRTP transport. Note: Stopping the transport may alter
         * internal state, hence we have already stored the required policies.
         */
        pjmedia_transport_srtp_stop((pjmedia_transport*)srtp);

        /* Restart the SRTP transport with the preserved crypto policies.
         * This ensures that the correct state is restored, preventing a fallback
         * to basic RTP.
         */
        status = pjmedia_transport_srtp_start((pjmedia_transport*)srtp,
                                              &tx, &rx);
        if (status != PJ_SUCCESS) {
            PJ_LOG(5, (srtp->pool->obj_name, "Failed to restart SRTP, err=%s",
                       get_libsrtp_errstr(err)));
        } else if (!srtp->bypass_srtp) {
            err = srtp_unprotect(srtp->srtp_rx_ctx, (pj_uint8_t*)pkt, &len);
        }
