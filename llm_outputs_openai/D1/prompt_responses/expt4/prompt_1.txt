         * Restarting SRTP is attempted to regain correct sequence handling.
         *
         * IMPORTANT SECURITY FIX:
         * Do not allow a fallback to basic RTP (insecure) by skipping SRTP unprotect.
         * Regardless of any bypass flag settings, always attempt to process the packet
         * through SRTP unprotect after a successful restart. If the restart fails,
         * the packet must be dropped rather than processed as basic RTP.
         */
        pjmedia_srtp_crypto tx, rx;
        pj_status_t status;

        /* Stop SRTP first, so that pjmedia_transport_srtp_start()
         * reinitializes all context (including the rollover counter).
         */
        pjmedia_transport_srtp_stop((pjmedia_transport*)srtp);

        tx = srtp->tx_policy;
        rx = srtp->rx_policy;
        status = pjmedia_transport_srtp_start((pjmedia_transport*)srtp,
                                              &tx, &rx);
        if (status != PJ_SUCCESS) {
            PJ_LOG(5, (srtp->pool->obj_name, "Failed to restart SRTP, err=%s",
                       get_libsrtp_errstr(err)));

            return PJ_EINVALIDOP;
