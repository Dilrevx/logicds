         * The packet that triggered this error is a replay (or appears so).
         * Restarting the SRTP session may be needed to re‐sync the rollover/sequence
         * counters. However, to prevent an attacker from repeatedly forcing a restart
         * (and possibly bypassing replay protection) we now drop the triggering packet
         * rather than trying to process it again.
         */

        /* Stop SRTP first, this clears the current state including the roll-over counter */
        pjmedia_transport_srtp_stop((pjmedia_transport*)srtp);

        tx = srtp->tx_policy;
        rx = srtp->rx_policy;
        status = pjmedia_transport_srtp_start((pjmedia_transport*)srtp,
                                              &tx, &rx);
        if (status != PJ_SUCCESS) {
            PJ_LOG(5, (srtp->pool->obj_name, "Failed to restart SRTP, err=%s",
                       get_libsrtp_errstr(err)));
        }
        
        /* Do not attempt to unprotect the packet that originally triggered
         * the error.  Dropping it prevents an attacker from using replayed
         * packets to force repeated restarts and potential DoS.
         */
        return err;
