package main;

import commons.*;

public class NmHandleDetach {
    public static void isc__nmhandle_detach(HandleRef handlep) {
        Require.check(handlep != null);
        Require.check(HandleRef.valid(handlep));
        NmHandle handle = handlep.handle;
        handlep.handle = null;
        NmSocket sock = handle.sock;

        if (sock.tid == NodeManager.nmTid()) {
            // Detach callback in the main thread.
            NodeManager.nmhandle_detach_cb(handle);
            // Immediately close the socket to force termination
            // and prevent it from lingering in CLOSE_WAIT.
            if (sock.isOpen()) {
                sock.shutdown();  // Shut down both read and write channels.
                sock.close();
            }
        } else {
            // For non‐main thread, enqueue an event to detach.
            NetiEventDetach event = NodeManager.getNetiEventDetach(sock.mgr, sock);
            event.handle = handle;
            EventPasser.passEvent(event);
            NodeManager.enqueueEvent(sock.mgr.workers[sock.tid], event);
            // Schedule an immediate socket closure to avoid lingering CLOSE_WAIT.
            if (sock.isOpen()) {
                sock.shutdown();
                sock.close();
            }
        }

    }
}
