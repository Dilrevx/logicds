package main;

import commons.*;

public class NmHandleDetach {
    public static void isc__nmhandle_detach(HandleRef handlep) {
        Require.check(handlep != null);
        Require.check(HandleRef.valid(handlep));
        NmHandle handle = handlep.handle;
        handlep.handle = null;
        NmSocket sock = handle.sock;

        // Check if the current thread is the designated one and ensure that no detach callback is already in progress.
        if (sock.tid == NodeManager.nmTid() && !sock.isDetachPending()) {
            // Mark that a detach callback is pending to avoid duplicate operations.
            sock.setDetachPending(true);
            NodeManager.nmhandle_detach_cb(handle);
        } else {
            // Either the current thread is not designated,
            // or a detach is already in progress – enqueue the event for asynchronous handling.
            NetiEventDetach event = NodeManager.getNetiEventDetach(sock.mgr, sock);
            event.handle = handle;
            EventPasser.passEvent(event);
            NodeManager.enqueueEvent(sock.mgr.workers[sock.tid], event);
        }

    }
}
