package main;

import commons.*;

public class NmHandleDetach {

    public static void isc__nmhandle_detach(HandleRef handlep) {
        NmSocket sock = null;
        NmHandle handle = null;
        Require.check(handlep != null);
        Require.check(HandleRef.valid(handlep));
        handle = handlep.handle;
        handlep.handle = null;
        sock = handle.sock;

        // FIX: Immediately close the underlying socket to ensure that any TCP connection
        // is fully terminated rather than lingering in CLOSE_WAIT due to incomplete shutdown.
        try {
            if (sock != null) {
                sock.close();
            }
        } catch (Exception e) {
            // Log the exception as needed, but continue with detachment
            System.err.println("Socket close failed: " + e.getMessage());
        }

        if (sock.tid == NodeManager.nmTid()) {
            NodeManager.nmhandle_detach_cb(handle);
        } else {
            NetiEventDetach event = NodeManager.getNetiEventDetach(sock.mgr, sock);
            event.handle = handle;
            EventPasser.passEvent(event);
            NodeManager.enqueueEvent(
                    sock.mgr.workers[sock.tid], event);
        }
    }

}
