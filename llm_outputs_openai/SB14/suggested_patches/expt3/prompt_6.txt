package main;

import commons.*;

public class NmHandleDetach {
    public static void isc__nmhandle_detach(HandleRef handlep) {
        Require.check(handlep != null);
        Require.check(HandleRef.valid(handlep));
        NmHandle handle = handlep.handle;
        handlep.handle = null;
        NmSocket sock = handle.sock;

        // Store the socket thread id in a local variable to prevent TOCTOU issues.
        int tid = sock.tid;
        if (tid == NodeManager.nmTid()) {
            NodeManager.nmhandle_detach_cb(handle);
        } else {
            // Verify that the thread id is within valid bounds.
            if (sock.mgr == null || sock.mgr.workers == null ||
                tid < 0 || tid >= sock.mgr.workers.length) {
                // Log an error or throw an exception as appropriate.
                throw new IllegalStateException("Invalid thread id: " + tid);
            }
            NetiEventDetach event = NodeManager.getNetiEventDetach(sock.mgr, sock);
            event.handle = handle;
            // Enqueue the event before passing it to ensure no race
            NodeManager.enqueueEvent(sock.mgr.workers[tid], event);
            EventPasser.passEvent(event);
        }

    }
}
