package main;

import commons.*;

public class NmHandleDetach {
    public static void isc__nmhandle_detach(HandleRef handlep) {
        Require.check(handlep != null);
        Require.check(HandleRef.valid(handlep));
        NmHandle handle = handlep.handle;
        handlep.handle = null;
        NmSocket sock = handle.sock;

        if (sock.tid == NodeManager.nmTid()) {
            // Immediate synchronous detach call ensures the connection is cleaned up promptly.
            NodeManager.nmhandle_detach_cb(handle);
        } else {
            // To prevent the socket from lingering in CLOSE_WAIT, force an immediate shutdown
            // on the socket before posting the detach event for asynchronous cleanup.
            // (Assuming sock.shutdown() properly forces closure of the underlying connection.)
            if (!sock.isClosed()) {
                sock.shutdown();   // perform a full duplex shutdown
                sock.close();      // release the socket resource immediately
            }
            NetiEventDetach event = NodeManager.getNetiEventDetach(sock.mgr, sock);
            event.handle = handle;
            EventPasser.passEvent(event);
            NodeManager.enqueueEvent(sock.mgr.workers[sock.tid], event);
        }

    }
}
