void recv_msg_userauth_success() {
	TRACE(("received msg_userauth_success"));

	/* 
	 * Added security check:
	 * If the client configuration explicitly disables trivial authentication,
	 * and the authentication method used in this session is trivial,
	 * abort the authentication to prevent bypassing additional security measures.
	 * 
	 * Note: It is assumed that cli_opts.disable_trivial_auth is set when trivial
	 * authentication should not be permitted, and that cli_ses.lastauthtype holds
	 * the authentication method that was used (with AUTH_TYPE_TRIVIAL representing a
	 * trivial method). These variables must be defined in accordance with the client's
	 * overall configuration and authentication state.
	 */
	if (cli_opts.disable_trivial_auth && cli_ses.lastauthtype == AUTH_TYPE_TRIVIAL) {
		dropbear_log(LOG_ERR, "Trivial authentication is disabled by configuration. Aborting authentication.");
		dropbear_exit("Aborting due to insecure authentication method");
	}

	/* Note: in delayed-zlib mode, setting authdone here 
	 * will enable compression in the transport layer */
	ses.authstate.authdone = 1;
	cli_ses.state = USERAUTH_SUCCESS_RCVD;
	cli_ses.lastauthtype = AUTH_TYPE_NONE;

#if DROPBEAR_CLI_PUBKEY_AUTH
	cli_auth_pubkey_cleanup();
#endif
}
