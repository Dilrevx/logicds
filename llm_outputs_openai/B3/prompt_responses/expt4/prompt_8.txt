void recv_msg_userauth_success() {
	TRACE(("received msg_userauth_success"));

	/* 
	 * Additional security check: prevent the acceptance of a trivial authentication 
	 * if such methods have been explicitly disallowed via client configuration.
	 * The check below verifies that if trivial authentication was used (indicated by 
	 * cli_ses.lastauthtype being set to AUTH_TYPE_TRIVIAL) and the client option does not
	 * allow trivial methods (cli_opts.allow_trivial_auth is false), then the session is 
	 * immediately terminated.
	 *
	 * Note: The exact names of the configuration variable (cli_opts.allow_trivial_auth)
	 * and the trivial authentication type (AUTH_TYPE_TRIVIAL) might differ in your codebase.
	 * Ensure they match your actual definitions.
	 */
	if (!cli_opts.allow_trivial_auth && cli_ses.lastauthtype == AUTH_TYPE_TRIVIAL) {
		dropbear_log(LOG_WARN, "Trivial authentication method is disallowed, terminating session");
		dropbear_exit("Authentication failed: insecure authentication method used");
	}

	/* In delayed-zlib mode, setting authdone here will enable compression in the transport layer */
	ses.authstate.authdone = 1;
	cli_ses.state = USERAUTH_SUCCESS_RCVD;
	cli_ses.lastauthtype = AUTH_TYPE_NONE;

#if DROPBEAR_CLI_PUBKEY_AUTH
	cli_auth_pubkey_cleanup();
#endif
}
