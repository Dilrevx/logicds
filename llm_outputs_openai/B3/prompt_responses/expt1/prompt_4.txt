void recv_msg_userauth_success() {
    TRACE(("received msg_userauth_success"));

    /* Prevent processing a duplicate success message.
     * In delayed-zlib mode, reprocessing could inappropriately re-enable
     * compression or lead to unpredictable state changes.
     */
    if (ses.authstate.authdone) {
        /* Already authenticated – ignore duplicate message */
        return;
    }

    /* Mark authentication as complete only once. This ensures that
     * any state changes (e.g. enabling compression) happen only once.
     */
    ses.authstate.authdone = 1;
    cli_ses.state = USERAUTH_SUCCESS_RCVD;
    cli_ses.lastauthtype = AUTH_TYPE_NONE;
#if DROPBEAR_CLI_PUBKEY_AUTH
    /* Additional public key auth handling (if any) can follow here */
#endif
}
