    STACK_OF(X509_CRL) *sub_crls = X509_STORE_CTX_get1_crls(store_ctx, subject);
    STACK_OF(X509_CRL) *iss_crls = X509_STORE_CTX_get1_crls(store_ctx, issuer);
    int i;

    if (sub_crls == NULL)
        return 0;

    if (iss_crls != NULL) {
        for (i = 0; i < sk_X509_CRL_num(iss_crls); i++) {
            X509_CRL *crl = sk_X509_CRL_value(iss_crls, i);
            /* Increase reference count as it will be in two stacks */
            CRYPTO_add(&crl->references, 1, CRYPTO_LOCK_X509_CRL);
            if (!sk_X509_CRL_push(sub_crls, crl)) {
                sk_X509_CRL_pop_free(iss_crls, X509_CRL_free);
                return 0;
            }
        }
        sk_X509_CRL_pop_free(iss_crls, X509_CRL_free);
    }
    crls = sub_crls;
}
