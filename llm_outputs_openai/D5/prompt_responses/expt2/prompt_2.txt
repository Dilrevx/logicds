private AdminManager() {
    // Load the admin provider.
    initProvider(ADMIN_PROVIDER.getValue());
    
    // Register a listener for user deletion events.
    // This ensures that if an admin account is deleted, any residual admin privilege
    // associated with the username is immediately revoked.
    UserEventManager.registerListener(new UserEventListener() {
        @Override
        public void onUserDeleted(UserEvent event) {
            String deletedUser = event.getUserName();
            // Explicitly revoke any administrative privileges associated with the deleted username.
            // This prevents a recreated account with the same name from inheriting old privileges.
            AdminPrivileges.revokePrivileges(deletedUser);
        }
        
        @Override
        public void onUserCreated(UserEvent event) {
            // No special handling needed on user creation.
        }
        
        // Other event methods can be added here if required.
    });
}
