private AdminManager() {
    // Load an admin provider securely.
    // Retrieve the provider value from configuration and validate it against a whitelist
    String provider = ADMIN_PROVIDER.getValue();
    if (!isValidAdminProvider(provider)) {
        throw new SecurityException("Invalid admin provider configuration");
    }
    initProvider(provider);
}

/**
 * Validates that the provided admin provider is one of the allowed values.
 * This prevents remote attackers from injecting arbitrary provider names via system properties.
 */
private boolean isValidAdminProvider(String provider) {
    // Define a whitelist of allowed provider names. Adjust as necessary.
    return "defaultProvider".equals(provider) || "trustedAdminProvider".equals(provider);
}
