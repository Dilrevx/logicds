private AdminManager() {
    // Load an admin provider.
    initProvider(ADMIN_PROVIDER.getValue());
    
    // Register a listener for user deletion events.
    // When a user is deleted, remove any associated administrative privileges.
    UserEventManager.registerUserDeletionListener(new UserDeletionListener() {
        @Override
        public void onUserDeleted(User user) {
            // Prevent the re-creation of a user with automatically inherited admin privileges.
            revokeAdminPrivileges(user.getUsername());
        }
    });
}

/**
 * Revokes admin privileges for the specified user.
 * This method explicitly removes the username from the collection of authorized admin identifiers,
 * ensuring that a new account with the same user name does not inadvertently inherit admin rights.
 *
 * @param username the username whose admin privileges need to be revoked
 */
private void revokeAdminPrivileges(String username) {
    // Example removal: AdminPrivileges is assumed to manage the admin.authorizedJIDs setting.
    // The actual implementation should ensure that the mapping for this user is removed.
    AdminPrivileges.removeAuthorizedJID(username);
}
