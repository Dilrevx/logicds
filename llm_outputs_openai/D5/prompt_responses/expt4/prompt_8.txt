    private AdminManager() {
        // Load an admin provider.
        initProvider(ADMIN_PROVIDER.getValue());

        // Register a listener to remove any lingering admin privileges when a user is deleted.
        // This prevents a deleted user from later regaining elevated rights by being re-created.
        UserEventDispatcher.addListener(new UserEventListener() {
            @Override
            public void userDeleted(User user, Map<String, Object> params) {
                String username = user.getUsername();
                // Remove the deleted username from the authorized admin JIDs.
                // Note: AuthorizedJIDsManager.removeAuthorizedAdminJID should perform the removal.
                AuthorizedJIDsManager.removeAuthorizedAdminJID(username);
                Log.info("Administrative privileges removed for deleted user: " + username);
            }

            // Provide no-op implementations for other user events.
            @Override
            public void userCreated(User user, Map<String, Object> params) {
                // No action required on user creation.
            }

            @Override
            public void userUpdated(User user, Map<String, Object> params) {
                // No action required on user update.
            }
        });
    }
