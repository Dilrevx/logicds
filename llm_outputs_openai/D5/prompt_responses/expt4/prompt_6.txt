private AdminManager() {
    // Load an admin provider.
    initProvider(ADMIN_PROVIDER.getValue());
    
    // Register a listener to handle user deletion events.
    // This listener ensures that when a user is deleted,
    // any lingering administrative privileges associated with that username are removed.
    UserEventManager.registerListener(new UserEventListener() {
        @Override
        public void onUserDeleted(String username) {
            // Remove any association of admin privileges for the deleted user.
            removeAdminPrivileges(username);
        }
    });
}
