    private AdminManager() {
        // Load an admin provider.
        initProvider(ADMIN_PROVIDER.getValue());
        
        // Register a listener for user deletion events to prevent privilege retention.
        // When a user is deleted, we deliberately remove any admin designation tied to the username.
        UserEventDispatcher.registerUserDeletionListener(new UserDeletionListener() {
            @Override
            public void userDeleted(String username) {
                // Remove the association of admin privileges with the deleted user account.
                AdminPrivilegesManager.revokeAdminPrivileges(username);
                Log.info("Admin privileges removed for deleted user: " + username);
            }
        });
    }
