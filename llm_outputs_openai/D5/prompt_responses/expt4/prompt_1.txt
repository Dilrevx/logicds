private AdminManager() {
    // Retrieve the admin provider from the trusted configuration source.
    String provider = ADMIN_PROVIDER.getValue();
    
    // Validate that the provider is one of the approved providers.
    // This check prevents an attacker from specifying an unauthorized provider via system properties.
    // Replace "com.example.TrustedAdminProvider" with your valid provider identifier if necessary.
    if (!"com.example.TrustedAdminProvider".equals(provider)) {
        throw new SecurityException("Unauthorized admin provider specified.");
    }
    
    initProvider(provider);
}
