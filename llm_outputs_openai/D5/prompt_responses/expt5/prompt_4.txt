    private AdminManager() {
        // Load the administrative provider.
        initProvider(ADMIN_PROVIDER.getValue());

        // Register a listener for user deletion events.
        // This listener ensures that if an administrator account is deleted,
        // any residual administrative privileges associated with that username
        // are explicitly revoked.
        UserEventManager.registerUserDeletionListener(new UserDeletionListener() {
            @Override
            public void onUserDeleted(String username) {
                // Revoke admin rights so that a re-created user will start without elevated privileges.
                revokeAdminPrivileges(username);
            }
        });
    }
</repair>
