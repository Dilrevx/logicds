    private AdminManager() {
        // Load an admin provider.
        initProvider(ADMIN_PROVIDER.getValue());
        
        // Register an event listener to remove lingering admin privileges on user deletion.
        UserEventDispatcher.registerListener(new UserEventListener() {
            @Override
            public void userDeleting(User user) {
                // Remove admin privileges associated with the user's username.
                removeAdminPrivilegesForUser(user.getUsername());
            }

            @Override
            public void userCreated(User user) {
                // No action required on user creation.
            }

            @Override
            public void userModified(User user) {
                // No action required on user modification.
            }
        });
    }
    
    // Helper function to remove administrative privileges from a user.
    private void removeAdminPrivilegesForUser(String username) {
        // Implementation to remove admin.authorizedJIDs or any associated admin roles.
        AdminPrivilegesStore.removeAdminAssociation(username);
    }
