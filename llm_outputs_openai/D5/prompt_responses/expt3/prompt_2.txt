private AdminManager() {
    // Load the admin provider.
    initProvider(ADMIN_PROVIDER.getValue());
    
    // Register a listener for user deletion events.
    // This listener ensures that when a user holding admin privileges is deleted,
    // their associated elevated rights are explicitly revoked.
    UserEventDispatcher.registerListener(new UserDeletionListener() {
        @Override
        public void onUserDeleted(String username) {
            // Remove any lingering admin privileges associated with the deleted user.
            // This prevents a deleted admin account from regaining privileges when recreated.
            removeAdminPrivileges(username);
        }
    });
}
