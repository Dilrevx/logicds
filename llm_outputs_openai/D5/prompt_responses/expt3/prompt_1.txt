private AdminManager() {
    // Load an admin provider from a trusted configuration rather than relying directly on a system property.
    // This prevents an attacker from supplying a malicious provider value via "admin.authorizedJIDs".
    String providerValue = ADMIN_PROVIDER.getValue();
    
    // Validate that the provider value is one of the approved, hard‐coded safe values.
    // For example, only "secureAdminProvider" is allowed.
    if (!"secureAdminProvider".equals(providerValue)) {
        // Optionally log the incident as a warning.
        // log.warn("Unauthorized admin provider value detected. Falling back to secure default.");
        providerValue = "secureAdminProvider";
    }
    
    initProvider(providerValue);
}
