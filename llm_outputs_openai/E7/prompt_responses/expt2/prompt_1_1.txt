  if (sec_hdr_type == LIBLTE_MME_SECURITY_HDR_TYPE_PLAIN_NAS) {
    bool allow_plain = false;
    switch (msg_type) {
      // Only allow messages where plain processing is truly safe.
      // Identity Request is permitted because additional IMSI checking is done in parse_identity_request().
      case LIBLTE_MME_MSG_TYPE_IDENTITY_REQUEST:
      case LIBLTE_MME_MSG_TYPE_AUTHENTICATION_REQUEST:
      case LIBLTE_MME_MSG_TYPE_AUTHENTICATION_REJECT:
        allow_plain = true;
        break;

      // For Attach/Service Reject messages, only allow plaintext if the EMM cause is exactly 25.
      case LIBLTE_MME_MSG_TYPE_ATTACH_REJECT:
      case LIBLTE_MME_MSG_TYPE_SERVICE_REJECT:
        if (emm_cause == 25) {
          allow_plain = true;
        }
        break;

      // Do not allow plain processing for messages that previously were accepted insecurely
      // such as Detach Request, EMM Information, EMM Status, Detach Accept, and Tracking Area Update Reject.
      default:
        break;
    }

    if (!allow_plain) {
      logger.error("Not handling NAS message MSG_TYPE=%02X with SEC_HDR_TYPE=%02X without integrity protection!",
                   msg_type,
                   sec_hdr_type);
      return;
    }
  }
