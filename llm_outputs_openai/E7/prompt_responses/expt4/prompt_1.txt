  if (sec_hdr_type == LIBLTE_MME_SECURITY_HDR_TYPE_PLAIN_NAS) {
    /* Only a limited set of NAS messages are allowed in plaintext.
       In particular, messages that have been identified as vulnerable
       – Detach Request, EMM Information, EMM Status, and Attach/Service Reject
       (or Service Reject) with emm_cause == 25 – must NOT be accepted without integrity protection. */
    switch (msg_type) {
      case LIBLTE_MME_MSG_TYPE_IDENTITY_REQUEST: // Allowed: parsed with special IMSI check
      case LIBLTE_MME_MSG_TYPE_AUTHENTICATION_REQUEST:
      case LIBLTE_MME_MSG_TYPE_AUTHENTICATION_REJECT:
      case LIBLTE_MME_MSG_TYPE_DETACH_ACCEPT:
      case LIBLTE_MME_MSG_TYPE_TRACKING_AREA_UPDATE_REJECT:
        /* These messages are allowed in plaintext. */
        break;
      case LIBLTE_MME_MSG_TYPE_ATTACH_REJECT:
      case LIBLTE_MME_MSG_TYPE_SERVICE_REJECT:
        /* For Attach/Service Reject, reject if the emm_cause is 25. */
        if (emm_cause == 25) {
          logger.error("Rejecting plaintext %s with disallowed emm_cause=%d",
                       (msg_type == LIBLTE_MME_MSG_TYPE_ATTACH_REJECT ? "Attach Reject" : "Service Reject"),
                       emm_cause);
          return;
        }
        break;
      /* Do not allow plaintext Detach Request, EMM Information, or EMM Status. */
      case LIBLTE_MME_MSG_TYPE_DETACH_REQUEST:
      case LIBLTE_MME_MSG_TYPE_EMM_INFORMATION:
      case LIBLTE_MME_MSG_TYPE_EMM_STATUS:
        logger.error("Plaintext NAS message %02X is not allowed without integrity protection.",
                     msg_type);
        return;
      default:
        logger.error("Not handling NAS message MSG_TYPE=%02X with SEC_HDR_TYPE=%02X without integrity protection!",
                     msg_type,
                     sec_hdr_type);
        return;
    }
  }
