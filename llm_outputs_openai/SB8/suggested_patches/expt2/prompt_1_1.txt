package main;

import commons.*;

public class Tls13ServerHello {
    public static int DoTls13ServerHello(TlsContext ssl, byte[] input, int[] inOutIdx, int helloSz, byte[] extMsgType) {
        int ret;
        byte[] suite = new byte[2];
        byte tls12minor = CommonsConstants.TLSv1_2_MINOR;
        if (ssl == null || ssl.arrays == null)
            return CommonsConstants.BAD_FUNC_ARG;
        ret = resetState(ssl);
        int idx = inOutIdx[0];
        if (helloSz < CommonsConstants.OPAQUE16_LEN)
            return CommonsConstants.BUFFER_ERROR;
        idx += CommonsConstants.OPAQUE16_LEN;
        if (idx > helloSz)
            return CommonsConstants.BUFFER_ERROR;

    if (extMsgType[0] == CommonsConstants.server_hello) {
        PreSharedKey psk = null;
        // Look for the PSK extension.
        TlsExtension extPsk = TLSX.find(ssl.extensions, CommonsConstants.TLSX_PRE_SHARED_KEY);
        if (extPsk != null)
            psk = (PreSharedKey) extPsk.data;

        // Also look for the key share extension.
        TlsExtension keyShareExt = TLSX.find(ssl.extensions, CommonsConstants.TLSX_KEY_SHARE);

        /*
         * SECURITY FIX:
         * If neither a valid PSK extension nor a key share extension is present, do not fall
         * back to using a default, predictable IKM. Instead, abort the handshake.
         */
        if (psk == null && keyShareExt == null) {
            /* Abort handshake as both required extensions are missing.
               Replace '-1' with an appropriate error code or TLS alert as needed. */
            return -1;
        }

        int ret = 0;
        while (psk != null && !psk.chosen)
            psk = psk.next;

        if (psk == null) {
            /*
             * No valid PSK was chosen. With the presence of a key share extension, proceed
             * using key share–based key agreement instead. This avoids falling back to a
             * default insecure IKM buffer.
             */
            ret = SetupKseKey(ssl, keyShareExt, 0);
            if (ret != 0)
                return ret;
            ssl.options.keyShareNegotiated = true;
        }
        else {
            ret = SetupPskKey(ssl, psk, 0);
            if (ret != 0)
                return ret;
            ssl.options.pskNegotiated = true;
        }
        ssl.keys.encryptionOn = true;
        ssl.options.serverState = CommonsConstants.SERVER_HELLO_COMPLETE;
    }

        else {
            ssl.options.tls1_3 = true;
            ssl.options.serverState = CommonsConstants.SERVER_HELLO_RETRY_REQUEST_COMPLETE;
        }
        return 0;
    }

    private static int resetState(TlsContext ssl) {
        ssl.options.asyncState = CommonsConstants.TLS_ASYNC_BEGIN;
        return 0;
    }

    private static int SetupPskKey(TlsContext ssl, PreSharedKey psk, int zero) {
        return 0;
    }
}
