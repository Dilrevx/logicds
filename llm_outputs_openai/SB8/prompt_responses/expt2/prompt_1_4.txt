               Instead, abort the handshake with an error to prevent using predictable key material. */
            TlsExtension keyShare = TLSX.find(ssl.extensions, CommonsConstants.TLSX_KEY_SHARE);
            if (psk == null && keyShare == null) {
                /* Neither PSK nor Key Share extension is present.
                   Aborting handshake prevents generation of a master secret from a predictable IKM. */
                return TLS_ERR_MISSING_REQUIRED_EXTENSION;
            }
