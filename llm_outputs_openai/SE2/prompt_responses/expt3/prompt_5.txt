int process_nas_message(PDU &pdu, SecurityContext &security_ctx)
{
    NasMessage nas_msg;

    // Extract the incoming sequence number from the NAS message header.
    // (Assuming nas_msg.header.sequence_num is set appropriately before unpacking.)
    uint32_t incoming_seq = nas_msg.header.sequence_num;
    if (incoming_seq < security_ctx.expected_seq_num)
    {
        logger.error("Received NAS message with sequence number " + std::to_string(incoming_seq) +
                     " below expected " + std::to_string(security_ctx.expected_seq_num) +
                     ", possible replay attack.");
        return SRSRAN_ERROR;
    }
    // Update the expected sequence number (this update strategy may vary according to specification).
    security_ctx.expected_seq_num = incoming_seq + 1;

    // Unpack the outer NAS header; if this fails, abort further processing.
    if (nas_msg.unpack_outer_hdr(pdu) != SRSRAN_SUCCESS)
    {
        logger.error("Unable to unpack outer NAS header");
        return SRSRAN_ERROR;
    }

    // Record the current security header type for auditing/tracing.
    auto current_hdr_type = nas_msg.hdr.security_header_type;

    switch (current_hdr_type)
    {
        case Nas5gsHdr.security_header_type_opts.plain_5gs_nas_message:
            // Reject plain messages if a secure context is already established.
            if (security_ctx.is_established)
            {
                logger.error("Plain NAS message not accepted after security context establishment");
                return SRSRAN_ERROR;
            }
            break;

        case Nas5gsHdr.security_header_type_opts.integrity_protected:
            if (!integrity_check(pdu.get()))
            {
                logger.error("Integrity check failed for NAS message");
                return SRSRAN_ERROR;
            }
            break;

        case Nas5gsHdr.security_header_type_opts.integrity_protected_and_ciphered:
            if (!integrity_check(pdu.get()))
            {
                logger.error("Integrity check failed for ciphered NAS message");
                return SRSRAN_ERROR;
            }
            cipher_decrypt(pdu.get());
            break;

        case Nas5gsHdr.security_header_type_opts.integrity_protected_with_new_5G_nas_context:
            // Process the message appropriately; additional context setup steps may be required.
            break;

        case Nas5gsHdr.security_header_type_opts.integrity_protected_and_ciphered_with_new_5G_nas_context:
            logger.error("Received NAS message with unsupported new NAS context configuration");
            return SRSRAN_ERROR;

        default:
            logger.error("Not handling NAS message with unknown security header");
            return SRSRAN_ERROR;
    }

    /* Further NAS message processing can be performed here... */

    return SRSRAN_SUCCESS;
}
