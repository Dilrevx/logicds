        // Verify the incoming NAS message's sequence number to prevent replay attacks.
        // Assume that 'nas_msg.sequence_number' contains the message's sequence number,
        // and 'security_context.expected_sequence_number' holds the next expected number.
        if (nas_msg.sequence_number < security_context.expected_sequence_number)
        {
            logger.error("Received NAS message with an old sequence number; possible replay attack.");
            return SRSRAN_ERROR;
        }
        // Update expected sequence number for the next message.
        security_context.expected_sequence_number = nas_msg.sequence_number + 1;

        if (nas_msg.unpack_outer_hdr(pdu) != SRSRAN_SUCCESS)
        {
            logger.error("Unable to unpack outer NAS header");
            return SRSRAN_ERROR;
        }

        switch (nas_msg.hdr.security_header_type)
        {
            case Nas5gsHdr.security_header_type_opts.plain_5gs_nas_message:
                // Do not allow plain NAS messages if a security context is already established.
                if (security_context.established)
                {
                    logger.error("Rejected plain NAS message after registration complete");
                    return SRSRAN_ERROR;
                }
                break;

            case Nas5gsHdr.security_header_type_opts.integrity_protected:
                if (!integrity_check(pdu.get()))
                {
                    logger.error("Not handling NAS message with integrity check error");
                    return SRSRAN_ERROR;
                }
                break;

            case Nas5gsHdr.security_header_type_opts.integrity_protected_and_ciphered:
                if (!integrity_check(pdu.get()))
                {
                    logger.error("Not handling NAS message with integrity check error");
                    return SRSRAN_ERROR;
                }
                else
                {
                    cipher_decrypt(pdu.get());
                }
                break;

            case Nas5gsHdr.security_header_type_opts.integrity_protected_with_new_5G_nas_context:
                // Reject new context messages if already registered to prevent potential misuse in replay scenarios.

                    return SRSRAN_ERROR;
                }
                break;

                return SRSRAN_ERROR;

            default:
                logger.error("Not handling NAS message with unknown security header");
                return SRSRAN_ERROR;
