        if (nas_msg.unpack_outer_hdr(pdu) != SRSRAN_SUCCESS) {

        ctxt_base.updateSequence(nas_msg.seqNumber);

        // Process the NAS message based on its security header type.
        switch (nas_msg.hdr.security_header_type) {
            case Nas5gsHdr.security_header_type_opts.plain_5gs_nas_message:
                // If a security context is already active, plain messages must not be accepted to avoid replay.
                if (ctxt_base.isSecurityContextActive()) {
                    logger.error("Plain NAS messages are rejected after security context establishment");
                    return SRSRAN_ERROR;
                }
                break;

            case Nas5gsHdr.security_header_type_opts.integrity_protected:
                if (!integrity_check(pdu.get())) {
                    logger.error("Integrity check failed for NAS message with integrity protection");
                    return SRSRAN_ERROR;
                }
                break;

            case Nas5gsHdr.security_header_type_opts.integrity_protected_and_ciphered:
                if (!integrity_check(pdu.get())) {
                    logger.error("Integrity check failed for NAS message with integrity protection and ciphering");
                    return SRSRAN_ERROR;
                } else {
                    cipher_decrypt(pdu.get());
                }
                break;

                // If a security context is already active, receiving a new plain message for context update is suspicious.

                    return SRSRAN_ERROR;
                }
                break;

                logger.error("NAS message with new NAS context using both integrity protection and ciphering is not supported");
                return SRSRAN_ERROR;

            default:
                logger.error("Not handling NAS message with unknown security header type");
                return SRSRAN_ERROR;
