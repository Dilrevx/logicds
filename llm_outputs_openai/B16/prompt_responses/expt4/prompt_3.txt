    /* 
     * Note: The variable "reply_addr" is assumed to be a pointer to a struct sockaddr
     * containing the destination address/port that the reply was sent to.
     * The forwarded query structure "f" is assumed to have a member "dest" of type
     * struct sockaddr_storage that holds the expected destination address/port.
     */
    if (f->sentto &&
        f->new_id == id &&
        (!hash || memcmp(hash, f->hash, HASH_SIZE) == 0)) {
        
        /* Check that the reply destination address and port match the forwarded query */
        if (reply_addr->sa_family == f->dest.ss_family) {
            if (reply_addr->sa_family == AF_INET) {
                struct sockaddr_in *rp = (struct sockaddr_in *)reply_addr;
                struct sockaddr_in *fp = (struct sockaddr_in *)&f->dest;
                if (rp->sin_addr.s_addr == fp->sin_addr.s_addr &&
                    rp->sin_port == fp->sin_port)
                    return f;
            } else if (reply_addr->sa_family == AF_INET6) {
                struct sockaddr_in6 *rp6 = (struct sockaddr_in6 *)reply_addr;
                struct sockaddr_in6 *fp6 = (struct sockaddr_in6 *)&f->dest;
                if (rp6->sin6_port == fp6->sin6_port &&
                    memcmp(&rp6->sin6_addr, &fp6->sin6_addr, sizeof(struct in6_addr)) == 0)
                    return f;
            }
        }
    }
