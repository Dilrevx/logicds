static struct frec *lookup_frec(unsigned short id, void *hash, int recv_fd)
{
    struct frec *f;

    for (f = daemon->frec_list; f; f = f->next) {
        /* Only consider queries that were actually sent */
        if (!f->sentto)
            continue;

        /* Check the query identifier and hash (if provided) */
        if (f->new_id != id)
            continue;
        if (hash && memcmp(hash, f->hash, HASH_SIZE) != 0)
            continue;

        /*
         * New check: verify that the reply is arriving on the same socket
         * from which the query was forwarded.
         *
         * For IPv4 and IPv6, the daemon should record the socket (or file
         * descriptor) used in the original outbound query (stored here in
         * f->socket_fd). An attacker forging a response would have to guess or
         * control this attribute, which is highly unlikely.
         */
        if (f->socket_fd != recv_fd)
            continue;

        return f;
    }
    return NULL;
}
