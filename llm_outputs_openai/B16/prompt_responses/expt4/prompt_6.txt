static struct frec *lookup_frec(unsigned short id, void *hash, int reply_fd)
{
    struct frec *f;

    for(f = daemon->frec_list; f; f = f->next) {
        /* First, apply the basic matching criteria:
           - The record must have been sent (f->sentto not NULL)
           - The reply’s ID must match the query’s new_id
           - And if a hash was provided, it must match.
        */
        if (!f->sentto || f->new_id != id)
            continue;
        if (hash && memcmp(hash, f->hash, HASH_SIZE) != 0)
            continue;

        /* Now, apply the additional check on the socket file descriptor to ensure
           the reply is coming in on the same channel that was used for sending the query.
           
           The check is threefold:
           1. If the query was sent using an upstream-bound socket (f->bound_fd is valid),
              then the reply must be received on that socket.
           2. For queries using the default IPv4 socket (when f->af == AF_INET),
              the reply must arrive on daemon->ipv4_fd.
           3. For queries using the default IPv6 socket (when f->af == AF_INET6),
              the reply must arrive on daemon->ipv6_fd.
        */

        if (f->bound_fd >= 0) {
            if (reply_fd != f->bound_fd)
                continue;
        }
        else if (f->af == AF_INET) {
            if (reply_fd != daemon->ipv4_fd)
                continue;
        }
        else if (f->af == AF_INET6) {
            if (reply_fd != daemon->ipv6_fd)
                continue;
        }
        else {
            /* If the address family is unknown, reject the query.
               This adds a safeguard against unexpected behavior.
            */
            continue;
        }

        return f;
    }
    return NULL;
}
