     * Ensure all query parameters are matched according to RFC5452.
     * In addition to checking the query ID and optional hash,
     * we now verify that the response's source IP,
     * destination IP, and destination port exactly match the original forwarded query.
     * It is assumed that the following variables are available:
     *   reply_src: the sender IP address of the DNS reply.
     *   reply_dst: the destination IP address of the DNS reply.
     *   reply_dst_port: the destination port of the DNS reply.
     * Also, it is assumed that f->sentto->addr holds the original destination IP 
     * (i.e. the DNS server's IP), and that f->local_ip and f->local_port hold the local 
     * resolver's IP and port used when sending the query.
     */
    if (f->sentto &&
        f->new_id == id &&
        ip_equal(reply_src, f->sentto->addr) &&          /* Verify source address */
        ip_equal(reply_dst, f->local_ip) &&                /* Verify destination address */
