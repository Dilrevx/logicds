static int constant_time_compare(const void *a, const void *b, size_t size)
{
    const unsigned char *p1 = a, *p2 = b;
    unsigned char result = 0;
    size_t i;
    for (i = 0; i < size; i++) {
        result |= p1[i] ^ p2[i];
    }
    return result;
}

static struct frec *lookup_frec(unsigned short id, void *hash)
{
    struct frec *f;

    /* Validate inputs and daemon state to prevent NULL pointer dereference */
    if (!daemon || !daemon->frec_list || !hash)
        return NULL;

    /* Iterate safely over the frec list */
    for (f = daemon->frec_list; f; f = f->next) {
        /* Check if the id matches */
        if (f->id == id) {
            /*
             * Use a constant time comparison for the hash to mitigate timing attacks.
             * Assume that the size of the hash is known and stored in HASH_SIZE.
             */
            if (constant_time_compare(f->hash, hash, HASH_SIZE) == 0)
                return f;
        }
    }

    return NULL;
}
