static struct frec *lookup_frec(unsigned short id, void *hash)
{
    struct frec *f;

    for (f = daemon->frec_list; f; f = f->next) {
        /* First, check that the ID matches */
        if (f->id != id)
            continue;

        /* Now, be sure both hash pointers are non‐NULL before comparing their contents.
           Depending on what “hash” points to (a string or binary blob), use the appropriate comparison.
           Here we assume the hash is a null‐terminated string.
           If both pointers are NULL, we consider them equal. */
        if (f->hash && hash) {
            if (strcmp((char *)f->hash, (char *)hash) == 0)
                return f;
        } else if (f->hash == hash) {
            return f;
        }
    }
    return NULL;
}
