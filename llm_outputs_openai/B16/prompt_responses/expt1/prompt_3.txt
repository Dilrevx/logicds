static struct frec *lookup_frec(unsigned short id, void *hash)
{
    struct frec *f;

    /* Always validate the input.
       If hash is supposed to be non-NULL, then bail out early if it is NULL.
       Also, if daemon or its frec_list is unexpectedly NULL, we safely handle that. */
    if (!hash || !daemon || !daemon->frec_list)
        return NULL;

    /*
     * For each record in the linked list, check that both the id as well as the
     * hash field match. This repair fixes the logical vulnerability where the
     * hash was not being verified, thus allowing a possible bypass if only the
     * id was checked.
     *
     * Note: Replace HASH_SIZE with the correct expected byte-size of the hash.
     */
    for (f = daemon->frec_list; f; f = f->next) {
        if (f->id == id && memcmp(f->hash, hash, HASH_SIZE) == 0) {
            return f;
        }
    }

    return NULL;
}
