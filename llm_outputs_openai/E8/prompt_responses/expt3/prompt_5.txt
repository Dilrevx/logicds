switch (msg_type)
{
    // New branch: handle authentication reject messages securely.
    case LIBLTE_MME_MSG_TYPE_AUTHENTICATION_REJECT:
    {
        if (is_security_context_active())
        {
            // If a security context is already active, do not process an authentication reject in plaintext.
            log_error("Received authentication reject in active security context; dropping message.");
            return;
        }
        // No secure context exists, so process the authentication reject normally.
        parse_authentication_reject(lcid, std::move(pdu));
        break;
    }

    // Adjusted branch for Service Reject messages.
    case LIBLTE_MME_MSG_TYPE_SERVICE_REJECT:
    {
        if (is_security_context_active())
        {
            // Optionally, log the error if we receive a service reject while a security context is active.
            log_error("Received Service Reject in active security context; dropping message.");
            return;
        }
        // Process the service reject message normally.
        parse_service_reject(lcid, std::move(pdu));
        break;
    }

    // Existing branch for ESM Information Request messages.
    case LIBLTE_MME_MSG_TYPE_ESM_INFORMATION_REQUEST:
    {
        parse_esm_information_request(lcid, std::move(pdu));
        break;
    }

    // ... additional cases ...
}
