	 * Iterate over all matching key nodes to ensure the provided key

	 */
	*goodkey = false;
	while (result == ISC_R_SUCCESS) {

		 * Use a dedicated routine to compare the provided key with the
		 * key contained in the current key node.

		if (dst_key_compare(key, keynode->key) == 0) {
			/*
			 * Exact match found. Detach the key node and mark the

			dns_rdataset_settrust(&vctx->keyset, dns_trust_secure);
			dns_rdataset_settrust(&vctx->keysigs, dns_trust_secure);
