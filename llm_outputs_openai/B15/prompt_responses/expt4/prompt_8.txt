static isc_result_t
check_dnskey_sigs(vctx_t *vctx, const dns_rdata_dnskey_t *dnskey,
                  dns_rdata_t *rdata, bool is_ksk)
{
    unsigned char *active_keys, *standby_keys;
    dns_keynode_t *keynode = NULL;
    bool *goodkey;
    dst_key_t *key = NULL;
    isc_result_t result;

    active_keys = (is_ksk ? vctx->ksk_algorithms : vctx->zsk_algorithms);
    standby_keys = (is_ksk ? vctx->standby_ksk : vctx->standby_zsk);
    goodkey = (is_ksk ? &vctx->goodksk : &vctx->goodzsk);

    if (dns_dnssec_selfsigns(rdata, vctx->origin, &vctx->keyset,
                             &vctx->keysigs, false, vctx->mctx))
    {
        if (active_keys[dnskey->algorithm] != 255) {
            active_keys[dnskey->algorithm]++;
        }
    } else if (!is_ksk &&
               dns_dnssec_signs(rdata, vctx->origin, &vctx->soaset,
                                &vctx->soasigs, false, vctx->mctx))
    {
        if (active_keys[dnskey->algorithm] != 255) {
            active_keys[dnskey->algorithm]++;
        }
        return (ISC_R_SUCCESS);
    } else {
        if (standby_keys[dnskey->algorithm] != 255) {
            standby_keys[dnskey->algorithm]++;
        }
        return (ISC_R_SUCCESS);
    }

    /*
     * If a trust anchor table was not supplied, a correctly self-signed
     * DNSKEY RRset is good enough.
     */
    if (vctx->secroots == NULL) {
        *goodkey = true;
        return (ISC_R_SUCCESS);
    }

    /*
     * Obtain a DST key from the supplied DNSKEY rdata.
     */
    result = dns_dnssec_keyfromrdata(vctx->origin, rdata, vctx->mctx, &key);
    if (result != ISC_R_SUCCESS) {
        return (result);
    }

    /*
     * Iteratively search the trust anchor table for a matching key.
     * Instead of accepting the first lookup result, the code now loops
     * through all potential matching key nodes and explicitly compares
     * the keys using the dedicated key comparison function.
     */
    bool found = false;
    while ((result = dns_keytable_findkeynode(vctx->secroots, vctx->origin,
                                               dst_key_alg(key),
                                               dst_key_id(key),
                                               &keynode)) == ISC_R_SUCCESS)
    {
        /* Assume dst_key_compare returns 0 when keys match.
         * Also assume keynode->key is accessible via a helper function.
         */
        if (dst_key_compare(key, keynode->key) == 0) {
            found = true;
            dns_keytable_detachkeynode(vctx->secroots, &keynode);
            break;
        }
        dns_keytable_detachkeynode(vctx->secroots, &keynode);
    }
    /* Regardless of the loop outcome, we set result to success so that
     * partial matches or not-found results do not cause error propagation.
     */
    result = ISC_R_SUCCESS;

    if (found) {
        dns_rdataset_settrust(&vctx->keyset, dns_trust_secure);
        dns_rdataset_settrust(&vctx->keysigs, dns_trust_secure);
        *goodkey = true;
    }

    /*
     * Clean up.
     */
    dst_key_free(&key);
    return (result);
}
