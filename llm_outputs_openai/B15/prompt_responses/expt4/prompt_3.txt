	case ISC_R_SUCCESS:
		/*
		 * The supplied key is a trust anchor.
		 * Detach the keynode and stamp the keyset and key signatures as secure.
		 */
		dns_keytable_detachkeynode(vctx->secroots, &keynode);
		dns_rdataset_settrust(&vctx->keyset, dns_trust_secure);
		dns_rdataset_settrust(&vctx->keysigs, dns_trust_secure);
		*goodkey = true;
		break;
	case ISC_R_NOTFOUND:
		/*
		 * The supplied key is not present in the trust anchor table.
		 * In this case we do not treat the key as secure.
		 */
		result = ISC_R_SUCCESS;
		break;
	/*
	 * IMPORTANT SECURITY FIX:
	 *
	 * Do NOT treat a partial match in the trust anchor table as success.
	 * A DNS_R_PARTIALMATCH indicates that key parameters did not match exactly,
	 * which means the incoming data cannot be considered fully validated.
	 * Returning a success here would erroneously allow forged data to be accepted.
	 * Thus, in the case of a DNS_R_PARTIALMATCH (or any other error), we return the error.
	 */
	default:
		/* For errors (including DNS_R_PARTIALMATCH) just return the error. */
		break;
	}

	 * Clean up.

	dst_key_free(&key);

	return (result);
