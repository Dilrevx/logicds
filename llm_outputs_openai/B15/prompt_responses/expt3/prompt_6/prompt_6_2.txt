	/* Always initialize the goodkey flag. */
	*goodkey = false;

	result = dns_keytable_findkeynode(vctx->secroots, vctx->origin,
	                                  dst_key_alg(key), dst_key_id(key),
	                                  &keynode);
	switch (result) {
	case ISC_R_SUCCESS:
		/*
		 * The supplied key is a trust anchor.
		 */
		dns_keytable_detachkeynode(vctx->secroots, &keynode);
		dns_rdataset_settrust(&vctx->keyset, dns_trust_secure);
		dns_rdataset_settrust(&vctx->keysigs, dns_trust_secure);
		*goodkey = true;
		break;
	case ISC_R_NOTFOUND:
		/*
		 * The supplied key is not present in the trust anchor table,
		 * so it is not a trust anchor.
		 * Do not modify result (it remains ISC_R_SUCCESS so that
		 * other DNSKEYs may be considered valid).
		 */
		result = ISC_R_SUCCESS;
		break;
	case DNS_R_PARTIALMATCH:
		/*
		 * A partial match was found.
		 *
		 * This case indicates that the key parameters did not
		 * exactly match those in the trust anchor.  For security,
		 * we now treat a partial match as a failure rather than
		 * silently accepting the key.
		 */
		/* Do not convert DNS_R_PARTIALMATCH to ISC_R_SUCCESS. */
		break;
	default:
		/*
		 * An error occurred while searching the trust anchor table.
		 */
		break;
	}

	/*
	 * Clean up.
	 */
	dst_key_free(&key);

	return (result);
