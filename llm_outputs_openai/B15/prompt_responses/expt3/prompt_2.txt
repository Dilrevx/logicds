	result = dns_keytable_findkeynode(vctx->secroots, vctx->origin,
	                                   dst_key_alg(key), dst_key_id(key),
	                                   &keynode);
	if (result != ISC_R_SUCCESS) {
		/*
		 * If no keynode was found or a partial match was returned,
		 * do not alter the dns datasets.
		 */
		if (result == DNS_R_PARTIALMATCH || result == ISC_R_NOTFOUND)
			result = ISC_R_SUCCESS;
		goto cleanup;
	}

	/* Iterate over all matching key nodes to verify an exact key match. */
	while (keynode != NULL) {
		/*
		 * Compare the provided key with the candidate key.
		 * Assume dns_key_compare returns 0 if the keys match.
		 */
		if (dns_key_compare(key, dns_keynode_getkey(keynode)) == 0) {
			/* Found an exact match: detach the keynode and mark as trusted. */
			dns_keytable_detachkeynode(vctx->secroots, &keynode);
			dns_rdataset_settrust(&vctx->keyset, dns_trust_secure);
			dns_rdataset_settrust(&vctx->keysigs, dns_trust_secure);
			*goodkey = true;
			break;
		}
		/*
		 * No match found with this key node.
		 * Detach it to avoid reprocessing and try to fetch the next candidate.
		 */
		dns_keytable_detachkeynode(vctx->secroots, &keynode);
		result = dns_keytable_findnextkeynode(vctx->secroots, vctx->origin,
		                                      dst_key_alg(key), dst_key_id(key),
		                                      &keynode);
		/*
		 * If no more keynodes are found, break out of the loop.
		 * Treat the situation as non-fatal.
		 */
		if (result != ISC_R_SUCCESS)
			break;
	}

	/* Always return success as the key processing is non-fatal.
	   A key that does not match is treated as "not a trust anchor". */
	result = ISC_R_SUCCESS;

cleanup:
	/*
	 * Clean up the key resource.
	 */
	dst_key_free(&key);
	return (result);
