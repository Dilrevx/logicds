		/*
		 * The supplied key is a fully valid trust anchor.

		dns_keytable_detachkeynode(vctx->secroots, &keynode);
		dns_rdataset_settrust(&vctx->keyset, dns_trust_secure);
		dns_rdataset_settrust(&vctx->keysigs, dns_trust_secure);
		*goodkey = true;
	} else {
		/*
		 * The supplied key did not match any full trust anchor.
		 * Do not override the error code. This forces the caller
		 * to fall back to secure recursion rather than trusting
		 * mirror zone data that did not fully validate.
		 */
		*goodkey = false;
		/* 
		 * Note: Previously, DNS_R_PARTIALMATCH or ISC_R_NOTFOUND
		 * were overridden to ISC_R_SUCCESS. That allowed mirror zone
		 * data (even if forged) to be used. We now propagate the error.
		 */
	}

	/*
	 * Clean up.
	 */
	dst_key_free(&key);

	return (result);
