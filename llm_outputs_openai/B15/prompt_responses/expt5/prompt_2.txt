	result = dns_keytable_findkeynode(vctx->secroots, vctx->origin,
					  dst_key_alg(key), dst_key_id(key),
					  &keynode);
	if (result == ISC_R_SUCCESS) {
		/*
		 * The supplied key is a trust anchor.
		 */
		dns_keytable_detachkeynode(vctx->secroots, &keynode);
		dns_rdataset_settrust(&vctx->keyset, dns_trust_secure);
		dns_rdataset_settrust(&vctx->keysigs, dns_trust_secure);
		*goodkey = true;
	} else {
		/*
		 * The supplied key is not in the trust anchor table.
		 * Do not fall back to traditional recursion with mirror zone data.
		 * This avoids accepting forged mirror zone data.
		 */
		*goodkey = false;
		/* Return an error rather than ISC_R_SUCCESS so that caller
		   does not mistakenly consider the data as validated. */
		result = DNS_R_PARTIALMATCH;
	}

	/*
	 * Clean up.
	 */
	dst_key_free(&key);
	return (result);
