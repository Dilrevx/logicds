package master;

import common.TlsState;
import common.HandshakeStates;
import common.HandshakeException;

public class TlsClient {
    public TlsState state;

    public TlsClient() {
        this.state = new TlsState();
        this.state.handshakeStage = 0;
        this.state.flags = 0;
        this.state.handshakeComplete = false;
        this.state.masterSecretDerived = false;
        this.state.SSL3_FLAGS_CCS_OK = 0x0080;
    }


    public int ssl3_connect() {
        int ret = -1;
        
        // Securely validate that the state object is properly initialized
        if (state == null) {
            throw new IllegalStateException("SSL state is not initialized.");
        }
        
        switch (state.handshakeStage) {
            case 0:
                // Log message via a secure logging mechanism rather than plain stdout (if available)
                System.out.println("Client: Sending ClientHello");
                state.handshakeStage = HandshakeStates.SSL3_ST_CR_FINISHED_A;
                ret = 1;
                break;
                
            // Example handling for the next expected handshake phase.
            case HandshakeStates.SSL3_ST_CR_FINISHED_A:
                // Process the server's Finished message with proper validation steps.
                System.out.println("Client: Processing Server Finished");
                // Insert cryptographic validations and state updates as required.
                // For example, validate the finished message and update handshake state.
                state.handshakeStage = HandshakeStates.SSL3_ST_DONE;
                ret = 1;
                break;
                
            case HandshakeStates.SSL3_ST_DONE:
                System.out.println("SSL handshake is complete.");
                ret = 0;
                break;
                
            default:
                // Handle unexpected state values securely.
                System.err.println("Error: Unknown handshake stage " + state.handshakeStage);
                // Optionally, clean up any sensitive state or abort the connection.
                // For added security, you might invoke a shutdown procedure here.
                ret = -1;
                break;
        }
        
        // After the switch, perform a final check.
        if (ret < 0) {
            // Log the handshake failure using a secure logging system.
            System.err.println("SSL handshake failed.");
            // Depending on your security policy, you might throw an exception or perform additional cleanup.
        }
        
        return ret;
    }


    private int ssl3_get_finished() {
        System.out.println("Client: ssl3_get_finished() succeeded");
        return 1;
    }

    public int readChangeCipherSpec() {
        if ((state.flags & state.SSL3_FLAGS_CCS_OK) == 0) {
           return 0;
        }
        System.out.println("Client: CCS accepted");
        state.flags &= ~state.SSL3_FLAGS_CCS_OK;
        return 1;
    }

    public void readServerFinished() {
        if (state.handshakeStage != HandshakeStates.SSL_ST_OK) {
            throw new HandshakeException("Client: Not ready for Finished");
        }
        System.out.println("Client: Received Finished; handshake complete");
        state.handshakeComplete = true;
    }
}
