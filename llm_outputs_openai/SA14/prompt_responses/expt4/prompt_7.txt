    public int ssl3_connect() {
        int ret = -1;
        switch (state.handshakeStage) {
            case 0:
                System.out.println("Client: Sending ClientHello");
                // In a correct handshake, the CCS message is only allowed later.
                // Set the CCS_OK flag to indicate that an in-order CCS is expected.
                state.flags |= state.SSL3_FLAGS_CCS_OK;
                state.handshakeStage = HandshakeStates.SSL3_ST_CR_FINISHED_A;
                ret = 1;
                break;
            case HandshakeStates.SSL3_ST_CR_FINISHED_A:
            case HandshakeStates.SSL3_ST_CR_FINISHED_B:
                // Only process a ChangeCipherSpec (CCS) message if it is expected.
                if ((state.flags & state.SSL3_FLAGS_CCS_OK) == 0) {
                    throw new HandshakeException("Client: Unexpected ChangeCipherSpec received");
                }
                // Clear the flag so that a duplicate CCS message is not accepted.
                state.flags &= ~state.SSL3_FLAGS_CCS_OK;
                System.out.println("Client: In Finished state, processing Finished message");
                ret = ssl3_get_finished();
                if (ret <= 0) {
                    return ret;
                }
                state.handshakeStage = HandshakeStates.SSL_ST_OK;
                break;
            case HandshakeStates.SSL_ST_OK:
                state.handshakeComplete = true;
                ret = 1;
                break;
            default:
                throw new HandshakeException("Client: Unknown handshake state");
        }
        return ret;
    }
