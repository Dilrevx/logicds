    public int ssl3_connect() {
        int ret = -1;
        switch (state.handshakeStage) {
            case 0:
                System.out.println("Client: Sending ClientHello");
                state.handshakeStage = HandshakeStates.SSL3_ST_CR_FINISHED_A;
                ret = 1;
                break;
            case HandshakeStates.SSL3_ST_CR_FINISHED_A:
            case HandshakeStates.SSL3_ST_CR_FINISHED_B:
                System.out.println("Client: In Finished state, verifying Finished message");
                ret = ssl3_get_finished();
                // Added strict check: if Finished message is not properly validated, abort the handshake.
                if (ret <= 0) {
                    System.err.println("Client: Finished message validation failed");
                    return ret;
                }
                state.handshakeStage = HandshakeStates.SSL_ST_OK;
                break;
            case HandshakeStates.SSL_ST_OK:
                System.out.println("Client: Handshake already in SSL_ST_OK state");
                break;
            default:
                System.err.println("Client: Unknown handshake stage encountered");
                ret = -1;
                break;
        }
        return ret;
    }
