public int ssl3_connect() {
    int ret = -1;
    switch (state.handshakeStage) {
        case 0:
            // Starting handshake: send ClientHello only when in the correct state.
            System.out.println("Client: Sending ClientHello");
            // Advance to the next expected stage.
            state.handshakeStage = HandshakeStates.SSL3_ST_CR_FINISHED_A;
            ret = 1;
            break;
        case HandshakeStates.SSL3_ST_CR_FINISHED_A:
            // Process the expected Server response.
            System.out.println("Client: Processing Server response for ClientHello");
            // Validate that the response is as expected (this could include certificate validation, etc.).
            // If the server response was valid, move to the next stage.
            state.handshakeStage = HandshakeStates.SSL3_ST_CR_FINISHED_B;
            ret = 1;
            break;
        case HandshakeStates.SSL3_ST_CR_FINISHED_B:
            // Final handshake stage: finish up.
            System.out.println("Client: Finishing handshake");
            // Final state update; handshake is complete.
            state.handshakeStage = HandshakeStates.FINISHED;
            ret = 1;
            break;
        default:
            // If the handshake stage is not recognized, do not try to continue.
            System.err.println("Error: Invalid handshake state (" + state.handshakeStage + "). Aborting connection.");
            ret = -1;
            break;
    }
    return ret;
}
