import com.example.User;
import com.example.UserEventListener;
import com.example.UserEventDispatcher;
import com.example.AdminPrivilegesManager;

initProvider(ADMIN_PROVIDER.getValue());

UserEventDispatcher.addListener(new UserEventListener() {
    @Override
    public void onUserDeleted(User user) {
        if (user != null) {
            cleanupAdminPrivileges(user);
        }
    }

    @Override
    public void onUserCreated(User user) {
        // No action needed on user creation
    }

    @Override
    public void onUserModified(User user) {
        // No action needed on user modification
    }
});

private void cleanupAdminPrivileges(User user) {
    AdminPrivilegesManager.revokePrivileges(user.getId());
}
</repair>
