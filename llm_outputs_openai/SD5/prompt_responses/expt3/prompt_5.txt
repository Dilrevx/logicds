    /* Original initialization of the admin provider */
    initProvider(ADMIN_PROVIDER.getValue());
    
    /* Secure fix: Register an event listener for user deletion events */
    eventDispatcher.registerListener(new UserDeletionListener() {
        @Override
        public void onUserDeleted(UserDeletionEvent event) {
            User user = event.getUser();
            // Ensure the user is valid before processing
            if (user != null) {
                // Invoke the clean-up method to revoke admin privileges
                removeAdminPrivileges(user);
            }
        }
    });
    
    /**
     * Helper method to remove any lingering administrative privileges that
     * may have been assigned to the deleted user. This helps safeguard against
     * attackers trying to escalate privileges via compromised user accounts.
     */
    private void removeAdminPrivileges(User user) {
        // Example: Check if the user is marked with an admin role or listed in the admin.authorizedJIDs
        if (user.hasRole("admin") || ADMIN_PROVIDER.getAuthorizedJIDs().contains(user.getJID())) {
            // Remove the admin role and/or clean up the admin.authorizedJIDs listing
            user.removeRole("admin");
            ADMIN_PROVIDER.getAuthorizedJIDs().remove(user.getJID());
            // Optionally log this removal for auditing purposes
            logger.info("Removed administrative privileges for user: " + user.getJID());
        }
    }
