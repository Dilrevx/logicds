initProvider(ADMIN_PROVIDER.getValue());

// Register an event listener to handle user deletion events and automatically revoke admin privileges.
EventDispatcher.registerListener(UserDeletionEvent.class, new UserDeletionListener() {
    @Override
    public void onUserDeleted(UserDeletionEvent event) {
        User user = event.getUser();
        // Ensure the event carries a valid user before attempting cleanup.
        if (user != null) {
            // Remove any administrative privileges associated with the deleted user.
            removeAdminPrivileges(user);
        }
    }
});
