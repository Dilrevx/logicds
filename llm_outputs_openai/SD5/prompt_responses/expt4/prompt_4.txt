        // Initialize admin provider maintaining existing functionality
        initProvider(ADMIN_PROVIDER.getValue());

        // Register an event listener for user deletion events.
        // When a user is deleted, remove any associated administrative privileges.
        UserEventDispatcher.registerDeletionListener(new UserDeletionListener() {
            @Override
            public void onUserDeleted(User user) {
                // Verify that the user is not null before proceeding
                if (user != null) {
                    // Revoke admin privileges associated with the deleted user.
                    // This prevents privilege escalation from lingering admin configurations.
                    AdminManager.revokeAdminPrivileges(user);
                }
            }
        });
