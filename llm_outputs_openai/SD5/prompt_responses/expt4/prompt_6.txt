    private AdminManager() {
        // Preserve original provider initialization.
        initProvider(ADMIN_PROVIDER.getValue());
        
        // Register a listener for user deletion events to ensure that any lingering
        // administrative privileges are automatically revoked when a user is removed.
        EventDispatcher.registerListener(EventType.USER_DELETION, new UserDeletionListener() {
            @Override
            public void onUserDeletion(User user) {
                // Ensure the user object is valid to avoid null pointer exceptions.
                if (user != null) {
                    // Remove any administrative privileges associated with the user.
                    AdminPrivilegesManager.removeAdminPrivileges(user);
                }
            }
        });
    }
