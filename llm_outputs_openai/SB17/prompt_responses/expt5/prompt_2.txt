    /* Retrieve the CRLs not just based on the subject but also on the issuer.
       This ensures that both the certificate’s subject and issuer fields are checked. */
    X509_NAME issuer = X509_get_issuer_name(cert);
    if (Commons.OPENSSL_VERSION_NUMBER >= Commons.OPENSSL_1_1 && !Commons.HAVE_LIBRESSL) {
        List<CRL> crlsSubject = storeCtx.get1_crls(subject);
        List<CRL> crlsIssuer = storeCtx.get1_crls(issuer);
        if (crlsSubject == null) {
            crls = crlsIssuer;
        } else {
            if (crlsIssuer != null) {
                crlsSubject.addAll(crlsIssuer);
            }
            crls = crlsSubject;
        }
    }
    else if (Commons.OPENSSL_VERSION_NUMBER >= Commons.OPENSSL_1_0 && !Commons.HAVE_LIBRESSL) {
        List<CRL> crlsSubject = Commons.get1_crls(storeCtx, subject);
        List<CRL> crlsIssuer = Commons.get1_crls(storeCtx, issuer);
        if (crlsSubject == null) {
            crls = crlsIssuer;

            if (crlsIssuer != null) {
                crlsSubject.addAll(crlsIssuer);
            }
            crls = crlsSubject;
        }
