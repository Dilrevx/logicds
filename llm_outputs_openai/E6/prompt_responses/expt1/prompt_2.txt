void nas::write_pdu(uint32_t lcid, unique_byte_buffer_t pdu)
{
    uint8 pd           = 0;
    uint8 msg_type     = 0;
    uint8 sec_hdr_type = 0;

    // Validate that the PDU pointer is not null.
    if (!pdu) {
        logger.error("write_pdu: Received null PDU pointer for LCID %u", lcid);
        return;
    }

    // Validate that the PDU contains at least the minimum expected number of bytes.
    // Assuming that LIBLTE_BYTE_MSG_STRUCT represents the minimum header to be parsed.
    if (pdu->N_bytes < sizeof(LIBLTE_BYTE_MSG_STRUCT)) {
        logger.error("write_pdu: Insufficient PDU length: %u bytes (expected at least %zu) for LCID %u",
                     pdu->N_bytes, sizeof(LIBLTE_BYTE_MSG_STRUCT), lcid);
        return;
    }

    logger.info(pdu->msg, pdu->N_bytes, "DL %s PDU", rrc->get_rb_name(lcid));

    // Parse the message security header.
    // Casting to LIBLTE_BYTE_MSG_STRUCT is now safe after our validations.
    liblte_mme_parse_msg_sec_header((LIBLTE_BYTE_MSG_STRUCT*)pdu.get(), &pd, &sec_hdr_type);

    return;
}
