package main;

import commons.X509;
import commons.X509_VERIFY_PARAM;
import commons.X509_STORE_CTX;
import commons.VerifyCallback;
import commons.Helper;
import commons.Constants;
import java.util.List;
import java.util.ArrayList;

public class X509CertVerifier {
    public static int X509_verify_cert(X509_STORE_CTX ctx) {
        X509 x, xtmp, xtmp2, chain_ss = null;
        int bad_chain = 0;
        X509_VERIFY_PARAM param = ctx.param;
        int depth, i, ok = 0;
        int num, j, retry;
        VerifyCallback cb = ctx.verify_cb;
        List<X509> sktmp = null;
        if (ctx.cert == null) {
            Helper.raiseError("X509_R_NO_CERT_SET_FOR_US_TO_VERIFY");
            return -1;
        }
        if (ctx.chain == null) {
            ctx.chain = new ArrayList<X509>();
            ctx.chain.add(ctx.cert);
            ctx.last_untrusted = 1;
        }
        if (ctx.untrusted != null) {
            sktmp = new ArrayList<X509>(ctx.untrusted);
        }
        num = ctx.chain.size();
        x = ctx.chain.get(num - 1);
        depth = param.depth;
        for (;;) {
            if (depth < num) break;
            if (Helper.cert_self_signed(x)) break;
            if ((param.flags & Constants.X509_V_FLAG_TRUSTED_FIRST) != 0) {
                ok = ctx.get_issuer(x);
                if (ok < 0) return ok;
                if (ok > 0) {
                    Helper.free(x);
                    break;
                }
            }
            if (ctx.untrusted != null) {
                xtmp = Helper.find_issuer(ctx, sktmp, x);
                if (xtmp != null) {
                    ctx.chain.add(xtmp);
                    ctx.last_untrusted++;
                    x = xtmp;
                    num++;
                    sktmp.remove(xtmp);
                    continue;
                }
            }
            break;
        }
        j = num;

        do {
            // Get current chain size and the last certificate in the chain.
            int chainSize = ctx.chain.size();
            x = ctx.chain.get(chainSize - 1);
            // Handle self-signed certificate cases.
            if (Helper.cert_self_signed(x)) {
                if (chainSize == 1) {
                    // For a single certificate that is self-signed,
                    // try to locate its issuer. If it fails or the issuer differs, report an error.
                    ok = ctx.get_issuer(x);
                    if (ok <= 0 || !Helper.equals(x, Helper.lastIssuer)) {
                        ctx.error = Constants.X509_V_ERR_DEPTH_ZERO_SELF_SIGNED_CERT;
                        ctx.current_cert = x;
                        ctx.error_depth = chainSize - 1;
                        if (ok == 1) {
                            Helper.free(Helper.lastIssuer);
                        }
                        bad_chain = 1;
                        ok = cb.call(0, ctx);
                        if (!Helper.booleanOk(ok)) {
                            return ok;
                        }
                    } else {
                        // If the issuer lookup succeeded and matches, replace certificate.
                        Helper.free(x);
                        x = Helper.lastIssuer;
                        ctx.chain.set(chainSize - 1, x);
                        // Update last_untrusted to reflect the current chain size.
                        ctx.last_untrusted = ctx.chain.size();
                    }
                } else {
                    // If there are multiple certificates and the last one is self-signed,
                    // remove it from the chain and update last_untrusted.
                    chain_ss = ctx.chain.remove(chainSize - 1);
                    ctx.last_untrusted = ctx.chain.size();
                    // Update x to be the new last certificate.
                    x = ctx.chain.get(ctx.chain.size() - 1);
                }
            }
  
            // Attempt to extend the certificate chain using valid issuer information.
            // This loop builds an alternative chain, ensuring intermediate certificates are properly validated.
            while (true) {
                // Stop if the current certificate is self-signed.
                if (Helper.cert_self_signed(x)) {
                    break;
                }
                ok = ctx.get_issuer(x);
                if (ok < 0) {
                    return ok;
                }
                // If no valid issuer is found, then no further chain building is possible.
                if (ok == 0) {
                    break;
                }
                // Append the obtained issuer to the chain
                x = Helper.lastIssuer;
                ctx.chain.add(x);
            }
  
            // Check whether the current chain is trusted.
            i = Helper.check_trust(ctx);
            if (i == Constants.X509_TRUST_REJECTED) {
                break;
            }
  
            retry = 0;
            // If the chain is not fully trusted and alternative chains are allowed,
            // then try to generate an alternative path.
            if (i != Constants.X509_TRUST_TRUSTED
                    && (param.flags & Constants.X509_V_FLAG_TRUSTED_FIRST) == 0
                    && (param.flags & Constants.X509_V_FLAG_NO_ALT_CHAINS) == 0) {
                // Iterate backward over the chain (except the first certificate)
                for (int j = ctx.chain.size(); j > 1; j--) {
                    xtmp2 = ctx.chain.get(j - 2);
                    ok = ctx.get_issuer(xtmp2);
                    if (ok < 0) {
                        return ok;
                    }
                    if (ok > 0) {
                        // Free the temporary issuer as it was used only in computation.
                        Helper.free(Helper.lastIssuer);
                        // Remove all certificates added beyond the point j - 1,
                        // ensuring the chain reflects a valid state.
                        while (ctx.chain.size() > j - 1) {
                            X509 removed = ctx.chain.remove(ctx.chain.size() - 1);
                            Helper.free(removed);
                        }
                        // Ensure the last_untrusted field correctly reflects the new chain size.
                        ctx.last_untrusted = ctx.chain.size();
                        // Signal that a valid alternative chain path was found.
                        retry = 1;
                        break;
                    }
                }
            }
        } while (retry != 0);


        return ok;
    }
}
