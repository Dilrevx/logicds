package main;

import commons.X509;
import commons.X509_VERIFY_PARAM;
import commons.X509_STORE_CTX;
import commons.VerifyCallback;
import commons.Helper;
import commons.Constants;
import java.util.List;
import java.util.ArrayList;

public class X509CertVerifier {
    public static int X509_verify_cert(X509_STORE_CTX ctx) {
        X509 x, xtmp, xtmp2, chain_ss = null;
        int bad_chain = 0;
        X509_VERIFY_PARAM param = ctx.param;
        int depth, i, ok = 0;
        int num, j, retry;
        VerifyCallback cb = ctx.verify_cb;
        List<X509> sktmp = null;
        if (ctx.cert == null) {
            Helper.raiseError("X509_R_NO_CERT_SET_FOR_US_TO_VERIFY");
            return -1;
        }
        if (ctx.chain == null) {
            ctx.chain = new ArrayList<X509>();
            ctx.chain.add(ctx.cert);
            ctx.last_untrusted = 1;
        }
        if (ctx.untrusted != null) {
            sktmp = new ArrayList<X509>(ctx.untrusted);
        }
        num = ctx.chain.size();
        x = ctx.chain.get(num - 1);
        depth = param.depth;
        for (;;) {
            if (depth < num) break;
            if (Helper.cert_self_signed(x)) break;
            if ((param.flags & Constants.X509_V_FLAG_TRUSTED_FIRST) != 0) {
                ok = ctx.get_issuer(x);
                if (ok < 0) return ok;
                if (ok > 0) {
                    Helper.free(x);
                    break;
                }
            }
            if (ctx.untrusted != null) {
                xtmp = Helper.find_issuer(ctx, sktmp, x);
                if (xtmp != null) {
                    ctx.chain.add(xtmp);
                    ctx.last_untrusted++;
                    x = xtmp;
                    num++;
                    sktmp.remove(xtmp);
                    continue;
                }
            }
            break;
        }
        j = num;

/* Repaired version of the certificate chain building/verification snippet */
do {
    i = ctx.chain.size();
    x = ctx.chain.get(i - 1);

    // 1. Handle self‐signed certificate case in a safe way.
    if (Helper.cert_self_signed(x)) {
        if (ctx.chain.size() == 1) {
            ok = ctx.get_issuer(x);
            if (ok <= 0 || !Helper.equals(x, Helper.lastIssuer)) {
                ctx.error = Constants.X509_V_ERR_DEPTH_ZERO_SELF_SIGNED_CERT;
                ctx.current_cert = x;
                ctx.error_depth = i - 1;
                // Free allocated issuer if previously set
                if (ok == 1) {
                    Helper.free(Helper.lastIssuer);
                    Helper.lastIssuer = null;
                }
                bad_chain = 1;
                ok = cb.call(0, ctx);
                if (!Helper.booleanOk(ok))
                    return ok;
            } else {
                // Replace self-signed certificate with its issuer after freeing original
                Helper.free(x);
                x = Helper.lastIssuer;
                ctx.chain.set(i - 1, x);
                ctx.last_untrusted = 0;
            }
        } else {
            // Removing the self-signed certificate from the chain safely.
            X509 removedCert = ctx.chain.remove(ctx.chain.size() - 1);
            Helper.free(removedCert);  // free any allocated memory
            // Update last_untrusted to match the new chain length
            ctx.last_untrusted = ctx.chain.size();
            num--;
            j--;
            x = ctx.chain.get(num - 1);
        }
    }

    // 2. Alternative chain building: follow the chain until a candidate is found or a self‐signed cert is reached.
    while (depth >= num) {
        if (Helper.cert_self_signed(x))
            break;
        ok = ctx.get_issuer(x);
        if (ok < 0)
            return ok;
        if (ok == 0)
            break;
        // Append the issuer to the chain
        x = Helper.lastIssuer;
        ctx.chain.add(x);
        num++;
    }

    // 3. Check trust for the current chain state.
    i = Helper.check_trust(ctx);
    if (i == Constants.X509_TRUST_REJECTED)
        break;

    retry = 0;
    // If the chain is not yet trusted and alternative chain search is allowed...
    if (i != Constants.X509_TRUST_TRUSTED &&
          ((param.flags & Constants.X509_V_FLAG_TRUSTED_FIRST) == 0) &&
          ((param.flags & Constants.X509_V_FLAG_NO_ALT_CHAINS) == 0))
    {
        // Process potential alternative chains. Do not reinitialize the chain container.
        while (j-- > 1) {
            X509 xtmp2 = ctx.chain.get(j - 1);
            ok = ctx.get_issuer(xtmp2);
            if (ok < 0)
                return ok;
            if (ok > 0) {
                // Free the temporary issuer immediately upon trust confirmation.
                Helper.free(Helper.lastIssuer);
                Helper.lastIssuer = null;
                // 4. Remove all certificates beyond the acceptable index and update last_untrusted.
                while (num > j) {
                    X509 removed = ctx.chain.remove(ctx.chain.size() - 1);
                    Helper.free(removed);
                    num--;
                }
                ctx.last_untrusted = ctx.chain.size();
                retry = 1;
                break;
            }
        }
    }
} while (retry != 0);

// 5. Final adjustment to ensure chain state integrity.
ctx.last_untrusted = ctx.chain.size();


        return ok;
    }
}
