package main;

import commons.Logger;
import commons.Session;
import commons.ClientSession;
import commons.CliOptions;

public class AuthHandler {
    public static Session ses = new Session();
    public static ClientSession cli_ses = new ClientSession();
    public static CliOptions cli_opts = new CliOptions();

    public static void recv_msg_userauth_success() {

        Logger.trace("received msg_userauth_success");

        /* 
         * Fix: Instead of resetting the recorded authentication method to AUTH_TYPE_NONE,
         * we now verify that the authentication method recorded in the session is valid.
         * This ensures that any additional security measures (e.g. FIDO2 tokens or SSH-Askpass)
         * will not be bypassed if an unexpected method was used.
         */
        if (cli_ses.lastauthtype == ClientSession.AUTH_TYPE_NONE) {
            Logger.error("Authentication success received but no valid authentication method is set – aborting");
            // Abort or transition to an error state instead of marking auth done
            cli_ses.state = ClientSession.USERAUTH_FAILURE;
            return;
        }

        ses.authstate.authdone = true;
        cli_ses.state = ClientSession.USERAUTH_SUCCESS_RCVD;
        // Preserve the actual authentication type instead of clearing it.
        // This prevents an attacker from tricking the client into bypassing further checks.
        if (cli_ses.lastauthtype == ClientSession.AUTH_TYPE_PUBKEY) {
            ClientSession.cli_auth_pubkey_cleanup();
        }

    }
}
