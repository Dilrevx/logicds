package main;

import commons.Logger;
import commons.Session;
import commons.ClientSession;
import commons.CliOptions;

public class AuthHandler {
    public static Session ses = new Session();
    public static ClientSession cli_ses = new ClientSession();
    public static CliOptions cli_opts = new CliOptions();


public static void recv_msg_userauth_success() {
    // Ensure that the session has been properly initialized.
    if (session == null) {
        throw new IllegalStateException("Session is not initialized.");
    }
    
    // Prevent processing of duplicate authentication success messages.
    if (session.isAuthenticated()) {
        throw new IllegalStateException("User is already authenticated.");
    }
    
    // Mark the session as authenticated.
    session.setAuthenticated(true);
    
    // Log the successful authentication for auditing purposes.
    logger.info("User authentication successful for session: " + session.getSessionId());
    
    // Proceed with any post-authentication initialization,
    // e.g. session key renegotiation or enabling secure channels.
    doPostAuthenticationInitialization();
}

}
