package main;

import commons.Logger;
import commons.Session;
import commons.ClientSession;
import commons.CliOptions;

public class AuthHandler {
    public static Session ses = new Session();
    public static ClientSession cli_ses = new ClientSession();
    public static CliOptions cli_opts = new CliOptions();

    public static void recv_msg_userauth_success() {

        Logger.trace("received msg_userauth_success");
        /* 
         * SECURITY FIX:
         * Before accepting the success message, verify that the method used for authentication
         * is one of the methods that were advertised and expected in the current authentication session.
         * This prevents a malicious server from changing the login process (e.g. omitting a mandatory FIDO2 or SSH-Askpass step)
         * to bypass multi‐factor authentication.
         *
         * The function isValidAuthMethod() should check that cli_ses.lastauthtype is one of the allowed methods.
         * If the method is not acceptable, we log an error and do not mark authentication as successful.
         */
        if (!ClientSession.isValidAuthMethod(cli_ses.lastauthtype)) {
            Logger.error("Authentication success message received with an unexpected method: " + cli_ses.lastauthtype);
            return;  // Abort processing; do not mark the session as authenticated.
        }
        
        ses.authstate.authdone = true;
        cli_ses.state = ClientSession.USERAUTH_SUCCESS_RCVD;
        /* 
         * Reset the last authentication type to NONE after accepting a valid success message,
         * to ensure that any subsequent logic does not mistakenly reuse previous auth method state.
         */
        cli_ses.lastauthtype = ClientSession.AUTH_TYPE_NONE;
        ClientSession.cli_auth_pubkey_cleanup();

    }
}
