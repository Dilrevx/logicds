package main;

import commons.Logger;
import commons.Session;
import commons.ClientSession;
import commons.CliOptions;

public class AuthHandler {
    public static Session ses = new Session();
    public static ClientSession cli_ses = new ClientSession();
    public static CliOptions cli_opts = new CliOptions();

    public static void recv_msg_userauth_success() {

        Logger.trace("received msg_userauth_success");

        /* 
         * SECURITY FIX: Do not accept a success message when no valid authentication method was used.
         * The "none" method is reserved and MUST NOT be used for actual authentication.
         * If the last authentication type is still AUTH_TYPE_NONE, then the authentication process 
         * did not use a proper method; abort authentication rather than accepting the success message.
         */
        if (cli_ses.lastauthtype == ClientSession.AUTH_TYPE_NONE) {
            Logger.error("Authentication success received using invalid 'none' method. Terminating session.");
            /* 
             * Terminate the session securely. The exact cleanup/disconnect function will depend on 
             * the overall design of the client code.
             */
            secure_disconnect(cli_ses);
            return;
        }

        ses.authstate.authdone = true;
        cli_ses.state = ClientSession.USERAUTH_SUCCESS_RCVD;
        ClientSession.cli_auth_pubkey_cleanup();

    }
}
