    public static void recv_msg_userauth_success() {
        Logger.trace("received msg_userauth_success");
        // Guard condition: If trivial authentication is explicitly disabled and the client is using a trivial authentication method,
        // then do not proceed with marking the authentication as successful.
        if (cli_opts.disableTrivialAuth && cli_ses.lastauthtype == ClientSession.AUTH_TYPE_TRIVIAL) {
            throw new RuntimeException("Trivial authentication method is disabled by configuration; aborting authentication success processing.");
        }
        
        ses.authstate.authdone = true;
        cli_ses.state = ClientSession.USERAUTH_SUCCESS_RCVD;
        cli_ses.lastauthtype = ClientSession.AUTH_TYPE_NONE;
        ClientSession.cli_auth_pubkey_cleanup();
    }
