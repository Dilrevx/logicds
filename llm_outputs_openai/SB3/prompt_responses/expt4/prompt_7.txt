    public static void recv_msg_userauth_success() {
        Logger.trace("received msg_userauth_success");
        
        // Verify that the authentication method used is one of the allowed ones.
        // In previous versions the auth type was reset to AUTH_TYPE_NONE, essentially losing
        // track of what method was actually used. This made additional security measures (like FIDO2 tokens)
        // impossible to enforce. Now, we require that lastauthtype contains a valid (non‐NONE) value.
        if (cli_ses.lastauthtype == ClientSession.AUTH_TYPE_NONE) {
            Logger.error("Invalid user authentication: no proper auth method was used.");
            return;
        }
        
        // If additional security measures are required (e.g. FIDO2 tokens), verify that the correct method was used.
        // Assume cli_opts.requireFIDO2 is true if the user demanded a hardware-backed auth.
        if (cli_opts.requireFIDO2 && cli_ses.lastauthtype != ClientSession.AUTH_TYPE_FIDO2) {
            Logger.error("FIDO2 authentication required but not used; rejecting login attempt.");
            return;
        }
        
        ses.authstate.authdone = true;
        cli_ses.state = ClientSession.USERAUTH_SUCCESS_RCVD;
        // Do not reset the authentication type; preserving it lets later checks enforce proper
        // post-authentication policies (e.g. detecting abuse of forwarded agents).
        ClientSession.cli_auth_pubkey_cleanup();
    }
