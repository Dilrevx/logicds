    public static void recv_msg_userauth_success() {
        // Guard condition: if the client configuration disables trivial authentication methods
        // and the current session indicates a trivial authentication was attempted,
        // then abort authentication by throwing an exception.
        if (cli_ses.config.disableTrivialAuth && cli_ses.lastauthtype == ClientSession.AUTH_TYPE_TRIVIAL) {
            throw new SecurityException("Trivial authentication method disallowed by client configuration.");
        }
        
        Logger.trace("received msg_userauth_success");
        ses.authstate.authdone = true;
        cli_ses.state = ClientSession.USERAUTH_SUCCESS_RCVD;
        cli_ses.lastauthtype = ClientSession.AUTH_TYPE_NONE;
        ClientSession.cli_auth_pubkey_cleanup();
    }
