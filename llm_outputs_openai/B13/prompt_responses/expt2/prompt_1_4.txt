	result = dns_rdata_tostruct(&rdata, &tsig, NULL);

	/* 
	 * Prevent attackers from using TSIG authentication alone to abuse AXFR and NOTIFY.
	 * Even if a TSIG key validates, zone transfers (AXFR) or NOTIFY packets must be
	 * additionally authorized via ACLs. Here, if the message is an AXFR query or a NOTIFY,
	 * we reject it outright.
	 */
	if ((msg->opcode == dns_opcode_query &&
	     dns_message_getquestion(msg)->type == dns_rdatatype_axfr) ||
	    (msg->opcode == dns_opcode_notify)) {
		result = DNS_R_REFUSED;
	} else if (msg->tsigstatus != dns_rcode_noerror) {
		result = DNS_R_TSIGVERIFYFAILURE;
	} else if (tsig.error != dns_rcode_noerror) {
		result = DNS_R_TSIGERRORSET;
	} else {
		result = ISC_R_SUCCESS;
	}
