	 * SECURITY PATCH:
	 *
	 * To prevent an attacker from bypassing TSIG authentication for
	 * AXFR (zone transfer) or NOTIFY requests, an additional ACL check
	 * is now performed.  Relying solely on TSIG keys for these types of
	 * requests is insufficient, because an attacker with the TSIG key
	 * name (even if not the actual key) could craft a packet that
	 * appears TSIG‐signed.
	 *
	 * The following check calls a hypothetical function "authorized_by_acl()"
	 * which must verify that the source (for example, by its IP address)
	 * of this request is allowed to perform the requested operation.
	 * If the request is an AXFR or a NOTIFY, and the ACL check fails,
	 * the request is rejected.
	 */
	if ((msg->opcode == dns_opcode_notify) ||
	    (msg->querytype == dns_rdatatype_axfr)) {
		if (!authorized_by_acl(msg)) {
			result = DNS_R_REFUSED;
			goto cleanup;
		}
	}
