    // Only disable OCSP verification if the certificate actually belongs to the chain being validated.
    // The NoCheck extension should only cancel the OCSP check for certificates that are part of the chain.
    if (cert.ocspNoCheckSet && verify == Commons.VERIFY_OCSP && chainContainsCertificate(chain, cert)) {
        verify = Commons.NO_VERIFY;
    }
</repair>
