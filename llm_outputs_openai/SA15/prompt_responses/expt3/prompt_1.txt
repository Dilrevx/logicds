        /*
         * The original code would accept an EAP_TLV_SUCCESS during Phase 1,
         * which can be exploited to bypass Phase 2 authentication.
         * To mitigate this attack vector, we now force a failure decision
         * when receiving an EAP_TLV_SUCCESS, regardless of the 'force_failure'
         * flag. This ensures that a success message sent out-of-turn
         * (used by an adversary to impersonate an Enterprise Wi-Fi network)
         * does not result in an authentication bypass.
         */
        if (tlvStatus == EAP_TLV_RESULT_SUCCESS) {
            resp_status = EAP_TLV_RESULT_FAILURE;
            ret.decision = EapMethodRet.DECISION_FAIL;
        } else if (tlvStatus == EAP_TLV_RESULT_FAILURE) {
            resp_status = EAP_TLV_RESULT_FAILURE;
            ret.decision = EapMethodRet.DECISION_FAIL;
        } else {
            resp_status = EAP_TLV_RESULT_FAILURE;
            ret.decision = EapMethodRet.DECISION_FAIL;
        }
