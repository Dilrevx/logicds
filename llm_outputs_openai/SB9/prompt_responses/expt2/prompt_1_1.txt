                /* Ensure that the certificate’s algorithm matches the signature algorithm in the CertificateVerify message.
                   The peer certificate must have been recorded with its signature algorithm in ssl.peerCertSigAlgo.
                   If they differ, reject the handshake. */
                if (ssl.peerCertSigAlgo != args.sigAlgo) {
                    Wslib.WOLFSSL_MSG("Signature algorithm mismatch between certificate message and certificate_verify message");
                    return CommonsConstants.SIG_VERIFY_E;
                }

                if (args.sigAlgo == CommonsConstants.ed25519_sa_algo && !ssl.peerEd25519KeyPresent) {
                    Wslib.WOLFSSL_MSG("Peer sent ED25519 signature but ED25519 certificate not found");
                    return CommonsConstants.SIG_VERIFY_E;
                }
                if (args.sigAlgo == CommonsConstants.ed448_sa_algo && !ssl.peerEd448KeyPresent) {
                    Wslib.WOLFSSL_MSG("Peer sent ED448 signature but ED448 certificate not found");
                    return CommonsConstants.SIG_VERIFY_E;
                }
                if (args.sigAlgo == CommonsConstants.ecc_dsa_sa_algo && !ssl.peerEccDsaKeyPresent) {
                    Wslib.WOLFSSL_MSG("Peer sent ECC signature but ECC certificate not found");
                    return CommonsConstants.SIG_VERIFY_E;
                }
                if (args.sigAlgo == CommonsConstants.falcon_level1_sa_algo && !ssl.peerFalconKeyPresent) {
                    Wslib.WOLFSSL_MSG("Peer sent Falcon Level 1 signature but matching certificate not found");
                    return CommonsConstants.SIG_VERIFY_E;
                }
                if (args.sigAlgo == CommonsConstants.falcon_level5_sa_algo && !ssl.peerFalconKeyPresent) {
                    Wslib.WOLFSSL_MSG("Peer sent Falcon Level 5 signature but matching certificate not found");
                    return CommonsConstants.SIG_VERIFY_E;
                }
                if (args.sigAlgo == CommonsConstants.rsa_sa_algo) {
                    Wslib.WOLFSSL_MSG("Peer sent PKCS#1.5 signature algorithm which is not allowed for the certificate");
                    return CommonsConstants.INVALID_PARAMETER;
                }
                if (args.sigAlgo == CommonsConstants.rsa_pss_sa_algo && (ssl.peerRsaKey == null || !ssl.peerRsaKeyPresent)) {

                    return CommonsConstants.SIG_VERIFY_E;
