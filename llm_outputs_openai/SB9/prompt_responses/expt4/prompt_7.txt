    public static int DoTls13CertificateVerify(TlsContext ssl, byte[] input, int[] inOutIdx, int totalSz) {
        int ret = 0;
        Buffer sig = ssl.buffers.sig;
        Dcv13Args args = new Dcv13Args();
        ssl.options.asyncState = CommonsConstants.TLS_ASYNC_BEGIN;
        args.hashAlgo = CommonsConstants.sha_mac;
        args.sigAlgo = CommonsConstants.anonymous_sa_algo;
        args.idx   = inOutIdx[0];
        args.begin = inOutIdx[0];

        switch (ssl.options.asyncState) {
            case CommonsConstants.TLS_ASYNC_BEGIN:
                ssl.options.asyncState = CommonsConstants.TLS_ASYNC_BUILD;
                // Intentional fall‐through.
            case CommonsConstants.TLS_ASYNC_BUILD:
                if ((args.idx - args.begin) + CommonsConstants.ENUM_LEN + CommonsConstants.ENUM_LEN > totalSz) {
                    return CommonsConstants.BUFFER_ERROR;
                }
                ret = CommonUtils.DecodeTls13SigAlg(input, args);
                if (ret < 0)
                    return ret;
                args.idx += CommonsConstants.OPAQUE16_LEN;
                if ((args.idx - args.begin) + CommonsConstants.OPAQUE16_LEN > totalSz)
                    return CommonsConstants.BUFFER_ERROR;
                args.sz = CommonUtils.ato16(input, args.idx);
                args.idx += CommonsConstants.OPAQUE16_LEN;
                if ((args.idx - args.begin) + args.sz > totalSz || args.sz > CommonsConstants.ENCRYPT_LEN)
                    return CommonsConstants.BUFFER_ERROR;

                // Do not allow legacy RSA PKCS#1.5 signature algorithm.
                if (args.sigAlgo == CommonsConstants.rsa_sa_algo) {
                    Wslib.WOLFSSL_MSG("Peer sent PKCS#1.5 algo but not in certificate");
                    return CommonsConstants.INVALID_PARAMETER;
                }

                /*
                 * New check: ensure that the signature algorithm indicated in the
                 * Certificate Verify message is consistent with the certificate's public key type.
                 * This prevents an attacker from choosing an algorithm that does not match what was
                 * offered in the certificate message.
                 */
                int expectedSigAlgo = -1;
                if (ssl.peerEd25519KeyPresent) {
                    expectedSigAlgo = CommonsConstants.ed25519_sa_algo;
                } else if (ssl.peerEd448KeyPresent) {
                    expectedSigAlgo = CommonsConstants.ed448_sa_algo;
                } else if (ssl.peerEccDsaKeyPresent) {
                    expectedSigAlgo = CommonsConstants.ecc_dsa_sa_algo;
                } else if (ssl.peerFalconKeyPresent) {
                    // Accept either Falcon level 1 or level 5.
                    if (args.sigAlgo != CommonsConstants.falcon_level1_sa_algo &&
                        args.sigAlgo != CommonsConstants.falcon_level5_sa_algo) {
                        Wslib.WOLFSSL_MSG("Mismatch: Certificate uses Falcon but Certificate Verify message uses different signature algorithm");
                        return CommonsConstants.SIG_VERIFY_E;
                    }
                    // For Falcon, the provided algorithm is acceptable.
                    expectedSigAlgo = args.sigAlgo;
                } else if (ssl.peerRsaKey != null && ssl.peerRsaKeyPresent) {
                    // Only RSA-PSS is allowed for TLS 1.3.
                    expectedSigAlgo = CommonsConstants.rsa_pss_sa_algo;
                } else {
                    Wslib.WOLFSSL_MSG("No valid certificate key present for signature verification");
                    return CommonsConstants.SIG_VERIFY_E;
                }
                if (expectedSigAlgo != -1 && expectedSigAlgo != args.sigAlgo) {
                    Wslib.WOLFSSL_MSG("Mismatch between certificate's public key type and the signature algorithm in Certificate Verify message");
                    return CommonsConstants.SIG_VERIFY_E;
                }

                // The rest of the processing (e.g. signature verification) would occur here.
                return 0;
            default:
                return 0;
        }
    }
