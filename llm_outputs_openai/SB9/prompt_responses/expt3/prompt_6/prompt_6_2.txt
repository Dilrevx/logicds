if (args.sigAlgo == CommonsConstants.ed25519_sa_algo) {
    if (!ssl.peerEd25519KeyPresent) {
        Wslib.WOLFSSL_MSG("Peer sent ED25519 sig but not ED25519 cert");
        return CommonsConstants.SIG_VERIFY_E;
    }
}
else if (args.sigAlgo == CommonsConstants.ed448_sa_algo) {
    if (!ssl.peerEd448KeyPresent) {
        Wslib.WOLFSSL_MSG("Peer sent ED448 sig but not ED448 cert");
        return CommonsConstants.SIG_VERIFY_E;
    }
}
else if (args.sigAlgo == CommonsConstants.ecc_dsa_sa_algo) {
    if (!ssl.peerEccDsaKeyPresent) {
        Wslib.WOLFSSL_MSG("Peer sent ECC sig but not ECC cert");
        return CommonsConstants.SIG_VERIFY_E;
    }
}
else if (args.sigAlgo == CommonsConstants.falcon_level1_sa_algo) {
    if (!ssl.peerFalconKeyPresent) {
        Wslib.WOLFSSL_MSG("Peer sent Falcon Level 1 sig but not Falcon cert");
        return CommonsConstants.SIG_VERIFY_E;
    }
}
else if (args.sigAlgo == CommonsConstants.falcon_level5_sa_algo) {
    if (!ssl.peerFalconKeyPresent) {
        Wslib.WOLFSSL_MSG("Peer sent Falcon Level 5 sig but not Falcon cert");
        return CommonsConstants.SIG_VERIFY_E;
    }
}
else if (args.sigAlgo == CommonsConstants.rsa_sa_algo) {
    /* Reject PKCS#1.5 signatures outright since they are not allowed */
    Wslib.WOLFSSL_MSG("Peer sent PKCS#1.5 sig algo which is unsupported");
    return CommonsConstants.INVALID_PARAMETER;
}
else if (args.sigAlgo == CommonsConstants.rsa_pss_sa_algo) {
    if (ssl.peerRsaKey == NULL || !ssl.peerRsaKeyPresent) {
        Wslib.WOLFSSL_MSG("Peer sent RSA-PSS sig but no valid RSA cert is present");
        return CommonsConstants.SIG_VERIFY_E;
    }
}
else {
    Wslib.WOLFSSL_MSG("Peer sent unknown signature algorithm");
    return CommonsConstants.INVALID_PARAMETER;
}
