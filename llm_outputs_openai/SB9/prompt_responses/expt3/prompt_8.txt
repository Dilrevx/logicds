boolean validKeyFound = false;
switch (args.sigAlgo) {
    case CommonsConstants.ed25519_sa_algo:
        if (ssl.peerEd25519KeyPresent) {
            validKeyFound = true;
        } else {
            Wslib.WOLFSSL_MSG("Peer sent ED25519 sig but not ED25519 cert");
        }
        break;
    case CommonsConstants.ed448_sa_algo:
        if (ssl.peerEd448KeyPresent) {
            validKeyFound = true;
        } else {
            Wslib.WOLFSSL_MSG("Peer sent ED448 sig but not ED448 cert");
        }
        break;
    case CommonsConstants.ecc_dsa_sa_algo:
        if (ssl.peerEccDsaKeyPresent) {
            validKeyFound = true;
        } else {
            Wslib.WOLFSSL_MSG("Peer sent ECC sig but not ECC cert");
        }
        break;
    case CommonsConstants.falcon_level1_sa_algo:
        if (ssl.peerFalconKeyPresent) {
            validKeyFound = true;
        } else {
            Wslib.WOLFSSL_MSG("Peer sent Falcon Level 1 sig but not Falcon cert");
        }
        break;
    case CommonsConstants.falcon_level5_sa_algo:
        if (ssl.peerFalconKeyPresent) {
            validKeyFound = true;
        } else {
            Wslib.WOLFSSL_MSG("Peer sent Falcon Level 5 sig but not Falcon cert");
        }
        break;
    case CommonsConstants.rsa_sa_algo:
        Wslib.WOLFSSL_MSG("Peer sent PKCS#1.5 algo but not in certificate");
        return CommonsConstants.INVALID_PARAMETER;
    case CommonsConstants.rsa_pss_sa_algo:
        if (ssl.peerRsaKey != null && ssl.peerRsaKeyPresent) {
            validKeyFound = true;
        } else {
            Wslib.WOLFSSL_MSG("Peer sent RSA sig but not RSA cert");
        }
        break;
    default:
        Wslib.WOLFSSL_MSG("Unsupported signature algorithm");
        break;
}
if (!validKeyFound) {
    Wslib.WOLFSSL_MSG("Peer's signature algorithm does not match the certificate");
    return CommonsConstants.SIG_VERIFY_E;
}
