package main;

import commons.*;

public class Tls13CertificateVerify {
    public static int DoTls13CertificateVerify(TlsContext ssl, byte[] input, int[] inOutIdx, int totalSz) {
        int ret = 0;
        Buffer sig = ssl.buffers.sig;
        Dcv13Args args = new Dcv13Args();
        ssl.options.asyncState = CommonsConstants.TLS_ASYNC_BEGIN;
        args.hashAlgo = CommonsConstants.sha_mac;
        args.sigAlgo = CommonsConstants.anonymous_sa_algo;
        args.idx = inOutIdx[0];
        args.begin = inOutIdx[0];
        switch (ssl.options.asyncState) {
            case CommonsConstants.TLS_ASYNC_BEGIN:
                ssl.options.asyncState = CommonsConstants.TLS_ASYNC_BUILD;
            case CommonsConstants.TLS_ASYNC_BUILD:
                if ((args.idx - args.begin) + CommonsConstants.ENUM_LEN + CommonsConstants.ENUM_LEN > totalSz)
                    return CommonsConstants.BUFFER_ERROR;
                ret = CommonUtils.DecodeTls13SigAlg(input, args);
                if (ret < 0)
                    return ret;
                args.idx += CommonsConstants.OPAQUE16_LEN;
                if ((args.idx - args.begin) + CommonsConstants.OPAQUE16_LEN > totalSz)
                    return CommonsConstants.BUFFER_ERROR;
                args.sz = CommonUtils.ato16(input, args.idx);
                args.idx += CommonsConstants.OPAQUE16_LEN;
                if ((args.idx - args.begin) + args.sz > totalSz || args.sz > CommonsConstants.ENCRYPT_LEN)
                    return CommonsConstants.BUFFER_ERROR;

                int validSigCert = 0;  /* Flag to indicate a valid certificate-key match for the given sigAlgo */

                if (args.sigAlgo == CommonsConstants.ed25519_sa_algo) {
                    Wslib.WOLFSSL_MSG("Checking ED25519: verifying certificate contains ED25519 key");
                    if (ssl.peerEd25519KeyPresent) {
                        validSigCert = 1;
                    } else {
                        Wslib.WOLFSSL_MSG("Peer sent ED25519 signature but certificate lacks ED25519 key");
                    }
                } else if (args.sigAlgo == CommonsConstants.ed448_sa_algo) {
                    Wslib.WOLFSSL_MSG("Checking ED448: verifying certificate contains ED448 key");
                    if (ssl.peerEd448KeyPresent) {
                        validSigCert = 1;
                    } else {
                        Wslib.WOLFSSL_MSG("Peer sent ED448 signature but certificate lacks ED448 key");
                    }
                } else if (args.sigAlgo == CommonsConstants.ecc_dsa_sa_algo) {
                    Wslib.WOLFSSL_MSG("Checking ECC DSA: verifying certificate contains ECC key");
                    if (ssl.peerEccDsaKeyPresent) {
                        validSigCert = 1;
                    } else {
                        Wslib.WOLFSSL_MSG("Peer sent ECC signature but certificate lacks ECC key");
                    }
                } else if (args.sigAlgo == CommonsConstants.falcon_level1_sa_algo) {
                    Wslib.WOLFSSL_MSG("Checking Falcon Level 1: verifying certificate contains Falcon Level 1 key");
                    if (ssl.peerFalconKeyPresent) {
                        validSigCert = 1;
                    } else {
                        Wslib.WOLFSSL_MSG("Peer sent Falcon Level 1 signature but certificate mismatch");
                    }
                } else if (args.sigAlgo == CommonsConstants.falcon_level5_sa_algo) {
                    Wslib.WOLFSSL_MSG("Checking Falcon Level 5: verifying certificate contains Falcon Level 5 key");
                    if (ssl.peerFalconKeyPresent) {
                        validSigCert = 1;
                    } else {
                        Wslib.WOLFSSL_MSG("Peer sent Falcon Level 5 signature but certificate mismatch");
                    }
                } else if (args.sigAlgo == CommonsConstants.rsa_sa_algo) {
                    Wslib.WOLFSSL_MSG("Peer sent PKCS#1.5 RSA algorithm, which is not permitted in TLS 1.3");
                    return CommonsConstants.INVALID_PARAMETER;
                } else if (args.sigAlgo == CommonsConstants.rsa_pss_sa_algo) {
                    Wslib.WOLFSSL_MSG("Checking RSA-PSS: verifying certificate contains RSA key");
                    if (ssl.peerRsaKey != null && ssl.peerRsaKeyPresent) {
                        validSigCert = 1;
                    } else {
                        Wslib.WOLFSSL_MSG("Peer sent RSA-PSS signature but certificate lacks valid RSA key");
                    }
                } else {
                    Wslib.WOLFSSL_MSG("Unknown signature algorithm provided");
                }

                if (!validSigCert) {
                    Wslib.WOLFSSL_MSG("Certificate key does not match the provided signature algorithm. Aborting handshake.");
                    return CommonsConstants.SIG_VERIFY_E;
                }

                return 0;
            default:
                return 0;
        }
    }
}
