package main;

import commons.*;

public class Tls13CertificateVerify {

    public static int DoTls13CertificateVerify(TlsContext ssl, byte[] input, int[] inOutIdx, int totalSz) {
        int ret = 0;
        Buffer sig = ssl.buffers.sig;
        Dcv13Args args = new Dcv13Args();

        // Set up initial async state and parameters.
        ssl.options.asyncState = CommonsConstants.TLS_ASYNC_BEGIN;
        args.hashAlgo = CommonsConstants.sha_mac;
        args.sigAlgo = CommonsConstants.anonymous_sa_algo;
        args.idx   = inOutIdx[0];
        args.begin = inOutIdx[0];

        switch (ssl.options.asyncState) {
            case CommonsConstants.TLS_ASYNC_BEGIN:
                ssl.options.asyncState = CommonsConstants.TLS_ASYNC_BUILD;
            case CommonsConstants.TLS_ASYNC_BUILD:
                // Check boundaries for reading signature algorithm and length fields.
                if ((args.idx - args.begin) + CommonsConstants.ENUM_LEN + CommonsConstants.ENUM_LEN > totalSz)
                    return CommonsConstants.BUFFER_ERROR;
                
                // Decode the TLS 1.3 signature algorithm.
                ret = CommonUtils.DecodeTls13SigAlg(input, args);
                if (ret < 0)
                    return ret;
                
                args.idx += CommonsConstants.OPAQUE16_LEN;
                
                if ((args.idx - args.begin) + CommonsConstants.OPAQUE16_LEN > totalSz)
                    return CommonsConstants.BUFFER_ERROR;
                
                // Obtain the signature data size.
                args.sz = CommonUtils.ato16(input, args.idx);
                args.idx += CommonsConstants.OPAQUE16_LEN;
                
                if ((args.idx - args.begin) + args.sz > totalSz || args.sz > CommonsConstants.ENCRYPT_LEN)
                    return CommonsConstants.BUFFER_ERROR;
                
                // Use a unified validation flag to ensure that the certificate matches the signature algorithm.
                boolean validCert = false;
                
                if (args.sigAlgo == CommonsConstants.ed25519_sa_algo) {
                    Wslib.WOLFSSL_MSG("Peer used ED25519 signature algorithm.");
                    validCert = ssl.peerEd25519KeyPresent;
                } else if (args.sigAlgo == CommonsConstants.ed448_sa_algo) {
                    Wslib.WOLFSSL_MSG("Peer used ED448 signature algorithm.");
                    validCert = ssl.peerEd448KeyPresent;
                } else if (args.sigAlgo == CommonsConstants.ecc_dsa_sa_algo) {
                    Wslib.WOLFSSL_MSG("Peer used ECC DSA signature algorithm.");
                    validCert = ssl.peerEccDsaKeyPresent;
                } else if (args.sigAlgo == CommonsConstants.falcon_level1_sa_algo) {
                    Wslib.WOLFSSL_MSG("Peer used Falcon Level 1 signature algorithm.");
                    validCert = ssl.peerFalconKeyPresent;
                } else if (args.sigAlgo == CommonsConstants.falcon_level5_sa_algo) {
                    Wslib.WOLFSSL_MSG("Peer used Falcon Level 5 signature algorithm.");
                    validCert = ssl.peerFalconKeyPresent;
                } else if (args.sigAlgo == CommonsConstants.rsa_pss_sa_algo) {
                    Wslib.WOLFSSL_MSG("Peer used RSA PSS signature algorithm.");
                    validCert = (ssl.peerRsaKey != null && ssl.peerRsaKeyPresent);
                } else if (args.sigAlgo == CommonsConstants.rsa_sa_algo) {
                    // Immediately reject RSA PKCS#1.5 signature algorithm, which is not allowed in TLS 1.3.
                    Wslib.WOLFSSL_MSG("Peer sent PKCS#1.5 signature algorithm, which is invalid in TLS 1.3.");
                    return CommonsConstants.INVALID_PARAMETER;
                } else {
                    // If the signature algorithm is unsupported, reject it.
                    Wslib.WOLFSSL_MSG("Unsupported signature algorithm received.");
                    return CommonsConstants.SIG_VERIFY_E;
                }
                
                // Enforce the final check – if the certificate key does not correspond to the signature algorithm, fail.
                if (!validCert) {
                    Wslib.WOLFSSL_MSG("Certificate does not match the signature algorithm used.");
                    return CommonsConstants.SIG_VERIFY_E;
                }
                return 0;
                
            default:
                return 0;
        }
    }

}
