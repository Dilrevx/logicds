        }

        p = OSSL_PARAM_locate_const(params, OSSL_CIPHER_PARAM_IVLEN);
        if (p != NULL && OSSL_PARAM_get_size_t(p, &iv_length)) {
            local_params[count++] = OSSL_PARAM_construct_size_t(OSSL_CIPHER_PARAM_IVLEN, &iv_length);
        }

        /* Terminate the parameters array */
        local_params[count] = OSSL_PARAM_construct_end();
