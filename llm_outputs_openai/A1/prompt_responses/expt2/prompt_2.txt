        /* Prepare a temporary parameter array for up to 2 parameters:

         * The array is terminated with OSSL_PARAM_END.
         */
        OSSL_PARAM upd_params[3] = { OSSL_PARAM_END, OSSL_PARAM_END, OSSL_PARAM_END };
        const OSSL_PARAM *p = NULL;

        /* Locate and copy the key length parameter, if present */
        p = OSSL_PARAM_locate_const(params, OSSL_CIPHER_PARAM_KEYLEN);
        if (p != NULL)
            upd_params[0] = *p;

        /* Locate and copy the IV length parameter.
         * Some implementations use "ivlen" as a synonymous key.
         */
        p = OSSL_PARAM_locate_const(params, OSSL_CIPHER_PARAM_IVLEN);
        if (p == NULL)
            p = OSSL_PARAM_locate_const(params, "ivlen");
        if (p != NULL)
            upd_params[1] = *p;

        if (!EVP_CIPHER_CTX_set_params(ctx, upd_params)) {

            return 0;
