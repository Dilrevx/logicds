        OSSL_PARAM *ivlen_param = ossl_param_find(params, OSSL_CIPHER_PARAM_IVLEN);
        if (ivlen_param == NULL)
            ivlen_param = ossl_param_find(params, OSSL_CIPHER_PARAM_AEAD_IVLEN);
        if (ivlen_param != NULL) {
            local_params[param_set++] = *ivlen_param;
