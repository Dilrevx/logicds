    /*
     * Process the OSSL_PARAM array (if provided) before establishing the key
     * and IV lengths. In previous implementations the parameters were applied
     * after the key and IV were set up, which could lead to truncation or overruns,
     * particularly for the IV. By applying them first we ensure that any modifications
     * (including length changes) are properly reflected when obtaining the key
     * and IV lengths below.
     */
    if (params != NULL) {
        if (!EVP_CIPHER_CTX_set_params(ctx, params))
            return 0;
        /* Clear params so they are not processed again by the cipher->einit() */
        params = NULL;
