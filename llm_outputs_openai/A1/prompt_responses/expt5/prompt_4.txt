     * OSSL_PARAM array (or their AEAD alternative) are applied to the context

    if (enc && params != NULL) {
        OSSL_PARAM local_params[3] = { OSSL_PARAM_END, OSSL_PARAM_END, OSSL_PARAM_END };
        const OSSL_PARAM *p;

        /* Look for the key length parameter */
        p = OSSL_PARAM_locate_const(params, OSSL_CIPHER_PARAM_KEYLEN);
        if (p != NULL)
            local_params[0] = *p;

        /* Look for the IV length parameter, or its AEAD equivalent */
        p = OSSL_PARAM_locate_const(params, OSSL_CIPHER_PARAM_IVLEN);
        if (p == NULL)
            p = OSSL_PARAM_locate_const(params, OSSL_CIPHER_PARAM_AEAD_IVLEN);
        if (p != NULL)
            local_params[1] = *p;

        if (!EVP_CIPHER_CTX_set_params(ctx, local_params)) {
            ERR_raise(ERR_LIB_EVP, EVP_R_INVALID_LENGTH);
            return 0;
