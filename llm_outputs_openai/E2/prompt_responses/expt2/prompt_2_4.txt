  // Replay Attack Prevention based on Sequence Number.
  // Assume that:
  //   - nas_msg.hdr.sequence_number is the sequence number provided in the message.
  //   - context.rx_count is the expected receive counter.
  //   - MAX_SEQ_NUM is defined as the maximum sequence number (e.g., 0xFF).
  //   - registration_complete is a flag indicating that the security context has
  //     been set up (registration complete).
  // If the sequence number is not the maximum allowed and is less than the expected
  // sequence number, then it is treated as a replay and rejected.
  // ---------------------------------------------------------------------------
  const uint8_t MAX_SEQ_NUM = 0xFF;
  uint8_t expected_seq = context.rx_count; // derived expected count from current context
  if ((nas_msg.hdr.sequence_number != MAX_SEQ_NUM) &&
      (nas_msg.hdr.sequence_number < expected_seq))
  {
    logger.warning("Potential NAS command replay detected: received sequence %u, expected at least %u",
                   nas_msg.hdr.sequence_number, expected_seq);
    return SRSRAN_ERROR;
  }

  // ---------------------------------------------------------------------------
  // Initialize a flag to track which security header type is processed.
  // '-1' indicates an invalid/unset value.
  // ---------------------------------------------------------------------------
  int current_msg_header_flag = -1;

  // Process the message according to its security header type.
  switch (nas_msg.hdr.security_header_type)
  {
    case nas_5gs_hdr::security_header_type_opts::plain_5gs_nas_message:
      // For plain NAS messages, ensure that they are not accepted after registration.
      // Once a security context is established, plain messages might be replayed to
      // trigger NAS signaling commands.
      if (registration_complete)
      {
        logger.error("Rejecting plain NAS message after registration complete");
        return SRSRAN_ERROR;
      }
      current_msg_header_flag = 0; // flag indicating plain message processed
      break;

    case nas_5gs_hdr::security_header_type_opts::integrity_protected:
      // Verify integrity of the message.
      if (!integrity_check(pdu.get()))
      {
        logger.error("Not handling NAS message with integrity check error");
        return SRSRAN_ERROR;
      }
      current_msg_header_flag = 1; // flag for integrity protected message
      break;

    case nas_5gs_hdr::security_header_type_opts::integrity_protected_and_ciphered:
      if (!integrity_check(pdu.get()))
      {
        logger.error("Not handling NAS message with integrity check error");
        return SRSRAN_ERROR;
      }
      else
      {
        // Message passes integrity verification; proceed with decryption.
        cipher_decrypt(pdu.get());
        current_msg_header_flag = 2; // flag for integrity protected and ciphered message
      }
      break;

    case nas_5gs_hdr::security_header_type_opts::integrity_protected_with_new_5G_nas_context:
      current_msg_header_flag = 3; // new NAS context with integrity protection
      break;

    case nas_5gs_hdr::security_header_type_opts::integrity_protected_and_ciphered_with_new_5G_nas_context:
      logger.error("Rejecting integrity protected and ciphered NAS message with new 5G NAS context");
      return SRSRAN_ERROR;

    default:
      logger.error("Not handling NAS message with unknown security header");
      break;
  }
