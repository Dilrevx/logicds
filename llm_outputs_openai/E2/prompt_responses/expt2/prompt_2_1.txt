  // Initialize a flag for the current message header status; -1 indicates not yet set.
  int current_security_flag = -1;

  // Assume these helper functions/definitions are provided elsewhere:
  // get_expected_seq_num() returns the expected incoming NAS message sequence number.
  // security_context_established() returns true if a security context has been set.
  // MAX_SEQ_NUM is the maximum valid sequence number value.
  const uint32_t MAX_SEQ_NUM = 0xFFFFFFFF;
  uint32_t expected_seq_num = get_expected_seq_num();

  if (nas_msg.unpack_outer_hdr(pdu) != SRSRAN_SUCCESS) {
    logger.error("Unable to unpack outer NAS header");
    return SRSRAN_ERROR;
  }

  // Check the sequence number to prevent replay attacks.
  // If the incoming sequence number is not the maximum allowed value and is less
  // than the expected count, treat it as a potential replay.
  uint32_t incoming_seq_num = nas_msg.hdr.sequence_number;
  if ((incoming_seq_num != MAX_SEQ_NUM) && (incoming_seq_num < expected_seq_num)) {
    logger.warn("Potential NAS replay attack detected: received seq %u, expected at least %u", incoming_seq_num, expected_seq_num);
    return SRSRAN_ERROR;
  }

  switch (nas_msg.hdr.security_header_type) {
    case nas_5gs_hdr::security_header_type_opts::plain_5gs_nas_message:
      // If a security context is already established, reject plain NAS messages
      // to avoid replaying previous valid commands after registration.
      if (security_context_established()) {
        logger.error("Rejecting plain NAS message because a security context is already established");
        return SRSRAN_ERROR;
      }
      current_security_flag = 0;
      break;

    case nas_5gs_hdr::security_header_type_opts::integrity_protected:
      if (!integrity_check(pdu.get())) {
        logger.error("Not handling NAS message with integrity check error");
        return SRSRAN_ERROR;
      }
      current_security_flag = 1;
      break;

    case nas_5gs_hdr::security_header_type_opts::integrity_protected_and_ciphered:
      if (!integrity_check(pdu.get())) {
        logger.error("Not handling NAS message with integrity check error");
        return SRSRAN_ERROR;
      } else {
        cipher_decrypt(pdu.get());
      }
      current_security_flag = 2;
      break;

    case nas_5gs_hdr::security_header_type_opts::integrity_protected_with_new_5G_nas_context:
      // Even though this case does not verify integrity here,
      // we update the flag to keep processing consistency.
      current_security_flag = 3;
      break;

    case nas_5gs_hdr::security_header_type_opts::integrity_protected_and_ciphered_with_new_5G_nas_context:
      // This message type is explicitly rejected.
      return SRSRAN_ERROR;

    default:
      logger.error("Not handling NAS message with unknown security header");
      return SRSRAN_ERROR;
  }
