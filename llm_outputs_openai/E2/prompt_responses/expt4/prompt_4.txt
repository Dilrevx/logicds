  if (nas_msg.unpack_outer_hdr(pdu) != SRSRAN_SUCCESS) {
    logger.error("Unable to unpack outer NAS header");

  // Initialize a flag for the current NAS header type.
  // Using an invalid default value helps track message processing.
  int cur_sec_header_type = -1;

  // ----------------------------------------------------------------------------
  // Replay-Attack Prevention: Check message sequence number
  // ----------------------------------------------------------------------------
  // It is assumed that:
  // - nas_msg.hdr.sequence_number holds the sequence number of the incoming message.
  // - this->expected_seq is the expected sequence number maintained by the NAS context.
  // - MAX_SEQ_VAL is a pre-defined constant representing the maximum valid sequence number,
  //   allowing wrap-around scenarios.
  //
  // If the incoming sequence number is not the maximum allowed value and is less than the
  // expected value, then this is treated as a replay attack.
  if ((nas_msg.hdr.sequence_number != MAX_SEQ_VAL) &&
      (nas_msg.hdr.sequence_number < this->expected_seq)) {
    logger.warn("NAS message replay detected: expected sequence number %d, received %d",
                this->expected_seq, nas_msg.hdr.sequence_number);
    return SRSRAN_ERROR;
  }

  // ----------------------------------------------------------------------------
  // Process based on the security header type
  // ----------------------------------------------------------------------------
  switch (nas_msg.hdr.security_header_type) {
    // For plain NAS messages, we reject the message if a secure context has already
    // been established (i.e., registration is complete). This prevents replay of prior
    // valid commands.
    case nas_5gs_hdr::security_header_type_opts::plain_5gs_nas_message:
      if (this->registration_complete) {
        logger.error("Rejecting plain NAS message after registration complete (possible replay attack)");
        return SRSRAN_ERROR;
      }
      cur_sec_header_type = nas_5gs_hdr::security_header_type_opts::plain_5gs_nas_message;
      break;

    // Integrity protected messages: verify integrity before proceeding.
    case nas_5gs_hdr::security_header_type_opts::integrity_protected:
      if (integrity_check(pdu.get()) == false) {
        logger.error("Not handling NAS message with integrity check error");
        return SRSRAN_ERROR;
      }
      cur_sec_header_type = nas_5gs_hdr::security_header_type_opts::integrity_protected;
      break;

    // Integrity protected and ciphered messages: check integrity and decrypt.
    case nas_5gs_hdr::security_header_type_opts::integrity_protected_and_ciphered:
      if (integrity_check(pdu.get()) == false) {
        logger.error("Not handling NAS message with integrity check error");
        return SRSRAN_ERROR;
      } else {
        cipher_decrypt(pdu.get());
      }
      cur_sec_header_type = nas_5gs_hdr::security_header_type_opts::integrity_protected_and_ciphered;
      break;

    // Messages introducing a new 5G NAS context:
    case nas_5gs_hdr::security_header_type_opts::integrity_protected_with_new_5G_nas_context:
      cur_sec_header_type = nas_5gs_hdr::security_header_type_opts::integrity_protected_with_new_5G_nas_context;
      break;

    // For messages that are integrity protected and ciphered with a new 5G NAS context,
    // the repair immediately rejects them.
    case nas_5gs_hdr::security_header_type_opts::integrity_protected_and_ciphered_with_new_5G_nas_context:
      return SRSRAN_ERROR;

    default:
      logger.error("Not handling NAS message with unknown security header");
      return SRSRAN_ERROR;
  }
