package main;

import commons.*;

public class CertImporter {

        result = GnutlsUtils.asn1_der_decoding(cert.cert, cert.der.data, cert.der.size, null);
        if (result < 0) {
            /* Decoding failed: revert expanded flag and free allocated DER data to avoid leaving
               the certificate in an inconsistent state */
            cert.expanded = false;
            GnutlsUtils.gnutls_assert();
            GnutlsUtils._gnutls_free_datum(cert.der);
            return result;
        }
        return result;

}
