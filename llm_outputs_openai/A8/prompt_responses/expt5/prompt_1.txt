    /*
     * Only process the first-kex packet if both sides agree that it should be used.
     * This prevents a malicious client from forcing the use of the first-kex packet
     * when the server did not intend to send one.
     */
    if (session->send_first_kex_follows && first_kex_packet_follows) {
        char **client_methods = crypto->client_kex.methods;
        char **server_methods = crypto->server_kex.methods;
        
        /* Check that the method arrays are valid to avoid null pointer dereference */
        if (!client_methods || !server_methods ||
            !client_methods[SSH_KEX] || !server_methods[SSH_KEX] ||
            !client_methods[SSH_HOSTKEYS] || !server_methods[SSH_HOSTKEYS]) {
            SSH_LOG(SSH_LOG_ERROR, "KEX method arrays are not properly initialized.");
            session->first_kex_follows_guess_wrong = 1;
        } else {
            session->first_kex_follows_guess_wrong =
                cmp_first_kex_algo(client_methods[SSH_KEX],
                                   server_methods[SSH_KEX]) ||
                cmp_first_kex_algo(client_methods[SSH_HOSTKEYS],
                                   server_methods[SSH_HOSTKEYS]);
            SSH_LOG(SSH_LOG_DEBUG, "The initial guess was %s.",
                    session->first_kex_follows_guess_wrong ? "wrong" : "right");
        }
    }
