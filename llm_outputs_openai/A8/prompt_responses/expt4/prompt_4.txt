/* Helper function to check if a given key exchange algorithm supports strict mode.
   This function checks if the algorithm is one of a set of predefined secure (strict) methods. */
static int strict_kex_supported(const char *algo) {
    if (algo == NULL)
        return 0;

    /* List of key exchange algorithms that support strict security enforcement.
       This list can be adjusted to include all algorithms validated for strict mode. */
    const char *strict_algos[] = {
        "chacha20-poly1305@openssh.com",
        "aes256-gcm@openssh.com",
        "aes128-gcm@openssh.com",
        "diffie-hellman-group-exchange-sha256",
        "diffie-hellman-group18-sha512",
        "diffie-hellman-group16-sha512",
        "diffie-hellman-group14-sha256",
        NULL
    };

    for (int i = 0; strict_algos[i] != NULL; i++) {
        if (strcmp(algo, strict_algos[i]) == 0)
            return 1;
    }
    return 0;
}

if (first_kex_packet_follows || session->send_first_kex_follows) {
    char **client_methods = crypto->client_kex.methods;
    char **server_methods = crypto->server_kex.methods;
    session->first_kex_follows_guess_wrong =
        cmp_first_kex_algo(client_methods[SSH_KEX],
                           server_methods[SSH_KEX]) ||
        cmp_first_kex_algo(client_methods[SSH_HOSTKEYS],
                           server_methods[SSH_HOSTKEYS]);

    /* Added extra verification for strict key exchange enforcement */
    if (server_methods && server_methods[SSH_KEX]) {
        if (strict_kex_supported(server_methods[SSH_KEX])) {
            session->strict_kex_enabled = 1;
            SSH_LOG(SSH_LOG_DEBUG, "Strict key exchange enabled based on server method.");
        } else {
            session->strict_kex_enabled = 0;
            SSH_LOG(SSH_LOG_DEBUG, "Server method does not support strict key exchange.");
        }
    } else if (client_methods && client_methods[SSH_KEX]) {
        if (strict_kex_supported(client_methods[SSH_KEX])) {
            session->strict_kex_enabled = 1;
            SSH_LOG(SSH_LOG_DEBUG, "Strict key exchange enabled based on client method.");
        } else {
            session->strict_kex_enabled = 0;
            SSH_LOG(SSH_LOG_DEBUG, "Client method does not support strict key exchange.");
        }
    } else {
        session->strict_kex_enabled = 0;
        SSH_LOG(SSH_LOG_WARN, "Unable to determine key exchange method for strict verification.");
    }

    SSH_LOG(SSH_LOG_DEBUG, "The initial guess was %s.",
            session->first_kex_follows_guess_wrong ? "wrong" : "right");
}
