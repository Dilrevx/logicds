    /*
     * When the first_kex_packet_follows flag is set, ignore any initial guess of key
     * exchange algorithms. In order to prevent attackers from bypassing integrity checks
     * (the Terrapin attack), we disable the initial guess by unconditionally marking the
     * guess as wrong. This forces a proper key exchange handshake without reusing any
     * potentially compromised or incomplete negotiation data.
     */
    if (first_kex_packet_follows || session->send_first_kex_follows) {
        SSH_LOG(SSH_LOG_DEBUG, "first_kex_packet_follows set; disabling initial guess");
        session->first_kex_follows_guess_wrong = 1;
    }
