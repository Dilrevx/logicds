     * This block checks if either party supports the strict key exchange feature.
     * If the strict key exchange algorithm (strict-kex@openssh.com) appears in either the
     * client's or server's key exchange method lists, we enable the strict kex flag.

    {
        int i;
        int strict_kex_found = 0;
        const char *strict_algo = "strict-kex@openssh.com";

        /* Check if the client's methods include the strict KEX algorithm */
        for (i = 0; crypto->client_kex.methods[i] != NULL; i++) {
            if (strcmp(crypto->client_kex.methods[i], strict_algo) == 0) {
                strict_kex_found = 1;
                break;
            }
        }

        /* If not found on the client side, check the server's methods */
        if (!strict_kex_found) {
            for (i = 0; crypto->server_kex.methods[i] != NULL; i++) {
                if (strcmp(crypto->server_kex.methods[i], strict_algo) == 0) {
                    strict_kex_found = 1;
                    break;
                }
            }
        }

        if (strict_kex_found) {
            SSH_LOG(SSH_LOG_DEBUG, "Strict key exchange support detected - enforcing strict KEX mode.");
