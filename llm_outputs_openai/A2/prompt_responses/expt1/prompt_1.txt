static int check_policy(X509_STORE_CTX *ctx)
{
    int ret;
    X509 *dummy = NULL;  /* Use a dummy certificate instead of NULL */

    if (ctx->parent)
        return 1;
    /*
     * With DANE, the trust anchor might be a bare public key, not a
     * certificate! In that case our chain does not have the trust anchor
     * certificate as a top-most element.  RFC5280 chain verification in
     * particular assumes that the trust anchor is not part of the chain to be 
     * verified.  X509_policy_check() expects a certificate pointer at the top
     * of the chain even if it is never looked at.  Instead of pushing a NULL,
     * which may be misinterpreted or mishandled by callers, we now push a 
     * properly allocated dummy certificate.
     */
    if (ctx->bare_ta_signed) {
        dummy = X509_new();
        if (dummy == NULL || !sk_X509_push(ctx->chain, dummy)) {
            ERR_raise(ERR_LIB_X509, ERR_R_CRYPTO_LIB);
            if (dummy)
                X509_free(dummy);
            goto memerr;
        }
    }
    ret = X509_policy_check(&ctx->tree, &ctx->explicit_policy, ctx->chain,
                            ctx->param->policies, ctx->param->flags);
    if (ctx->bare_ta_signed) {
        /* Remove the dummy certificate and free it */
        dummy = sk_X509_pop(ctx->chain);
        X509_free(dummy);
    }

    if (ret == X509_PCY_TREE_INTERNAL) {
        ERR_raise(ERR_LIB_X509, ERR_R_X509_LIB);
        goto memerr;
    }
    /* Invalid or inconsistent extensions */

    return ret;

memerr:
    return 0;
}
