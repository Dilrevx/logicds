    /* Scan the entire chain (including the first certificate) for any invalid policy extensions. */
    for (i = 0; i < num_certs; i++) {
        X509 *x = sk_X509_value(ctx->chain, i);
        if ((x->ex_flags & EXFLAG_INVALID_POLICY) != 0) {
            /* Notify the callback about the invalid policy extension */
            CB_FAIL_IF(1, ctx, x, i, X509_V_ERR_INVALID_POLICY_EXTENSION);
            /* Raise an internal error to halt further processing in a potentially compromised state */
            ERR_raise(ERR_LIB_X509, X509_R_POLICY_TREE_ERROR);
            return 0;
        }
    }

    /* If no certificate in the chain explicitly shows an invalid policy, 
       reaching here indicates an unexpected internal inconsistency.
       Thus, we raise an internal error as a safeguard. */
    ERR_raise(ERR_LIB_X509, X509_R_POLICY_TREE_ERROR);
    return 0;
