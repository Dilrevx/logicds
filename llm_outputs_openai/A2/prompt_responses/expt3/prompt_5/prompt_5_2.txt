if (ret == X509_PCY_TREE_INVALID) {
    int i;
    int num = sk_X509_num(ctx->chain);

    /* Check every certificate in the entire chain (starting at index 0) */
    for (i = 0; i < num; i++) {
        X509 *x = sk_X509_value(ctx->chain, i);

        if ((x->ex_flags & EXFLAG_INVALID_POLICY) != 0) {
            /* Trigger the callback failure and raise an internal error immediately */
            CB_FAIL_IF(1, ctx, x, i, X509_V_ERR_INVALID_POLICY_EXTENSION);
            ERR_raise(ERR_LIB_X509, X509_R_INVALID_CERTIFICATE_POLICY);
            return 0;
        }
    }

    /* If no specific certificate triggered the callback but the policy tree is still invalid,
       raise an error as a safeguard */
    ERR_raise(ERR_LIB_X509, X509_R_INVALID_CERTIFICATE_POLICY);
    return 0;
