public static void srtp_rtp_cb(Pjmedia_tp_cb_param param) {
    TransportSrtp srtp = (TransportSrtp) param.user_data;
    Object pkt = param.pkt;
    int len = param.size;
    int err = srtp_unprotect(srtp.srtp_rx_ctx, pkt, len);

    // If a SRTP restart is needed due to replay errors, force a secure restart.
    if (srtp.probation_cnt > 0 &&
            (err == SRTP_ERR_REPLAY_OLD || err == SRTP_ERR_REPLAY_FAIL)) {

        // Stop the current SRTP so that we can attempt restarting it.
        pjmedia_transport_srtp_stop(srtp);

        SrtpCrypto tx = srtp.tx_policy;
        SrtpCrypto rx = srtp.rx_policy;
        int status = pjmedia_transport_srtp_start(srtp, tx, rx);
        if (status != PJ_SUCCESS) {
            // If restart fails, drop the packet rather than risking fallback to insecure RTP.
            return;
        }

        // After successful restart, re-run unprotect.
        err = srtp_unprotect(srtp.srtp_rx_ctx, pkt, len);
        if (err != PJ_SUCCESS) {
            // If unprotection still fails, do not process further.
            return;
        }
    }

    // Only pass the packet to the RTP callback if SRTP unprotection is successful.
    if (err == PJ_SUCCESS && srtp.rtp_cb != null) {
        srtp.rtp_cb.rtp_cb(srtp.user_data, pkt, len);
    }
}
