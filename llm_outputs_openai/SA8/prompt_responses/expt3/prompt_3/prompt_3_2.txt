String kexMethods = clientMethods[SSHConstants.SSH_KEX];
String[] methods = kexMethods.split(",");
StringBuilder safeMethods = new StringBuilder();
for (String method : methods) {
    // Skip the unwanted extension entry while rebuilding the list
    if (!SSHConstants.KEX_EXTENSION_CLIENT.equals(method)) {
        if (safeMethods.length() > 0) {
            safeMethods.append(",");
        }
        safeMethods.append(method);
    }
}
clientMethods[SSHConstants.SSH_KEX] = safeMethods.toString();
