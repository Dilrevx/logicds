// Safely remove the extension algorithm without truncating the entire list.
// The Terrapin attack is based on improperly dropping parts of the algorithm list,
// so instead of truncating the string at the extension's position, we remove just the
// extension algorithm from the list. This ensures that if extra algorithms follow the
// extension identifier, they remain available for negotiation.
//
// Helper function to remove a specific algorithm from a comma‐separated list.
function safeRemoveAlgorithm(methodList, algorithm) {
    var algorithms = methodList.split(",");
    var filtered = [];
    for (var i = 0; i < algorithms.length; i++) {
        // Trim any extra whitespace for safe comparison.
        if (algorithms[i].trim() !== algorithm) {
            filtered.push(algorithms[i].trim());
        }
    }
    return filtered.join(",");
}

if (extStart !== null) {
    // Remove only the extension algorithm rather than truncating the list.
    clientMethods[SSHConstants.SSH_KEX] =
        safeRemoveAlgorithm(clientMethods[SSHConstants.SSH_KEX], SSHConstants.KEX_EXTENSION_CLIENT);
}
