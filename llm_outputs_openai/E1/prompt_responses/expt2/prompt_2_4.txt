  // Determine the current message header type for further processing (e.g., for sequence number checks)
  int current_message_header_type = 0;

  switch (nas_msg.hdr.security_header_type)
  {
    case nas_5gs_hdr::security_header_type_opts::plain_5gs_nas_message:
      // Reject plain NAS messages if a security context is already active
      if (is_security_context_active())
      {
        logger.error("Plain NAS message received after security context activation. Rejecting message.");
        return SRSRAN_ERROR;
      }
      current_message_header_type = 1; // Identifier for plain NAS message
      break;

    case nas_5gs_hdr::security_header_type_opts::integrity_protected:
      current_message_header_type = 2; // Identifier for integrity protected messages
      if (integrity_check(pdu.get()) == false)
      {
        logger.error("Not handling NAS message with integrity check error");
        return SRSRAN_ERROR;
      }
      // TODO: Implement sequence number check to prevent replay attacks.
      break;

    case nas_5gs_hdr::security_header_type_opts::integrity_protected_and_ciphered:
      current_message_header_type = 3; // Identifier for integrity protected and ciphered messages
      if (integrity_check(pdu.get()) == false)
      {
        logger.error("Not handling NAS message with integrity check error");
        return SRSRAN_ERROR;
      }
      else
      {
        cipher_decrypt(pdu.get());
      }
      // TODO: Implement sequence number check to prevent replay attacks.
      break;

    case nas_5gs_hdr::security_header_type_opts::integrity_protected_with_new_5G_nas_context:
      current_message_header_type = 4; // Identifier for integrity protected with new 5G NAS context
      // TODO: Implement sequence number check to prevent replay attacks.
      break;

    case nas_5gs_hdr::security_header_type_opts::integrity_protected_and_ciphered_with_new_5G_nas_context:
      logger.error("Not handling NAS message: integrity protected and ciphered with new 5G NAS context is not supported");
      return SRSRAN_ERROR;

    default:
      logger.error("Not handling NAS message with unknown security header");
      return SRSRAN_ERROR;
  }
