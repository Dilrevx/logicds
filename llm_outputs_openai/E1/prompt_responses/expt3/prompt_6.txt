switch (nas_msg.hdr.security_header_type) {
  case nas_5gs_hdr::security_header_type_opts::plain_5gs_nas_message:
    // No security processing necessary for plain messages.
    break;
    
  case nas_5gs_hdr::security_header_type_opts::integrity_protected:
    if (!integrity_check(pdu.get())) {
      logger.error("Not handling NAS message due to integrity check error (integrity_protected).");
      return SRSRAN_ERROR;
    }
    break;
    
  case nas_5gs_hdr::security_header_type_opts::integrity_protected_and_ciphered:
    if (!integrity_check(pdu.get())) {
      logger.error("Not handling NAS message due to integrity check error (integrity_protected_and_ciphered).");
      return SRSRAN_ERROR;
    } else {
      cipher_decrypt(pdu.get());
    }
    break;
    
  // NEW: Both new NAS context types must perform the integrity check.
  case nas_5gs_hdr::security_header_type_opts::integrity_protected_with_new_5G_nas_context:
    if (!integrity_check(pdu.get())) {
      logger.error("Not handling NAS message due to integrity check error (integrity_protected_with_new_5G_nas_context).");
      return SRSRAN_ERROR;
    }
    break;
    
  case nas_5gs_hdr::security_header_type_opts::integrity_protected_and_ciphered_with_new_5G_nas_context:
    if (!integrity_check(pdu.get())) {
      logger.error("Not handling NAS message due to integrity check error (integrity_protected_and_ciphered_with_new_5G_nas_context).");
      return SRSRAN_ERROR;
    } else {
      cipher_decrypt(pdu.get());
    }
    break;
    
  default:
    // Unrecognized security header type: log and return error.
    logger.error("Not handling NAS message with unknown security header.");
    return SRSRAN_ERROR;
}
