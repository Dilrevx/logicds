uint8_t current_message_header = 0;

// Optional: early check for inactive security context if needed in your design
if (!is_security_context_activated())
{
  // When no security context is active, plain messages may be allowed.
  // No additional action is needed here.

switch (nas_msg.hdr.security_header_type)
{
  case nas_5gs_hdr::security_header_type_opts::plain_5gs_nas_message:
    // If a plain NAS message is received after a security context is active, reject it.
    if (is_security_context_activated())
    {
      logger.error("Rejecting plain NAS message after security context activation");
      return SRSRAN_ERROR;
    }
    current_message_header = 0x00; // Assign header value for plain messages
    break;

  case nas_5gs_hdr::security_header_type_opts::integrity_protected:
    current_message_header = 0x01; // Assign header value for integrity-protected messages
    if (!integrity_check(pdu.get()))
    {
      logger.error("Not handling NAS message with integrity check error");
      return SRSRAN_ERROR;
    }
    // NOTE: Consider adding sequence number validation here to prevent replay attacks.
    break;

  case nas_5gs_hdr::security_header_type_opts::integrity_protected_and_ciphered:
    current_message_header = 0x02; // Assign header value for integrity-protected and ciphered messages
    if (!integrity_check(pdu.get()))
    {
      logger.error("Not handling NAS message with integrity check error");
      return SRSRAN_ERROR;
    }
    else
    {
      cipher_decrypt(pdu.get());
    }
    // NOTE: Consider adding sequence number validation here as well.
    break;

  case nas_5gs_hdr::security_header_type_opts::integrity_protected_with_new_5G_nas_context:
    current_message_header = 0x03; // Assign header value for messages with a new 5G NAS context
    // NOTE: Sequence number checks for replay protection should be added here.
    break;

  case nas_5gs_hdr::security_header_type_opts::integrity_protected_and_ciphered_with_new_5G_nas_context:
    logger.error("Not handling NAS message with integrity protected and ciphered new 5G NAS context");
    return SRSRAN_ERROR;

  default:
    logger.error("Not handling NAS message with unknown security header");
    return SRSRAN_ERROR; // It is safer to return an error here.
}
