int mosquitto_acl_check_default(struct mosquitto_db *db, struct mosquitto *context, const char *topic, int access)
{
    char *local_acl = NULL;
    struct mosquitto__acl *acl_root;
    bool result = false;
    int i;
    int tlen;
    struct mosquitto__security_options *security_opts = NULL;
    /* Define an upper bound for topic length so that integer overflow or huge allocations cannot occur.
       (This value should be chosen based on the system limits and application requirements.) */
    const size_t MAX_TOPIC_LENGTH = 65535;

    if(!db || !context || !topic) return MOSQ_ERR_INVAL;
    if(context->bridge) return MOSQ_ERR_SUCCESS;

    if(db->config->per_listener_settings){
        if(!context->listener)
            return MOSQ_ERR_ACL_DENIED;
        security_opts = &context->listener->security_options;
    } else {
        security_opts = &db->config->security_options;
    }

    /* Check that the topic length is within a safe range. Using strnlen prevents reading past the intended area.
       If the topic is too long, we deny access. */
    tlen = (int)strnlen(topic, MAX_TOPIC_LENGTH + 1);
    if((size_t)tlen > MAX_TOPIC_LENGTH){
        return MOSQ_ERR_ACL_DENIED;
    }

    /* Allocate a local buffer for the topic, ensuring we have room including the terminating NUL byte.
       Check for allocation failure before using the buffer. */
    local_acl = malloc(tlen + 1);
    if(!local_acl) {
        return MOSQ_ERR_NOMEM;
    }
    memcpy(local_acl, topic, tlen);
    local_acl[tlen] = '\0';

    /*
     * Vulnerability Repair:
     * Earlier versions directly copied and used the topic name without appropriate sanity checks.
     * This repair ensures that the topic length is bounded and that allocation is checked.
     *
     * After sanitizing the topic, we iterate over the ACL list (which in this example is assumed to be
     * stored in security_opts->acl_list) and use a secure matching function.
     */
    acl_root = security_opts->acl_list;
    for(i = 0; acl_root != NULL; i++){
        if(mosquitto_topic_matches_sub(acl_root->topic, local_acl)){
            if(acl_root->access & access){
                result = true;
                break;
            }
        }
        acl_root = acl_root->next;
    }

    free(local_acl);
    return (result == true) ? MOSQ_ERR_SUCCESS : MOSQ_ERR_ACL_DENIED;
}
