#ifdef HAVE_OCSP
    /* Only if performing OCSP verification, disable further checks based on the NoCheck extension
     * provided that the responder certificate is truly linked to the certificate chain.
     * This additional check prevents an attacker from supplying an OCSP response signed by a
     * certificate that merely has the NoCheck extension but is not actually related to the chain.
     */
    if (verify == VERIFY_OCSP) {
        /* OCSP_responder_id_matches_chain() should return true only if 'cert' is actually a valid,
         * trusted OCSP responder certificate for the certificate chain in question.
         */
        if (cert->ocspNoCheckSet && OCSP_responder_id_matches_chain(cert, chain)) {
            verify = NO_VERIFY;
        }
    }
#endif
