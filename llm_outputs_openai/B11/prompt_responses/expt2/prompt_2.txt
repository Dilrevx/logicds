    /* Determine the proper verification state for the OCSP responder’s cert.
     * Instead of turning off all checks simply because the NoCheck extension
     * is present, we now branch into two controlled paths:
     *
     * - If this is the OCSP response for the responder’s certificate and it
     *   contains the NoCheck extension, then we switch to the standard certificate
     *   verification state, ensuring the cert still undergoes proper checking.
     *
     * - Otherwise (i.e. if NoCheck is not set), we move to a dedicated OCSP
     *   verification state that enforces full, rigorous checks on the response.
     */
    if (verify == VERIFY_OCSP) {
        if (cert->ocspNoCheckSet)
            verify = VERIFY_CERT;     /* regular certificate verification workflow */
        else
            verify = VERIFY_OCSP_FULL;/* enforce full OCSP-specific validation */
