#ifdef HAVE_OCSP
    /* Only disable OCSP verification (“no check”) for a certificate that’s really a responder.
       This patch ensures that we only bypass verification when the certificate both carries
       the noCheck extension and its Extended Key Usage indicates it is allowed to sign OCSP responses. */
    if (cert->ocspNoCheckSet && verify == VERIFY_OCSP) {
        /* Verify that the certificate is allowed for OCSP signing.
           X509_check_eku() returns 1 if the expected EKU is present.
           (The OpenSSL NID for OCSP signing is NID_OCSP_sign.) */
        if (X509_check_eku(cert, NID_OCSP_sign) == 1) {
            verify = NO_VERIFY;
        }
    }
#endif
