    if (verify == VERIFY_OCSP_RESP) {
        if (cert->ocspNoCheckSet) {
            /* The certificate contains the NoCheck extension.
             * Instead of completely disabling verification (which would be dangerous),
             * we switch to the normal OCSP verification state so that standard checks are still applied.
             */
            verify = VERIFY_OCSP;
        } else {
            /* If no NoCheck flag is present, enforce full OCSP responder certificate validation.
             * VERIFY_OCSP_FULL is a dedicated state that ensures rigorous verification.
             */
            verify = VERIFY_OCSP_FULL;
        }
