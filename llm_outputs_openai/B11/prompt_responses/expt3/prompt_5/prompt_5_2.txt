#ifdef HAVE_OCSP
    /*
     * Check that we are processing an OCSP responder certificate.
     * Use a helper function (is_ocsp_responder) to verify the certificate's context.
     */
    if (is_ocsp_responder(cert)) {
        if (cert->ocspNoCheckSet) {
            /*
             * Although the certificate indicates the NoCheck extension,
             * do not bypass all verifications. Instead, switch to standard OCSP checks.
             */
            verify = VERIFY_OCSP_STANDARD;
        } else {
            /*
             * In the absence of the NoCheck flag, enforce full OCSP validation.
             */
            verify = VERIFY_OCSP_FULL;
        }
    }
#endif
