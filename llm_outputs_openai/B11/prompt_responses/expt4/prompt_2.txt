#ifdef HAVE_OCSP
    /*
     * Refined OCSP responder verification logic:
     *
     * Previously, if a certificate possessed the NoCheck extension while in an OCSP
     * verification context (VERIFY_OCSP), verification was fully disabled by setting the
     * state to NO_VERIFY. This was dangerous because it allowed an unrelated responder certificate
     * to bypass proper checks.
     *
     * In this revised logic, if we're processing an OCSP responder certificate (indicated by
     * verify == VERIFY_OCSP) and the certificate has the NoCheck extension, we no longer disable
     * verification altogether. Instead, we downgrade to standard certificate verification (VERIFY)
     * rather than completely bypassing it.
     *
     * If the NoCheck flag is not set, we continue to use full OCSP-specific verification.
     */
    if (verify == VERIFY_OCSP) {
        if (cert->ocspNoCheckSet)
            verify = VERIFY;      /* Use standard verification, not a complete bypass */
        else
            verify = VERIFY_OCSP; /* Continue with full OCSP verification */
    }
#endif
