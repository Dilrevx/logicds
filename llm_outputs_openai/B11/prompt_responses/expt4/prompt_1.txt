       if and only if the certificate used in the response is the one
       that is properly associated with the OCSP responder in the chain.
       This prevents an unrelated cert—even one carrying the NoCheck
       extension—from bypassing OCSP validation.
       In this example we assume that the variable "ocsp_resp_cert" holds
       the expected OCSP responder certificate (as determined from the chain).
       Adjust or replace this check as needed for your integration. */
    if (cert->ocspNoCheckSet && verify == VERIFY_OCSP) {
        if (ocsp_resp_cert != NULL && X509_cmp(ocsp_resp_cert, cert) == 0) {
            verify = NO_VERIFY;
        }
    }
