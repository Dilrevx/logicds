#ifdef HAVE_OCSP
    /* For OCSP responses we must not fully bypass verification even if the NoCheck extension is present.
       Instead, if we are processing an OCSP responder certificate, then when the certificate has the
       NoCheck extension we switch to a normal verification state rather than disabling all checks.
       This ensures that the responder cert is still validated properly. */
    if (verify == VERIFY_OCSP) {
        if (cert->ocspNoCheckSet)
            /* Instead of disabling verification entirely (as was done by setting NO_VERIFY),
               we change to a normal certificate verification state to perform full checks. */
            verify = VERIFY;
        else
            /* Keep using the dedicated OCSP verification state to enforce full OCSP checks */
            verify = VERIFY_OCSP;
    }
