void rrc_nr::handle_security_mode_command(const asn1::rrc_nr::security_mode_cmd_s& smc)
{
  transaction_id = smc.rrc_transaction_id;

  const auto& sec_algo_cfg = smc.crit_exts.security_mode_cmd().security_cfg_smc.security_algorithm_cfg;
  sec_cfg.cipher_algo = static_cast<CIPHERING_ALGORITHM_ID_ENUM>(sec_algo_cfg.ciphering_algorithm.value);

  // Check whether an integrity protection algorithm is provided.
  if (!sec_algo_cfg.integrity_prot_algorithm_present)
  {
    logger.error("Missing Integrity Algorithm Config. Rejecting Security Mode Command.");
    send_security_mode_reject();
    return;
  }

  sec_cfg.integ_algo = static_cast<INTEGRITY_ALGORITHM_ID_ENUM>(sec_algo_cfg.integrity_prot_algorithm.value);

  // Reject any command that specifies an insecure integrity algorithm (e.g. NIA0 plaintext)
  if (sec_cfg.integ_algo == INTEGRITY_ALGORITHM_ID_ENUM::NIA0)
  {
    logger.error("Insecure Integrity Algorithm (NIA0) is not allowed. Rejecting Security Mode Command.");
    send_security_mode_reject();
    return;
  }

  logger.info("Received Security Mode Command nea: %s, nia: %s",
              ciphering_algorithm_id_nr_text[sec_cfg.cipher_algo],
              integrity_algorithm_id_nr_text[sec_cfg.integ_algo]);

  // Generate AS security keys only after successful security configuration.
  generate_as_keys();
  security_is_activated = true;

  // Configure PDCP for security protection.
  uint32_t lcid = srb_to_lcid(nr_srb::srb1);
  pdcp->config_security(lcid, sec_cfg);
  pdcp->enable_integrity(lcid, DIRECTION_TXRX);

  // Acknowledge successful security mode activation.
  send_security_mode_complete();

  pdcp->enable_encryption(lcid, DIRECTION_TXRX);
}
