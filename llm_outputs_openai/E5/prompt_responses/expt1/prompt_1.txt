void rrc_nr::handle_security_mode_command(const asn1::rrc_nr::security_mode_cmd_s& smc)
{
    transaction_id = smc.rrc_transaction_id;

    // Check that the security_mode_cmd extension is present.
    if (!smc.crit_exts.has_security_mode_cmd())
    {
        // Log an error and safely return.
        LOG_ERROR("Security mode command missing in crit_exts.");
        return;
    }
    
    const auto& sec_mode_cmd   = smc.crit_exts.security_mode_cmd();
    const auto& sec_cfg_smc    = sec_mode_cmd.security_cfg_smc;
    const auto& sec_algo_cfg   = sec_cfg_smc.security_algorithm_cfg;

    // Validate that the ciphering algorithm value is within the allowed enum bounds.
    // Assume MAX_CIPHERING_ALGORITHM_ID is defined elsewhere as the maximum valid enum value.
    unsigned int cipher_value = sec_algo_cfg.ciphering_algorithm.value;
    if (cipher_value > MAX_CIPHERING_ALGORITHM_ID)
    {
        LOG_ERROR("Invalid ciphering algorithm value: %u", cipher_value);
        // Safely reject this command. In production, you might want to generate an error response.
        return;
    }

    // Use static_cast instead of a C-style cast to translate the value to the enum.
    sec_cfg.cipher_algo = static_cast<CIPHERING_ALGORITHM_ID_ENUM>(cipher_value);

    // ... continue processing the security mode command as required ...
}
