package main;

import commons.*;

public class Nas5g {
    public static final int SRSRAN_SUCCESS = 0;
    public static final int SRSRAN_ERROR = -1;
    static Logger logger = new Logger();
    static Pcap pcap;
    private boolean has_sec_ctxt;
    private int current_message_header;


        logger.info(pdu.msg, pdu.N_bytes, "DL PDU (length %d)", pdu.N_bytes);

        commons.Nas5gsMsg nas_msg = new commons.Nas5gsMsg();

        if (nas_msg.unpack_outer_hdr(pdu) != SRSRAN_SUCCESS) {
            logger.error("Unable to unpack outer NAS header");
            return SRSRAN_ERROR;
        }

        // Track the current security header type for debugging and clarity.
        int currentSecHeader = nas_msg.hdr.security_header_type;

        switch (currentSecHeader) {
            case commons.Nas5gsHdr.security_header_type_opts.plain_5gs_nas_message:
                // Prevent processing of plain NAS messages when a security context is already set.
                if (this.has_sec_ctxt) {
                    logger.error("Plain NAS message received after security activation");
                    return SRSRAN_ERROR;
                }
                // Optionally set a tracking flag or variable here for plain messages.
                break;

            case commons.Nas5gsHdr.security_header_type_opts.integrity_protected:
                // Set tracking variable as needed for the integrity protected case.
                if (!integrity_check(pdu.get())) {
                    logger.error("Not handling NAS message with integrity check error");
                    return SRSRAN_ERROR;
                }
                break;

            case commons.Nas5gsHdr.security_header_type_opts.integrity_protected_and_ciphered:
                // Set tracking variable as needed for messages that are integrity protected and ciphered.
                if (!integrity_check(pdu.get())) {
                    logger.error("Not handling NAS message with integrity check error");
                    return SRSRAN_ERROR;
                } else {
                    cipher_decrypt(pdu.get());
                }
                break;

            case commons.Nas5gsHdr.security_header_type_opts.integrity_protected_with_new_5G_nas_context:
                // Set tracking variable accordingly if needed.
                break;

            case commons.Nas5gsHdr.security_header_type_opts.integrity_protected_and_ciphered_with_new_5G_nas_context:
                // This mode is not supported.
                logger.error("Not handling NAS message with integrity protection and ciphering in new 5G context");
                return SRSRAN_ERROR;

            default:
                logger.error("Not handling NAS message with unknown security header");
                return SRSRAN_ERROR;
        }

        if (pcap != null) {
            pcap.write_nas(pdu.msg, pdu.N_bytes);
        }

        logger.info(pdu.msg, pdu.N_bytes, "Decrypted DL PDU (length %d)", pdu.N_bytes);

        if (nas_msg.unpack(pdu) != SRSRAN_SUCCESS) {
            logger.error("Unable to unpack complete NAS pdu");
            return SRSRAN_ERROR;
        }

        switch (nas_msg.hdr.message_type) {
            case commons.MsgOpts.Options.registration_accept:
                handle_registration_accept(nas_msg.registration_accept());
                break;
            case commons.MsgOpts.Options.registration_reject:
                handle_registration_reject(nas_msg.registration_reject());
                break;
            case commons.MsgOpts.Options.authentication_reject:
                handle_authentication_reject(nas_msg.authentication_reject());
                break;
            case commons.MsgOpts.Options.authentication_request:
                handle_authentication_request(nas_msg.authentication_request());
                break;
            case commons.MsgOpts.Options.identity_request:
                handle_identity_request(nas_msg.identity_request());
                break;
            case commons.MsgOpts.Options.security_mode_command:
                handle_security_mode_command(nas_msg.security_mode_command(), pdu);
                break;
            case commons.MsgOpts.Options.service_accept:
                handle_service_accept(nas_msg.service_accept());
                break;
            case commons.MsgOpts.Options.service_reject:
                handle_service_reject(nas_msg.service_reject());
                break;
            case commons.MsgOpts.Options.deregistration_accept_ue_terminated:
                handle_deregistration_accept_ue_terminated(nas_msg.deregistration_accept_ue_terminated());
                break;
            case commons.MsgOpts.Options.deregistration_request_ue_terminated:
                handle_deregistration_request_ue_terminated(nas_msg.deregistration_request_ue_terminated());
                break;
            case commons.MsgOpts.Options.dl_nas_transport:
                handle_dl_nas_transport(nas_msg.dl_nas_transport());
                break;
            case commons.MsgOpts.Options.deregistration_accept_ue_originating:
                handle_deregistration_accept_ue_originating(nas_msg.deregistration_accept_ue_originating());
                break;
            case commons.MsgOpts.Options.configuration_update_command:
                handle_configuration_update_command(nas_msg.configuration_update_command());
                break;
            default:
                logger.error("Not handling NAS message type: " + nas_msg.hdr.message_type);
                break;
        }
        return SRSRAN_SUCCESS;


    private boolean integrity_check(byte[] data) {
        return false;
    }

    private void cipher_decrypt(byte[] data) {}

    private void handle_registration_accept(Object o) {}
    private void handle_registration_reject(Object o) {}
    private void handle_authentication_reject(Object o) {}
    private void handle_authentication_request(Object o) {}
    private void handle_identity_request(Object o) {}
    private void handle_security_mode_command(Object o, commons.UniqueByteBuffer pdu) {}
    private void handle_service_accept(Object o) {}
    private void handle_service_reject(Object o) {}
    private void handle_deregistration_accept_ue_terminated(Object o) {}
    private void handle_deregistration_request_ue_terminated(Object o) {}
    private void handle_dl_nas_transport(Object o) {}
    private void handle_deregistration_accept_ue_originating(Object o) {}
    private void handle_configuration_update_command(Object o) {}
}
