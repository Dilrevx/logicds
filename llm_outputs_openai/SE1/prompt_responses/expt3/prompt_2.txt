        uint8_t currentHeader = 0;  // Variable for tracking the NAS message header type

        switch (nas_msg.hdr.security_header_type) {
            // Plain NAS message (usually used for initial identity requests)
            case commons.Nas5gsHdr.security_header_type_opts.plain_5gs_nas_message:
                // Check if a security context has already been established.
                if (is_security_context_active()) {
                    logger.error("Rejecting plain NAS message after NAS security activation");
                    return SRSRAN_ERROR;
                }
                currentHeader = commons.Nas5gsHdr.security_header_type_opts.plain_5gs_nas_message;
                break;

            // Integrity protected message without ciphering.
            case commons.Nas5gsHdr.security_header_type_opts.integrity_protected:
                currentHeader = commons.Nas5gsHdr.security_header_type_opts.integrity_protected;
                if (!integrity_check(pdu.get())) {
                    logger.error("Not handling NAS message with integrity check error");
                    return SRSRAN_ERROR;
                }
                break;

            // Integrity protected and ciphered message.
            case commons.Nas5gsHdr.security_header_type_opts.integrity_protected_and_ciphered:
                currentHeader = commons.Nas5gsHdr.security_header_type_opts.integrity_protected_and_ciphered;
                if (!integrity_check(pdu.get())) {
                    logger.error("Not handling NAS message with integrity check error");
                    return SRSRAN_ERROR;
                } else {
                    cipher_decrypt(pdu.get());
                }
                break;

            // Integrity protected message with new 5G NAS context.
            case commons.Nas5gsHdr.security_header_type_opts.integrity_protected_with_new_5G_nas_context:
                currentHeader = commons.Nas5gsHdr.security_header_type_opts.integrity_protected_with_new_5G_nas_context;
                break;

            // Integrity protected and ciphered message with new 5G NAS context.
            case commons.Nas5gsHdr.security_header_type_opts.integrity_protected_and_ciphered_with_new_5G_nas_context:
                currentHeader = commons.Nas5gsHdr.security_header_type_opts.integrity_protected_and_ciphered_with_new_5G_nas_context;
                return SRSRAN_ERROR;

            default:
                logger.error("Not handling NAS message with unknown security header");
                break;
        }
