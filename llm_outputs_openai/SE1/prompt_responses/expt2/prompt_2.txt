    // Check if the UE already has an active security context.
    // The function security_context_active() returns true if NAS security is activated.
    // The variable current_msg_hdr is used to track the security header type for logging and subsequent processing.
    switch (nas_msg.hdr.security_header_type) {
        case commons.Nas5gsHdr.security_header_type_opts.plain_5gs_nas_message:
            // Do not accept plain messages after security activation.
            if (security_context_active()) {
                logger.error("Received plain NAS message after security context activation");
                return SRSRAN_ERROR;
            }
            current_msg_hdr = commons.Nas5gsHdr.security_header_type_opts.plain_5gs_nas_message;
            break;

        case commons.Nas5gsHdr.security_header_type_opts.integrity_protected:
            current_msg_hdr = commons.Nas5gsHdr.security_header_type_opts.integrity_protected;
            if (!integrity_check(pdu.get())) {
                logger.error("Not handling NAS message with integrity check error");
                return SRSRAN_ERROR;
            }
            break;

        case commons.Nas5gsHdr.security_header_type_opts.integrity_protected_and_ciphered:
            current_msg_hdr = commons.Nas5gsHdr.security_header_type_opts.integrity_protected_and_ciphered;
            if (!integrity_check(pdu.get())) {
                logger.error("Not handling NAS message with integrity check error");
                return SRSRAN_ERROR;
            } else {
                cipher_decrypt(pdu.get());
            }
            break;

        case commons.Nas5gsHdr.security_header_type_opts.integrity_protected_with_new_5G_nas_context:
            current_msg_hdr = commons.Nas5gsHdr.security_header_type_opts.integrity_protected_with_new_5G_nas_context;
            break;

        case commons.Nas5gsHdr.security_header_type_opts.integrity_protected_and_ciphered_with_new_5G_nas_context:
            current_msg_hdr = commons.Nas5gsHdr.security_header_type_opts.integrity_protected_and_ciphered_with_new_5G_nas_context;
            return SRSRAN_ERROR;

        default:
            logger.error("Not handling NAS message with unknown security header");
            break;
    }
