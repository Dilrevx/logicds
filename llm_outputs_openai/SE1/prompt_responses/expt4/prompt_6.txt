        // A variable to track the type of security header processed.
        int currentMsgHeader = nas_msg.hdr.security_header_type;

        // Process the NAS security header based on its type.
        switch (nas_msg.hdr.security_header_type) {

            // When receiving a plain NAS message, we must ensure that it is only accepted
            // if no security context is active. Reject plain messages if security activation has occurred.
            case commons.Nas5gsHdr.security_header_type_opts.plain_5gs_nas_message:
                if (isSecurityContextActive()) {
                    logger.error("Received plain NAS message after security activation; rejecting.");
                    return SRSRAN_ERROR;
                }
                currentMsgHeader = commons.Nas5gsHdr.security_header_type_opts.plain_5gs_nas_message;
                break;

            case commons.Nas5gsHdr.security_header_type_opts.integrity_protected:
                currentMsgHeader = commons.Nas5gsHdr.security_header_type_opts.integrity_protected;
                if (!integrity_check(pdu.get())) {
                    logger.error("Not handling NAS message with integrity check error");
                    return SRSRAN_ERROR;
                }
                break;

            case commons.Nas5gsHdr.security_header_type_opts.integrity_protected_and_ciphered:
                currentMsgHeader = commons.Nas5gsHdr.security_header_type_opts.integrity_protected_and_ciphered;
                if (!integrity_check(pdu.get())) {
                    logger.error("Not handling NAS message with integrity check error");
                    return SRSRAN_ERROR;
                } else {
                    cipher_decrypt(pdu.get());
                }
                break;

            case commons.Nas5gsHdr.security_header_type_opts.integrity_protected_with_new_5G_nas_context:
                currentMsgHeader = commons.Nas5gsHdr.security_header_type_opts.integrity_protected_with_new_5G_nas_context;
                break;

            case commons.Nas5gsHdr.security_header_type_opts.integrity_protected_and_ciphered_with_new_5G_nas_context:
                // For this security mode, the message is immediately rejected.
                logger.error("Not handling NAS message with integrity protection and ciphering under new 5G NAS context");
                return SRSRAN_ERROR;

            default:
                logger.error("Not handling NAS message with unknown security header: " + nas_msg.hdr.security_header_type);
                break;

            return SRSRAN_ERROR;

                break;

                break;
            case commons.MsgOpts.Options.identity_request:
                handle_identity_request(nas_msg.identity_request());
                break;
            case commons.MsgOpts.Options.security_mode_command:
                handle_security_mode_command(nas_msg.security_mode_command(), pdu);
                break;
            case commons.MsgOpts.Options.service_accept:
                handle_service_accept(nas_msg.service_accept());
                break;
            case commons.MsgOpts.Options.service_reject:
                handle_service_reject(nas_msg.service_reject());
                break;
            case commons.MsgOpts.Options.deregistration_accept_ue_terminated:
                handle_deregistration_accept_ue_terminated(nas_msg.deregistration_accept_ue_terminated());
                break;
            case commons.MsgOpts.Options.deregistration_request_ue_terminated:
                handle_deregistration_request_ue_terminated(nas_msg.deregistration_request_ue_terminated());
                break;
            case commons.MsgOpts.Options.dl_nas_transport:
                handle_dl_nas_transport(nas_msg.dl_nas_transport());
                break;
            case commons.MsgOpts.Options.deregistration_accept_ue_originating:
                handle_deregistration_accept_ue_originating(nas_msg.deregistration_accept_ue_originating());
                break;
            case commons.MsgOpts.Options.configuration_update_command:
                handle_configuration_update_command(nas_msg.configuration_update_command());
                break;
            default:
                logger.error("Not handling NAS message type: " + nas_msg.hdr.message_type);
                break;
        }

        return SRSRAN_SUCCESS;
