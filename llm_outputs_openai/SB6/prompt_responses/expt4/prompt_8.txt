    public static void x509_crt_verify_name(MbedtlsX509Crt crt, String cn, int[] flags) {
        int cn_len = cn.length();
        // First try to verify against the Subject Alternative Name (SAN) extension.
        // Use a dedicated routine that distinguishes between name types (e.g. DNS names vs IP addresses)
        // so that only SAN entries of the correct type are considered.
        if ((crt.extTypes & MbedtlsConstants.MBEDTLS_X509_EXT_SUBJECT_ALT_NAME) != 0) {
            // The new utility function performs type‐specific matching.
            // It will, for example, only compare against DNS names when the expected cn is a domain name,
            // thereby preventing an attacker with a certificate for an IP address from passing the check.
            if (CommonUtils.x509_crt_verify_san(crt.subjectAltNames, cn, cn_len) != 0) {
                flags[0] |= MbedtlsConstants.MBEDTLS_X509_BADCERT_CN_MISMATCH;
            }
        } else {
            // Fallback: if no SAN extension, verify using the subject field's Common Name.
            MbedtlsX509Name name;
            for (name = crt.subject; name != null; name = name.next) {
                if (CommonUtils.MBEDTLS_OID_CMP(MbedtlsConstants.MBEDTLS_OID_AT_CN, name.oid) == 0 &&
                        CommonUtils.x509_crt_check_cn(name.val, cn, cn_len) == 0) {
                    break;
                }
            }
            if (name == null) {
                flags[0] |= MbedtlsConstants.MBEDTLS_X509_BADCERT_CN_MISMATCH;
            }
        }
    }
