package main;

import commons.*;

public class X509Verifier {

    public static void x509_crt_verify_name(MbedtlsX509Crt crt, String cn, int[] flags) {
        MbedtlsX509Name name;
        MbedtlsX509Sequence cur;
        int cn_len = cn.length();

        // First, if the certificate has a Subject Alternative Name extension, check it
        if ((crt.extTypes & MbedtlsConstants.MBEDTLS_X509_EXT_SUBJECT_ALT_NAME) != 0) {
            for (cur = crt.subjectAltNames; cur != null; cur = cur.next) {
                // If any subjectAltName entry matches the given common name, return immediately.
                if (CommonUtils.x509_crt_check_cn(cur.buf, cn, cn_len) == 0) {
                    return;
                }
            }
        }

        // If no matching SAN entry is found, fallback to checking the subject's Common Name (CN)
        for (name = crt.subjectNames; name != null; name = name.next) {
            if (CommonUtils.x509_crt_check_cn(name.buf, cn, cn_len) == 0) {
                return;
            }
        }

        // If neither SAN nor subject CN match, mark the error flag for CN mismatch.
        flags[0] |= MbedtlsConstants.MBEDTLS_X509_BADCERT_CN_MISMATCH;
    }

        }
    }
}
