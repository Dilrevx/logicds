package main;

import commons.*;

public class X509Verifier {

    public static void x509_crt_verify_name(MbedtlsX509Crt crt, String cn, int[] flags) {
        int cn_len = cn.length();
        boolean valid = false;

        // If the certificate contains the Subject Alternative Name (SAN) extension,
        // perform a secure and type-specific check on each SAN entry.
        if ((crt.extTypes & MbedtlsConstants.MBEDTLS_X509_EXT_SUBJECT_ALT_NAME) != 0) {
            for (MbedtlsX509Sequence cur = crt.subjectAltNames; cur != null; cur = cur.next) {
                /*
                 * The new helper function verifySANEntry is expected to check:
                 * - That the SAN entry is of the appropriate type (for example, a DNS name if a domain name is expected).
                 * - That the expected name exactly matches the SAN entry.
                 *
                 * It returns 0 on a successful match, or a nonzero value otherwise.
                 * This prevents an attacker from misusing non-DNS name types (e.g. IP addresses) to impersonate another identity.
                 */
                if (CommonUtils.verifySANEntry(cur.buf, cn, cn_len) == 0) {
                    valid = true;
                    break;
                }
            }
        }
        // Otherwise fall back to the traditional common name (CN) verification.
        else {
            for (MbedtlsX509Name name = crt.subject; name != null; name = name.next) {
                if (CommonUtils.MBEDTLS_OID_CMP(MbedtlsConstants.MBEDTLS_OID_AT_CN, name.oid) == 0 &&
                        CommonUtils.x509_crt_check_cn(name.val, cn, cn_len) == 0) {
                    valid = true;
                    break;
                }
            }
        }

        if (!valid) {
            flags[0] |= MbedtlsConstants.MBEDTLS_X509_BADCERT_CN_MISMATCH;
        }
    }

        }
    }
}
