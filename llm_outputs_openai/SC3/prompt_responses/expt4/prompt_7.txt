    public static MosqErr mosquittoAclCheckDefault(
            MosquittoDb db,
            MosquittoContext context,
            String topic,
            Access access
    ) {
        if (db == null || context == null || topic == null) {
            return MosqErr.MOSQ_ERR_INVAL;
        }
        if (context.bridge) {
            return MosqErr.MOSQ_ERR_SUCCESS;
        }

        MosquittoSecurityOptions securityOpts;
        if (db.config.perListenerSettings) {
            if (context.listener == null) {
                return MosqErr.MOSQ_ERR_ACL_DENIED;
            }
            securityOpts = context.listener.securityOptions;
        } else {
            securityOpts = db.config.securityOptions;
        }

        /* 
         * If no ACL rules or patterns were loaded then we have two possibilities:
         *   1) no ACL file was configured: in that case we defer to the plugin via
         *      MOSQ_ERR_PLUGIN_DEFER so that access is allowed.
         *   2) an ACL file was specified (even if empty or only comments/blank lines) so 
         *      the effective rule set is empty and access should be denied.
         *
         * We assume that when an ACL file is configured, the configuration object (db.config)
         * will have its aclFile field set to a non-null value.
         */
        if (securityOpts.aclList == null && securityOpts.aclPatterns == null) {
            if (db.config.aclFile != null) {
                return MosqErr.MOSQ_ERR_ACL_DENIED;
            }
            return MosqErr.MOSQ_ERR_PLUGIN_DEFER;
        }

        return MosqErr.MOSQ_ERR_ACL_DENIED;
    }
