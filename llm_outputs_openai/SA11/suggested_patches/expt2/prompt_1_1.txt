package main;

import commons.*;

public class X509CrtRevocationChecker {
    public static int mbedtls_x509_crt_is_revoked(
            MbedtlsX509Crt crt, MbedtlsX509Crl crl) {
        MbedtlsX509CrlEntry cur = crl.entry;
        while (cur != null && cur.serial.len != 0) {

            if (crt.serial.len == cur.serial.len
                    && MemUtil.memcmp(crt.serial.p, cur.serial.p, crt.serial.len) == 0) {
                /* 
                 * Since the certificate's serial number appears in the CRL,
                 * it must be considered revoked regardless of the revocation date.
                 * Relying on a local clock can be manipulated by an attacker.
                 */
                return 1;
            }

            cur = cur.next;
        }
        return 0;
    }
}
