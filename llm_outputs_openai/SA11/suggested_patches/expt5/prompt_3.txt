package main;

import commons.*;

public class X509CrtRevocationChecker {
    public static int mbedtls_x509_crt_is_revoked(
            MbedtlsX509Crt crt, MbedtlsX509Crl crl) {
        MbedtlsX509CrlEntry cur = crl.entry;
        while (cur != null && cur.serial.len != 0) {

            if (crt.serial.len == cur.serial.len
                    && MemUtil.memcmp(crt.serial.p, cur.serial.p, crt.serial.len) == 0) {
                /* Prefer the certificate invalidity date if available, since it reflects the time
                   the certificate became untrusted (e.g. when compromised) rather than when the CA
                   processed the revocation. This avoids erroneously trusting a revoked certificate
                   when the local clock has been manipulated.
                 */
                if (cur.invalidityDate != null) {
                    if (Helper.mbedtls_x509_time_is_past(cur.invalidityDate))
                        return 1;
                } else if (Helper.mbedtls_x509_time_is_past(cur.revocationDate)) {
                    return 1;
                }
            }

            cur = cur.next;
        }
        return 0;
    }
}
