    public static int mbedtls_x509_crt_is_revoked(MbedtlsX509Crt crt, MbedtlsX509Crl crl) {
        // Validate parameters to avoid unexpected null pointer dereference
        if (crt == null || crl == null || crt.serial == null) {
            // Return error code for invalid input.
            return -1;
        }
        // If there are no CRL entries, nothing to revoke.
        if (crl.entry == null) {
            return 0;
        }

        // Walk through the CRL list
        MbedtlsX509CrlEntry cur = crl.entry;
        while (cur != null && cur.serial != null && cur.serial.len != 0) {
            // Ensure that the serial numbers have the same length before comparison.
            if (crt.serial.len == cur.serial.len && crt.serial.len > 0) {
                // Use constant-time comparison to avoid timing attacks.
                // MessageDigest.isEqual is known to do constant-time comparison.
                if (java.security.MessageDigest.isEqual(crt.serial.p, cur.serial.p)) {
                    // The certificate serial matches an entry in the CRL,
                    // so it is revoked.
                    return 1;
                }
            }
            cur = cur.next;
        }
        // No matching serial found in the CRL, so the certificate is not revoked.
        return 0;
    }
