    public static int mbedtls_x509_crt_is_revoked(
            MbedtlsX509Crt crt, MbedtlsX509Crl crl) {
        MbedtlsX509CrlEntry cur = crl.entry;
        while (cur != null && cur.serial.len != 0) {
            // Ensure the certificate's serial length exactly matches the CRL entry's serial length.
            // Then compare the serial numbers in constant time to avoid timing side-channels.
            if (crt.serial.len == cur.serial.len) {
                boolean equal = true;
                for (int i = 0; i < crt.serial.len; i++) {
                    if (crt.serial.p[i] != cur.serial.p[i]) {
                        equal = false;
                        break;
                    }
                }
                if (equal) {
                    // Certificate is revoked.
                    return 1;
                }
            }
            cur = cur.next;
        }
        return 0;
    }
