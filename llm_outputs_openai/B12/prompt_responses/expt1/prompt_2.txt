    /* Check for public key of required type.
     *
     * Bugfix: previously the code did not validate that the peer certificate
     * actually contains a public key of the correct type matching the signature
     * algorithm provided. This allowed an attacker to supply a mismatched key.
     * The fix below verifies that the peer certificate exists and that its
     * public key type (RSA vs. ECC) matches the algorithm indicated in args->sigAlgo.
     */
    if (ssl->peerCert == NULL || ssl->peerCert->publicKey == NULL) {
        ret = SIG_VERIFY_ERROR;
        ERROR_OUT(ret, exit_dcv);
    }
    
    switch(args->sigAlgo) {
        case TLS_SIG_RSA:
            if (ssl->peerCert->publicKey->type != RSA_KEY_TYPE) {
                ret = SIG_VERIFY_ERROR;
                ERROR_OUT(ret, exit_dcv);
            }
            break;
        case TLS_SIG_ECDSA:
            if (ssl->peerCert->publicKey->type != ECC_KEY_TYPE) {
                ret = SIG_VERIFY_ERROR;
                ERROR_OUT(ret, exit_dcv);
            }
            break;
        default:
            ret = SIG_VERIFY_ERROR;
            ERROR_OUT(ret, exit_dcv);
    }
    
    /* Perform the signature verification using the validated public key.
     * This function is assumed to handle the MAC and digest computations
     * as specified by TLS 1.3 and compare against the signature buffer.
     */
    ret = wolfSSL_Tls13SignatureVerify(ssl, args, input, totalSz);
    if (ret < 0)
        goto exit_dcv;
    
    /* On success, update the consumed index for the certificate verify message */
    *inOutIdx = args->idx;
    
exit_dcv:
    WOLFSSL_LEAVE("DoTls13CertificateVerify");
    return ret;
