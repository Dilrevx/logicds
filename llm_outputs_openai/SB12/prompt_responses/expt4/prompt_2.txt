    if (!ssl.peerEd448KeyPresent) {
        Wslib.WOLFSSL_MSG("Error: ED448 signature received but the corresponding ED448 certificate is missing");
        return Commons.INVALID_SIGNATURE;
    }

if (args.sigAlgo == Commons.ecc_dsa_sa_algo) {
    if (!ssl.peerEccDsaKeyPresent) {
        Wslib.WOLFSSL_MSG("Error: ECC DSA signature received but the corresponding ECC certificate is missing");
        return Commons.INVALID_SIGNATURE;
    }
}

    Wslib.WOLFSSL_MSG("Error: PKCS#1.5 signature received; this algorithm is not supported in this context");
    return Commons.INVALID_PARAMETER;
}
