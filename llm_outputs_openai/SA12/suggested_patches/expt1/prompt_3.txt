package main;

import commons.*;

public class Tls13FinishedHandler {

    public static int DoTls13Finished(WOLFSSL ssl, byte[] input, int[] inOutIdx,
                                        int size, int totalSz, int sniff) {
        int ret;
        int finishedSz = 0;
        byte[] secret;
        byte[] mac = new byte[32];

        // Begin operation trace
        Tls13Helper.WOLFSSL_START(Constants.WC_FUNC_FINISHED_DO);

        // Validate function parameters to avoid logical vulnerabilities
        if (ssl == null || input == null || inOutIdx == null || inOutIdx.length < 1) {
            return Constants.BAD_FUNC_ARGUMENT;
        }

        // Use the current index from inOutIdx safely
        int currentIndex = inOutIdx[0];
        if (currentIndex < 0 || currentIndex >= totalSz) {
            return Constants.INVALID_INDEX;
        }
        if (currentIndex + size > input.length) {
            // The input buffer does not contain enough data for processing
            return Constants.BUFFER_TOO_SMALL;
        }
        if (totalSz < size) {
            return Constants.INVALID_ARGUMENT;
        }

        // Derive the finished message MAC. This function should perform any necessary secret derivation
        // and fill the 'mac' array with the expected value.
        ret = Tls13Helper.DeriveFinishedMessage(ssl, mac);
        if (ret != Constants.SUCCESS) {
            return ret;
        }

        // Securely compare the computed MAC with the segment from the input buffer.
        // This function should implement a constant-time comparison to prevent timing attacks.
        boolean macEqual = Tls13Helper.secureCompare(mac, input, currentIndex, size);
        if (!macEqual) {
            return Constants.FINISHED_VERIFICATION_FAILED;
        }
        finishedSz = size;

        // Update the index pointer after successful processing
        inOutIdx[0] = currentIndex + finishedSz;

        // (Optional) Additional processing, such as session state update, can be done here.

        return finishedSz;
    }

}
