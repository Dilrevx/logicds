package main;

import commons.*;

public class Tls13FinishedHandler {

        Tls13Helper.WOLFSSL_ENTER("DoTls13Finished");

        /* 
         * FIX: Enforce mutual authentication.
         * If the server requires a client certificate (mutual auth)
         * then the Certificate Verify message must have been received.
         * Otherwise we reject the handshake.
         *
         * Note: “ssl.reqMutualAuth” and “ssl.certVerifyReceived” are assumed
         * to be boolean fields in the ssl object representing whether client
         * authentication is required and whether a valid Certificate Verify
         * message has been processed respectively.
         */
        if (ssl.reqMutualAuth && !ssl.certVerifyReceived)
            return Constants.CERT_VERIFY_MISSING;

        if (inOutIdx[0] + size + ssl.keys.padSz > totalSz)
            return Constants.BUFFER_E;

        // Process the Finished message and compute expected MAC.
        ret = ComputeTls13FinishedMac(ssl, input, inOutIdx[0], size, mac);
        if (ret != Constants.SUCCESS)
            return ret;

        // Constant-time comparison of computed MAC vs expected value.
        if (!ConstantTimeEqual(mac, ssl.expectedFinishedMac, mac.length))
            return Constants.FINISHED_VERIFY_ERROR;

        finishedSz = size + ssl.keys.padSz;
        inOutIdx[0] += finishedSz;

        Tls13Helper.WOLFSSL_LEAVE("DoTls13Finished");
        return finishedSz;
    }

}
