/* Entry point for handling TLSv1.3 Finished message */
Tls13Helper.WOLFSSL_ENTER("DoTls13Finished");

/* Early mutual-authentication check:
   - Verify that if we are running as a server with mutual authentication enabled,
     the client's certificate must have been verified (unless the session is resumed).
*/
if (ssl.isServer && ssl.config.requireMutualAuth && !ssl.peerCertVerified &&
    !ssl.sessionResumed) {
    /* Trigger a fatal alert indicating that a valid client certificate is required */
    wolfSSL_Alert(ssl, WOLFSSL_TLS_FATAL, WOLFSSL_TLS_ALERT_CERTIFICATE_REQUIRED);
    return Constants.PEER_CERT_REQUIRED_E;
}

/* Check that the output buffer has enough available space */
if (inOutIdx[0] + size + ssl.keys.padSz > totalSz) {
    return Constants.BUFFER_E;
}
