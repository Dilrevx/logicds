             * Only a limited set of NAS messages is allowed in unprotected (plain) mode.
             * The allowed messages must be those sent by the UE before a security context is established.
             * In particular, the plain NAS messages from the UE SHOULD only be:
             *   - Attach Request
             *   - Service Request
             *   - Tracking Area Update Request
             *   - Detach Request
             *   - EMM Status
             *
             * Reject any other NAS messages coming in plain (i.e. without integrity protection).
             */
            switch (msg_type) {
                case LIBLTE_MME_MSG_TYPE_ATTACH_REQUEST:
                case LIBLTE_MME_MSG_TYPE_SERVICE_REQUEST:
                case LIBLTE_MME_MSG_TYPE_TRACKING_AREA_UPDATE_REQUEST:
                case LIBLTE_MME_MSG_TYPE_DETACH_REQUEST:
                case LIBLTE_MME_MSG_TYPE_EMM_STATUS:
                    break;
                default:
                    logger.error(
                            "Rejecting NAS message MSG_TYPE=%02X with SEC_HDR_TYPE=%02X: integrity protection required!",
                            msg_type,
                            sec_hdr_type
