    if (sec_hdr_type == LIBLTE_MME_SECURITY_HDR_TYPE_PLAIN_NAS) {
        switch (msg_type) {
            case LIBLTE_MME_MSG_TYPE_IDENTITY_REQUEST:
            case LIBLTE_MME_MSG_TYPE_AUTHENTICATION_REQUEST:
            case LIBLTE_MME_MSG_TYPE_AUTHENTICATION_REJECT:
            case LIBLTE_MME_MSG_TYPE_DETACH_ACCEPT:
            case LIBLTE_MME_MSG_TYPE_TRACKING_AREA_UPDATE_REJECT:
                /* These messages are allowed in plaintext. */
                break;
            case LIBLTE_MME_MSG_TYPE_ATTACH_REJECT:
            case LIBLTE_MME_MSG_TYPE_SERVICE_REJECT:
                /* Only allow Attach/Service Reject in plaintext if emm_cause equals 25.
                 * Any other emm_cause value must be rejected because these messages
                 * require integrity protection.
                 */
                if (emm_cause != 25) {
                    logger.error("Plain NAS message MSG_TYPE=%02X with SEC_HDR_TYPE=%02X requires emm_cause=#25 when sent without integrity protection!", msg_type, sec_hdr_type);
                    return SRSRAN_ERROR;
                }
                break;
            default:
                /* Disallow messages which used to be accepted in plaintext but should require integrity protection. 
                 * Specifically, this removes the plaintext acceptance of Detach Request, EMM Information, and EMM Status.
                 */
                logger.error("Not handling NAS message MSG_TYPE=%02X with SEC_HDR_TYPE=%02X without integrity protection!", msg_type, sec_hdr_type);
                return SRSRAN_ERROR;
        }
    }
