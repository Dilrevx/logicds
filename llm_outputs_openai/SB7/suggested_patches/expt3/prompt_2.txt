package main;

import commons.*;
public class X509Verifier {
    public static int x509_verify_ctx_add_chain(X509VerifyCtx ctx, X509VerifyChain chain, String name) {
        int depth;
        X509 last = X509Utils.x509_verify_chain_last(chain);
        X509 leaf = X509Utils.x509_verify_chain_leaf(chain);
        depth = X509Utils.sk_X509_num(chain.certs);
        if (depth > 0)
            depth--;
        if (ctx.chains_count >= ctx.max_chains)
            return X509Utils.x509_verify_cert_error(ctx, last, depth, X509Constants.X509_V_ERR_CERT_CHAIN_TOO_LONG, 0);
        if (chain.cert_errors[depth] == X509Constants.X509_V_ERR_UNABLE_TO_GET_ISSUER_CERT_LOCALLY)
            chain.cert_errors[depth] = X509Constants.X509_V_OK;
        if (!X509Utils.x509_verify_ctx_validate_legacy_chain(ctx, chain, depth))
            return 0;

    /* First, verify the certificate validity and hostname before duplicating the chain */
    if (!X509Utils.x509_verify_cert_valid(ctx, leaf, NULL))
        return 0;
    if (!X509Utils.x509_verify_cert_hostname(ctx, leaf, name))
        return 0;

    /*
     * For the leaf certificate (depth == 0), capture its error
     * unless the error is the specific issuer error (here assumed to be X509_V_ERR_ISSUER_CERT_UNKNOWN).
     * This preserves the original error even if a callback tells us to continue.
     */
    int saved_leaf_error = X509Constants.X509_V_OK;
    if (depth == 0 && ctx.error != X509Constants.X509_V_ERR_ISSUER_CERT_UNKNOWN)
        saved_leaf_error = ctx.error;

    /* Now duplicate the certificate chain after the certificate validations succeed */
    X509Chain *dupChain = (ctx.chains_count < ctx.max_chains)
                          ? X509Utils.x509_verify_chain_dup(chain)
                          : NULL;
    if (dupChain == NULL) {
        return X509Utils.x509_verify_cert_error(ctx, last, depth,
                           X509Constants.X509_V_ERR_OUT_OF_MEM, 0);
    }

    /* Save the duplicated chain */
    ctx.chains[ctx.chains_count++] = dupChain;

    /* Reset the context error to indicate success */
    ctx.error = X509Constants.X509_V_OK;

    /*
     * Record the error details in the duplicated chain's error mapping.
     * For the leaf certificate, if an error was captured, store it at index 0.
     * Also, update the error for the current depth.
     */
    if (depth == 0 && saved_leaf_error != X509Constants.X509_V_OK)
        dupChain->errors[0] = saved_leaf_error;
    dupChain->errors[depth] = ctx.error;

        ctx.error_depth = depth;
        return 1;
    }
}
