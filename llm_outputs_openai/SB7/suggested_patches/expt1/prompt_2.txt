package main;

import commons.*;
public class X509Verifier {

    if (!X509Utils.x509_verify_ctx_validate_legacy_chain(ctx, chain, depth))
        return 0;
    
    /*
     * Repaired logical vulnerability:
     * Do not unconditionally override the error indicating that the issuer certificate was not found.
     * Instead, only clear the error if the last certificate is actually a trusted anchor.
     */
    if (chain.cert_errors[depth] == X509Constants.X509_V_ERR_UNABLE_TO_GET_ISSUER_CERT_LOCALLY) {
        // Ensure that we only clear the error if the certificate is trusted.
        if (!X509Utils.x509_is_trusted_cert(last)) {
            return X509Utils.x509_verify_cert_error(ctx,
                                                    last,
                                                    depth,
                                                    X509Constants.X509_V_ERR_UNABLE_TO_GET_ISSUER_CERT_LOCALLY,
                                                    0);
        }
        chain.cert_errors[depth] = X509Constants.X509_V_OK;
    }
    
    /* Add the validated chain to the context and update the count */
    ctx.chains[ctx.chains_count++] = chain;
    return X509Constants.X509_V_OK;

}
