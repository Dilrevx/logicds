package main;

import commons.*;

public class OcspRespChecker {

    ret = Helper.gnutls_x509_crt_get_serial(crt, cserial, tHolder);
    if (ret < 0) {
        return ret;
    }
    /* Ensure that the certificate’s serial number length exactly matches the one
       in the OCSP response before proceeding. A mismatch here would indicate an
       invalid or manipulated response. */
    if (rserial.size != cserial.size) {
        return ErrorCodes.GNUTLS_E_OCSP_RESPONSE_ERROR;
    }
    /* Perform a constant‐time comparison between the certificate serial and the
       OCSP response serial to avoid timing attacks. */
    if (!Helper.constant_time_memcmp(rserial.data, cserial.data, cserial.size)) {
        return ErrorCodes.GNUTLS_E_OCSP_RESPONSE_ERROR;
    }
    return 0;

}
