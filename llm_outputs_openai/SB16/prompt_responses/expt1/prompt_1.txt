public static Frec lookup_frec(int id, byte[] hash, int family, int fd) {
    // Validate input to avoid null pointer dereferences.
    if (hash == null) {
        throw new IllegalArgumentException("The 'hash' parameter cannot be null.");
    }
    // Iterate through the linked list of Frec records.
    for (Frec f = Daemon.frec_list; f != null; f = f.next) {
        // First, ensure that the integer and other properties match.
        if (f.id == id && f.family == family && f.fd == fd) {
            // Also ensure the stored hash is not null and has the same length.
            if (f.hash != null && f.hash.length == hash.length) {
                // Use constant-time comparison to avoid timing attacks.
                if (java.security.MessageDigest.isEqual(f.hash, hash)) {
                    return f;
                }
            }
        }
    }
    return null;
}
