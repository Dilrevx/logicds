package main;
import commons.*;
import java.util.Arrays;
public class LookupFrec {
    public static Frec lookup_frec(int id, byte[] hash, int family, int fd) {
        for (Frec f = Daemon.frec_list; f != null; f = f.next) {

            if (f.sentto != null
                    && f.new_id == id
                    && (hash == null || Arrays.equals(hash, f.hash))) {

                // The reply’s file descriptor (replyFD) should have been obtained earlier
                // from the received packet's network metadata.
                boolean validNetworkPath = false;

                // Check the address family used for the original query.
                // For IPv4, ensure the IPv4-specific file descriptor matches.
                if (f.sentto.family == AF_INET) {
                    if (f.ipv4_fd != 0 && f.ipv4_fd == replyFD) {
                        validNetworkPath = true;
                    }
                }
                // For IPv6, ensure the IPv6-specific file descriptor matches.
                else if (f.sentto.family == AF_INET6) {
                    if (f.ipv6_fd != 0 && f.ipv6_fd == replyFD) {
                        validNetworkPath = true;
                    }
                }

                // As a fallback, check if the sent-to structure carries a specific socket
                // and if its file descriptor matches the reply's FD.
                if (!validNetworkPath && f.sentto.socket_fd != 0) {
                    if (f.sentto.socket_fd == replyFD) {
                        validNetworkPath = true;
                    }
                }

                if (validNetworkPath) {
                    return f;
                }
            }

        }
        return null;
    }
}
