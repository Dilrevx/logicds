package main;
import commons.*;
import java.util.Arrays;
public class LookupFrec {
    public static Frec lookup_frec(int id, byte[] hash, int family, int fd) {
        for (Frec f = Daemon.frec_list; f != null; f = f.next) {

            if (f.sentto != null
                    && f.new_id == id
                    && (hash == null || Arrays.equals(hash, f.hash))) {
                /* Verify that the reply was received on the same interface as the forwarded query. */
                if ((reply_family == AF_INET && f.ipv4_fd != -1 && f.ipv4_fd == reply_fd) ||
                    (reply_family == AF_INET6 && f.ipv6_fd != -1 && f.ipv6_fd == reply_fd) ||
                    (f.generic_fd != -1 && f.generic_fd == reply_fd)) {
                    return f;
                }
            }

        }
        return null;
    }
}
