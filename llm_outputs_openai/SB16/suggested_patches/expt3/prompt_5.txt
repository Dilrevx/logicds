package main;
import commons.*;
import java.util.Arrays;
public class LookupFrec {
    public static Frec lookup_frec(int id, byte[] hash, int family, int fd) {
        for (Frec f = Daemon.frec_list; f != null; f = f.next) {

/* Assume reply_fd and reply_af are provided.
   reply_af might be AF_INET for IPv4 or AF_INET6 for IPv6. */
if (f.sentto != null
        && f.new_id == id
        && (hash == null || Arrays.equals(hash, f.hash))) {

    boolean is_valid = false;

    /* Check for address family specific match. 
       For IPv4, verify that the expected IPv4 file descriptor is set and matches reply_fd */
    if (reply_af == AF_INET) {
        if (f.ipv4_fd != 0 && f.ipv4_fd == reply_fd) {
            is_valid = true;
        }
    }
    /* For IPv6, do the same using the IPv6 file descriptor */
    else if (reply_af == AF_INET6) {
        if (f.ipv6_fd != 0 && f.ipv6_fd == reply_fd) {
            is_valid = true;
        }
    }

    /* Additional fallback: if the sent-to structure includes a socket file descriptor,
       ensure that it also matches the reply file descriptor */
    if (!is_valid && f.sentto.sock_fd != 0 && f.sentto.sock_fd == reply_fd) {
        is_valid = true;
    }

    if (is_valid) {
        return f;
    }
}

        }
        return null;
    }
}
