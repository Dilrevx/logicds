package main;
import commons.*;
import java.util.Arrays;
public class LookupFrec {
    public static Frec lookup_frec(int id, byte[] hash, int family, int fd) {
        for (Frec f = Daemon.frec_list; f != null; f = f.next) {

            if (f.sentto != null
                    && f.new_id == id
                    && (hash == null || Arrays.equals(hash, f.hash))) {

                // Added extra checks to ensure that the reply comes from the same network interface/socket
                // that was used when forwarding the query.
                boolean validReply = false;

                // Check if the reply comes in via an IPv4 socket.
                if (family == AddressFamily.AF_INET && f.ipv4SocketFd >= 0) {
                    if (f.ipv4SocketFd == incomingFd) {
                        validReply = true;
                    }
                }
                // Check if the reply comes in via an IPv6 socket.
                else if (family == AddressFamily.AF_INET6 && f.ipv6SocketFd >= 0) {
                    if (f.ipv6SocketFd == incomingFd) {
                        validReply = true;
                    }
                }
                // Fallback: if no specific IPv4/IPv6 socket is set, use a generic interface/socket check.
                else if (f.socketFd >= 0) {
                    if (f.socketFd == incomingFd) {
                        validReply = true;
                    }
                }

                if (validReply) {
                    return f;
                }
            }

        }
        return null;
    }
}
