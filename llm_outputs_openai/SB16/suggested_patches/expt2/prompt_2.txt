package main;
import commons.*;
import java.util.Arrays;
public class LookupFrec {
    public static Frec lookup_frec(int id, byte[] hash, int family, int fd) {
        for (Frec f = Daemon.frec_list; f != null; f = f.next) {

            if (f.sentto != null
                    && f.new_id == id
                    && (hash == null || Arrays.equals(hash, f.hash))) {
                boolean valid = false;
                
                // Check that the reply came via the proper IPv4 channel.
                if (f.sentto.getFamily() == AddressFamily.INET) {  // assuming AddressFamily.INET indicates IPv4
                    if (f.sentto.getIPv4Fd() != -1 && replyFd == f.sentto.getIPv4Fd()) {
                        valid = true;
                    }
                }
                // Check that the reply came via the proper IPv6 channel.
                else if (f.sentto.getFamily() == AddressFamily.INET6) {  // assuming AddressFamily.INET6 indicates IPv6
                    if (f.sentto.getIPv6Fd() != -1 && replyFd == f.sentto.getIPv6Fd()) {
                        valid = true;
                    }
                }
                
                // Fallback: verify via an explicit socket (if present).
                if (!valid && f.sentto.getSocket() != null) {
                    if (replyFd == f.sentto.getSocket().getFd()) {
                        valid = true;
                    }
                }
                
                if (valid) {
                    return f;
                }
            }

        }
        return null;
    }
}
