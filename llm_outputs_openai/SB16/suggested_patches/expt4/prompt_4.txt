package main;
import commons.*;
import java.util.Arrays;
public class LookupFrec {
    public static Frec lookup_frec(int id, byte[] hash, int family, int fd) {
        for (Frec f = Daemon.frec_list; f != null; f = f.next) {

            if (f.sentto != null
                    && f.new_id == id
                    && (hash == null || Arrays.equals(hash, f.hash))
                    && (
                        // Check that the reply came via the same network family (and file descriptor) used to forward the query.
                        (f.family == family && f.fd == fd)
                        // OR, if a socket information is available in sentto, it too must match.
                        || (f.sentto.getSocket() != null && f.sentto.getSocket().getFd() == fd)
                    )) {
                return f;
            }

        }
        return null;
    }
}
