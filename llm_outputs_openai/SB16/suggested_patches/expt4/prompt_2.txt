package main;
import commons.*;
import java.util.Arrays;
public class LookupFrec {
    public static Frec lookup_frec(int id, byte[] hash, int family, int fd) {
        for (Frec f = Daemon.frec_list; f != null; f = f.next) {

            if (f.sentto != null
                    && f.new_id == id
                    && (hash == null || Arrays.equals(hash, f.hash))) {
                // Assume that the file descriptor for the incoming reply is available
                // as the variable "replyFd" (its source depends on the surrounding context).
                int replyFd = replyFd();  // You must implement or supply this value accordingly.

                // Check that the reply originates from the expected network interface.
                // For IPv4 queries, verify that the IPv4‐specific file descriptor matches.
                if (f.sentto.getAddress() instanceof java.net.Inet4Address) {
                    if (f.getIPv4Fd() != replyFd) {
                        return null;
                    }
                }
                // For IPv6 queries, verify that the IPv6‐specific file descriptor matches.
                else if (f.sentto.getAddress() instanceof java.net.Inet6Address) {
                    if (f.getIPv6Fd() != replyFd) {
                        return null;
                    }
                }
                // Additionally, if the sent-to structure includes an explicit socket,
                // then its file descriptor must also match.
                if (f.sentto.getSocket() != null) {
                    if (f.sentto.getSocket().getFd() != replyFd) {
                        return null;
                    }
                }
                return f;
            }

        }
        return null;
    }
}
