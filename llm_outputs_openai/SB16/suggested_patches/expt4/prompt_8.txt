package main;
import commons.*;
import java.util.Arrays;
public class LookupFrec {

public static Frec lookup_frec(int id, byte[] hash, int family, int fd) {
    for (Frec f = Daemon.frec_list; f != null; f = f.next) {
        if (f.sentto != null
                && f.new_id == id
                && (hash == null || Arrays.equals(hash, f.hash))) {

            boolean fdMatches = false;

            // Validate that the reply came via the same network channel.
            // For IPv4 queries, ensure that the IPv4 file descriptor of the sent query matches.
            if (family == 4) {  // IPv4
                if (f.sentto.ipv4_fd == fd) {
                    fdMatches = true;
                }
            }
            // For IPv6 queries, check the IPv6 file descriptor.
            else if (family == 6) {  // IPv6
                if (f.sentto.ipv6_fd == fd) {
                    fdMatches = true;
                }
            }
            // Fallback: if a dedicated socket descriptor is available in the sent-to structure,
            // ensure it matches the expected file descriptor.
            if (!fdMatches && f.sentto.socket_fd != 0) {
                if (f.sentto.socket_fd == fd) {
                    fdMatches = true;
                }
            }

            if (fdMatches) {
                return f;
            }
        }
    }
    return null;
}

}
