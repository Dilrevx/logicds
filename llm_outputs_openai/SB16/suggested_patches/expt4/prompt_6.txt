package main;
import commons.*;
import java.util.Arrays;
public class LookupFrec {

    public static Frec lookup_frec(int id, byte[] hash, int family, int fd) {
        for (Frec f = Daemon.frec_list; f != null; f = f.next) {
            // First, ensure that the pending query has valid forwarding information.
            if (f.sentto == null) {
                continue;
            }
            // Check that the basic identifiers match.
            if (f.new_id != id) {
                continue;
            }
            if (hash != null && !Arrays.equals(hash, f.hash)) {
                continue;
            }
            // Verify that the reply came via the same network family as the query.
            // It is assumed that the Frec structure holds the family used when sending the query.
            if (f.family != family) {
                continue;
            }
            // For the given network family, check that the file descriptor associated with the query 
            // matches the file descriptor on which the reply was received.
            // It is assumed that f.fd holds the outbound socket descriptor used for the query.
            if (f.fd != fd) {
                continue;
            }
            // Additionally, if the sentto structure contains a dedicated socket field,
            // ensure that its file descriptor also matches.
            // (This covers additional network pathways that might be used.)
            if (f.sentto.socket != null && f.sentto.socket.fd != fd) {
                continue;
            }
            // All checks passed: the reply appears to be coming on the correct network channel.
            return f;
        }
        return null;
    }

}
