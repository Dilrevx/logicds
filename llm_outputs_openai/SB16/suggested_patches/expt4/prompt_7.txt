package main;
import commons.*;
import java.util.Arrays;
public class LookupFrec {

    public static Frec lookup_frec(int id, byte[] hash, int family, int fd) {
        for (Frec f = Daemon.frec_list; f != null; f = f.next) {
            // Ensure the query has a valid destination and that its
            // reply destination (address family and port) matches the reply.
            if (f.sentto != null 
                    && f.new_id == id
                    && f.sentto.getPort() == fd
                    && (   (family == 4 && f.sentto.getAddress() instanceof java.net.Inet4Address)
                        || (family == 6 && f.sentto.getAddress() instanceof java.net.Inet6Address))
                    && (hash == null || java.util.Arrays.equals(hash, f.hash))) {
                return f;
            }
        }
        return null;
    }

}
