    public static int BuildMessage(TlsContext ssl,
                                   byte[] output,
                                   int outSz,
                                   byte[] input,
                                   int inSz,
                                   int type,
                                   int hashOutput,
                                   int sizeOnly,
                                   int asyncOkay,
                                   int epochOrder) {
        BuildMsgArgs args;
        BuildMsgArgs lcl_args = new BuildMsgArgs();
        
        // Validate required pointers and size parameters.
        if (ssl == null || output == null || input == null) {
            return CommonsConstants.BAD_FUNC_ARG;
        }
        if (outSz < 0 || inSz < 0 || output.length < outSz || input.length < inSz) {
            return CommonsConstants.BAD_FUNC_ARG;
        }
        
        // Ensure that the input message is within an acceptable boundary.
        if (inSz > CommonsConstants.MAX_MESSAGE_LENGTH) {
            return CommonsConstants.BAD_MESSAGE_LENGTH;
        }
        
        args = lcl_args;
        ssl.options.buildMsgState = CommonsConstants.BUILD_MSG_BEGIN;
        
        if (ssl.options.tls1_1) {
            args.ivSz = CommonsConstants.BLOCK_SIZE;
        }
        
        // Bugfix: fixed Logical Vulnerability by ensuring the output buffer is large enough
        // to account for both the header and the message payload.
        // Assume the header size is defined in CommonsConstants.
        int headerSize = CommonsConstants.MSG_HEADER_SIZE;
        int requiredLength = headerSize + inSz;
        if (outSz < requiredLength) {
            return CommonsConstants.OUTPUT_BUFFER_TOO_SMALL;
        }
        
        // Build the message header.
        // For example, write the 'type' and 'epochOrder' into the header in big-endian order.
        int offset = 0;
        output[offset++] = (byte) type;
        output[offset++] = (byte) ((epochOrder >> 24) & 0xFF);
        output[offset++] = (byte) ((epochOrder >> 16) & 0xFF);
        output[offset++] = (byte) ((epochOrder >> 8) & 0xFF);
        output[offset++] = (byte) (epochOrder & 0xFF);
        
        // Optionally, populate additional header fields as required.
        // For instance, if there is room, you might fill in length, hash, flags, etc.
        
        // Copy the input data safely into the output buffer after the header.
        System.arraycopy(input, 0, output, offset, inSz);
        
        // Return the total number of bytes written.
        return requiredLength;
    }
