    /* First try to validate against the Subject Alternative Names (SAN) using a dedicated helper
       that enforces strict type checking (e.g. only DNS names are considered valid).
       If the helper returns 0, a valid SAN match has been found and no further check is needed. */
    if( crt->ext_types & MBEDTLS_X509_EXT_SUBJECT_ALT_NAME )
    {
        if( x509_crt_verify_san( crt, cn, cn_len ) == 0 )
        {
            /* A valid SAN was found: certificate is accepted */
            return;
        }
        /* If the SAN check did not yield a valid match, fall back to checking the subject CN */
    }
    
    /* Fallback: Iterate through the subject's common name (CN) entries.
       Only entries with the proper CN OID are evaluated, ensuring type-safe matching. */
    for( name = &crt->subject; name != NULL; name = name->next )
    {
        if( MBEDTLS_OID_CMP( MBEDTLS_OID_AT_CN, &name->oid ) == 0 &&
            x509_crt_check_cn( &name->val, cn, cn_len ) == 0 )
        {
            /* Valid subject CN found: certificate is accepted */
            return;
        }
    }
    
    /* Neither a valid SAN nor subject CN was found: mark the certificate as having a CN mismatch */
    *flags |= MBEDTLS_X509_BADCERT_CN_MISMATCH;
