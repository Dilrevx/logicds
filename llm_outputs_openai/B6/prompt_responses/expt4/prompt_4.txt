    /* First, if the certificate contains a Subject Alternative Name (SAN) extension,
       use a specialized helper function to verify the expected name against only
       the permitted SAN entry types (e.g. DNS names). */
    if( crt->ext_types & MBEDTLS_X509_EXT_SUBJECT_ALT_NAME )
    {
        if( x509_crt_verify_san( crt, cn, cn_len ) == 0 )
            return;
    }

    /* Fallback: Check the subject field for a Common Name (CN) match. */
    for( name = &crt->subject; name != NULL; name = name->next )
    {
        if( MBEDTLS_OID_CMP( MBEDTLS_OID_AT_CN, &name->oid ) == 0 &&
            x509_crt_check_cn( &name->val, cn, cn_len ) == 0 )
        {
            return;
        }
    }

    /* If neither the securely verified SAN nor the subject CN matches,
       mark the certificate as having a common name mismatch. */
    *flags |= MBEDTLS_X509_BADCERT_CN_MISMATCH;
