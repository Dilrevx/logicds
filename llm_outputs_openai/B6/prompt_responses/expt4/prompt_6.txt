static int x509_crt_verify_san( const mbedtls_x509_crt *crt,
                                const char *cn, size_t cn_len )
{
    const mbedtls_x509_sequence *cur;
    /*
     * Iterate over the Subject Alternative Names (SAN) and only consider
     * entries of an allowed type. Here we assume that only DNS names (as indicated
     * by MBEDTLS_X509_SAN_DNS) are permitted for name matching.
     * (Note: In your actual implementation, ensure that MBEDTLS_X509_SAN_DNS
     * exactly corresponds to the correct ASN.1 tag for DNS names as parsed into cur->buf.)
     */
    for( cur = crt->subject_alt_names; cur != NULL; cur = cur->next )
    {
        if( cur->buf.tag == MBEDTLS_X509_SAN_DNS &&
            x509_crt_check_cn( &cur->buf, cn, cn_len ) == 0 )
        {
            /* A proper DNSName match was found in the SAN extension */
            return 0;
        }
    }
    return -1;
}

static void x509_crt_verify_name( const mbedtls_x509_crt *crt,
                                  const char *cn,
                                  uint32_t *flags )
{
    const mbedtls_x509_name *name;
    size_t cn_len = strlen( cn );

    /* If the certificate has a Subject Alternative Name (SAN) extension,
       use the new helper function to perform type‐aware verification.
       This ensures that only permitted SAN types (e.g. DNS names) are compared. */
    if( crt->ext_types & MBEDTLS_X509_EXT_SUBJECT_ALT_NAME )
    {
        if( x509_crt_verify_san( crt, cn, cn_len ) == 0 )
        {
            /* Valid SAN match found: no further check required. */
            return;
        }
    }

    /* Fallback to checking the subject's common name */
    for( name = crt->subject; name != NULL; name = name->next )
    {
        if( MBEDTLS_OID_CMP( MBEDTLS_OID_AT_CN, &name->oid ) == 0 &&
            x509_crt_check_cn( &name->val, cn, cn_len ) == 0 )
        {
            /* Found a matching common name in the subject field */
            return;
        }
    }

    /* No valid name was found – set the flag to indicate a Common Name mismatch */
    *flags |= MBEDTLS_X509_BADCERT_CN_MISMATCH;
}
