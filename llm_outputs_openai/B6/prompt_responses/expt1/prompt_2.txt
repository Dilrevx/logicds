    /* First, check if a Subject Alternative Name (SAN) is present and matches */

        {

            if( cur->buf.len == cn_len &&
                memcmp( cur->buf.p, cn, cn_len ) == 0 )
            {
                /* Found a matching SAN, certificate name verified */
                return;
            }
        }
    }

    /* If no SAN matched, fall back to verifying the Common Name (CN) in the subject */
    for( name = &crt->subject; name != NULL; name = name->next )
    {
        /* Check if this subject name entry corresponds to the Common Name OID.
           MBEDTLS_OID_AT_CN is used for the CN.
           (Depending on your configuration this OID may be defined elsewhere.) */
        if( MBEDTLS_OID_CMP( MBEDTLS_OID_AT_CN, &name->oid ) == 0 )
        {
            if( name->val.len == cn_len &&
                memcmp( name->val.p, cn, cn_len ) == 0 )
            {
                /* Found a matching CN, certificate name verified */
                return;
            }
        }
    }

    /* If neither the SAN nor the CN matches the given name, mark a mismatch */
    *flags |= MBEDTLS_X509_BADCERT_CN_MISMATCH;
