/* Helper function to verify SAN entries of the appropriate type (only DNS names are considered valid for CN matching) */
static int x509_crt_check_san( const mbedtls_x509_sequence *san_list,
                               const unsigned char *cn, size_t cn_len )
{
    const mbedtls_x509_sequence *cur;

    for( cur = san_list; cur != NULL; cur = cur->next )
    {
        /*
         * Only process SAN entries that are DNS names.
         * We assume that the x509 buffer (cur->buf) carries an identifier for the SAN type.
         * The field 'tag' here is used to denote the type.
         * Replace MBEDTLS_X509_SAN_DNS_NAME with the appropriate constant for DNS names.
         */
        if( cur->buf.tag == MBEDTLS_X509_SAN_DNS_NAME )
        {
            if( x509_crt_check_cn( &cur->buf, cn, cn_len ) == 0 )
                return 0; /* Valid match found */
        }
    }

    return -1; /* No valid SAN match was found */
}

if( crt->ext_types & MBEDTLS_X509_EXT_SUBJECT_ALT_NAME )
{
    /* Attempt to match the expected name against valid DNS names in the SAN */
    if( x509_crt_check_san( &crt->subject_alt_names, cn, cn_len ) == 0 )
    {
        /* Successful match found: return early or continue as valid */
    }
    else
    {
        *flags |= MBEDTLS_X509_BADCERT_CN_MISMATCH;
    }
}
else
{
    const mbedtls_x509_sequence *name;
    for( name = &crt->subject; name != NULL; name = name->next )
    {
        if( MBEDTLS_OID_CMP( MBEDTLS_OID_AT_CN, &name->oid ) == 0 &&
            x509_crt_check_cn( &name->val, cn, cn_len ) == 0 )
        {
            /* Valid subject common name match found; stop further processing */
            break;
        }
    }

    if( name == NULL )
        *flags |= MBEDTLS_X509_BADCERT_CN_MISMATCH;
}
