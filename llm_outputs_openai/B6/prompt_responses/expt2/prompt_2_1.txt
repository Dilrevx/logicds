    /*
     * If the certificate contains a Subject Alternative Name (SAN) extension,
     * use a specialized function to verify that one of the SAN entries matches
     * the expected common name. This function is responsible for filtering out
     * unsuitable SAN types (e.g. numeric IP addresses) so that only valid names
     * are considered.
     */
    if( crt->ext_types & MBEDTLS_X509_EXT_SUBJECT_ALT_NAME )
    {
        if( x509_crt_check_subject_alt_name( crt, cn, cn_len ) == 0 )
        {
            /* A valid match has been found in the SAN extension.
             * Verification succeeds so we can return without further processing.
             */
            return;
        }
        else
        {
            /* None of the acceptable SAN entries matched. */
            *flags |= MBEDTLS_X509_BADCERT_CN_MISMATCH;
            return;
        }
    }
    else
    {
        /* Fallback to checking the certificate's subject field if SAN is absent. */
        for( name = &crt->subject; name != NULL; name = name->next )
        {
            /* Only examine names with the common name OID. */
            if( MBEDTLS_OID_CMP( MBEDTLS_OID_AT_CN, &name->oid ) == 0 &&
                x509_crt_check_cn( &name->val, cn, cn_len ) == 0 )
            {
                /* Common name in subject matches expected value. */
                return;
            }
        }
        /* No matching common name found in the subject field. */
        *flags |= MBEDTLS_X509_BADCERT_CN_MISMATCH;
        return;
    }
