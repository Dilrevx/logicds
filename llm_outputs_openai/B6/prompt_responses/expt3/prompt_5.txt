/* Helper function: verifies acceptable SAN entries (e.g., DNS names) only.
   Returns 0 if an acceptable match for the provided cn is found,
   or a non-zero value otherwise.
   Note: You must update the condition in the SAN type check if needed.
*/
int x509_crt_verify_san( const mbedtls_x509_crt *crt,
                         const char *cn, size_t cn_len )
{
    mbedtls_x509_sequence *cur;
    for( cur = &crt->subject_alt_names; cur != NULL; cur = cur->next )
    {
        /* Process only permitted SAN types (e.g., DNS names). 
           The condition here should ensure that, for example,
           raw IP addresses are skipped.
           For this example, we assume that cur->buf contains a DNS name,
           so the check always runs. Update this condition based on your rules.
        */
        if( /* check that cur->buf contains a proper DNS name, not an IP address */ 1 )
        {
            if( x509_crt_check_cn( &cur->buf, cn, cn_len ) == 0 )
                return 0;
        }
    }
    return -1;
}

/* Main verification function that checks the certificate's CN.
   It first attempts to verify using the SAN entries (if present) and then
   falls back to verifying the subject's CN.
*/
void verify_certificate_cn( const mbedtls_x509_crt *crt,
                            const char *cn, size_t cn_len,
                            unsigned int *flags )
{
    mbedtls_x509_name *name;

    /* If the certificate includes a SAN extension, verify SAN entries first */
    if( crt->ext_types & MBEDTLS_X509_EXT_SUBJECT_ALT_NAME )
    {
        if( x509_crt_verify_san( crt, cn, cn_len ) == 0 )
        {
            /* A valid SAN match was found, certificate accepted */
            return;
        }
    }

    /* Fallback: iterate over the subject's distinguished name to find a matching CN */
    for( name = &crt->subject; name != NULL; name = name->next )
    {
        if( MBEDTLS_OID_CMP( MBEDTLS_OID_AT_CN, &name->oid ) == 0 &&
            x509_crt_check_cn( &name->val, cn, cn_len ) == 0 )
        {
            /* A valid CN in the subject was found, certificate accepted */
            return;
        }
    }

    /* Neither SAN nor subject CN had a valid match; flag the certificate as invalid */
    *flags |= MBEDTLS_X509_BADCERT_CN_MISMATCH;
}
