	 * Verify the leaf certificate.  If the certificate is not valid,
	 * capture the error and return it via x509_verify_cert_error.
	 */
	if (!x509_verify_cert_valid(ctx, leaf, NULL)) {
		int err = ctx->error;
		if (err == X509_V_OK)
			err = X509_V_ERR_CERT_INVALID;  /* default error if none set */
		return x509_verify_cert_error(ctx, leaf, depth, err, 0);
	}

	/*
	 * Check the hostname.  If the hostname check fails, record the error.
	 */
	if (!x509_verify_cert_hostname(ctx, leaf, name)) {
		int err = ctx->error;
		if (err == X509_V_OK)
			err = X509_V_ERR_HOSTNAME_MISMATCH;
		return x509_verify_cert_error(ctx, leaf, depth, err, 0);
	}

	ctx->chains_count++;
	ctx->error = X509_V_OK;
	ctx->error_depth = depth;
