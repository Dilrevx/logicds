static int
x509_verify_ctx_add_chain(struct x509_verify_ctx *ctx,
    struct x509_verify_chain *chain, char *name)
{
    size_t depth;
    X509 *last = x509_verify_chain_last(chain);
    X509 *leaf = x509_verify_chain_leaf(chain);

    depth = sk_X509_num(chain->certs);
    if (depth > 0)
        depth--;

    if (ctx->chains_count >= ctx->max_chains)
        return x509_verify_cert_error(ctx, last, depth,
                                      X509_V_ERR_CERT_CHAIN_TOO_LONG, 0);

    /* Clear a get issuer failure for a root certificate. */
    if (chain->cert_errors[depth] == X509_V_ERR_UNABLE_TO_GET_ISSUER_CERT_LOCALLY)
        chain->cert_errors[depth] = X509_V_OK;

    if (!x509_verify_ctx_validate_legacy_chain(ctx, chain, depth))
        return 0;

    /*
     * Verify leaf certificate validity before duplicating the chain.
     * If verification fails and an error has been recorded in the context,
     * capture that error in the chain’s error array (unless it is the specific
     * "unable to get issuer cert locally" error already cleared above).
     */
    if (!x509_verify_cert_valid(ctx, leaf, NULL)) {
        if (chain->cert_errors[depth] == X509_V_OK &&
            ctx->error != X509_V_OK &&
            ctx->error != X509_V_ERR_UNABLE_TO_GET_ISSUER_CERT_LOCALLY)
            chain->cert_errors[depth] = ctx->error;
        return 0;
    }

    if (!x509_verify_cert_hostname(ctx, leaf, name)) {
        if (chain->cert_errors[depth] == X509_V_OK &&
            ctx->error != X509_V_OK &&
            ctx->error != X509_V_ERR_UNABLE_TO_GET_ISSUER_CERT_LOCALLY)
            chain->cert_errors[depth] = ctx->error;
        return 0;
    }

    /*
     * Now that the leaf certificate has been successfully verified and any
     * error captured, duplicate the chain. This ensures that errors during the
     * leaf verification are not lost if the verification callback advised
     * continued processing on an invalid certificate.
     */
    ctx->chains[ctx->chains_count] = x509_verify_chain_dup(chain);
    if (ctx->chains[ctx->chains_count] == NULL)
        return x509_verify_cert_error(ctx, last, depth,
                                      X509_V_ERR_OUT_OF_MEM, 0);

    ctx->chains_count++;
    ctx->error = X509_V_OK;
    ctx->error_depth = depth;
    return 1;
}
