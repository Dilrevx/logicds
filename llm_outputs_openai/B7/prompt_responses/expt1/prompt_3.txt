	/* Clear a get issuer failure for a root certificate only if it is truly self‐issued.
	 * Without this check a non–self-signed certificate might wrongly have its error cleared.
	 */
	if (X509_check_issued(last, last) == X509_V_OK &&
	    chain->cert_errors[depth] == X509_V_ERR_UNABLE_TO_GET_ISSUER_CERT_LOCALLY)
		chain->cert_errors[depth] = X509_V_OK;

	/* Add the chain to the context.  We assume that ctx->chains is a stack of chains. */
	if (!sk_X509_verify_chain_push(ctx->chains, chain))
		return x509_verify_cert_error(ctx, last, depth,
		    X509_V_ERR_OUT_OF_MEM, 1);

	ctx->chains_count++;
	return 1;
}
