The repair removes the explicit allowance for the message type "integrity and ciphered with new EPS security context" from the secure header switch so that any message using that header type will be caught by the default case and rejected. In the original code, this problematic header type was mistakenly treated similarly to “plain NAS” and “service request” messages, which meant that the code did not enforce any integrity check on these messages. This oversight could allow messages to bypass the necessary validation step, opening the door to potential attacks.

In the repaired code, the case for LIBLTE_MME_SECURITY_HDR_TYPE_INTEGRITY_WITH_NEW_EPS_SECURITY_CONTEXT has been removed from the list of cases that simply break out of the switch. Now, only the explicitly safe types (like plain NAS and service requests) are allowed without further integrity checking. For headers that require integrity verification (such as LIBLTE_MME_SECURITY_HDR_TYPE_INTEGRITY and LIBLTE_MME_SECURITY_HDR_TYPE_INTEGRITY_AND_CIPHERED), the code remains unchanged: it performs an integrity check and, for ciphered messages, proceeds with decryption if the check passes. Any incoming message with a header type not explicitly permitted (including the previously problematic new EPS security context type) falls into the default case, which logs the error and rejects the message outright.

By forcing messages with the new EPS security header through the default error path rather than allowing them through unchecked, the repair ensures that every message undergoes the appropriate integrity verification. This effectively mitigates the vulnerability because now there is no possibility for a message using that header type to bypass integrity checking. The overall security posture is improved by guaranteeing that only messages with fully validated security properties are processed further.

    public void write_pdu(int lcid, UniqueByteBuffer pdu) {
        int pd = 0;
        int msg_type = 0;
        int sec_hdr_type = 0;

        logger.info(pdu.msg, pdu.N_bytes, "DL %s PDU", rrc.getRbName(lcid));

        RefHolder pdHolder = new RefHolder();
        RefHolder secHdrHolder = new RefHolder();
        LiblteMme.parseMsgSecHeader(pdu.get(), pdHolder, secHdrHolder);
        sec_hdr_type = secHdrHolder.value;
        switch (sec_hdr_type) {
            case LIBLTE_MME_SECURITY_HDR_TYPE_PLAIN_NAS:
            case LIBLTE_MME_SECURITY_HDR_TYPE_SERVICE_REQUEST:
                break;
            case LIBLTE_MME_SECURITY_HDR_TYPE_INTEGRITY:
            case LIBLTE_MME_SECURITY_HDR_TYPE_INTEGRITY_AND_CIPHERED:
                if (integrity_check(pdu.get())) {
                    if (sec_hdr_type == LIBLTE_MME_SECURITY_HDR_TYPE_INTEGRITY_AND_CIPHERED) {
                        cipher_decrypt(pdu.get());
                    }
                    break;
                } else {
                    logger.error("Not handling NAS message with integrity check error");
                    return;
                }
            default:
                logger.error("Not handling NAS message with SEC_HDR_TYPE=%02X", sec_hdr_type);
                return;
        }

        if (pcap != null) {
            pcap.writeNas(pdu.msg, pdu.N_bytes);
        }

        RefHolder msgTypeHolder = new RefHolder();
        LiblteMme.parseMsgHeader(pdu.get(), pdHolder, msgTypeHolder);
        msg_type = msgTypeHolder.value;
        logger.info(pdu.msg, pdu.N_bytes, "DL %s Decrypted PDU", rrc.getRbName(lcid));

        if (sec_hdr_type == LIBLTE_MME_SECURITY_HDR_TYPE_PLAIN_NAS) {
            switch (msg_type) {
                case LIBLTE_MME_MSG_TYPE_IDENTITY_REQUEST:
                case LIBLTE_MME_MSG_TYPE_EMM_INFORMATION:
                case LIBLTE_MME_MSG_TYPE_EMM_STATUS:
                case LIBLTE_MME_MSG_TYPE_AUTHENTICATION_REQUEST:
                case LIBLTE_MME_MSG_TYPE_AUTHENTICATION_REJECT:
                case LIBLTE_MME_MSG_TYPE_ATTACH_REJECT:
                case LIBLTE_MME_MSG_TYPE_DETACH_REQUEST:
                case LIBLTE_MME_MSG_TYPE_DETACH_ACCEPT:
                case LIBLTE_MME_MSG_TYPE_TRACKING_AREA_UPDATE_REJECT:
                case LIBLTE_MME_MSG_TYPE_SERVICE_REJECT:
                    break;
                default:
                    logger.error(
                            "Not handling NAS message MSG_TYPE=%02X with SEC_HDR_TYPE=%02X without integrity protection!",
                            msg_type,
                            sec_hdr_type
                    );
                    return;
            }
        }

        // The previous specific handling for LIBLTE_MME_SECURITY_HDR_TYPE_INTEGRITY_WITH_NEW_EPS_SECURITY_CONTEXT
        // has been removed to enforce integrity checks for such messages.

        switch (msg_type) {
            case LIBLTE_MME_MSG_TYPE_ATTACH_ACCEPT:
                parseAttachAccept(lcid, pdu);
                break;
            case LIBLTE_MME_MSG_TYPE_ATTACH_REJECT:
                parseAttachReject(lcid, pdu);
                break;
            case LIBLTE_MME_MSG_TYPE_AUTHENTICATION_REQUEST:
                parseAuthenticationRequest(lcid, pdu, sec_hdr_type);
                break;
            default:
                logger.error("Not handling NAS message with MSG_TYPE=%02X", msg_type);
                return;
        }
    }
