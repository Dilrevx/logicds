		/* Generate a new client random when:
		   - The handshake flag HSK_HRR_RECEIVED is not set, AND
		   - Either the session is not DTLS, or it is DTLS with no hello verify request processed.
		   In other words, only skip generating a new random when in a DTLS session that has already processed one or more hello verify requests.
		*/
		if (!(session->internals.hsk_flags & HSK_HRR_RECEIVED) &&
		    !(IS_DTLS(session) && session->internals.dtls.hsk_hello_verify_requests != 0)) {
			ret = _gnutls_gen_client_random(session);
			if (ret < 0) {
				gnutls_assert();
				goto cleanup;
			}
		}
