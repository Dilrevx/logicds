		/* If not resuming or retrying the handshake, generate a fresh client random.
		 * For DTLS sessions, we only skip generating a new random value if a Hello Verify Request
		 * has already been processed (indicated by hsk_hello_verify_requests != 0).
		 * This ensures that a new, random client random value is generated for all new handshakes,

		if (!(session->internals.hsk_flags & HSK_HRR_RECEIVED) &&
		    !(IS_DTLS(session) && session->internals.dtls.hsk_hello_verify_requests != 0)) {
			ret = _gnutls_gen_client_random(session);
			if (ret < 0) {
				gnutls_assert();
				goto cleanup;
			}
		}
