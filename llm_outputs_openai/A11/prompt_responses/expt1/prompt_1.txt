int mbedtls_x509_crt_is_revoked( const mbedtls_x509_crt *crt, const mbedtls_x509_crl *crl )
{
    /* Defensive checks to ensure the pointers are valid */
    if( crt == NULL || crl == NULL )
        return( 0 );

    /*
     * Iterate through the CRL chain. The original code never advanced to the next entry,
     * which could lead to an infinite loop and potential denial‐of‐service.
     */
    const mbedtls_x509_crl_entry *cur = &crl->entry;

    while( cur != NULL && cur->serial.len != 0 )
    {
        /* Compare the certificate's serial number with the current CRL entry's serial */
        if( crt->serial.len == cur->serial.len &&
            mbedtls_platform_memequal( crt->serial.p, cur->serial.p, crt->serial.len ) )
        {
            return( 1 );  /* Match found: Certificate is revoked. */
        }

        /* Advance to the next CRL entry to avoid an infinite loop */
        cur = cur->next;
    }

    return( 0 );  /* No matching CRL entry was found, so certificate isn't revoked. */
}
