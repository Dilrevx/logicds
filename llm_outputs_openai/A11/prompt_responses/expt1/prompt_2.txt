int mbedtls_x509_crt_is_revoked( const mbedtls_x509_crt *crt, const mbedtls_x509_crl *crl )
{
    const mbedtls_x509_crl_entry *cur = &crl->entry;

    while( cur != NULL )
    {
        /*
         * The CRL entries are chained and the last (or an invalid) entry is indicated
         * by a zero length serial. Explicitly checking this inside the loop (instead
         * of as part of the while condition) avoids a logical flaw when processing
         * valid entries.
         */
        if( cur->serial.len == 0 )
            break;

        if( crt->serial.len == cur->serial.len &&
            memcmp( crt->serial.p, cur->serial.p, crt->serial.len ) == 0 )
        {
            return( 1 );
        }

        cur = cur->next;
    }
    return( 0 );
}
