int mbedtls_x509_crt_is_revoked( const mbedtls_x509_crt *crt, const mbedtls_x509_crl *crl )
{
    const mbedtls_x509_crl_entry *cur = &crl->entry;

    while( cur != NULL && cur->serial.len != 0 )
    {
        /* Compare the certificate's serial number with the CRL entry
           using both length and memory comparison. */
        if( crt->serial.len == cur->serial.len &&
            memcmp( crt->serial.p, cur->serial.p, crt->serial.len ) == 0 )
        {
            return 1;  /* Certificate is revoked */
        }

        /* Advance to the next entry in the CRL.
           The previous implementation did not update the pointer,
           which could lead to an infinite loop. */
        cur = cur->next;
    }

    return 0;  /* Certificate is not found in the CRL, hence not revoked */
}
