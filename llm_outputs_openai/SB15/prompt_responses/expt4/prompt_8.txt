    public static int check_dnskey_sigs(Vctx vctx,
                                        DnsRdataDnskey dnskey,
                                        RData rdata,
                                        boolean is_ksk)
    {
        byte[] active_keys  = is_ksk ? vctx.ksk_algorithms : vctx.zsk_algorithms;
        byte[] standby_keys = is_ksk ? vctx.standby_ksk    : vctx.standby_zsk;
        boolean[] goodkey   = is_ksk ? vctx.goodksk         : vctx.goodzsk;

        if (Dnssec.dns_dnssec_selfsigns(rdata,
                vctx.origin, vctx.keyset, vctx.keysigs, false, vctx.mctx))
        {
            if ((active_keys[dnskey.algorithm] & 0xFF) != 255) {
                active_keys[dnskey.algorithm]++;
            }
        }
        else if (!is_ksk &&
              Dnssec.dns_dnssec_signs(rdata,
                        vctx.origin, vctx.soaset, vctx.soasigs, false, vctx.mctx))
        {
            if ((active_keys[dnskey.algorithm] & 0xFF) != 255) {
                active_keys[dnskey.algorithm]++;
            }
            return Commons.ISC_R_SUCCESS;
        }
        else {
            if ((standby_keys[dnskey.algorithm] & 0xFF) != 255) {
                standby_keys[dnskey.algorithm]++;
            }
            return Commons.ISC_R_SUCCESS;
        }

        // If there are no secure roots configured, mark the key as good.
        if (vctx.secroots == null) {
            goodkey[0] = true;
            return Commons.ISC_R_SUCCESS;
        }

        // Extract the key from the provided rdata.
        DstKey[] keyOut = new DstKey[1];
        int result = Dnssec.dns_dnssec_keyfromrdata(
                vctx.origin, rdata, vctx.mctx, keyOut);
        if (result != Commons.ISC_R_SUCCESS) {
            // Free any allocated key (if any) and exit early.
            if (keyOut[0] != null) {
                Dst.dst_key_free(keyOut);
            }
            return Commons.ISC_R_SUCCESS;
        }
        DstKey key = keyOut[0];

        // Loop over all matching key nodes in the secure key table.
        boolean foundValidKey = false;
        while (true) {
            Keynode[] knOut = new Keynode[1];
            int searchResult = Keytable.dns_keytable_findkeynode(
                    vctx.secroots, vctx.origin,
                    Dst.dst_key_alg(key), Dst.dst_key_id(key),
                    knOut);
            if (searchResult != Commons.ISC_R_SUCCESS) {
                // No more key nodes found.
                break;
            }
            Keynode keynode = knOut[0];
            // Use a secure comparison function to verify that this key node matches the expected key.
            if (secureCompare(keynode, key)) {
                // Detach the matching key node as it is now verified.
                Keytable.dns_keytable_detachkeynode(vctx.secroots, new Keynode[]{keynode});
                Rdataset.dns_rdataset_settrust(vctx.keyset,  Commons.dns_trust_secure);
                Rdataset.dns_rdataset_settrust(vctx.keysigs,  Commons.dns_trust_secure);
                goodkey[0] = true;
                foundValidKey = true;
                break;
            } else {
                // Detach key nodes that do not match to remove any forged data.
                Keytable.dns_keytable_detachkeynode(vctx.secroots, new Keynode[]{keynode});
            }
        }

        // In case any lingering key nodes remain in the table, remove them.
        // This extra loop ensures cleanup even if no valid key was found.
        Keynode[] cleanupNode = new Keynode[1];
        while (Keytable.dns_keytable_findkeynode(vctx.secroots, vctx.origin,
                Dst.dst_key_alg(key), Dst.dst_key_id(key), cleanupNode) == Commons.ISC_R_SUCCESS) {
            Keytable.dns_keytable_detachkeynode(vctx.secroots, new Keynode[]{cleanupNode[0]});
        }

        // Free the memory associated with the key extracted from rdata.
        Dst.dst_key_free(keyOut);

        return Commons.ISC_R_SUCCESS;
    }

    private static boolean constantTimeArrayCompare(byte[] a, byte[] b) {
        if (a == null || b == null || a.length != b.length) {
            return false;
        }
        int result = 0;
        for (int i = 0; i < a.length; i++) {
