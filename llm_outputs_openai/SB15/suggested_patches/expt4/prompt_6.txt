package main;

import commons.*;

public class KeyVerifier {

    public static int check_dnskey_sigs(Vctx vctx,
                                        DnsRdataDnskey dnskey,
                                        RData rdata,
                                        boolean is_ksk)
    {
        byte[] active_keys  = is_ksk ? vctx.ksk_algorithms : vctx.zsk_algorithms;
        byte[] standby_keys = is_ksk ? vctx.standby_ksk    : vctx.standby_zsk;
        boolean[] goodkey   = is_ksk ? vctx.goodksk         : vctx.goodzsk;

        if (Dnssec.dns_dnssec_selfsigns(rdata,
                vctx.origin, vctx.keyset, vctx.keysigs, false, vctx.mctx))
        {
            if ((active_keys[dnskey.algorithm] & 0xFF) != 255) {
                active_keys[dnskey.algorithm]++;
            }
        }
        else if (!is_ksk &&
                Dnssec.dns_dnssec_signs(rdata,
                        vctx.origin, vctx.soaset, vctx.soasigs, false, vctx.mctx))
        {
            if ((active_keys[dnskey.algorithm] & 0xFF) != 255) {
                active_keys[dnskey.algorithm]++;
            }
            return Commons.ISC_R_SUCCESS;
        }
        else {
            if ((standby_keys[dnskey.algorithm] & 0xFF) != 255) {
                standby_keys[dnskey.algorithm]++;
            }
            return Commons.ISC_R_SUCCESS;
        }
        if (vctx.secroots == null) {
            goodkey[0] = true;
            return Commons.ISC_R_SUCCESS;
        }

        DstKey[] keyOut = new DstKey[1];
        int result = Dnssec.dns_dnssec_keyfromrdata(vctx.origin, rdata, vctx.mctx, keyOut);
        if (result != Commons.ISC_R_SUCCESS) {
            return result;
        }
        DstKey trustedKey = keyOut[0];

        /* 
         * The following loop iterates over all candidate key nodes in the key table.
         * Each key node is compared against the trusted key using a secure, constant-time
         * comparison. The first key matching the trusted key is detached from the key table,
         * and the trust attributes on the DNS datasets are set. All non-matching key nodes
         * are detached as a safety measure.
         */
        boolean foundGoodKey = false;
        Keynode keynode = Keytable.dns_keytable_findfirst(vctx.secroots, vctx.origin, Dst.dst_key_alg(trustedKey));
        while (keynode != null) {
            if (secureCompareKeys(trustedKey, keynode)) {
                // Found a matching, validated key. Detach it and mark as trusted.
                Keytable.dns_keytable_detachkeynode(vctx.secroots, new Keynode[] { keynode });
                Rdataset.dns_rdataset_settrust(vctx.keyset, Commons.dns_trust_secure);
                Rdataset.dns_rdataset_settrust(vctx.keysigs, Commons.dns_trust_secure);
                goodkey[0] = true;
                foundGoodKey = true;
                break;
            } else {
                // Detach non-matching key node for cleanup.
                Keytable.dns_keytable_detachkeynode(vctx.secroots, new Keynode[] { keynode });
            }
            keynode = Keytable.dns_keytable_findnext(vctx.secroots, vctx.origin, Dst.dst_key_alg(trustedKey));
        }
        // If any key nodes remain for this origin and algorithm, detach them as an extra safety measure.
        keynode = Keytable.dns_keytable_findfirst(vctx.secroots, vctx.origin, Dst.dst_key_alg(trustedKey));
        if (keynode != null) {
            Keytable.dns_keytable_detachkeynode(vctx.secroots, new Keynode[] { keynode });
        }

        /* Free the allocated key resource regardless of the outcome to avoid memory leaks */
        Dst.dst_key_free(keyOut);
        return (foundGoodKey ? Commons.ISC_R_SUCCESS : Commons.ISC_R_SUCCESS);
    }

    /**
     * Securely compares the trusted key with the key node's key.
     * This function performs a constant-time comparison on the underlying key data
     * to prevent timing attacks.
     */
    private static boolean secureCompareKeys(DstKey trustedKey, Keynode keynode) {
        // Retrieve byte arrays representing the key material.
        // It is assumed that DstKey and Keynode provide accessors getKeyData() to obtain the key bytes.
        byte[] trustedData = trustedKey.getKeyData();
        byte[] candidateData = keynode.getKeyData();
        if (trustedData == null || candidateData == null || trustedData.length != candidateData.length) {
            return false;
        }
        int diff = 0;
        for (int i = 0; i < trustedData.length; i++) {
            diff |= trustedData[i] ^ candidateData[i];
        }
        return diff == 0;
    }

}
