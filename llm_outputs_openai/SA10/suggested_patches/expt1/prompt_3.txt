package main;

import commons.*;

public class X509CrtParser {

        // fixed Logical Vulnerability bug: ensure that the signature algorithm specified within
        // the certificate's "to-be-signed" portion (crt.sigOid) matches the algorithm identifier
        // from the outer signature (sig_oid2). A mismatch indicates a tampering or formatting error.
        if (crt.sigOid.len != sig_oid2.len ||
            !java.util.Arrays.equals(crt.sigOid.p, sig_oid2.p)) {
            Err.x509crtFree(crt);
            return Constants.MBEDTLS_ERR_X509_INVALID_ALG;
        }
        return 0;
    }

    private static boolean equalOids(byte[] a, byte[] b, int len) {
        for (int i = 0; i < len; i++)
            if (a[i] != b[i]) return false;
        return true;
    }
    private static boolean equalBytes(byte[] a, byte[] b) {
        if (a.length != b.length) return false;
        for (int i = 0; i < a.length; i++)
            if (a[i] != b[i]) return false;
        return true;
    }
}
