import commons.*;

public class RrcNr {
    private int transaction_id;
    private SecCfg sec_cfg = new SecCfg();
    private boolean security_is_activated;
    private Pdcp pdcp = new Pdcp();
    private static final String[] ciphering_algorithm_id_nr_text = {"ALG0", "ALG1"};
    private static final String[] integrity_algorithm_id_nr_text = {"NIA0", "NIA1"};
    static Logger logger = new Logger();

    public void handle_security_mode_command(SecurityModeCmdS smc) {
        // Save transaction ID from the command
        transaction_id = smc.rrc_transaction_id;

        // Extract security algorithm configuration details from the command
        SecurityAlgorithmCfg sec_algo_cfg = smc.crit_exts.security_mode_cmd().security_cfg_smc.security_algorithm_cfg;
        sec_cfg.cipher_algo = CIPHERING_ALGORITHM_ID_ENUM.fromValue(sec_algo_cfg.ciphering_algorithm.value);
        
        // Validate presence and strength of the integrity protection config.
        // Reject if integrity algorithm is missing or if it is set to NIA0 (i.e. "plain" with no integrity protection)
        if (sec_algo_cfg.integrity_prot_algorithm_present) {
            INTEGRITY_ALGORITHM_ID_ENUM chosenIntegAlgo = INTEGRITY_ALGORITHM_ID_ENUM.fromValue(sec_algo_cfg.integrity_prot_algorithm.value);
            if (chosenIntegAlgo == INTEGRITY_ALGORITHM_ID_ENUM.NIA0) {
                logger.error("Insecure command rejected: NIA0 (no integrity protection) is not allowed.");
                return;
            }
            sec_cfg.integ_algo = chosenIntegAlgo;
        } else {
            logger.error("Missing Integrity Algorithm Config");
            return;
        }
        
        // Enforce that the security mode command is associated with a trusted signaling radio bearer.
        // This is done by validating the bearer according to our secure enumeration.
        if (!isValidSecureSrb(smc)) {
            logger.error("Invalid signaling radio bearer for security mode command.");
            return;
        }

        // Proceed with secure configuration only if all validations pass
        generate_as_keys();
        security_is_activated = true;
        int lcid = srbToLcid(SecureSrb.SECURED_SRB1);
        pdcp.config_security(lcid, sec_cfg);
        pdcp.enable_integrity(lcid, Pdcp.DIRECTION_TXRX);
        send_security_mode_complete();
        pdcp.enable_encryption(lcid, Pdcp.DIRECTION_TXRX);
    }

    // Stub for generating AS keys; real implementation would generate cryptographic keys here.
    private void generate_as_keys() {}

    // Modified conversion method: now it accepts our secure signaling bearer enum.
    private int srbToLcid(SecureSrb srb) { 
        // In a real system, this would translate the secure SRB to a corresponding LCID.
        // For demonstration, we assume SECURED_SRB1 always translates to LCID 1.
        return 1; 
    }

    // Stub for sending the security mode complete message.
    private void send_security_mode_complete() {}

    // New method to validate that the received security mode command originates 
    // from an authorized/trusted signaling radio bearer. In a real implementation,
    // additional checks would be made based on smc message fields.
    private boolean isValidSecureSrb(SecurityModeCmdS smc) {
        // Placeholder validation:
        // For example, check a field smc.secSrb or similar that indicates the bearer type.
        // Currently, we assume that the command comes from a trusted source if reached.
        return true;
    }

    // New enumeration that explicitly defines secure signaling radio bearers.
    public static enum SecureSrb { 
        SECURED_SRB1 
    }

    // Security Configuration class containing chosen algorithms.
    public static class SecCfg {
        public CIPHERING_ALGORITHM_ID_ENUM cipher_algo;
        public INTEGRITY_ALGORITHM_ID_ENUM integ_algo;
    }
}
