public class SecCfg {
    public CIPHERING_ALGORITHM_ID_ENUM cipher_algo;
    public INTEGRITY_ALGORITHM_ID_ENUM integ_algo;
    // New field to associate a validated, secure signaling radio bearer.
    public SecureNrSrb secureBearer;

    /**
     * Configures the security algorithms along with a trusted signaling radio bearer.
     * This method enforces that insecure or “plain” configurations (such as using NIA0 for integrity)
     * are rejected.
     *
     * @param cipher the ciphering algorithm
     * @param integ the integrity algorithm
     * @param bearer the validated secure signaling radio bearer
     * @throws IllegalArgumentException if an insecure integrity algorithm or invalid bearer is used
     */
    public void configure(CIPHERING_ALGORITHM_ID_ENUM cipher,
                          INTEGRITY_ALGORITHM_ID_ENUM integ,
                          SecureNrSrb bearer) {
        // Prevent insecure integrity configuration (NIA0 means no integrity protection)
        if (integ == INTEGRITY_ALGORITHM_ID_ENUM.NIA0) {
            throw new IllegalArgumentException("Insecure integrity algorithm (NIA0) is not allowed");
        }
        // Enforce that only commands tied to a trusted, validated bearer are accepted.
        if (bearer == null) {
            throw new IllegalArgumentException("A valid secureBearer must be provided");
        }
        this.cipher_algo = cipher;
        this.integ_algo = integ;
        this.secureBearer = bearer;
    }

    /**
     * Enumeration defining the allowed secure signaling radio bearers.
     * Commands provided outside of these secure bearers (e.g., plain/bypassed modes)
     * are rejected to enforce proper RRC layer security activation.
     */
    public static enum SecureNrSrb {
        // Only commands associated with SRB1_SECURE are allowed.
        SRB1_SECURE
    }
}
