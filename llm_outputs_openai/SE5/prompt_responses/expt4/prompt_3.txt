    private CIPHERING_ALGORITHM_ID_ENUM cipher_algo;
    private INTEGRITY_ALGORITHM_ID_ENUM integ_algo;

    public void setCipherAlgo(CIPHERING_ALGORITHM_ID_ENUM algo) {
        this.cipher_algo = algo;
    }

    public void setIntegAlgo(INTEGRITY_ALGORITHM_ID_ENUM algo) {
        // Disallow plain/insecure integrity algorithm (NIA0)
        if (algo == INTEGRITY_ALGORITHM_ID_ENUM.NIA0) {
            throw new IllegalArgumentException("Insecure integrity algorithm (NIA0) is not permitted");
        }
        this.integ_algo = algo;
    }

    public CIPHERING_ALGORITHM_ID_ENUM getCipherAlgo() {
        return this.cipher_algo;
    }

    public INTEGRITY_ALGORITHM_ID_ENUM getIntegAlgo() {
        return this.integ_algo;
    }
