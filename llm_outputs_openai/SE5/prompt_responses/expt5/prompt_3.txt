public enum SECURE_SIGNALLING_BEARER_ENUM
{
    // This is the only trusted signaling radio bearer allowed.
    VALID_NRSRB
}

public class RRC_SecurityConfig
{
    // Coupled secure bearer with ciphering and integrity algorithms.
    public SECURE_SIGNALLING_BEARER_ENUM secureBearer;
    public CIPHERING_ALGORITHM_ID_ENUM cipher_algo;
    public INTEGRITY_ALGORITHM_ID_ENUM integ_algo;

    // Constructor enforcing that only the validated signaling bearer is accepted.
    public RRC_SecurityConfig(SECURE_SIGNALLING_BEARER_ENUM bearer, 
                              CIPHERING_ALGORITHM_ID_ENUM cipher, 
                              INTEGRITY_ALGORITHM_ID_ENUM integ)
    {
        if (bearer != SECURE_SIGNALLING_BEARER_ENUM.VALID_NRSRB)
        {
            throw new ArgumentException("Command rejected: Unsupported signaling bearer for secure configuration.");
        }
        
        secureBearer = bearer;
        cipher_algo = cipher;
        integ_algo = integ;
    }
}
