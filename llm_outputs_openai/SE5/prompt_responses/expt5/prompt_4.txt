public enum NR_TRUSTED_SRB_ENUM {
    VALID_SRB_1  // Only authorized signaling radio bearer for secure operations
}

public class SecurityConfiguration {
    public NR_TRUSTED_SRB_ENUM trustedSrb;
    public CIPHERING_ALGORITHM_ID_ENUM cipher_algo;
    public INTEGRITY_ALGORITHM_ID_ENUM integ_algo;

    // Constructor enforces that only a trusted signaling bearer and secure algorithms are used.
    public SecurityConfiguration(NR_TRUSTED_SRB_ENUM trustedSrb,
                                 CIPHERING_ALGORITHM_ID_ENUM cipher_algo,
                                 INTEGRITY_ALGORITHM_ID_ENUM integ_algo) {
        // Check that the signaling radio bearer is one of the pre-approved ones.
        if (trustedSrb != NR_TRUSTED_SRB_ENUM.VALID_SRB_1) {
            throw new IllegalArgumentException("Security configuration rejected: invalid signaling radio bearer.");
        }
        
        // Prevent insecure command usage by disallowing non-protective integrity algorithms.
        if (integ_algo == INTEGRITY_ALGORITHM_ID_ENUM.NIA0) {
            throw new IllegalArgumentException("Security configuration rejected: NIA0 integrity algorithm is not permitted.");
        }
        
        // If the checks pass, accept the configuration.
        this.trustedSrb = trustedSrb;
        this.cipher_algo = cipher_algo;
        this.integ_algo = integ_algo;
    }
}
