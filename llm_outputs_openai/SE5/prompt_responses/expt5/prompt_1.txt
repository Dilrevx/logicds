public class CryptoParameters {
    // Make fields private and final to prevent unauthorized modification.
    private final CIPHERING_ALGORITHM_ID_ENUM cipher_algo;
    private final INTEGRITY_ALGORITHM_ID_ENUM integ_algo;

    // Constructor with validation to ensure non-null values.
    public CryptoParameters(CIPHERING_ALGORITHM_ID_ENUM cipher_algo, INTEGRITY_ALGORITHM_ID_ENUM integ_algo) {
        if (cipher_algo == null || integ_algo == null) {
            throw new IllegalArgumentException("Algorithm identifiers cannot be null.");
        }
        this.cipher_algo = cipher_algo;
        this.integ_algo = integ_algo;
    }

    // Read-only accessors.
    public CIPHERING_ALGORITHM_ID_ENUM getCipherAlgo() {
        return cipher_algo;
    }

    public INTEGRITY_ALGORITHM_ID_ENUM getIntegAlgo() {
        return integ_algo;
    }
}
