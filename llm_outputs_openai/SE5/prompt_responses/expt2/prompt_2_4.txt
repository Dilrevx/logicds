public enum AllowedSRB {
    SRB1; // Only permitted SRB value for security configuration.
}

public class SecCfg {
    private AllowedSRB srb;
    private CIPHERING_ALGORITHM_ID_ENUM cipher_algo;
    private INTEGRITY_ALGORITHM_ID_ENUM integ_algo;

    // Constructor enforces that security configuration is only applied for the allowed SRB.
    public SecCfg(AllowedSRB srb, CIPHERING_ALGORITHM_ID_ENUM cipher_algo, INTEGRITY_ALGORITHM_ID_ENUM integ_algo) {
        if (srb != AllowedSRB.SRB1) {
            throw new IllegalArgumentException("Invalid SRB. Only SRB1 is allowed for security activation.");
        }
        this.srb = srb;
        this.cipher_algo = cipher_algo;
        this.integ_algo = integ_algo;
    }

    public AllowedSRB getSrb() {
        return srb;
    }

    public CIPHERING_ALGORITHM_ID_ENUM getCipherAlgo() {
        return cipher_algo;
    }

    public INTEGRITY_ALGORITHM_ID_ENUM getIntegAlgo() {
        return integ_algo;
    }

    public void setCipherAlgo(CIPHERING_ALGORITHM_ID_ENUM cipher_algo) {
        this.cipher_algo = cipher_algo;
    }

    public void setIntegAlgo(INTEGRITY_ALGORITHM_ID_ENUM integ_algo) {
        this.integ_algo = integ_algo;
    }
}
