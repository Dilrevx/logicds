public class SecurityParameters {
    private CIPHERING_ALGORITHM_ID_ENUM cipher_algo;
    private INTEGRITY_ALGORITHM_ID_ENUM integ_algo;

    public SecurityParameters(CIPHERING_ALGORITHM_ID_ENUM cipher_algo, INTEGRITY_ALGORITHM_ID_ENUM integ_algo) {
        // Disallow the insecure "NIA0" algorithm values during instantiation
        if (cipher_algo == CIPHERING_ALGORITHM_ID_ENUM.NIA0) {
            throw new IllegalArgumentException("Insecure ciphering algorithm NIA0 is not permitted.");
        }
        if (integ_algo == INTEGRITY_ALGORITHM_ID_ENUM.NIA0) {
            throw new IllegalArgumentException("Insecure integrity algorithm NIA0 is not permitted.");
        }
        this.cipher_algo = cipher_algo;
        this.integ_algo = integ_algo;
    }

    public CIPHERING_ALGORITHM_ID_ENUM getCipherAlgo() {
        return cipher_algo;
    }

    public void setCipherAlgo(CIPHERING_ALGORITHM_ID_ENUM cipher_algo) {
        if (cipher_algo == CIPHERING_ALGORITHM_ID_ENUM.NIA0) {
            throw new IllegalArgumentException("Insecure ciphering algorithm NIA0 is not permitted.");
        }
        this.cipher_algo = cipher_algo;
    }

    public INTEGRITY_ALGORITHM_ID_ENUM getIntegAlgo() {
        return integ_algo;
    }

    public void setIntegAlgo(INTEGRITY_ALGORITHM_ID_ENUM integ_algo) {
        if (integ_algo == INTEGRITY_ALGORITHM_ID_ENUM.NIA0) {
            throw new IllegalArgumentException("Insecure integrity algorithm NIA0 is not permitted.");
        }
        this.integ_algo = integ_algo;
    }
}
