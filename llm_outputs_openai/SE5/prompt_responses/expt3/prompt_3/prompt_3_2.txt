public enum CIPHERING_ALGORITHM_ID_ENUM {
    CIPHERING_ALGORITHM_NIA0_PLAIN,
    CIPHERING_ALGORITHM_SECURE
    // ... add additional secure variants here
}

public enum INTEGRITY_ALGORITHM_ID_ENUM {
    INTEGRITY_ALGORITHM_NIA0_PLAIN,
    INTEGRITY_ALGORITHM_SECURE
    // ... add additional secure variants here
}

public class RrcSecurityConfig {
    public CIPHERING_ALGORITHM_ID_ENUM cipher_algo;
    public INTEGRITY_ALGORITHM_ID_ENUM integ_algo;

    /**
     * Sets the security algorithms after ensuring that insecure
     * plain variants (which bypass RRC security activation) are not allowed.
     *
     * @param cipher The ciphering algorithm.
     * @param integ The integrity algorithm.
     * @throws IllegalArgumentException if an insecure algorithm is provided.
     */
    public void setSecurityAlgorithms(CIPHERING_ALGORITHM_ID_ENUM cipher,
                                      INTEGRITY_ALGORITHM_ID_ENUM integ) {
        // Validate that insecure plain variants are not used.
        if (cipher == CIPHERING_ALGORITHM_ID_ENUM.CIPHERING_ALGORITHM_NIA0_PLAIN ||
            integ == INTEGRITY_ALGORITHM_ID_ENUM.INTEGRITY_ALGORITHM_NIA0_PLAIN) {
            throw new IllegalArgumentException("Insecure algorithm configuration is not allowed: " +
                                               "RRC security activation must not be bypassed.");
        }
        
        // Assign the secure algorithms.
        this.cipher_algo = cipher;
        this.integ_algo = integ;
    }
}
