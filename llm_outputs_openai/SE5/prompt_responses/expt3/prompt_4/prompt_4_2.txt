public class CipherSettings
{
    private CIPHERING_ALGORITHM_ID_ENUM cipher_algo;
    private INTEGRITY_ALGORITHM_ID_ENUM integ_algo;

    public CIPHERING_ALGORITHM_ID_ENUM CipherAlgo
    {
        get { return cipher_algo; }
        set
        {
            // Validate that the assigned enum value is defined.
            if (!Enum.IsDefined(typeof(CIPHERING_ALGORITHM_ID_ENUM), value))
            {
                throw new ArgumentOutOfRangeException(nameof(value), "Invalid ciphering algorithm provided.");
            }
            cipher_algo = value;
        }
    }

    public INTEGRITY_ALGORITHM_ID_ENUM IntegAlgo
    {
        get { return integ_algo; }
        set
        {
            // Validate that the assigned enum value is defined.
            if (!Enum.IsDefined(typeof(INTEGRITY_ALGORITHM_ID_ENUM), value))
            {
                throw new ArgumentOutOfRangeException(nameof(value), "Invalid integrity algorithm provided.");
            }
            integ_algo = value;
        }
    }

    // Constructor with validation applied through the property setters.
    public CipherSettings(CIPHERING_ALGORITHM_ID_ENUM cipherAlgo, INTEGRITY_ALGORITHM_ID_ENUM integAlgo)
    {
        CipherAlgo = cipherAlgo;
        IntegAlgo = integAlgo;
    }
}
