public class RrcSecurityParameters {
    // These enum values should never be set to the “plain” (i.e. NIA0) variants during security activation.
    public CIPHERING_ALGORITHM_ID_ENUM cipher_algo;
    public INTEGRITY_ALGORITHM_ID_ENUM integ_algo;

    /**
     * Validates that security activation parameters do not request a plain (non-secured) configuration.
     * The "plain" (e.g. NIA0) selection is not allowed since it bypasses RRC layer security activation.
     *
     * @throws SecurityException if a plain security configuration is attempted.
     */
    public void validateParameters() {
        if (cipher_algo == CIPHERING_ALGORITHM_ID_ENUM.NIA0_PLAIN ||
            integ_algo == INTEGRITY_ALGORITHM_ID_ENUM.NIA0_PLAIN) {
            throw new SecurityException("RRC layer security activation rejected: plain configuration (NIA0) is not permitted.");
        }
    }
    
    // In the security activation flow, call validateParameters() before proceeding.
}
