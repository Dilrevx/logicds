public enum NR_Trusted_Signaling_RB {
    // This enumeration explicitly limits the trusted signaling radio bearers.
    VALID_SRB,      // Commands coming from a validated, secure signaling bearer.
    INVALID_SRB     // Any non-trusted bearer should be rejected.
}

public class SecurityConfiguration {
    // The signalingRB field forces any security configuration to declare its source.
    public NR_Trusted_Signaling_RB signalingRB;
    public CIPHERING_ALGORITHM_ID_ENUM cipher_algo;
    public INTEGRITY_ALGORITHM_ID_ENUM integ_algo;

    /**
     * Checks whether the security configuration is valid.
     * In particular, this check enforces that:
     *   - Only commands with a VALID_SRB are accepted.
     *   - The integrity algorithm is not set to NIA0 (which implies no integrity protection).
     *
     * @return true if the configuration is secure; false otherwise.
     */
    public boolean isSecurityConfigurationValid() {
        // Reject any command not coming from a trusted signaling radio bearer,
        // or if the integrity algorithm is insecure (NIA0).
        if (signalingRB != NR_Trusted_Signaling_RB.VALID_SRB || integ_algo == INTEGRITY_ALGORITHM_ID_ENUM.NIA0) {
            return false;
        }
        return true;
    }
}
