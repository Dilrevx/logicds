public enum SRB {
    SRB1
}

public class SecCfg {
    private final SRB srb;
    private CIPHERING_ALGORITHM_ID_ENUM cipherAlgo;
    private INTEGRITY_ALGORITHM_ID_ENUM integAlgo;

    public SecCfg(SRB srb, CIPHERING_ALGORITHM_ID_ENUM cipherAlgo, INTEGRITY_ALGORITHM_ID_ENUM integAlgo) {
        if (srb != SRB.SRB1) {
            throw new IllegalArgumentException("Invalid SRB. Only SRB1 is allowed.");
        }
        this.srb = srb;
        this.cipherAlgo = cipherAlgo;
        this.integAlgo = integAlgo;
    }

    public SRB getSrb() {
        return srb;
    }

    public CIPHERING_ALGORITHM_ID_ENUM getCipherAlgo() {
        return cipherAlgo;
    }

    public void setCipherAlgo(CIPHERING_ALGORITHM_ID_ENUM cipherAlgo) {
        this.cipherAlgo = cipherAlgo;
    }

    public INTEGRITY_ALGORITHM_ID_ENUM getIntegAlgo() {
        return integAlgo;
    }

    public void setIntegAlgo(INTEGRITY_ALGORITHM_ID_ENUM integAlgo) {
        this.integAlgo = integAlgo;
    }
}
