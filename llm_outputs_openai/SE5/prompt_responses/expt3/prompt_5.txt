/* Define an enumeration listing only approved secure signaling radio bearers */
public enum SECURE_SIGNALLING_RB_ENUM
{
    VALID_NR_SIGNALING_BEARER  // Only this bearer is allowed for security configuration
    // Additional secure bearer types can be added here if needed.
}

/* Enumeration for ciphering algorithms */
public enum CIPHERING_ALGORITHM_ID_ENUM
{
    // Define your ciphering algorithms here.
}

/* Enumeration for integrity algorithms */
public enum INTEGRITY_ALGORITHM_ID_ENUM
{
    NIA0,      // No integrity protection (insecure)
    // Define additional integrity algorithms here.
}

/* A secure configuration class that couples the algorithms with an allowed signaling bearer */
public class SecurityConfiguration
{
    public SECURE_SIGNALLING_RB_ENUM secureSrb;
    public CIPHERING_ALGORITHM_ID_ENUM cipher_algo;
    public INTEGRITY_ALGORITHM_ID_ENUM integ_algo;

    // Constructor performs the validation of the signaling radio bearer and the integrity algorithm.
    public SecurityConfiguration(
        SECURE_SIGNALLING_RB_ENUM secureSrb,
        CIPHERING_ALGORITHM_ID_ENUM cipherAlgo,
        INTEGRITY_ALGORITHM_ID_ENUM integAlgo)
    {
        // Check if the command comes from an allowed radio bearer.
        if (secureSrb != SECURE_SIGNALLING_RB_ENUM.VALID_NR_SIGNALING_BEARER)
        {
            throw new InvalidOperationException("Command rejected: signaling radio bearer is not authorized.");
        }

        // Reject insecure commands that use the NIA0 (plain/insecure) integrity algorithm.
        if (integAlgo == INTEGRITY_ALGORITHM_ID_ENUM.NIA0)
        {
            throw new InvalidOperationException("Command rejected: insecure integrity algorithm NIA0 is not allowed.");
        }

        this.secureSrb = secureSrb;
        this.cipher_algo = cipherAlgo;
        this.integ_algo = integAlgo;
    }
}
