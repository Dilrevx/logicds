package main;


import commons.*;

public class RrcNr {
    private int transaction_id;
    private SecCfg sec_cfg = new SecCfg();
    private boolean security_is_activated;
    private Pdcp pdcp = new Pdcp();
    private static final String[] ciphering_algorithm_id_nr_text = {"ALG0", "ALG1"};
    private static final String[] integrity_algorithm_id_nr_text = {"NIA0", "NIA1"};
    static Logger logger = new Logger();

    public void handle_security_mode_command(SecurityModeCmdS smc) {
        transaction_id = smc.rrc_transaction_id;
        SecurityAlgorithmCfg sec_algo_cfg = smc.crit_exts.security_mode_cmd().security_cfg_smc.security_algorithm_cfg;
        sec_cfg.cipher_algo = CIPHERING_ALGORITHM_ID_ENUM.fromValue(sec_algo_cfg.ciphering_algorithm.value);
        
        // Only process security mode command if an integrity algorithm is present
        if (sec_algo_cfg.integrity_prot_algorithm_present) {
            sec_cfg.integ_algo = INTEGRITY_ALGORITHM_ID_ENUM.fromValue(sec_algo_cfg.integrity_prot_algorithm.value);
            // Reject commands that use the insecure "plain" integrity algorithm (NIA0)
            if (sec_cfg.integ_algo == INTEGRITY_ALGORITHM_ID_ENUM.NIA0) {
                logger.error("Insecure integrity algorithm NIA0 is not permitted. Security mode command rejected.");
                return;
            }
        } else {
            logger.error("Missing Integrity Algorithm Config. Security mode command rejected.");
            return;
        }
        
        // Validate that the command is associated with a secure signaling radio bearer.
        if (!isValidSecureBearer(smc)) {
            logger.error("Security mode command is not associated with a valid secure signaling radio bearer. Command rejected.");
            return;
        }
        
        generate_as_keys();
        security_is_activated = true;
        // Map the secure signaling radio bearer to its logical channel ID.
        int lcid = secureSrbToLcid(SecureNrSrb.srb1_secure);
        pdcp.config_security(lcid, sec_cfg);
        pdcp.enable_integrity(lcid, Pdcp.DIRECTION_TXRX);
        send_security_mode_complete();
        pdcp.enable_encryption(lcid, Pdcp.DIRECTION_TXRX);
    }

    // New method to validate that the command comes from a trusted signaling radio bearer.
    // In a real-world scenario, this method would inspect properties of smc to confirm bearer identity.
    private boolean isValidSecureBearer(SecurityModeCmdS smc) {
        // For illustration, assume a valid command must have a non-negative transaction ID.
        return smc.rrc_transaction_id >= 0;
    }

    private void generate_as_keys() {}

    // Maps our secure signaling radio bearer to an LCID.
    private int secureSrbToLcid(SecureNrSrb srb) {
        // In our simplified mapping, we convert our validated secure bearer into the internal SRB representation.
        // This example simply uses srb1 from the existing enumeration.
        return srb_to_lcid(NrSrb.srb1);
    }

    private int srb_to_lcid(NrSrb srb) {
        // Mapping logic: returns the logical channel id corresponding to the signaling radio bearer.
        return 0; 
    }

    private void send_security_mode_complete() {}

    public static enum NrSrb { srb1 }

    // New enumeration that explicitly defines secure signaling radio bearers.
    public static enum SecureNrSrb { srb1_secure }

    public static class SecCfg {
        public CIPHERING_ALGORITHM_ID_ENUM cipher_algo;
        public INTEGRITY_ALGORITHM_ID_ENUM integ_algo;
    }
}
