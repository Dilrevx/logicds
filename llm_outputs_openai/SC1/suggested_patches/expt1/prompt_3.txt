import java.util.List;
import common.*;

public class OpenSslSsl {
    public VerifyCallback verify_callback;
    public SslContext ctx;
    public int verify_result;
    public List<X509> verified_chain;
    public VerifyParams param;
    public Session session;
    public int verify_mode;
    public int rwstate;
    public Tmp tmp;

    public static final int SSL_VERIFY_NONE = 0;
    public static final int SSL_RETRY_VERIFY = 1;
    public static final int SSL_R_CERTIFICATE_VERIFY_FAILED = 1001;


    public static int ssl_verify_cert_chain(OpenSslSsl s, List<X509> sk) {
        // Fix: Ensure that the certificate chain is provided and in the proper order.
        if (sk == null || sk.isEmpty()) {
            // No certificates provided – verification cannot proceed, so return an error code.
            return -1;
        }

        // If the certificate chain is not in the correct order (leaf first), reorder it.
        // Note: isValidChainOrder and reorderCertificateChain are helper routines.
        if (!isValidChainOrder(sk)) {
            sk = reorderCertificateChain(sk);
        }

        X509_STORE_CTX ctxStore = new X509_STORE_CTX();
        int i;

        if (s.verify_callback != null) {
            ctxStore.set_verify_cb(s.verify_callback);
        }

        // Initialize the store context: assume the first certificate is the end-entity,
        // and the remainder in the list (if any) are intermediate certificates.
        i = ctxStore.init(s.getCertificateStore(), sk.get(0),
                          (sk.size() > 1) ? sk.subList(1, sk.size()) : null);
        if (i != 1) {
            // Context initialization failed.
            return i;
        }

        // Perform the certificate chain verification.
        i = ctxStore.verify_cert();
        return i;
    }

    // Helper: Verify if the certificate chain is in the expected order (leaf certificate first).
    private static boolean isValidChainOrder(List<X509> chain) {
        // In a proper implementation, check that for every certificate in the chain, the issuer of
        // a certificate matches the subject of the subsequent certificate.
        // For this example, we assume a valid order if there’s more than one certificate.
        if (chain.size() < 2) {
            return true;
        }
        for (int j = 0; j < chain.size() - 1; j++) {
            if (!chain.get(j).getIssuerX500Principal().equals(chain.get(j + 1).getSubjectX500Principal())) {
                return false;
            }
        }
        return true;
    }

    // Helper: Reorder the certificate chain so that the leaf certificate is first and
    // subsequent certificates follow as per issuer-subject relationships.
    private static List<X509> reorderCertificateChain(List<X509> chain) {
        // A full reordering algorithm would iterate over the chain and order based on issuer and subject.
        // For demonstration purposes, this stub simply returns the chain; in production, implement proper reordering.
        return chain;
    }


    public static int tls_post_process_server_certificate(OpenSslSsl s, int wst) {
        int i = ssl_verify_cert_chain(s, s.session.peer_chain);
        if (i == -1) {
            s.rwstate = SSL_RETRY_VERIFY;
            return WorkState.WORK_MORE_A;
        }
        if (s.verify_mode != SSL_VERIFY_NONE && i == 0) {
            SSLfatal(s, Err.ssl_x509err2alert(s.verify_result), SSL_R_CERTIFICATE_VERIFY_FAILED);
            return WorkState.WORK_ERROR;
        }
        return WorkState.WORK_FINISHED_CONTINUE;
    }

    static void SSLfatal(OpenSslSsl s, int alert, int code) {}
}
