void
isc__nmhandle_detach(isc_nmhandle_t **handlep FLARG) {
	isc_nmsocket_t *sock = NULL;
	isc_nmhandle_t *handle = NULL;

	REQUIRE(handlep != NULL);
	REQUIRE(VALID_NMHANDLE(*handlep));

	handle = *handlep;
	*handlep = NULL;

	sock = handle->sock;
	if (sock->tid == isc_nm_tid()) {
		/* Same thread: run the detach callback directly */
		nmhandle_detach_cb(&handle FLARG_PASS);
	} else {
		/*
		 * Different thread: schedule an asynchronous detach.
		 * Note: To avoid situations where a crafted TCP stream might
		 * leave the connection lingering in CLOSE_WAIT indefinitely,
		 * ensure that the network thread is promptly woken-up to process
		 * the detach event.
		 */
		isc__netievent_detach_t *event =
			isc__nm_get_netievent_detach(sock->mgr, sock);
		/*
		 * We are using an implicit "attach" as the last reference;
		 * it will be explicitly destroyed in the asynchronous callback.
		 */
		event->handle = handle;
		FLARG_IEVENT_PASS(event);
		isc__nm_enqueue_ievent(&sock->mgr->workers[sock->tid],
				       (isc__netievent_t *)event);
		/* Force waking the worker thread so the queued detach is processed */
		uv_async_send(&sock->mgr->workers[sock->tid].async);
	}
}
