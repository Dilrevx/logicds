void
isc__nmhandle_detach(isc_nmhandle_t **handlep FLARG) {
    isc_nmsocket_t *sock = NULL;
    isc_nmhandle_t *handle = NULL;

    REQUIRE(handlep != NULL);
    REQUIRE(VALID_NMHANDLE(*handlep));

    handle = *handlep;
    *handlep = NULL;

    /*
     * Bugfix: Previously, the internal socket pointer was not cleared
     * before detaching the socket, leaving a potential race condition
     * or double-use if another thread tried to access it. In this fix,
     * we first grab and clear the socket reference from the handle,
     * then properly detach the socket.
     */
    if (handle->sock != NULL) {
        sock = handle->sock;
        handle->sock = NULL;
        isc_nmsocket_detach(&sock);
    }

    /*
     * If there are any other resources tied to the handle that are
     * only valid while the handle is "attached", release them here.
     * For example, if handle is part of a linked list or has allocated
     * ancillary memory, make sure to remove/free those resources now.
     */
    // Example: 
    // ISC_LIST_UNLINK(somelist, handle, link);

    /*
     * Finally, free the handle if it is no longer needed.
     * (Assuming this is the proper lifecycle operation.)
     */
    isc_mem_free(handle);

    return;
}
