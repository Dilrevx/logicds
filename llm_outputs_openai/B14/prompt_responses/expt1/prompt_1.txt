void
isc__nmhandle_detach(isc_nmhandle_t **handlep FLARG) {
	isc_nmsocket_t *sock = NULL;
	isc_nmhandle_t *handle = NULL;

	/*
	 * Instead of relying only on REQUIRE (which may be compiled out
	 * in production builds) we do an explicit run‐time check.  Then,
	 * to avoid a race condition between validation and use, we use an
	 * atomic pointer exchange to “detach” the handle.
	 */
	if (handlep == NULL) {
		/* Log error if needed */
		return;
	}

	/* Atomically clear the caller’s pointer and get the handle.
	   (Assumes isc_atomic_xchgptr() is available; otherwise,
	   one must properly protect this operation with locking.) */
	handle = isc_atomic_xchgptr((void * volatile *)handlep, NULL);
	if (handle == NULL)
		return;

	/* Validate the handle. Even though VALID_NMHANDLE might be a macro
	   that asserts in debug builds, we check at runtime to avoid
	   processing a corrupted handle in production. */
	if (!VALID_NMHANDLE(handle))
		return;

	/* Proceed to detach the associated nmsocket if any. */
	sock = handle->socket;
	if (sock != NULL)
		isc_nmsocket_detach(&sock);

	/*
	 * Any additional cleanup of 'handle' must be performed here.
	 * For example, releasing locks, freeing dynamic memory, etc.
	 * (Assuming that after detaching, the handle is no longer used.)
	 */

	/* End of function cleanup */
	return;
}
