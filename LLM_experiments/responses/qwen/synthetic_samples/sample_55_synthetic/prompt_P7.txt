    public static String getDnsKey() {
        String key = null;

        // Try to get the DNS key from an environment variable
        key = System.getenv("DNS_KEY");

        // If the environment variable is not set, try to get it from the configuration
        if (key == null || key.isEmpty()) {
            key = Config.Server.getDnsKey();
        }

        // Validate the key (this is a simple example; you might want more robust validation)
        if (key == null || key.isEmpty()) {
            throw new IllegalStateException("DNS key is not set or is invalid.");
        }

        return key;
    }
