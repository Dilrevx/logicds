        import java.util.regex.Pattern;
        import java.util.regex.Matcher;

        // Utility method to escape regex metacharacters
        public static String escapeRegex(String input) {
            if (input == null) return null;
            StringBuilder escaped = new StringBuilder();
            for (char c : input.toCharArray()) {
                if (Pattern.compile("[\\^$.|?*+()\\[\\]{}]").matcher(Character.toString(c)).find()) {
                    escaped.append("\\").append(c);
                } else {
                    escaped.append(c);
                }
            }
            return escaped.toString();
        }

        String sanitizedUser = escapeRegex(user);
        String sanitizedCollectionUrl = escapeRegex(collectionUrl);

        defaults.put("login", sanitizedUser);
        defaults.put("path", sanitizedCollectionUrl);
