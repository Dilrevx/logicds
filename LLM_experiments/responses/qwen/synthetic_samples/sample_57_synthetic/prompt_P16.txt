        import java.util.regex.Pattern;
        
        // Utility method to escape regex metacharacters
        String escapeRegex(String input) {
            return Pattern.compile("[\\W]").matcher(input).replaceAll("\\\\$0");
        }
        
        Map<String, String> defaults = new HashMap<String, String>();
        String sanitizedUser = escapeRegex(user);
        String sanitizedCollectionUrl = escapeRegex(collectionUrl);
        defaults.put("login", sanitizedUser);
        defaults.put("path", sanitizedCollectionUrl);
