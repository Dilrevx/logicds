#include <time.h>

#define TIME_WINDOW (24 * 60 * 60) // 24 hours in seconds

time_t currentTime = time(NULL);
time_t lowerBound = currentTime - TIME_WINDOW;
time_t upperBound = currentTime + TIME_WINDOW;

if (crt.serial.len == cur.serial.len
        && MemUtil.memcmp(crt.serial.p, cur.serial.p, crt.serial.len) == 0) {
    // Check if the revocation date is within the trusted time window
    if (cur.revocationDate >= lowerBound && cur.revocationDate <= upperBound) {
        // If the revocation date is within the window, use the system time to check
        if (Helper.mbedtls_x509_time_is_past(cur.revocationDate)) {
            return 1; // Certificate is revoked
        }
    } else {
        // If the revocation date is outside the window, consider it an error
        // or handle it according to your application's security policy
        // For now, we'll just log a warning and proceed
        printf("Warning: Revocation date is outside the trusted time window.\n");
    }
}
