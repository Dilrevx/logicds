    public VerifyResult verifyCaptcha(Context ctx, String key, String captcha) {
        CaptchaRepo.CaptchaResult fetched = captchaRepo.getCaptcha(ctx, key);
        String realCaptcha = fetched.value;
        Exception err = fetched.err;
        if (err != null) {
            return new VerifyResult(false, null);
        }
        // Add a check to ensure the captcha has not expired or been used before
        if (fetched.isExpired() || fetched.isUsed()) {
            return new VerifyResult(false, "Captcha expired or already used");
        }
        boolean isMatch = trimSpace(captcha).equals(realCaptcha);
        if (isMatch) {
            // Mark the captcha as used to prevent replay attacks
            captchaRepo.markCaptchaAsUsed(ctx, key);
        }
        return new VerifyResult(isMatch, null);
    }
