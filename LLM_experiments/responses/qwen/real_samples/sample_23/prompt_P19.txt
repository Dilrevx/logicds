            /* Encrypt message */
            ret = EncryptMessage(ssl, output, args->sz);

            ssl->options.buildMsgState = BUILD_MSG_DONE;

        FALL_THROUGH;

        {
            /* Message is fully built */
            ret = args->sz;
            goto exit_buildmsg;

            WOLFSSL_MSG("Invalid build message state");
            ERROR_OUT(BAD_STATE_E, exit_buildmsg);
        }
