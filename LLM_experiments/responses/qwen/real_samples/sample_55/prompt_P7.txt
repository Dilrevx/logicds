import (
    "errors"
    "os"
    "config" // Assuming this is your configuration package
)

// ErrDNSKeyNotSet is returned when the DNS key is not set via environment variable or configuration.
var ErrDNSKeyNotSet = errors.New("DNS key is not set")

func GetDNSKey() (string, error) {
    key := os.Getenv("DNS_KEY")
    if key == "" {
        key = config.Config.Server.DNSKey
        if key == "" {
            return "", ErrDNSKeyNotSet
        }
    }
    return key, nil
}
