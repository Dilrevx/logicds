import "os"

func GetDNSKey() string {
	var key string
	if envKey := os.Getenv("DNS_KEY"); envKey != "" {
		key = envKey
	} else if config.Config.Server.DNSKey != "" {
		key = config.Config.Server.DNSKey
	} else {
		key = "defaultkey" // Ensure there's always a default or handle the absence of a key appropriately
	}
	return key
}
