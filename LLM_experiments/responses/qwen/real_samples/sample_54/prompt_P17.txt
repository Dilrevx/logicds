    /*
     * Create a temporary directory for mounting proc
     */
    tmpdir = mkdtemp(template);
    if (!tmpdir) {
        fuse_log(FUSE_LOG_ERR, "mkdtemp(): %m\n");
        exit(1);
    }

    /*
     * Mount proc to the temporary directory
     */
    if (mount("proc", tmpdir, "proc",
              MS_NODEV | MS_NOEXEC | MS_NOSUID | MS_RELATIME, NULL) < 0) {
        fuse_log(FUSE_LOG_ERR, "mount(%s): %m\n", tmpdir);
        exit(1);
    }

    /*
     * Bind mount the proc from the temporary directory to /proc
     */
    if (mount(tmpdir, "/proc", NULL, MS_BIND, NULL) < 0) {
        fuse_log(FUSE_LOG_ERR, "mount(%s, /proc): %m\n", tmpdir);
        exit(1);
    }

    /*
     * Unmount the temporary directory
     */
    if (umount(tmpdir) < 0) {
        fuse_log(FUSE_LOG_ERR, "umount(%s): %m\n", tmpdir);
        exit(1);
    }

    /*
     * Remove the temporary directory
     */
    if (rmdir(tmpdir) < 0) {
        fuse_log(FUSE_LOG_ERR, "rmdir(%s): %m\n", tmpdir);
        exit(1);
    }
