    /* Create a temporary directory for mounting */
    tmpdir = mkdtemp(template);
    if (!tmpdir) {
        fuse_log(FUSE_LOG_ERR, "mkdtemp(): %m\n");
        exit(1);
    }

    /* Mount a tmpfs on the temporary directory */
    if (mount("tmpfs", tmpdir, "tmpfs", MS_NODEV | MS_NOEXEC | MS_NOSUID, NULL) < 0) {
        fuse_log(FUSE_LOG_ERR, "mount(tmpfs): %m\n");
        exit(1);
    }

    /* Bind mount /proc to the temporary directory */
    if (mount("/proc", tmpdir, NULL, MS_BIND, NULL) < 0) {
        fuse_log(FUSE_LOG_ERR, "mount(/proc, MS_BIND): %m\n");
        exit(1);
    }

    /* Remount /proc with the new pid namespace */
    if (mount("proc", tmpdir, "proc",
              MS_NODEV | MS_NOEXEC | MS_NOSUID | MS_RELATIME, NULL) < 0) {
        fuse_log(FUSE_LOG_ERR, "mount(/proc): %m\n");
        exit(1);
    }

    /* Chroot into the temporary directory */
    if (chroot(tmpdir) < 0) {
        fuse_log(FUSE_LOG_ERR, "chroot(): %m\n");
        exit(1);
    }

    /* Change the working directory to the root of the new chroot environment */
    if (chdir("/") < 0) {
        fuse_log(FUSE_LOG_ERR, "chdir(): %m\n");
        exit(1);
    }
