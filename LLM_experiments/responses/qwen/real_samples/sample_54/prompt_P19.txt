    /* Create a temporary directory for bind mounts */
    tmpdir = mkdtemp(template);
    if (!tmpdir) {
        fuse_log(FUSE_LOG_ERR, "mkdtemp(): %m\n");
        exit(1);
    }

    /* Bind mount /dev to the temporary directory */
    if (mount("/dev", tmpdir, NULL, MS_BIND, NULL) < 0) {
        fuse_log(FUSE_LOG_ERR, "mount(/dev, %s, MS_BIND): %m\n", tmpdir);
        exit(1);
    }

    /* Remount /dev as read-only */
    if (mount(NULL, tmpdir, NULL, MS_REMOUNT | MS_BIND | MS_RDONLY, NULL) < 0) {
        fuse_log(FUSE_LOG_ERR, "mount(%s, MS_REMOUNT|MS_BIND|MS_RDONLY): %m\n", tmpdir);
        exit(1);
    }

    /* Chroot into the temporary directory */
    if (chroot(tmpdir) < 0) {
        fuse_log(FUSE_LOG_ERR, "chroot(%s): %m\n", tmpdir);
        exit(1);
    }

    /* Change working directory to root after chroot */
    if (chdir("/") < 0) {
        fuse_log(FUSE_LOG_ERR, "chdir(/): %m\n");
        exit(1);
    }
</repair>
