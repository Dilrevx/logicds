#include <string.h>

// Function prototype for URI decoding (assuming it exists or you implement it)
char* mk_uri_decode(const char* uri);

if (sr->uri != NULL) {
    // Decode the URI
    char* decoded_uri = mk_uri_decode(sr->uri);
    if (decoded_uri == NULL) {
        // Handle error in decoding
        PLUGIN_TRACE("[FD %i] Close connection, URI decoding failed", cs->socket);
        mk_api->header_set_http_status(sr, MK_CLIENT_BAD_REQUEST);
        return MK_PLUGIN_RET_CLOSE_CONX;
    }

    // Perform security check on the decoded URI
    if (mk_security_check_url(decoded_uri) < 0) {
        PLUGIN_TRACE("[FD %i] Close connection, blocked URL", cs->socket);
        mk_api->header_set_http_status(sr, MK_CLIENT_FORBIDDEN);
        free(decoded_uri); // Free the allocated memory for decoded URI
        return MK_PLUGIN_RET_CLOSE_CONX;
    }

    // Free the allocated memory for decoded URI if not blocked
    free(decoded_uri);
}
