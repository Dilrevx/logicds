static void x509_crt_verify_name( const mbedtls_x509_crt *crt,
                                  const char *cn,
                                  uint32_t *flags )
{
    const mbedtls_x509_name *name;
    const mbedtls_x509_sequence *cur;
    size_t cn_len = strlen( cn );

    if (cn == NULL || cn_len == 0) {
        *flags |= MBEDTLS_X509_BADCERT_CN_MISMATCH;
        return;
    }

    for( name = &crt->subject; name!= NULL; name = mbedtls_x509_get_subject_alt_name( crt, name ) )
    {
        for( cur = name->oids; cur!= NULL; cur = mbedtls_x509_get_next_sequence( cur ) )
        {
            if( mbedtls_oid_get_numeric_string( &cur->oid, "2.5.4.3" ) == 0 )
            {
                if( mbedtls_x509_string_cmp( &cur->val, cn ) == 0 )
                {
                    *flags &= ~MBEDTLS_X509_BADCERT_CN_MISMATCH;
                    return;
                }
            }
        }
    }

    *flags |= MBEDTLS_X509_BADCERT_CN_MISMATCH;
}
