func VerifyGet(cfg schema.AuthenticationBackendConfiguration) middlewares.RequestHandler {
	refreshProfile, refreshProfileInterval := getProfileRefreshSettings(cfg)

	return func(ctx *middlewares.AutheliaCtx) {
		ctx.Logger.Tracef("Headers=%s", ctx.Request.Header.String())
		targetURL, err := ctx.GetOriginalURL()
		if err!= nil {
			ctx.Logger.Errorf("Failed to get original URL: %v", err)
			ctx.Error(http.StatusInternalServerError, "Failed to get original URL")
			return
		}

		if targetURL.Scheme!= "https" {
			ctx.Logger.Warnf("Target URL scheme is not HTTPS: %s", targetURL.Scheme)
			ctx.Error(http.StatusForbidden, "Target URL scheme is not HTTPS")
			return
		}
	}
}
