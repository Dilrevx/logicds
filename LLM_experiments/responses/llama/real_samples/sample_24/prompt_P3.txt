/* Add a function to check if the OCSP response is for the correct certificate */
bool cert_matches_responder(const cert_t *cert, const cert_t *responder_cert) {
    return (cert->serial == responder_cert->serial && 
            memcmp(cert->issuer_dn, responder_cert->issuer_dn, sizeof(cert->issuer_dn)) == 0);
}
