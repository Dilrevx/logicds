#define EXPECTED_SERIAL_NUMBER_LENGTH 20

ret = gnutls_x509_crt_get_serial(crt, cserial.data, &t);
if (ret < 0) {
    gnutls_assert();
    goto cleanup;
}

if (rserial.size!= EXPECTED_SERIAL_NUMBER_LENGTH) {
    ret = GNUTLS_E_OCSP_RESPONSE_ERROR;
    gnutls_assert();
    goto cleanup;
}

if (cserial.size > rserial.size) {
    if (memcmp(cserial.data + rserial.size, "\0", cserial.size - rserial.size)!= 0) {
        ret = GNUTLS_E_OCSP_RESPONSE_ERROR;
        gnutls_assert();
        goto cleanup;
    }
}

if (rserial.size!= cserial.size
    || memcmp(cserial.data, rserial.data, rserial.size)!= 0) {
    ret = GNUTLS_E_OCSP_RESPONSE_ERROR;
    gnutls_assert();
    goto cleanup;
}
