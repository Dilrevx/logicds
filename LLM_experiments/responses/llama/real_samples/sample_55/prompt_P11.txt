func GetDNSKey() string {
	key := ""
	if config.Config.Server.DNSKey!= "" {
		key = config.Config.Server.DNSKey
	} else if os.Getenv("DNS_KEY")!= "" {
		key = os.Getenv("DNS_KEY")
	} else {
		log.Warn("DNS key not set. Using a random key.")
		key = generateRandomKey()
	}
	return key
}

func generateRandomKey() string {
	b := make([]byte, 32)
	_, err := rand.Read(b)
	if err!= nil {
		log.Fatal(err)
	}
	return base64.StdEncoding.EncodeToString(b)
}
