func GetDNSKey() (string, error) {
	key := os.Getenv("DNS_KEY")
	if key == "" {
		return "", errors.New("DNS_KEY environment variable is not set")
	}
	// Validate the key to prevent potential attacks
	if!isValidDNSKey(key) {
		return "", errors.New("invalid DNS_KEY")
	}
	return key, nil
}

func isValidDNSKey(key string) bool {
	// Implement a validation logic for the DNS key
	// For example, check if the key is a valid UUID
	_, err := uuid.Parse(key)
	return err == nil
}
