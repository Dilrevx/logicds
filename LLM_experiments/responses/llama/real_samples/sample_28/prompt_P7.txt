result = dns_keytable_findkeynode(vctx->secroots, vctx->origin,
                                  dst_key_alg(key), dst_key_id(key),
                                  &keynode);
if (result == ISC_R_SUCCESS) {
    /*
     * The supplied key is a potential trust anchor.
     */
    dns_rdataset_t dnskey_rrset;
    dns_rdataset_init(&dnskey_rrset);

    // Get the DNSKEY RRset
    result = dns_db_find(vctx->db, vctx->origin, dns_rdatatype_dnskey, 0, &dnskey_rrset);
    if (result == ISC_R_SUCCESS) {
        // Validate the DNSKEY RRset
        result = dns_dnssec_validate(vctx->db, vctx->origin, &dnskey_rrset, vctx->now, vctx->keyset, vctx->keysigs);
        if (result == ISC_R_SUCCESS) {
            // The DNSKEY RRset is valid, so the key is a trust anchor
            dns_keytable_detachkeynode(vctx->secroots, &keynode);
            dns_rdataset_settrust(&vctx->keyset, dns_trust_secure);
            dns_rdataset_settrust(&vctx->keysigs, dns_trust_secure);
            *goodkey = true;
        } else {
            // The DNSKEY RRset is not valid, so the key is not a trust anchor
            *goodkey = false;
        }
        dns_rdataset_disassociate(&dnskey_rrset);
    } else {
        // An error occurred while getting the DNSKEY RRset
        *goodkey = false;
    }
} else if (result == DNS_R_PARTIALMATCH) {
    // The supplied key is not present in the trust anchor table,
    // but other keys signing the DNSKEY RRset may be, so this is
    // not an error, we just do not set 'vctx->good[kz]sk'.
    result = ISC_R_SUCCESS;
    *goodkey = false;
} else if (result == ISC_R_NOTFOUND) {
    // The supplied key is not present in the trust anchor table
    *goodkey = false;
} else {
    // An error occurred while searching the trust anchor table,
    // return it to the caller.
}

/*
 * Clean up.
 */
dst_key_free(&key);

return (result);
