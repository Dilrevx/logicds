static int evp_cipher_init_internal(EVP_CIPHER_CTX *ctx,
                                    const EVP_CIPHER *cipher,
                                    ENGINE *impl, const unsigned char *key,
                                    const unsigned char *iv, int enc,
                                    const OSSL_PARAM params[])
{
    int n;
#if!defined(OPENSSL_NO_ENGINE) &&!defined(FIPS_MODULE)
    ENGINE *tmpimpl = NULL;
#endif

    ctx->iv_len = -1;

    if (enc == -1) {
        if (ctx->encrypt == -1) {
            ERR_raise(ERR_LIB_EVP, EVP_R_NO_CIPHER_DIRECTION);
            return 0;
        }
        enc = ctx->encrypt;
    } else {
        if (enc!= 0 && enc!= 1) {
            ERR_raise(ERR_LIB_EVP, EVP_R_INVALID_OPERATION);
            return 0;
        }
        ctx->encrypt = enc;
    }

    if (cipher == NULL && ctx->cipher == NULL) {
        ERR_raise(ERR_LIB_EVP, EVP_R_NO_CIPHER_SET);
        return 0;
    }

    /* Code below to be removed when legacy support is dropped. */

#if!defined(OPENSSL_NO_ENGINE) &&!defined(FIPS_MODULE)
    if (ctx->engine && ctx->cipher
        && (cipher == NULL || cipher->nid == ctx->cipher->nid))
        goto skip_to_init;

    if (cipher!= NULL && impl == NULL) {
         /* Ask if an ENGINE is reserved for this job */
        tmpimpl = ENGINE_get_cipher_engine(cipher->nid);
    }
#endif

    if (cipher!= NULL && (cipher->flags & EVP_CIPH_FLAG_CUSTOM_IV) && iv == NULL) {
        ERR_raise(ERR_LIB_EVP, EVP_R_MISSING_IV);
        return 0;
    }

    if (cipher!= NULL && (cipher->flags & EVP_CIPH_FLAG_CUSTOM_KEY) && key == NULL) {
        ERR_raise(ERR_LIB_EVP, EVP_R_MISSING_KEY);
        return 0;
    }
}
