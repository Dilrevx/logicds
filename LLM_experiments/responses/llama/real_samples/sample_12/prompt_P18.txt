int DoTls13Finished(WOLFSSL* ssl, const byte* input, word32* inOutIdx,
                           word32 size, word32 totalSz, int sniff)
{
    int    ret;
    word32 finishedSz = 0;
    byte*  secret;
    byte   mac[WC_MAX_DIGEST_SIZE];

    if (ssl == NULL || input == NULL || inOutIdx == NULL || size == 0 || totalSz == 0) {
        return BAD_FUNC_ARG;
    }

    if (totalSz < TLS_FINISHED_SZ) {
        return BUFFER_E;
    }

    if (ssl->options.side == WOLFSSL_CLIENT_END) {
        secret = ssl->keys.client_finished_secret;
    } else {
        secret = ssl->keys.server_finished_secret;
    }

    if (wolfSSL_Hmac(ssl->macHmacEnc, secret, TLS_FINISHED_SECRET_SZ,
                     input, *inOutIdx, mac)!= 0) {
        return MEMORY_E;
    }

    finishedSz = TLS_FINISHED_SZ;
    if (XMEMCMP(mac, input + *inOutIdx, finishedSz)!= 0) {
        return DECRYPTION_E;
    }

    *inOutIdx += finishedSz;
    return 0;
}
