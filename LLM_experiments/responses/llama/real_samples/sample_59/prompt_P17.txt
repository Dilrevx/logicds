func (cs *CaptchaService) VerifyCaptcha(ctx context.Context, key, captcha string) (isCorrect bool, err error) {
	if key == "" {
		return false, errors.New("captcha key is empty")
	}
	if captcha == "" {
		return false, errors.New("captcha is empty")
	}
	realCaptcha, err := cs.captchaRepo.GetCaptcha(ctx, key)
	if err!= nil {
		return false, err
	}
	if realCaptcha == "" {
		return false, errors.New("captcha not found")
	}
	return captcha == realCaptcha, nil
}
