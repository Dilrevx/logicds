            /* Check for public key of required type. */
            if (ssl->keys->key_type!= rsa && ssl->keys->key_type!= ecc) {
                ERROR_OUT(INVALID_PARAMETER, exit_dcv);
            }
            if (args->sigAlgo == rsa_sa_algo && ssl->keys->key_type!= rsa) {
                ERROR_OUT(INVALID_PARAMETER, exit_dcv);
            }
            if (args->sigAlgo == ecdsa_sa_algo && ssl->keys->key_type!= ecc) {
                ERROR_OUT(INVALID_PARAMETER, exit_dcv);
            }
            if (args->sigAlgo == ed25519_sa_algo && ssl->keys->key_type!= ed25519) {
                ERROR_OUT(INVALID_PARAMETER, exit_dcv);
            }
            if (args->sigAlgo == ed448_sa_algo && ssl->keys->key_type!= ed448) {
                ERROR_OUT(INVALID_PARAMETER, exit_dcv);
            }
            if (args->sigAlgo == anonymous_sa_algo && ssl->keys->key_type!= anonymous) {
                ERROR_OUT(INVALID_PARAMETER, exit_dcv);

            if (args->sz > 0) {
                if (args->sz > buffer_size(sig)) {
                    ERROR_OUT(BUFFER_ERROR, exit_dcv);
                }
                XMEMCPY(sig->buffer, input + args->idx, args->sz);
                sig->length = args->sz;
            }

            /* Advance state and proceed */
            ssl->options.asyncState = TLS_ASYNC_VERIFY;
        } /* case TLS_ASYNC_BUILD */
        FALL_THROUGH;

        case TLS_ASYNC_VERIFY:
        {
            /* Verify signature. */
            ret = VerifyTls13Certificate(ssl, args->hashAlgo, args->sigAlgo,
                                         sig->buffer, sig->length);
            if (ret < 0)
                goto exit_dcv;

            /* Advance state and proceed */
            ssl->options.asyncState = TLS_ASYNC_DONE;
        } /* case TLS_ASYNC_VERIFY */
        FALL_THROUGH;

        case TLS_ASYNC_DONE:
        {
            /* Done with CertificateVerify. */
            ssl->options.asyncState = TLS_ASYNC_BEGIN;
            *inOutIdx = args->idx + args->sz;
            ret = 0;
        } /* case TLS_ASYNC_DONE */
        break;

        default:
            ERROR_OUT(ASYNC_FATAL_ERROR, exit_dcv);
    }

exit_dcv:
    WOLFSSL_LEAVE("DoTls13CertificateVerify", ret);
    WOLFSSL_END(WC_FUNC_CERTIFICATE_VERIFY_DO);
    return ret;
