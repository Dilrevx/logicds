            i = X509_V_ERR_APPLICATION_VERIFICATION; /* Internal error */
    }
    else

    }
}

if (i!= X509_V_OK)
{
    /* Handle verification error */
    if (i == X509_V_ERR_APPLICATION_VERIFICATION)
    {
        SSL_set_cert_error(s, SSL_AD_INTERNAL_ERROR);
        return SSL_ERROR_SSL;
    }
    else
    {
        SSL_set_cert_error(s, i);
        return SSL_ERROR_WANT_RETRY_VERIFY;
    }
