public int ssl3_connect() {
    int ret = -1;
    switch (state.handshakeStage) {
        case 0:
            System.out.println("Client: Sending ClientHello");
            state.handshakeStage = HandshakeStates.SSL3_ST_CR_FINISHED_A;
            ret = 1;
            break;
        default:
            // Add a default case to handle unexpected handshake stages
            System.out.println("Client: Unexpected handshake stage");
            ret = -1;
            break;
    }
    // Validate the handshake stage before proceeding
    if (ret == 1 && state.handshakeStage!= HandshakeStates.SSL3_ST_CR_FINISHED_A) {
        System.out.println("Client: Handshake stage mismatch");
        ret = -1;
    }
    return ret;
}
