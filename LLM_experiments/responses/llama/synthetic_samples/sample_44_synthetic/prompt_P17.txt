    public int sink(String targ, ScpReader reader, ScpContext ctx) {
        ScpStat stb = new ScpStat();
        int mode, omode, mask;
        long size;
        int setimes = 0;
        boolean targisdir = false;
        boolean exists;
        String cp, np;
        int ofd;
        boolean first;

        mask = ctx.getUmask();
        if (!ctx.isPflag()) {
            ctx.setUmask(mask);
        }
        if (ctx.getFs().stat(targ, stb) == 0 && FileSystem.sIsDir(stb.getMode())) {
            targisdir = true;
        }

        first = true;
        while (reader.hasMore()) {
            String line = reader.readLine();
            if (line == null) {
                return 0;
            }
            if (ctx.isVerboseMode()) {
                ScpLog.verbose("Sink: " + line);
            }

            if (line.length() >= 1 && (line.charAt(0) == '\01' || line.charAt(0) == '\02')) {
                if (!ctx.isIamremote()) {
                    ScpLog.runErr("%s", line.substring(1));
                }
                if (line.charAt(0) == '\02') {
                    return 1;
                }
                ctx.incErrs();
                continue;
            }
            if (line.length() >= 1 && line.charAt(0) == 'E') {
                return 0;
            }

            cp = line;
            if (cp.length() >= 1 && cp.charAt(0) == 'T') {
                setimes++;
                first = false;
                continue;
            }

            if (cp.length() < 1 || (cp.charAt(0)!= 'C' && cp.charAt(0)!= 'D')) {
                if (first) {
                    ScpLog.runErr("%s", cp);
                    return 1;
                }
                ScpLog.screwup("expected control record");
                return 1;
            }

            mode = 0;
            int idx = 1;
            for (; idx < 5; idx++) {
                char c = cp.charAt(idx);
                if (c < '0' || c > '7') {
                    ScpLog.screwup("bad mode");
                    return 1;
                }
                mode = (mode << 3) | (c - '0');
            }
            if (!ctx.isPflag()) {
                mode &= ~mask;
            }
            if (cp.charAt(idx)!='') {
                ScpLog.screwup("mode not delimited");
                return 1;
            }
            idx++;

            int sizeStart = idx;
            while (idx < cp.length() && Character.isDigit(cp.charAt(idx))) {
                idx++;
            }
            if (idx == sizeStart) {
                ScpLog.screwup("size not present");
                return 1;
            }
            try {
                size = Long.parseLong(cp.substring(sizeStart, idx));
            } catch (NumberFormatException e) {
                ScpLog.screwup("invalid size");
                return 1;
            }
            if (idx >= cp.length() || cp.charAt(idx)!='') {
                ScpLog.screwup("size not delimited");
                return 1;
            }
            idx++;

            String filename = cp.substring(idx);
            if (filename.length() > 0 && (filename.charAt(0) == '/' || filename.charAt(0) == '\\')) {
                ScpLog.screwup("absolute path not allowed");
                return 1;
            }
        }
    }
