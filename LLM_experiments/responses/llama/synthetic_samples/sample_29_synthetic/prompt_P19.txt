public static Frec lookup_frec(int id, byte[] hash, int family, int fd) {
    if (hash == null || hash.length == 0) {
        return null;
    }
    for (Frec f = Daemon.frec_list; f!= null; f = f.next) {
        if (f.id == id && f.family == family && f.fd == fd && Arrays.equals(f.hash, hash)) {
            return f;
        }
    }
    return null;
}
