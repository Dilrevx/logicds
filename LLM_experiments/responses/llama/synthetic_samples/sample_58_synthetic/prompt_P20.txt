    public void adminAuth(GinContext ctx) {
        String token = extractToken(ctx);
        if (token.length() == 0) {
            handleResponse(ctx, AuthError.unauthorized(AuthError.UNAUTHORIZED_REASON));
            ctx.abort();
            return;
        }

        AuthService.CacheResult result = authService.getAdminUserCacheInfo(ctx, token);
        UserInfo userInfo = result.userInfo;
        Exception err = result.err;

        if (err!= null) {
            handleResponse(ctx, AuthError.forbidden(AuthError.UNAUTHORIZED_REASON));
            ctx.abort();
            return;
        }
        if (userInfo == null) {
            handleResponse(ctx, AuthError.unauthorized(AuthError.UNAUTHORIZED_REASON));
            ctx.abort();
            return;
        }
        // Rest of the function remains the same
    }
