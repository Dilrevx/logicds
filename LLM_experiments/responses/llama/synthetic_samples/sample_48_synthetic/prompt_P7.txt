// Helper function for secure string comparison
private boolean secureCompare(String a, String b) {
    if (a == null || b == null) {
        return false;
    }
    int result = 0;
    for (int i = 0; i < a.length(); i++) {
        result |= a.charAt(i) ^ b.charAt(i);
    }
    for (int i = 0; i < b.length(); i++) {
        result |= a.charAt(i) ^ b.charAt(i);
    }
    return result == 0;
}
