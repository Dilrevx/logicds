public static void x509_crt_verify_name(MbedtlsX509Crt crt, String cn, int[] flags) {
    MbedtlsX509Name     name;
    MbedtlsX509Sequence cur;
    int                 cn_len = cn.length();
    if ((crt.extTypes & MbedtlsConstants.MBEDTLS_X509_EXT_SUBJECT_ALT_NAME)!= 0) {
        for (cur = crt.subjectAltNames; cur!= null; cur = cur.next) {
            if (cur.tag == MbedtlsConstants.MBEDTLS_ASN1_DNS) {
                String dnsName = new String(cur.data, 0, cur.len);
                if (dnsName.equals(cn)) {
                    return;
                }
            }
        }
    }
    name = crt.subject;
    for (cur = name; cur!= null; cur = cur.next) {
        if (cur.tag == MbedtlsConstants.MBEDTLS_OID_AT_COMMON_NAME) {
            if (cur.len == cn_len) {
                if (Arrays.equals(cur.data, cn.getBytes())) {
                    return;
                }
            }
        }
    }
    throw new MbedtlsException("Certificate common name does not match");
}
