    public static void x509_crt_verify_name(MbedtlsX509Crt crt, String cn, int[] flags) {
        MbedtlsX509Name     name;
        MbedtlsX509Sequence cur;
        int                 cn_len = cn.length();
        if ((crt.extTypes & MbedtlsConstants.MBEDTLS_X509_EXT_SUBJECT_ALT_NAME)!= 0) {
            for (cur = crt.subjectAltNames; cur!= null; cur = cur.next) {
                if (cur.type == MbedtlsConstants.MBEDTLS_X509_NAME_EMAIL || 
                    cur.type == MbedtlsConstants.MBEDTLS_X509_NAME_DNS || 
                    cur.type == MbedtlsConstants.MBEDTLS_X509_NAME_URI) {
                    if (CommonUtils.x509_crt_check_san(cur.buf, cn, cn_len) == 0)
                        break;
                }
            }
        } else {
            if (CommonUtils.x509_crt_check_cn(crt.subject.common_name, cn, cn_len) == 0)
                return;
        }
    }
