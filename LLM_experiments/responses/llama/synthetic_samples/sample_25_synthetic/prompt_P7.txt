    Wslib.WOLFSSL_MSG("Peer sent ED25519 key but not in verify");
    return Commons.INVALID_PARAMETER;
}
if (args.sigAlgo == Commons.ed448_sa_algo &&!ssl.peerEd448KeyPresent) {
    Wslib.WOLFSSL_MSG("Peer sent ED448 key but not in verify");
    return Commons.INVALID_PARAMETER;
}
if (args.sigAlgo == Commons.ecc_dsa_sa_algo &&!ssl.peerEccDsaKeyPresent) {
    Wslib.WOLFSSL_MSG("Peer sent ECC key but not in verify");
    return Commons.INVALID_PARAMETER;
}
if (args.sigAlgo == Commons.rsa_pss_sa_algo &&!ssl.peerRsaKeyPresent) {
    Wslib.WOLFSSL_MSG("Peer sent RSA key but not in verify");
    return Commons.INVALID_PARAMETER;
}
if (args.sigAlgo == Commons.rsa_sa_algo) {
    Wslib.WOLFSSL_MSG("RSA PKCS#1.5 signatures are not supported");
    return Commons.INVALID_PARAMETER;
}
