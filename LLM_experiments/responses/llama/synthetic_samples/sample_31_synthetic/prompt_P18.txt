public static int ssl_verify_cert_chain(OpenSslSsl s, List<X509> sk) {
    X509_STORE_CTX ctxStore = new X509_STORE_CTX();
    int i;
    if (s.verify_callback!= null) {
        ctxStore.set_verify_cb(s.verify_callback);
        ctxStore.set_verify_cb_arg(s);
    }
    ctxStore.set_cert_store(s.cert_store);
    ctxStore.set_chain(sk);
    ctxStore.set0_cert(s.peer_cert);
    i = X509_verify_cert(ctxStore);
    if (i <= 0) {
        int error = X509_STORE_CTX_get_error(ctxStore);
        if (error == X509_V_ERR_DEPTH_ZERO_SELF_SIGNED_CERT) {
            // Handle self-signed certificate
        } else if (error == X509_V_ERR_UNABLE_TO_GET_ISSUER_CERT_LOCALLY) {
            // Handle unable to get issuer certificate locally
        } else {
            // Handle other errors
        }
    }
    X509_STORE_CTX_free(ctxStore);
    return i;
}
