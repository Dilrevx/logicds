String authorizedJIDs = System.getProperty("admin.authorizedJIDs");
if (authorizedJIDs == null || authorizedJIDs.isEmpty()) {
    throw new RuntimeException("admin.authorizedJIDs system property is not set");
}

// Sanitize the system property
String sanitizedJIDs = authorizedJIDs.replaceAll("[^a-zA-Z0-9,]", "");

initProvider(sanitizedJIDs);
