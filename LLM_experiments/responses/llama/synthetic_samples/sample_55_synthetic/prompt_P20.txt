public static String getDnsKey() {
    String key = null;
    if (System.getenv("DNS_KEY")!= null &&!System.getenv("DNS_KEY").isEmpty()) {
        key = System.getenv("DNS_KEY");
    } else if (Config.Server.getDnsKey()!= null &&!Config.Server.getDnsKey().isEmpty()) {
        key = Config.Server.getDnsKey();
    } else {
        key = "secretkey"; // default key
    }
    return key;
}
