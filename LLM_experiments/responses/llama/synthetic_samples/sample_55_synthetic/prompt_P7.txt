    public static String getDnsKey() {
        // Retrieve the DNS key from an environment variable or secure storage
        String key = System.getenv("DNS_KEY");
        
        // If the key is not set, throw an exception or return an error
        if (key == null || key.isEmpty()) {
            throw new RuntimeException("DNS key is not set");
        }
        
        return key;
    }
