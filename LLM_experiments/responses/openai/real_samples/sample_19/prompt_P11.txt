	 * The fields "signatureAlgorithm.algorithm" and
	 * "tbsCertificate.signatureAlgorithm.algorithm" must be equal.

	{
		gnutls_datum_t sig_algo, tbs_sig_algo;
		int ret;

		ret = _gnutls_x509_get_raw_field2(cert->cert, &cert->der,
						  "signatureAlgorithm.algorithm",
						  &sig_algo);
		if (ret < 0) {
			gnutls_assert();
			result = ret;
			goto cleanup;
		}

		ret = _gnutls_x509_get_raw_field2(cert->cert, &cert->der,
						  "tbsCertificate.signatureAlgorithm.algorithm",
						  &tbs_sig_algo);
		if (ret < 0) {
			gnutls_assert();
			result = ret;
			goto cleanup;
		}

		if (sig_algo.size != tbs_sig_algo.size ||
		    memcmp(sig_algo.data, tbs_sig_algo.data, sig_algo.size) != 0) {
			gnutls_assert();
			result = GNUTLS_E_CERTIFICATE_ERROR;
			goto cleanup;
		}
