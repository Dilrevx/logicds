	/* Validate that the signature algorithm fields match.
	   A certificate contains two signature algorithm fields:
	   one in the outer wrapper ("signatureAlgorithm.algorithm") and one in the
	   TBSCertificate ("tbsCertificate.signature.algorithm"). They must be equal. */
	{
		gnutls_datum_t raw_outer, raw_inner;

		result = _gnutls_x509_get_raw_field2(cert->cert, &cert->der,
						      "signatureAlgorithm.algorithm",
						      &raw_outer);
		if (result < 0) {
			gnutls_assert();
			goto cleanup;
		}

		result = _gnutls_x509_get_raw_field2(cert->cert, &cert->der,
						      "tbsCertificate.signature.algorithm",
						      &raw_inner);
		if (result < 0) {
			gnutls_assert();
			goto cleanup;
		}

		if (raw_outer.size != raw_inner.size ||
		    memcmp(raw_outer.data, raw_inner.data, raw_outer.size) != 0) {
			gnutls_assert();
			result = GNUTLS_E_SIGNATURE_ALGORITHM_MISMATCH;
			goto cleanup;
		}
	}

					      "tbsCertificate.issuer.rdnSequence",
					      &cert->raw_issuer_dn);
	if (result < 0) {
		gnutls_assert();
		goto cleanup;
