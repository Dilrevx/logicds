    /* Remember whether 'first_kex_packet_follows' was set and the client guess was wrong.
     * In that case (on the server side), the next SSH_MSG_KEXDH_INIT message must be ignored.
     *
     * The bug was that even if first_kex_packet_follows was set, no explicit action was taken
     * to mark the session so that a premature SSH_MSG_KEXDH_INIT would be rejected.
     * Here we record the state by setting a session flag. The message handler for SSH_MSG_KEXDH_INIT
     * must check for SSH_SESSION_FLAG_IGNORE_FIRST_KEXDH_INIT and ignore the message accordingly.
     */
    if (server_kex) {
        if (first_kex_packet_follows) {
            SSH_LOG(SSH_LOG_DEBUG,
                    "Server: first_kex_packet_follows set and client guess is incorrect. "
                    "Marking to ignore next SSH_MSG_KEXDH_INIT.");
            session->flags |= SSH_SESSION_FLAG_IGNORE_FIRST_KEXDH_INIT;
        } else {
            session->flags &= ~SSH_SESSION_FLAG_IGNORE_FIRST_KEXDH_INIT;
        }
    }
