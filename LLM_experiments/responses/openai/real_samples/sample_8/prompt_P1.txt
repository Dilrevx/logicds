    if (first_kex_packet_follows || session->send_first_kex_follows) {
        /*
         * Mitigation for the Terrapin vulnerability:
         *
         * The original code attempted to “guess” the first key exchange algorithms
         * by comparing the client and server lists. However, due to missing packets
         * in the handshake (the Terrapin attack), an attacker can force a downgrade of
         * security features if the initial guess is trusted.
         *
         * To prevent this, we unconditionally treat the first guess as incorrect when
         * first_kex_packet_follows is used. This ensures that the client and server will
         * perform a full, secure key exchange rather than relying on the (possibly
         * manipulated) preliminary guess.
         */
        session->first_kex_follows_guess_wrong = 1;
        SSH_LOG(SSH_LOG_WARNING, "first_kex_packet_follows is set; initial guess forced to wrong to mitigate Terrapin attack");
    }
