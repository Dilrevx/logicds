func GetDNSKey() string {
	// Attempt to get the DNS key from the environment variable first.
	if key := os.Getenv("DNS_KEY"); key != "" {
		return key
	}

	// If not set in the environment, try the configuration.
	if config.Config.Server.DNSKey != "" {
		return config.Config.Server.DNSKey
	}

	// Neither source provided a valid DNS key. Avoid using a hardcoded default.
	// It's safer to fail fast than to use an insecure default key.
	log.Fatal("DNS key is not configured; please set the DNS_KEY environment variable or configure config.Config.Server.DNSKey")
	return ""
}
