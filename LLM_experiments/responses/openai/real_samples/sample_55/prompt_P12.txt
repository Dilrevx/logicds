func GetDNSKey() string {
	// First, check if the environment variable is set.
	if key := os.Getenv("DNS_KEY"); key != "" {
		return key
	}
	// Next, check if the configuration value is set.
	if key := config.Config.Server.DNSKey; key != "" {
		return key
	}
	// If no key is provided, do not return a hardcoded default.
	// Returning an empty string forces the caller to handle the missing key appropriately.
	return ""
}
