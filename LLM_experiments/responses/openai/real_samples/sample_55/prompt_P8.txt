func GetDNSKey() string {
	// First, try reading the DNS key from the environment.
	if key := os.Getenv("DNS_KEY"); key != "" {
		return key
	}
	// Next, check the configuration.
	if config.Config.Server.DNSKey != "" {
		return config.Config.Server.DNSKey
	}
	// If no key is provided, do not fall back to an insecure hard-coded default.
	// Instead, log the error and exit the application.
	log.Fatal("DNS key must be provided via the DNS_KEY environment variable or config file")
	return ""
}
