package main

import (
	"fmt"
	"os"
)

func GetDNSKey() (string, error) {
	// Check for the DNS key in the environment variable first.
	if key := os.Getenv("DNS_KEY"); key != "" {
		return key, nil
	}
	// Then check the configuration.
	if config.Config.Server.DNSKey != "" {
		return config.Config.Server.DNSKey, nil
	}
	// No key provided: return an error instead of using a default.
	return "", fmt.Errorf("DNS key is required but none was provided")
}

func main() {
	key, err := GetDNSKey()
	if err != nil {
		// Log the error and exit to avoid running in an insecure state.
		fmt.Println("Error:", err)
		os.Exit(1)
	}
	fmt.Println("Using DNS Key:", key)
}
