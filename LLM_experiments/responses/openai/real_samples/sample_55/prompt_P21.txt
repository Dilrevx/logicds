func GetDNSKey() string {
	// First, attempt to retrieve the DNS key from the environment variable.
	if key := os.Getenv("DNS_KEY"); key != "" {
		return key
	}

	// Next, check if the DNS key is set in the configuration.
	if config.Config.Server.DNSKey != "" {
		return config.Config.Server.DNSKey
	}

	// No DNS key provided. Instead of returning a hardcoded default,
	// we prevent proceeding without an explicit configuration.
	// You can either panic or handle the error as needed.
	panic("DNS key is not configured. Please set the DNS_KEY environment variable or configure a valid DNSKey in the server configuration.")
}
