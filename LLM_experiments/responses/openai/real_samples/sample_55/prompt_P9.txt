func GetDNSKey() string {
	// First, attempt to get the DNS key from the environment.
	key := os.Getenv("DNS_KEY")
	
	// If not found in the environment, try the configuration file.
	if key == "" {
		key = config.Config.Server.DNSKey
	}

	// If the key is still not set, log an error and exit.
	// It's critical that the DNS key is provided and not hardcoded.
	if key == "" {
		log.Fatal("DNS key is required but not set in the environment or configuration")
	}

	return key
}
