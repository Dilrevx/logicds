def update_user_profile(user, data):
    # Fixed Logical vulnerability bug:
    # Previously, an incorrect use of logical operators might have caused legitimate update requests to be rejected.
    # The following condition now correctly allows the update if the user is an administrator or if the user is updating their own profile.
    if not (user.is_admin() or user.id == data.get("user_id")):
        raise Exception("Unauthorized user: You must be an administrator or update your own profile.")

    # Proceed safely with the profile update.
    user.update_profile(data)
