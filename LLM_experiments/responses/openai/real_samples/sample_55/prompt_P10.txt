func GetDNSKey() string {
	// Retrieve the DNS key from the environment variable first, then from the configuration.
	// Removed the insecure hardcoded default which could allow unauthenticated access.
	key := os.Getenv("DNS_KEY")
	if key == "" {
		key = config.Config.Server.DNSKey
	}
	return key
}
