func GetDNSKey() string {
	// First use the trusted configuration value.
	key := config.Config.Server.DNSKey

	// Fall back to the environment variable only if the config value isn’t set.
	if key == "" {
		key = os.Getenv("DNS_KEY")
	}

	// Do not fall back to a hard‐coded “secretkey” default.
	// Rather, enforce that a proper key is provided.
	if key == "" {
		panic("DNS key is not configured")
	}

	return key
}
