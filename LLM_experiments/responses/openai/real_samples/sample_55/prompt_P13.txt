func GetDNSKey() string {
	// Remove hardcoded fallback to "secretkey" to ensure DNS key is explicitly configured.
	key := os.Getenv("DNS_KEY")
	if key != "" {
		return key
	}
	if config.Config.Server.DNSKey != "" {
		return config.Config.Server.DNSKey
	}
	// If neither source provides a valid DNS key, return an empty string.
	// Additional handling (e.g., logging, error returns, or failure to start) could be implemented as needed.
	return ""
}
