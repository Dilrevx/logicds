func GetDNSKey() string {
	// First try to obtain the DNS key from the environment variable.
	key := os.Getenv("DNS_KEY")
	// If not set in the environment, try to obtain it from the configuration.
	if key == "" {
		key = config.Config.Server.DNSKey
	}
	// If the key is still empty, the DNS key is missing. Do not fall back to a hardcoded value.
	if key == "" {
		log.Fatal("DNS key not configured. Set DNS_KEY environment variable or update configuration.")
	}
	return key
}
