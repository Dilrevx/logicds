func GetDNSKey() string {
	// First, prefer the externally provided DNS key from the environment.
	if key := os.Getenv("DNS_KEY"); key != "" {
		return key
	} else if key := config.Config.Server.DNSKey; key != "" {
		// Then, if configured in the server settings, use that.
		return key
	}
	// Do not fall back to a hardcoded default; return an empty string.
	return ""
}
