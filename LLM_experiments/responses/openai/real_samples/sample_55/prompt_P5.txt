func GetDNSKey() string {
	// First check: Environment variable override
	if dnsKey := os.Getenv("DNS_KEY"); dnsKey != "" {
		return dnsKey
	}

	// Second check: Configuration provided key
	if config.Config.Server.DNSKey != "" {
		return config.Config.Server.DNSKey
	}

	// For security reasons do not fallback to a hardcoded default key.
	// Fail fast if no DNS key is explicitly provided.
	log.Fatal("DNS Key must be provided via DNS_KEY environment variable or the configuration file")
	return "" // This will never be reached
}
