func GetDNSKey() (string, error) {
	// First check environment variable for the DNS key.
	if key := os.Getenv("DNS_KEY"); key != "" {
		return key, nil
	}

	// Next, check the configuration for the DNS key.
	if key := config.Config.Server.DNSKey; key != "" {
		return key, nil
	}

	// If no DNS key is set via environment or config, return an error rather than using a hardcoded default.
	return "", errors.New("DNS key is not configured")
}
