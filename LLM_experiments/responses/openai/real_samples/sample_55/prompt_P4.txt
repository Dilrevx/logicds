func GetDNSKey() string {
	keyFromEnv := os.Getenv("DNS_KEY")
	if keyFromEnv != "" {
		return keyFromEnv
	}
	if config.Config.Server.DNSKey != "" {
		return config.Config.Server.DNSKey
	}
	// No valid DNS key provided; return an empty string instead of using a hardcoded default.
	return ""
}
