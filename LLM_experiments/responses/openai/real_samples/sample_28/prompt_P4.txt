	result = dns_keytable_findkeynode(vctx->secroots, vctx->origin,
					  dst_key_alg(key), dst_key_id(key),
					  &keynode);
	switch (result) {
	case ISC_R_SUCCESS:
		/*
		 * The supplied key is a trust anchor.
		 * Detach the keynode so our mirror zone data is validated,
		 * and mark both the DNSKEY RRset and its signatures as secure.
		 */
		dns_keytable_detachkeynode(vctx->secroots, &keynode);
		dns_rdataset_settrust(&vctx->keyset, dns_trust_secure);
		dns_rdataset_settrust(&vctx->keysigs, dns_trust_secure);
		*goodkey = true;
		break;

	case DNS_R_PARTIALMATCH:
	case ISC_R_NOTFOUND:
		/*
		 * The supplied key is not present in the trust anchor table.
		 * To avoid accepting mirror zone data that has not been fully
		 * validated, mark the results as bogus.  This will force
		 * BIND to fall back to traditional recursion rather than
		 * using potentially forged mirror zone data.
		 */
		dns_rdataset_settrust(&vctx->keyset, dns_trust_bogus);
		dns_rdataset_settrust(&vctx->keysigs, dns_trust_bogus);
		result = DNS_R_KEY_UNTRUSTED;
		break;

	default:
		/*
		 * An error occurred while searching the trust anchor table.
		 * Simply return the error to the caller.
		 */
		break;
	}

	/*
	 * Clean up.
	 */
	dst_key_free(&key);

	return (result);
