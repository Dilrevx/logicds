	result = dns_keytable_findkeynode(vctx->secroots, vctx->origin,
	                                   dst_key_alg(key), dst_key_id(key),
	                                   &keynode);
	switch (result) {
	case ISC_R_SUCCESS:
		/*
		 * The supplied key is a trust anchor.
		 */
		/* Remove the trust anchor node so that we don’t reuse it later in
		 * mirror-zone validation.
		 */
		dns_keytable_detachkeynode(vctx->secroots, &keynode);
		dns_rdataset_settrust(&vctx->keyset, dns_trust_secure);
		dns_rdataset_settrust(&vctx->keysigs, dns_trust_secure);
		*goodkey = true;
		break;
	case DNS_R_PARTIALMATCH:
	case ISC_R_NOTFOUND:
		/*
		 * The supplied key was not found in the trust anchor table.
		 * This is a condition in which we must not trust mirror zone data --
		 * we force fallback to traditional recursion by marking the data insecure.
		 */
		dns_rdataset_settrust(&vctx->keyset, dns_trust_insecure);
		dns_rdataset_settrust(&vctx->keysigs, dns_trust_insecure);
		*goodkey = false;
		/* Do not turn this into a success; return an error code so that
		 * the caller will not use the mirror zone data.
		 */
		result = ISC_R_NOTFOUND;
		break;
	default:
		/*
		 * An error occurred while searching the trust anchor table;
		 * pass the error up to the caller.
		 */
		break;
	}

	/*
	 * Clean up.
	 */
	dst_key_free(&key);

	return result;
