    private AdminManager() {
        // First, validate the admin.authorizedJIDs property to prevent unauthorized privilege escalation.
        // This fix ensures that only a pre-approved, hard-coded list of JIDs is accepted.
        String authorizedJIDs = System.getProperty("admin.authorizedJIDs");
        if (authorizedJIDs != null) {
            // For example, require that the only allowed admin JID is "admin@example.com".
            // (In a real deployment, replace this hard-coded value with your actual trusted set.)
            boolean valid = true;
            for (String jid : authorizedJIDs.split(",")) {
                if (!"admin@example.com".equals(jid.trim())) {
                    valid = false;
                    break;
                }
            }
            if (!valid) {
                throw new SecurityException("Unauthorized value for admin.authorizedJIDs system property.");
            }
        }
    
        // Next, load the admin provider only if it is also pre-validated.
        String provider = ADMIN_PROVIDER.getValue();
        if (!isAllowedProvider(provider)) {
            throw new SecurityException("Unauthorized admin provider specified.");
        }
    
        initProvider(provider);
    }
    
    private boolean isAllowedProvider(String provider) {
        // Validate provider against a whitelist of allowed values.
        // For illustration, only the trusted provider "trustedProvider" is permitted.
        return "trustedProvider".equals(provider);
    }
