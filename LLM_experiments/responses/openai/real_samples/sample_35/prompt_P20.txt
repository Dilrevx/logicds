private AdminManager() {
    // Securely load an admin provider.
    String provider = ADMIN_PROVIDER.getValue();

    // Validate that the provider name is neither null nor empty.
    if (provider == null || provider.trim().isEmpty()) {
        throw new IllegalStateException("Admin provider is not defined.");
    }
    
    // Optional: Validate against a white-list of accepted providers to prevent misconfiguration.
    if (!isAllowedProvider(provider)) {
        throw new IllegalStateException("Unrecognized admin provider: " + provider);
    }
    
    // Initialize with the validated provider.
    initProvider(provider);
}

private boolean isAllowedProvider(String provider) {
    // List of allowed providers. Adjust this list as required.
    return "SecureAdminProvider".equals(provider) ||
           "BackupAdminProvider".equals(provider);
}
