    private AdminManager() {
        // Load an admin provider ensuring that only a trusted provider is used.
        Class<? extends AdminProvider> providerClass = ADMIN_PROVIDER.getValue();
        if (!isProviderAuthorized(providerClass)) {
            Log.error("Unauthorized admin provider attempted: " + providerClass.getName());
            throw new SecurityException("Unauthorized admin provider");
        }
        initProvider(providerClass);
    }
    
    /**
     * Verifies that the supplied admin provider is authorized.
     * In this repair we limit admin providers to the default provider only,
     * preventing an attacker from supplying a custom provider that could escalate privileges.
     */
    private boolean isProviderAuthorized(Class<? extends AdminProvider> providerClass) {
        // Only the default provider is allowed, which prevents remote manipulation via the
        // "provider.admin.className" system property.
        return providerClass.equals(DefaultAdminProvider.class);
    }
