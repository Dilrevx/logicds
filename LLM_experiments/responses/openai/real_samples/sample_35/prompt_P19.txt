def update_user_balance(user, amount):
    """
    Safely updates a user's balance by a specified amount.
    This version fixes a logical vulnerability where an update could inadvertently drop a balance below zero.
    """
    # Validate that amount is numeric.
    if not isinstance(amount, (int, float)):
        raise ValueError("Amount must be a number.")
    
    # Compute the new balance.
    new_balance = user.balance + amount

    # Logical check: ensure that deducting funds does not result in a negative balance.
    # This prevents the vulnerability where an update could lead to an overdrawn account.
    if new_balance < 0:
        raise ValueError("Insufficient funds: update would result in a negative balance.")
    
    # Update and return the new balance.
    user.balance = new_balance
    return user.balance
