private AdminManager() {
    // Validate the admin provider to prevent an attacker from forcing the use of a malicious provider.
    String provider = ADMIN_PROVIDER.getValue();
    
    // Only allow the provider if it is one of the explicitly whitelisted safe implementations.
    if (!isAllowedAdminProvider(provider)) {
        // Log the suspicious attempt and fall back to a secure default provider.
        System.err.println("Unauthorized admin provider provided: " + provider + ". Falling back to safe default.");
        provider = "com.mycompany.providers.SafeAdminProvider";
    }
    
    // Initialize the admin provider with the validated value.
    initProvider(provider);
}

private boolean isAllowedAdminProvider(String provider) {
    // Define the whitelist of allowed admin provider implementations.
    return "com.mycompany.providers.SafeAdminProvider".equals(provider)
        || "com.mycompany.providers.AnotherAllowedProvider".equals(provider);
}
