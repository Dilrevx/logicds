private AdminManager() {
    // Load an admin provider securely.
    String provider = ADMIN_PROVIDER.getValue();
    if (provider == null || provider.trim().isEmpty()) {
        throw new IllegalStateException("Admin provider value is not defined.");
    }
    // Validate that the provider string is an expected safe provider.
    // Only allow explicitly approved providers to mitigate provider injection risks.
    if (!"com.example.SafeAdminProvider".equals(provider)) {
        throw new SecurityException("Unauthorized or unexpected admin provider specified.");
    }
    // Proceed with the initialization using the verified provider.
    initProvider(provider);
}
