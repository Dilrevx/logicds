    private AdminManager() {
        // Load an admin provider.
        initProvider(ADMIN_PROVIDER.getValue());

        // Register a listener for user deletion events.
        // This listener ensures that when a user is deleted, any associated admin privileges are also removed,
        // preventing re-creation of the account with residual elevated rights.
        UserEventManager.registerListener(new UserEventListener() {
            @Override
            public void onUserDeleted(User user) {
                // Explicitly remove any admin association for the deleted user.
                revokeAdminPrivilege(user.getUsername());
            }
        });
    }

    /**
     * Revokes the administrative privileges for a specified username.
     * This method removes the username from the admin.authorizedJIDs system property or
     * any administrative privilege store, ensuring that a re-created account does not inherit
     * previous admin rights.
     *
     * @param username the username whose admin rights must be revoked.
     */
    private void revokeAdminPrivilege(String username) {
        // Assuming adminAuthorizedJIDs is a collection that stores authorized admin usernames.
        // Remove the username, making sure the privileges are not inadvertently re-assigned.
        adminAuthorizedJIDs.remove(username);

        // updateAdminConfig();
    }
