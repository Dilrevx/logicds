private AdminManager() {
    // Load and validate the admin provider.
    String providerValue = ADMIN_PROVIDER.getValue();
    if (providerValue == null || providerValue.trim().isEmpty()) {
        throw new IllegalStateException("Admin provider must not be null or empty.");
    }
    
    // Validate against an approved whitelist of providers to prevent injection or misconfiguration.
    if (!isValidProvider(providerValue)) {
        throw new SecurityException("Admin provider is not valid: " + providerValue);
    }
    
    initProvider(providerValue);
}

/**
 * Validates that the provider is one of the allowed admin providers.
 * Adjust or extend this whitelist as necessary.
 */
private boolean isValidProvider(String provider) {
    // Example whitelist; replace "TrustedAdminProvider" with the actual allowed provider.
    return "TrustedAdminProvider".equals(provider);
}
