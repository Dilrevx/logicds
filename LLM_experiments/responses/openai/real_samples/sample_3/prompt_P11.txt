           inadvertently substitute the NULL cipher from the default provider. */
        if (ctx->cipher != NULL && ctx->cipher->nid == NID_undef) {
            ERR_raise(ERR_LIB_EVP, EVP_R_UNSUPPORTED_CIPHER);
            return 0;
        }
        if (ctx->cipher == ctx->fetched_cipher)
            ctx->cipher = NULL;

        ctx->fetched_cipher = NULL;
        goto legacy;
