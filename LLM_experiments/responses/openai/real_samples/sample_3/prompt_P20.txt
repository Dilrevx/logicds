                                    const EVP_CIPHER *cipher,
                                    ENGINE *impl, const unsigned char *key,
                                    const unsigned char *iv, int enc,
                                    const OSSL_PARAM params[])
{
    int n;
#if !defined(OPENSSL_NO_ENGINE) && !defined(FIPS_MODULE)
    ENGINE *tmpimpl = NULL;
#endif

    ctx->iv_len = -1;

    /*
     * enc == 1 means we are encrypting.
     * enc == 0 means we are decrypting.
     * enc == -1 means use the previously initialised value for encrypt/decrypt.
     */
    if (enc == -1) {
        enc = ctx->encrypt;
    } else {
        if (enc)
            enc = 1;
        ctx->encrypt = enc;
    }

    if (cipher == NULL && ctx->cipher == NULL) {
        ERR_raise(ERR_LIB_EVP, EVP_R_NO_CIPHER_SET);
        return 0;
    }

#if !defined(OPENSSL_NO_ENGINE) && !defined(FIPS_MODULE)
    /*
     * If the context already has an associated engine and a cipher, and if we are
     * reusing the same cipher (or no new cipher is provided), then we can skip
     * reselecting an engine.
     */
    if (ctx->engine && ctx->cipher &&
        (cipher == NULL || cipher->nid == ctx->cipher->nid))
        goto skip_to_init;

    /*
     * If a cipher is provided and no specific ENGINE implementation (impl) is
     * given, then ask if an ENGINE is registered for handling this cipher.
     */
    if (cipher != NULL && impl == NULL)
         tmpimpl = ENGINE_get_cipher_engine(cipher->nid);
#endif

     * The legacy branch (which may have been taking over if any ENGINE-related pointer was non-NULL)

     * Normal (core) implementation is invoked below.

skip_to_init:
    /* Optionally, adjust engine-related settings if needed. */
    ;
#endif
