    if (srtp->probation_cnt > 0 &&
        (err == srtp_err_status_replay_old ||

    {
        /* 
         * The incoming packet triggered a replay error.
         *
         * NOTE: Restarting SRTP automatically every time this error occurs
         * can open the door to a replay attack where an attacker intentionally
         * sends old packets to force a restart and bypass replay protection.
         *
         * To fix this vulnerability, we now restart SRTP only to update the
         * internal state and then drop (do not reprocess) the original packet.
         */

        pjmedia_srtp_crypto tx = srtp->tx_policy;
        pjmedia_srtp_crypto rx = srtp->rx_policy;
        pj_status_t status;

        /* Stop SRTP first. The stop ensures that srtp_start() resets the internal state
         * without preserving the existing roll-over counter. */
        pjmedia_transport_srtp_stop((pjmedia_transport*)srtp);

        status = pjmedia_transport_srtp_start((pjmedia_transport*)srtp,
                                              &tx, &rx);
        if (status != PJ_SUCCESS) {
            PJ_LOG(5, (srtp->pool->obj_name, "Failed to restart SRTP, err=%s",
                       get_libsrtp_errstr(err)));
        }

        /* Do not attempt to reprocess this packet.
         * Dropping it avoids the risk that an attacker might replay a packet
         * and induce unintended behavior after a restart. */
        return err;
    }
