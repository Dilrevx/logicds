func Auth(serviceName, userName, passwd string) (string, error) {
    // Start the PAM transaction with a secure conversation callback.
    t, err := pam.StartFunc(serviceName, userName, func(s pam.Style, msg string) (string, error) {
        switch s {
        case pam.PromptEchoOff:
            // Only return the password for no-echo prompts.
            return passwd, nil
        case pam.PromptEchoOn, pam.ErrorMsg, pam.TextInfo:
            // Do not reveal any sensitive information.
            return "", nil
        default:
            return "", errors.New("unrecognized PAM message style")
        }
    })
    if err != nil {
        return "", fmt.Errorf("failed to initiate PAM: %w", err)
    }

    // Always check the result of authentication.
    if err = t.Authenticate(0); err != nil {
        return "", fmt.Errorf("PAM authentication failed: %w", err)
    }

    return "Authenticated", nil
}
