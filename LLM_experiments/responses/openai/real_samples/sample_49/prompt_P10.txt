func Auth(serviceName, userName, passwd string) (string, error) {
	t, err := pam.StartFunc(serviceName, userName, func(s pam.Style, msg string) (string, error) {
		switch s {
		case pam.PromptEchoOff:
			return passwd, nil
		case pam.PromptEchoOn, pam.ErrorMsg, pam.TextInfo:
			return "", nil
		}
		return "", errors.New("Unrecognized PAM message style")
	})
	if err != nil {
		return "", err
	}

	if err = t.Authenticate(0); err != nil {
		return "", err
	}

	// PAM login names might suffer transformations in the PAM stack.
	// We should take whatever the PAM stack returns for it.
	authenticatedUser, err := t.GetItem(pam.User)
	if err != nil {
		return "", err
	}

	// Additional authorization check: verify that the authenticated user is allowed access.
	// This is to ensure that even if authentication via PAM succeeds, only authorized users can proceed.
	if !isAuthorized(authenticatedUser) {
		return "", errors.New("user is not authorized to access the system")
	}

	return authenticatedUser, nil
}

// isAuthorized performs a basic authorization check for the user.
// In a real-world scenario, this function should verify the user's existence in the
// application's database, check user account status, group memberships, roles, etc.
func isAuthorized(user string) bool {
	// Reject empty usernames.
	if user == "" {
		return false
	}

	// Example authorization logic: block user with a specific name or prefix.
	// Modify this as needed for your application's authorization requirements.
	if user == "blockedUser" || (len(user) >= 5 && user[:5] == "guest") {
		return false
	}

	return true
}
