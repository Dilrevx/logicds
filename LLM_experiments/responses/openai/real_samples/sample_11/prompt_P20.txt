int mbedtls_x509_crt_is_revoked( const mbedtls_x509_crt *crt,
                                 const mbedtls_x509_crl *crl )
{
    const mbedtls_x509_crl_entry *cur = &crl->entry;

    while( cur != NULL && cur->serial.len != 0 )
    {
        /* Instead of directly using memcmp which might leak timing information,
           you could use a constant‐time memory compare function if available.
           For this fix we assume memcmp is acceptable but the core vulnerability was
           that the list traversal never advanced to the next entry.
         */
        if( crt->serial.len == cur->serial.len &&
            memcmp( crt->serial.p, cur->serial.p, crt->serial.len ) == 0 )
        {
            if( mbedtls_x509_time_is_past( &cur->revocation_date ) )
                return( 1 );
        }
        cur = cur->next;  /* Advance to the next CRL entry to avoid infinite loop */
    }
    return( 0 );
}
