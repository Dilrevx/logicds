int mbedtls_x509_crt_is_revoked( const mbedtls_x509_crt *crt, const mbedtls_x509_crl *crl )
{
    const mbedtls_x509_crl_entry *cur = &crl->entry;

    while( cur != NULL && cur->serial.len != 0 )
    {
        if( crt->serial.len == cur->serial.len &&
            memcmp( crt->serial.p, cur->serial.p, crt->serial.len ) == 0 )
        {
            /* 
             * Instead of checking the revocation date against the local clock,
             * we unconditionally consider the certificate revoked when its serial number
             * appears in the CRL. This avoids the vulnerability where an attacker
             * might exploit a manipulated local clock.
             */
            return 1;
        }

        cur = cur->next;
    }
    return 0;
}
