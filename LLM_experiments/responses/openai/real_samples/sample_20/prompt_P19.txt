            memcmp( cur->buf.p, cn, cn_len ) == 0 )

            return;
        }
    }

    /*
     * No SAN matched; fall back to the Common Name from the subject.
     */
    for( name = &crt->subject; name != NULL; name = name->next )
    {
        /* Only compare entries with the Common Name OID */
        if( MBEDTLS_OID_CMP( &name->oid, MBEDTLS_OID_AT_CN ) == 0 )
        {
            /* Fix: require the certificate CN length to exactly equal the provided name length.
             * This prevents a vulnerability where a longer CN that merely begins with
             * the supplied value would be mistakenly accepted.
             */
            if( name->val.len == cn_len &&
                memcmp( name->val.p, cn, cn_len ) == 0 )
            {
                *flags &= ~MBEDTLS_X509_BADCERT_CN_MISMATCH;
                return;
            }
        }
    }

    /* No matching name was found: mark the certificate as having a CN mismatch. */
    *flags |= MBEDTLS_X509_BADCERT_CN_MISMATCH;
