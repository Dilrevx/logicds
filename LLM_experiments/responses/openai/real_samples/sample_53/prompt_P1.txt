    /* Normalize the URI to decode percent-encoded characters (including encoded forward slashes)
       before performing the security check. This prevents attackers from bypassing access restrictions
       by using encoded characters in the URI. */
    #define MAX_URI_LEN 2048
    char normalized_uri[MAX_URI_LEN];

    /* Assume mk_normalize_uri decodes the URI and returns 0 on success and -1 on failure */
    if (mk_normalize_uri(sr->uri, normalized_uri, MAX_URI_LEN) < 0) {
        PLUGIN_TRACE("[FD %i] Close connection, failed to normalize URL", cs->socket);
        mk_api->header_set_http_status(sr, MK_CLIENT_FORBIDDEN);
        return MK_PLUGIN_RET_CLOSE_CONX;
    }

    if (mk_security_check_url(normalized_uri) < 0) {
        PLUGIN_TRACE("[FD %i] Close connection, blocked URL", cs->socket);
        mk_api->header_set_http_status(sr, MK_CLIENT_FORBIDDEN);
        return MK_PLUGIN_RET_CLOSE_CONX;
    }
