int _mkp_stage_30(struct plugin *p,
                  struct client_session *cs,
                  struct session_request *sr)
{
    mk_ptr_t referer;
    (void)p;
    (void)cs;

    PLUGIN_TRACE("[FD %i] Mandril validating URL", cs->socket);

    /* 
     * Fix: Validate that the request and the referer header are both present.
     * Without these checks, a NULL pointer or an empty header might allow an attacker
     * to bypass the URL validation logic.
     */
    if (!sr || sr->header_referer.data == NULL || sr->header_referer.len == 0) {
        PLUGIN_TRACE("[FD %i] Missing or invalid Referer header", cs->socket);
        // Return error code indicating rejection; adjust the value as needed.
        return -1;
    }

    /* 
     * Copy the referer from the session request.
     */
    referer = sr->header_referer;

    /*
     * Validate that the referer URL has a proper scheme.
     * This check ensures the referer begins with "http://" or "https://".
     */
    if (referer.len < 7) {
        PLUGIN_TRACE("[FD %i] Referer too short", cs->socket);
        return -1;
    }

    if (strncmp(referer.data, "http://", 7) != 0 &&
        (referer.len < 8 || strncmp(referer.data, "https://", 8) != 0)) {
        PLUGIN_TRACE("[FD %i] Invalid Referer URL scheme", cs->socket);
        return -1;
    }

    /* 
     * Additional validations (e.g., domain whitelisting) can be added here.
     */

    // If all validations pass, proceed with further processing.
    return 0;
}
