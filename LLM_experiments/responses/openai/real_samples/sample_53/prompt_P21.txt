int _mkp_stage_30(struct plugin *p,
        struct client_session *cs,
        struct session_request *sr)
{
    mk_ptr_t referer;
    (void) p;
    (void) cs;

    PLUGIN_TRACE("[FD %i] Mandril validating URL", cs->socket);

    /* 
     * Decode the URI before performing the security check.
     * This prevents attackers from bypassing restrictions by encoding characters like '/'
     */
    char *decoded_uri = mk_api->uri_decode(sr->uri);
    if (decoded_uri == NULL) {
        PLUGIN_TRACE("[FD %i] Failed to decode URL, blocked access", cs->socket);
        mk_api->header_set_http_status(sr, MK_CLIENT_FORBIDDEN);
        return MK_PLUGIN_RET_CLOSE_CONX;
    }

    if (mk_security_check_url(decoded_uri) < 0) {
        PLUGIN_TRACE("[FD %i] Close connection, blocked URL (after decoding)", cs->socket);
        free(decoded_uri);
        mk_api->header_set_http_status(sr, MK_CLIENT_FORBIDDEN);
        return MK_PLUGIN_RET_CLOSE_CONX;
    }

    free(decoded_uri);

    /* Continue processing as the URL is now validated */
    return MK_PLUGIN_RET_CONTINUE;
}
