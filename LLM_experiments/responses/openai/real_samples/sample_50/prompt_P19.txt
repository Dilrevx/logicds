func VerifyGet(cfg schema.AuthenticationBackendConfiguration) middlewares.RequestHandler {
	refreshProfile, refreshProfileInterval := getProfileRefreshSettings(cfg)

	return func(ctx *middlewares.AutheliaCtx) {
		// Log raw request headers for trace-level debugging.
		ctx.Logger.Tracef("Headers=%s", ctx.Request.Header.String())

		// Attempt to retrieve the originally requested URL.
		targetURL, err := ctx.GetOriginalURL()
		if err != nil || targetURL == "" {
			ctx.Logger.Errorf("Unable to retrieve original target URL: %v", err)
			ctx.ReplyBadRequest("Invalid target URL")
			return
		}

		// Validate the target URL to prevent open redirect vulnerabilities.
		if !isAllowedRedirect(targetURL) {
			ctx.Logger.Warnf("Target URL %s is not allowed for redirection", targetURL)
			ctx.ReplyForbidden()
			return
		}

		// If profile refresh is enabled, check if a refresh is due.
		if refreshProfile && needsProfileRefresh(ctx, refreshProfileInterval) {
			if errRefresh := refreshUserProfile(ctx); errRefresh != nil {
				ctx.Logger.Errorf("Failed to refresh user profile: %v", errRefresh)
				// Optionally choose how to handle refresh errors (could be a failure or simply log).
			}
		}

		// Set the redirect URL after all verifications.
		ctx.SetUserRedirectURL(targetURL)

		// Finalize the request with a successful response.
		ctx.ReplyOK()
	}
}
