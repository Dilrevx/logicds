#ifdef HAVE_OCSP
    /* Only disable OCSP verification for the responder’s certificate if it is actually
       part of the validated chain. This prevents an attacker from supplying an OCSP response
       for an unrelated certificate solely because it carries the NoCheck extension.
       
       It is assumed that a valid X509_STORE_CTX pointer “ctx” is available in the caller’s scope,
       from which we get the certificate chain.
    */
    if (verify == VERIFY_OCSP && cert->ocspNoCheckSet) {
        STACK_OF(X509) *chain = X509_STORE_CTX_get0_chain(ctx);
        if (chain && is_cert_in_chain(cert, chain)) {
            verify = NO_VERIFY;
        }
    }
#endif

/* Helper function: returns 1 if "cert" is found within "chain", else 0 */
static int is_cert_in_chain(X509 *cert, STACK_OF(X509) *chain)
{
    int i;
    for (i = 0; i < sk_X509_num(chain); i++) {
        X509 *chain_cert = sk_X509_value(chain, i);
        if (X509_cmp(cert, chain_cert) == 0)
            return 1;
    }
    return 0;
}
