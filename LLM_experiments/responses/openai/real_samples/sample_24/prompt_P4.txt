    /*
     * Only disable OCSP verification (i.e. set verify = NO_VERIFY) for the responder’s
     * certificate if and only if that certificate is actually part of the expected chain.
     * This prevents an attacker from supplying an OCSP response for an unrelated certificate
     * simply because it carries the NoCheck extension.
     *
     * Note: The helper function certificate_in_chain() must verify that 'cert' is a member
     *       of the certificate chain in question.
     */
    if (verify == VERIFY_OCSP && cert->ocspNoCheckSet) {
        if (certificate_in_chain(cert)) {
            verify = NO_VERIFY;
        }
    }
