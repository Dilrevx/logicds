    /* Process OCSP responder certificate with controlled verification.
     * When handling the OCSP responder, distinguish between certificates that
     * carry the NoCheck extension and those that do not.
     *
     * If the certificate has the NoCheck flag, do not disable verification entirely;
     * instead, fall back to the standard certificate verification (VERIFY_CERT),
     * ensuring that even trusted responders are checked.
     * Otherwise, if the certificate lacks the NoCheck flag, continue to enforce
     * the full OCSP-based certificate verification.
     */
    if (verify == VERIFY_OCSP) {
        if (cert->ocspNoCheckSet)
            verify = VERIFY_CERT;  /* Use the regular certificate verification workflow */
        else
            verify = VERIFY_OCSP;  /* Enforce dedicated, rigorous OCSP responder verification */
    }
