    /* 
     * Only disable verification via the OCSP NoCheck extension if the responder’s certificate 
     * is actually part of the certificate chain under validation. This prevents an attacker 
     * from supplying an unrelated OCSP response with a NoCheck extension that would bypass 
     * proper verification.
     *
     * Note: The helper function is_cert_in_chain() must be implemented to verify that 'cert' 
     * is one of the certificates in the chain under validation.
     */
    if (verify == VERIFY_OCSP && cert->ocspNoCheckSet) {
        if (is_cert_in_chain(cert, chain)) {
            verify = NO_VERIFY;
        }
    }
