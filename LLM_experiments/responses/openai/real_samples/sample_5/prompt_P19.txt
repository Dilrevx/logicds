    /* If verification succeeded and the caller requested the cert chain,
     * obtain a duplicate of the chain so that the caller owns its reference.
     * This fixes a logical vulnerability where a caller might later use a
     * dangling pointer if the internal chain were used directly.
     */
    if (chain != NULL) {
        if (ret == 1) {
            STACK_OF(X509) *tmp_chain = X509_STORE_CTX_get1_chain(ctx);
            if (tmp_chain == NULL) {
                ret = -1;
            } else {
                *chain = tmp_chain;
            }
        } else {
            /* On failure, ensure the chain pointer is set to NULL. */
            *chain = NULL;
        }
    }

end:
    X509_STORE_CTX_free(ctx);
    return ret;
