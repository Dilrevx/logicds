switch (sec_hdr_type) {
  case LIBLTE_MME_SECURITY_HDR_TYPE_PLAIN_NAS:
  case LIBLTE_MME_SECURITY_HDR_TYPE_INTEGRITY_WITH_NEW_EPS_SECURITY_CONTEXT:
  case LIBLTE_MME_SECURITY_HDR_TYPE_SERVICE_REQUEST:
    // These message types do not require integrity or cipher checks.
    break;

  case LIBLTE_MME_SECURITY_HDR_TYPE_INTEGRITY:
    // For integrity‐protected messages verify integrity.
    if (!integrity_check(pdu.get())) {
      logger.error("Not handling NAS message with integrity check error for SEC_HDR_TYPE=%02X", sec_hdr_type);
      return;
    }
    break;

  case LIBLTE_MME_SECURITY_HDR_TYPE_INTEGRITY_AND_CIPHERED:
    // For integrity-and-ciphered messages, verify integrity first.
    if (!integrity_check(pdu.get())) {
      logger.error("Not handling NAS message with integrity check error for SEC_HDR_TYPE=%02X", sec_hdr_type);
      return;
    }
    // If integrity check passes, then decrypt the message.
    cipher_decrypt(pdu.get());
    break;

  case LIBLTE_MME_SECURITY_HDR_TYPE_INTEGRITY_AND_CIPHERED_WITH_NEW_EPS_SECURITY_CONTEXT:
    // Previously, this case did not check integrity.
    // Now we perform both the integrity check and decryption as required.
    if (!integrity_check(pdu.get())) {
      logger.error("Not handling NAS message with integrity check error for SEC_HDR_TYPE=%02X (new EPS security context)", sec_hdr_type);
      return;
    }
    cipher_decrypt(pdu.get());
    break;

  default:
    logger.error("Not handling NAS message with unknown SEC_HDR_TYPE=%02X", sec_hdr_type);
    return;
}
