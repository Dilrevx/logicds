    /* Check that the signature OIDs match, and that the algorithm
     * parameters are either both present as a correctly‐encoded ASN.1 NULL.
     *
     * In previous versions a zero-length parameters field (an “empty” array)
     * was treated as equivalent to an ASN.1 NULL. That is incorrect: if any
     * parameters are provided they must be an explicit ASN.1 NULL (i.e. 2 bytes: 0x05 0x00).
     *
     * Thus, we require that:
     * - The signature OIDs are identical.
     * - The parameters buffers have equal length.
     * - And, if parameters are present, they must be exactly 2 bytes long and
     *   equal to { 0x05, 0x00 }.
     *
     * Certificates that omit parameters (i.e. zero-length fields) or that have
     * parameters that do not exactly match a DER-encoded NULL are rejected.
     */
    if( crt->sig_oid.len != sig_oid2.len ||
        memcmp( crt->sig_oid.p, sig_oid2.p, crt->sig_oid.len ) != 0 ||
        sig_params1.len != sig_params2.len ||
        /* Fail if no parameters are present (zero-length) */
        sig_params1.len == 0 ||
        /* Or if they are present but not exactly the DER encoding of NULL */
        sig_params1.len != 2 ||
        sig_params1.p[0] != 0x05 || sig_params1.p[1] != 0x00 )
    {
        mbedtls_x509_crt_free( crt );
        return( MBEDTLS_ERR_X509_SIG_MISMATCH );
    }
</repair>
