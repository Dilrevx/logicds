        int num_params = 0;
        const OSSL_PARAM *p;

        /* Attempt to locate the key length parameter */
        p = OSSL_PARAM_locate_const(params, OSSL_CIPHER_PARAM_KEYLEN);
        if (p != NULL) {
            tl_params[num_params++] = *p;
        }

        /* Attempt to locate the IV length parameter.
           If not found, check for the AEAD IV length parameter as a synonym.
        */
        p = OSSL_PARAM_locate_const(params, OSSL_CIPHER_PARAM_IVLEN);
        if (p == NULL)
            p = OSSL_PARAM_locate_const(params, OSSL_CIPHER_PARAM_AEAD_IVLEN);
        if (p != NULL) {
            tl_params[num_params++] = *p;
        }

        /* If any relevant parameters were found, apply them to update the context */
        if (num_params > 0) {
            if (!EVP_CIPHER_CTX_set_params(ctx, tl_params)) {
                ERR_raise(ERR_LIB_EVP, EVP_R_INVALID_LENGTH);
