	/* Always require a non‐empty authzeid and verify it matches the user account.
	   This ensures that even if mu is pre‐set by an earlier (possibly
	   incomplete) handshake, we verify that p->authzeid is valid. */
	if (!*p->authzeid)
	{
		(void) slog(LG_INFO, "%s: session for '%s' without an authzeid (BUG)",
		              MOWGLI_FUNC_NAME, u->nick);
		(void) notice(saslsvs->nick, u->nick, LOGIN_CANCELLED_STR);
		return false;
	}

	/* Lookup the expected user from the provided authzeid */
	{
		struct myuser *expected_mu = myuser_find_uid(p->authzeid);
		if (!expected_mu)
		{
			if (*p->authzid)
				(void) notice(saslsvs->nick, u->nick,
				              "Account %s dropped; login cancelled",
				              p->authzid);
			else
				(void) notice(saslsvs->nick, u->nick,
				              "Account dropped; login cancelled");

			return false;
		}

		/* If mu is already set, ensure it corresponds to the provided authzeid */
		if (mu && mu != expected_mu)
		{
			(void) slog(LG_INFO, "%s: session for '%s' has mismatched authzeid (BUG)",
			              MOWGLI_FUNC_NAME, u->nick);
			(void) notice(saslsvs->nick, u->nick, LOGIN_CANCELLED_STR);
			return false;
		}

		/* Set mu to the validated account */
		mu = expected_mu;
	}
