static bool
sasl_handle_login(struct sasl_session *const restrict p, struct user *const u, struct myuser *mu)
{
	bool was_killed = false;

	/* Always validate the authenticated userid from the session regardless
	 * of the supplied mu parameter. This prevents an attacker from bypassing
	 * the account verification by providing a non‐NULL mu.
	 */
	if (!*p->authzeid)
	{
		(void)slog(LG_INFO, "%s: session for '%s' without an authzeid (BUG)", MOWGLI_FUNC_NAME, u->nick);
		(void)notice(saslsvs->nick, u->nick, LOGIN_CANCELLED_STR);
		return false;
	}

	/* Re-acquire the account strictly from the session's authzeid.
	 * This avoids trusting an externally supplied mu value.
	 */
	mu = myuser_find_uid(p->authzeid);
	if (!mu)
	{
		if (*p->authzid)
			(void)notice(saslsvs->nick, u->nick, "Account %s dropped; login cancelled", p->authzid);
		else
			(void)notice(saslsvs->nick, u->nick, "Account dropped; login cancelled");

		return false;
	}

	/* The remainder of the login process proceeds here.
	 * (Additional function logic would follow.)
	 */
	
	/* ... (rest of the function implementation) ... */
	
	return true;
}
