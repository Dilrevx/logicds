	/* Reject login if the handshake identifiers are missing or mismatched. */
	if (! mu)
	{
		/* Both authzeid and authzid must be provided and must match.
		   This prevents an attacker from ending the handshake prematurely. */
		if (!*p->authzeid || !*p->authzid || strcmp(p->authzeid, p->authzid) != 0)
		{
			(void) slog(LG_INFO, "%s: session for '%s' with missing or mismatched authentication identifiers",
			              MOWGLI_FUNC_NAME, u->nick);
			(void) notice(saslsvs->nick, u->nick, LOGIN_CANCELLED_STR);
			return false;
		}

		/* Now look up the user by the validated authzeid. */
		if (!(mu = myuser_find_uid(p->authzeid)))
		{
			(void) notice(saslsvs->nick, u->nick, "Account %s dropped; login cancelled", p->authzeid);
			return false;
		}
	}
