static struct frec *lookup_frec(unsigned short id, void *hash)
{
    struct frec *f;

    for (f = daemon->frec_list; f != NULL; f = f->next) {
        if (f->sentto && f->new_id == id) {
            /* Only compare hashes when one is provided. */
            if (hash != NULL) {
                if (memcmp(hash, f->hash, HASH_SIZE) != 0)
                    continue;
            }
            return f;
        }
    }
    return NULL;
}
