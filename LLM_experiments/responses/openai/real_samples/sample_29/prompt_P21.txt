static struct frec *lookup_frec(unsigned short id, void *hash, struct sockaddr *reply_dst, socklen_t reply_dst_len)
{
    struct frec *f;

    /*
     * Iterate over the list of pending forwarded DNS queries.
     * In addition to checking that the reply’s query ID and optional hash match,
     * we now also verify that the reply arrived on the expected destination address/port.
     * This additional check is required by RFC5452 to ensure that a forged reply cannot be accepted.
     */
    for (f = daemon->frec_list; f; f = f->next) {
        if (f->sentto && f->new_id == id &&
            (!hash || memcmp(hash, f->hash, HASH_SIZE) == 0) &&
            (f->dest_len == reply_dst_len) &&
            (memcmp(&f->dest, reply_dst, reply_dst_len) == 0))
        {
            return f;
        }
    }
    return NULL;
}
