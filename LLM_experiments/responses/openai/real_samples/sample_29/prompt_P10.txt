static struct frec *lookup_frec(unsigned short id, void *hash, struct sockaddr *from)
{
    struct frec *f;

    for(f = daemon->frec_list; f; f = f->next) {
        /* Ensure we are only matching entries for which we have a valid destination */
        if (!f->sentto)
            continue;

        /* Check all attributes as specified by RFC5452: query ID, additional hash and
           importantly the expected reply destination (address and port). */
        if (f->new_id != id)
            continue;
        if (hash && memcmp(hash, f->hash, HASH_SIZE) != 0)
            continue;
        if (from && memcmp(from, f->sentto, sizeof(struct sockaddr)) != 0)
            continue;

        return f;
    }
      
    return NULL;
}
