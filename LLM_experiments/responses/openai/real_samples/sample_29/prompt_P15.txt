    /* Compare the forwarded query with the incoming reply info.
     * In addition to verifying the ID and optional query hash, we now
     * validate that the reply arrived on the correct network socket.
     *
     * We assume that the caller supplies the file descriptor (reply_fd)
     * on which the reply was received, and that the forwarded query object
     * (f) contains information about the socket used to send the query.
     *
     * The check is performed in three parts:
     * 1. If the query was sent via a bound upstream socket (f->bound_fd >= 0),
     *    then the reply must be received on that same socket.
     * 2. Otherwise, if the query was sent over IPv4 (f->sock_family == AF_INET),
     *    then the reply must originate on the IPv4 socket (f->fd_ipv4).
     * 3. Otherwise, for IPv6 (f->sock_family == AF_INET6),
     *    the reply must be on the IPv6 socket (f->fd_ipv6).
     *
     * If none of these conditions match, the reply is rejected.
     */
    if (f->sentto && f->new_id == id &&
        (!hash || memcmp(hash, f->hash, HASH_SIZE) == 0))
    {
        if (f->bound_fd >= 0) {
            /* If a specific upstream socket was bound, ensure it is the one that replied. */
            if (f->bound_fd != reply_fd)
                return NULL;
        }
        else if (f->sock_family == AF_INET) {
            /* IPv4: verify reply comes through the same IPv4 socket */
            if (f->fd_ipv4 != reply_fd)
                return NULL;
        }
        else if (f->sock_family == AF_INET6) {
            /* IPv6: verify reply comes through the same IPv6 socket */
            if (f->fd_ipv6 != reply_fd)
                return NULL;
        }
        else {
            /* Unrecognized address family; reject the reply */
            return NULL;
        }
        return f;
    }
    /* No matching forwarded query found */
    return NULL;
