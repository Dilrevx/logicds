		// Parse and validate the JWT token using the provided secret.
		parsedToken, err := jwt.Parse(token, func(t *jwt.Token) (interface{}, error) {
			// Ensure that the token's signing method is HMAC and not something else.
			if _, ok := t.Method.(*jwt.SigningMethodHMAC); !ok {
				return nil, fmt.Errorf("unexpected signing method: %v", t.Header["alg"])
			}
			return []byte(secret), nil
		})
		if err != nil || !parsedToken.Valid {
			return echo.NewHTTPError(http.StatusUnauthorized, "Invalid access token")
		}

		// Optionally, propagate claims to the context for downstream access.
		c.Set("user", parsedToken.Claims)
		return next(c)
	}
