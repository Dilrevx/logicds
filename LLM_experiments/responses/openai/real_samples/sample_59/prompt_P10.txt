func (cs *CaptchaService) VerifyCaptcha(ctx context.Context, key, captcha string) (isCorrect bool, err error) {
	realCaptcha, err := cs.captchaRepo.GetCaptcha(ctx, key)
	if err != nil {
		return false, err
	}

	// Remove the captcha regardless of verification result to prevent replay attacks.
	defer func() {
		// Log or handle the error from deletion only if no previous error occurred.
		if delErr := cs.captchaRepo.DeleteCaptcha(ctx, key); delErr != nil && err == nil {
			err = delErr
		}
	}()

	if strings.TrimSpace(captcha) != realCaptcha {
		return false, nil
	}

	return true, nil
}
