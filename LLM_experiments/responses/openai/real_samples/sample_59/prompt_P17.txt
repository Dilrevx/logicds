func (cs *CaptchaService) VerifyCaptcha(ctx context.Context, key, captcha string) (isCorrect bool, err error) {
	realCaptcha, err := cs.captchaRepo.GetCaptcha(ctx, key)
	if err != nil {
		return false, err
	}

	// Use constant time comparison to mitigate timing attacks.
	// If the lengths differ, the constant time compare would still run, but we combine a length check as well.
	if len(realCaptcha) != len(captcha) {
		// Even if lengths differ, perform a dummy constant time comparison
		// to keep the timing consistent.
		_ = subtle.ConstantTimeCompare([]byte(realCaptcha), []byte(captcha))
		return false, nil
	}

	isCorrect = (subtle.ConstantTimeCompare([]byte(realCaptcha), []byte(captcha)) == 1)

	// Optionally, delete the captcha once verified to prevent replay attacks.
	// For example:
	// err = cs.captchaRepo.DeleteCaptcha(ctx, key)
	// if err != nil {
	//	   return false, err
	// }

	return isCorrect, nil
}
