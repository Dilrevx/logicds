func (cs *CaptchaService) VerifyCaptcha(ctx context.Context, key, captcha string) (isCorrect bool, err error) {
	// Retrieve the stored captcha value from the repository
	realCaptcha, err := cs.captchaRepo.GetCaptcha(ctx, key)
	if err != nil {
		return false, err
	}

	// Delete the captcha after retrieval to prevent replay attacks,
	// regardless of whether the provided captcha is correct.
	// Note: If deletion fails, log the error and continue.
	if delErr := cs.captchaRepo.DeleteCaptcha(ctx, key); delErr != nil {
		// In production code consider using a logger instead of printing.
		// log.Printf("warning: failed to delete captcha for key %s: %v", key, delErr)
	}

	// Ensure both captcha strings have the same length.
	// This is important for the constant time comparison to work correctly.
	if len(realCaptcha) != len(captcha) {
		return false, nil
	}

	// Use constant time comparison to avoid timing attacks.
	if subtle.ConstantTimeCompare([]byte(realCaptcha), []byte(captcha)) == 1 {
		return true, nil
	}

	return false, nil
}
