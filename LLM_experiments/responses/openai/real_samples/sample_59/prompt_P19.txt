func (cs *CaptchaService) VerifyCaptcha(ctx context.Context, key, captcha string) (isCorrect bool, err error) {
	realCaptcha, err := cs.captchaRepo.GetCaptcha(ctx, key)
	if err != nil {
		return false, fmt.Errorf("failed to retrieve captcha for key %s: %w", key, err)
	}

	// If the returned captcha is empty then it's an error (e.g. expired or not found)
	if realCaptcha == "" {
		return false, fmt.Errorf("captcha for key %s not found or expired", key)
	}

	// Use a constant time comparison to mitigate timing attacks.
	if subtle.ConstantTimeCompare([]byte(captcha), []byte(realCaptcha)) == 1 {
		isCorrect = true

		// Optionally, remove the captcha from storage once it has been successfully verified,
		// preventing replay attacks.
		if err := cs.captchaRepo.DeleteCaptcha(ctx, key); err != nil {
			// Log the error if necessary but the captcha correct status is already true.
			// Depending on the design, you may choose to return an error instead.
			// Here, we simply log the deletion error.
			// log.Printf("failed to delete captcha with key %s: %v", key, err)
		}
	} else {
		isCorrect = false
	}

	return isCorrect, nil
}
