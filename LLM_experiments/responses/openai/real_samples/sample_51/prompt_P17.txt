Below is one recommended repair. In the original code the check

  if (NULL != r->headers_in.user.data) { … return NGX_OK; }

allowed an attacker who injected a "user" header to bypass SPNEGO. One fix is to remove that “shortcut” check so that authentication is never bypassed merely because a header already exists. For example, the change below completely removes the vulnerable check. (Note that the remainder of the SPNEGO negotiation logic would then follow.)

<repair>
    spnego_debug3("SSO auth handling IN: token.len=%d, head=%d, ret=%d",
            ctx->token.len, ctx->head, ctx->ret);

    if (ctx->token.len && ctx->head) {
        spnego_debug1("Found token and head, returning %d", ctx->ret);
        return ctx->ret;
    }

    /* Removed vulnerable check for r->headers_in.user.
     * Do not bypass authentication merely because a user header is present. */

    spnego_debug0("Begin auth");

    if (alcf->allow_basic) {
        spnego_debug0("Detect basic auth");
        ret = ngx_http_auth_basic_user(r);
        if (NGX_OK == ret) {
            spnego_debug0("Basic auth credentials supplied by client");
            /* If basic auth is enabled and basic credentials are supplied,
             * do not fall through to real SPNEGO */
            return ret;
        }
    }

    /* Additional SPNEGO authentication logic here */

    return ret;
}
</repair>
