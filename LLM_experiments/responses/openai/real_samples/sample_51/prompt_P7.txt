/* Helper function to verify that the username is syntactically valid.
   For example, we reject empty names and names containing spaces (or other disallowed characters). */
static ngx_int_t
spnego_username_valid(ngx_str_t *username)
{
    if (username == NULL || username->len == 0) {
        return 0;
    }
    /* Example check: disallow any username that contains a space.
       You can extend this to other invalid characters as required. */
    u_char *p = username->data, *end = username->data + username->len;
    for (; p < end; p++) {
        if (*p == ' ') {
            return 0;
        }
    }
    return 1;
}

ngx_int_t
ngx_http_auth_spnego_handler(ngx_http_request_t *r, spnego_ctx_t *ctx, spnego_alcf_t *alcf)
{
    if (NGX_DECLINED == ngx_http_auth_spnego_basic(r, ctx, alcf)) {
        spnego_debug0("Basic auth failed");

        /* New check: if the username is provided, ensure that it is syntactically valid.
           This prevents malformed usernames from bypassing authentication. */
        if (ctx->username != NULL) {
            if (!spnego_username_valid(ctx->username)) {
                spnego_debug0("Malformed username detected - rejecting request");
                return (ctx->ret = NGX_HTTP_UNAUTHORIZED);
            }
        }

        if (NGX_ERROR == ngx_http_auth_spnego_headers_basic_only(r, ctx, alcf)) {
            spnego_debug0("Error setting headers");
            return (ctx->ret = NGX_HTTP_INTERNAL_SERVER_ERROR);
        }
        return (ctx->ret = NGX_HTTP_UNAUTHORIZED);
    }

    /* Continue handling the request if basic authentication succeeded... */
    /* ... (rest of the processing code) ... */
    return NGX_OK;  /* This return may differ based on further logic */
}
