int nas_5g::handle_security_mode_command(security_mode_command_t& security_mode_command,
                                         srsran::unique_byte_buffer_t pdu)
{
  logger.info("Handling Security Mode Command");

  // Enforce that insecure (plain) algorithms are not accepted.
  // If either the ciphering or the integrity protection algorithm is “plain” (e.g., nas_sm_cmd_NIA0_plain),
  // then reject the command to ensure that the NAS layer security activation is not bypassed.
  if ((security_mode_command.selected_nas_security_algorithms.ciphering_algorithm.value == nas_sm_cmd_NIA0_plain) ||
      (security_mode_command.selected_nas_security_algorithms.integrity_protection_algorithm.value == nas_sm_cmd_NIA0_plain))
  {
    logger.error("Insecure algorithm selection detected: plain algorithm not allowed. Rejecting Security Mode Command.");
    send_security_mode_reject(cause_5gmm_t::cause_5gmm_type_::insecure_algorithm);
    return SRSRAN_ERROR;
  }

  // Set the approved security algorithms after validation.
  ctxt_base.cipher_algo = static_cast<CIPHERING_ALGORITHM_ID_ENUM>(security_mode_command.selected_nas_security_algorithms.ciphering_algorithm.value);
  ctxt_base.integ_algo  = static_cast<INTEGRITY_ALGORITHM_ID_ENUM>(security_mode_command.selected_nas_security_algorithms.integrity_protection_algorithm.value);

  // Verify the replayed UE security capabilities.
  if (!check_replayed_ue_security_capabilities(security_mode_command.replayed_ue_security_capabilities)) {
    logger.warning("Sending Security Mode Reject due to security capabilities mismatch");
    send_security_mode_reject(cause_5gmm_t::cause_5gmm_type_::ue_security_capabilities_mismatch);
    return SRSRAN_ERROR;
  }

  // If processing the initial security mode command, reset the related counters.
  if (initial_sec_command) {
    set_k_gnb_count(0);
    ctxt_base.tx_count = 0;
    initial_sec_command = false;
  }

  // Continue with additional security mode command handling...
  // (Insert further processing as needed)

  return SRSRAN_SUCCESS;
}
