int nas_5g::handle_security_mode_command(security_mode_command_t& security_mode_command,
                                         srsran::unique_byte_buffer_t pdu)
{
    logger.info("Handling Security Mode Command");

    // Validate that the provided ciphering algorithm is supported by the UE capabilities.
    if (!is_supported_cipher_algorithm(security_mode_command.selected_nas_security_algorithms.ciphering_algorithm.value)) {
        logger.warning("Invalid or unsupported ciphering algorithm received");
        send_security_mode_reject(cause_5gmm_t::cause_5gmm_type_::ciphering_algorithm_not_supported);
        return SRSRAN_ERROR;
    }

    // Validate that the provided integrity protection algorithm is supported.
    if (!is_supported_integ_algorithm(security_mode_command.selected_nas_security_algorithms.integrity_protection_algorithm.value)) {
        logger.warning("Invalid or unsupported integrity protection algorithm received");
        send_security_mode_reject(cause_5gmm_t::cause_5gmm_type_::integrity_protection_algorithm_not_supported);
        return SRSRAN_ERROR;
    }

    // Only after successful validation assign the algorithms.
    ctxt_base.cipher_algo = static_cast<CIPHERING_ALGORITHM_ID_ENUM>(
        security_mode_command.selected_nas_security_algorithms.ciphering_algorithm.value);
    ctxt_base.integ_algo  = static_cast<INTEGRITY_ALGORITHM_ID_ENUM>(
        security_mode_command.selected_nas_security_algorithms.integrity_protection_algorithm.value);

    // Check replayed UE security capabilities and reject if they do not match expected values.
    if (!check_replayed_ue_security_capabilities(security_mode_command.replayed_ue_security_capabilities)) {
        logger.warning("Sending Security Mode Reject due to security capabilities mismatch");
        send_security_mode_reject(cause_5gmm_t::cause_5gmm_type_::ue_security_capabilities_mismatch);
        return SRSRAN_ERROR;
    }

    // Reset counters if this is the initial security mode command.
    if (initial_sec_command) {
        set_k_gnb_count(0);
        ctxt_base.tx_count = 0;
        initial_sec_command = false;
    }

    // Continue with additional processing of the Security Mode Command as required.
    // ...

    return SRSRAN_SUCCESS;
}
