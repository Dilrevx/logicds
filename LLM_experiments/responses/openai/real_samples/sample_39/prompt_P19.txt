int nas_5g::handle_security_mode_command(security_mode_command_t &security_mode_command,
                                          srsran::unique_byte_buffer_t pdu)
{
    logger.info("Handling Security Mode Command");

    // Verify that the PDU has at least the minimum required length.
    const size_t MIN_PDU_LENGTH = 10; // adjust based on protocol specification
    if (pdu.size() < MIN_PDU_LENGTH)
    {
        logger.error("Security Mode Command PDU is too short.");
        return -1;  // return an error code indicating PDU is malformed
    }

    // Parse the security_mode_command from the buffer.
    // This function should safely extract and validate fields from the PDU.
    try
    {
        // Example parsing function that fills in security_mode_command.
        // It must not perform any unsafe memory copying operations.
        parse_security_mode_command(security_mode_command, pdu);
    }
    catch (const std::exception &ex)
    {
        logger.error(std::string("Failed to parse Security Mode Command: ") + ex.what());
        return -2;  // error during parsing the command
    }

    // Validate the MAC or integrity check value on the message to ensure that
    // the command is genuine and untampered.
    if (!verify_security_mode_mac(security_mode_command, pdu))
    {
        logger.error("Security Mode Command MAC verification failed.");
        return -3;  // error code for MAC verification failure
    }

    // Check that the cipher suite requested is supported.
    if (!is_supported_cipher_suite(security_mode_command.cipher_suite))
    {
        logger.error("Unsupported cipher suite in Security Mode Command.");
        return -4;  // error code for unsupported cipher suite
    }

    // If needed, perform further validations on security parameters.
    if (!validate_security_parameters(security_mode_command))
    {
        logger.error("Security parameters validation failed for Security Mode Command.");
        return -5;  // error code for invalid security parameters
    }

    // Update the security context of the session. This involves setting
    // the negotiated cipher suite, keys, and updating the security state.
    if (!update_security_context(security_mode_command))
    {
        logger.error("Failed to update security context based on Security Mode Command.");
        return -6;  // error code indicating failure to update context
    }

    logger.info("Security Mode Command processed successfully.");
    return 0;
}
