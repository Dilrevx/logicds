     * Open the temporary directory with extra restrictions to avoid symlink attacks.

     * possible device special file substitution.
     */
    lo->proc_self_fd = open(tmpdir, O_PATH | O_DIRECTORY | O_NOFOLLOW);
    if (lo->proc_self_fd == -1) {
        fuse_log(FUSE_LOG_ERR, "open(%s, O_PATH|O_DIRECTORY|O_NOFOLLOW): %m\n", tmpdir);
        exit(1);

        struct stat st;
        if (fstat(lo->proc_self_fd, &st) < 0) {
