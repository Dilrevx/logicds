     * Instead of creating a temporary directory and performing a bind mount—which
     * introduces a risk of host privilege escalation—we directly obtain a file
     * descriptor for /proc/self/fd.  This avoids the opportunity for a privileged
     * guest user to tamper with a temporary mountpoint.
     */
    lo->proc_self_fd = open("/proc/self/fd", O_PATH | O_CLOEXEC);
    if (lo->proc_self_fd == -1) {
        fuse_log(FUSE_LOG_ERR, "open(/proc/self/fd, O_PATH|O_CLOEXEC): %m\n");
