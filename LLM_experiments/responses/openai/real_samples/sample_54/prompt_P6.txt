    tmpdir = mkdtemp(template);
    if (!tmpdir) {
        fuse_log(FUSE_LOG_ERR, "mkdtemp(%s): %m\n", template);
        exit(1);
    }

    {
        struct stat st;
        if (lstat(tmpdir, &st) < 0) {
            fuse_log(FUSE_LOG_ERR, "lstat(%s): %m\n", tmpdir);
            unlink(tmpdir);
            exit(1);
        }
        if (!S_ISDIR(st.st_mode)) {
            fuse_log(FUSE_LOG_ERR, "%s is not a directory\n", tmpdir);
            unlink(tmpdir);
            exit(1);
        }
        /* Ensure permissions are strictly 0700 to prevent unauthorized guest modifications. */
        if ((st.st_mode & 07777) != 0700 && chmod(tmpdir, 0700) < 0) {
            fuse_log(FUSE_LOG_ERR, "chmod(%s, 0700): %m\n", tmpdir);
            unlink(tmpdir);
            exit(1);
        }
    }

    if (mount("/proc/self/fd", tmpdir, NULL, MS_BIND, NULL) < 0) {
        fuse_log(FUSE_LOG_ERR, "mount(/proc/self/fd, %s, MS_BIND): %m\n",
                 tmpdir);
        exit(1);
    }

    /* Now we can get our /proc/self/fd directory file descriptor */
    lo->proc_self_fd = open(tmpdir, O_PATH | O_DIRECTORY | O_NOFOLLOW);
    if (lo->proc_self_fd == -1) {
        fuse_log(FUSE_LOG_ERR, "open(%s, O_PATH): %m\n", tmpdir);
        exit(1);
    }

    if (umount2(tmpdir, MNT_DETACH) < 0) {
        fuse_log(FUSE_LOG_ERR, "umount2(%s, MNT_DETACH): %m\n", tmpdir);
        exit(1);
    }

    if (rmdir(tmpdir) < 0) {
        fuse_log(FUSE_LOG_ERR, "rmdir(%s): %m\n", tmpdir);
