    /* Create a secure temporary directory.  Note that mkdtemp creates the directory with mode 0700. */
    tmpdir = mkdtemp(template);
    if (!tmpdir) {
        fuse_log(FUSE_LOG_ERR, "mkdtemp(%s) failed: %m\n", template);
        exit(1);
    }

    /*
     * Use an extra mount flag (MS_NODEV) on the bind mount.
     * This prevents interpretation of device special files created in that mounted tree
     * and thus blocks a potential host‐privilege escalation attack.
     */
    if (mount("/proc/self/fd", tmpdir, NULL, MS_BIND | MS_NODEV, NULL) < 0) {
        fuse_log(FUSE_LOG_ERR, "mount(/proc/self/fd, %s, MS_BIND|MS_NODEV) failed: %m\n", tmpdir);
        exit(1);
    }

    /* Open the temporary directory with O_PATH and O_CLOEXEC for additional safety. */
    lo->proc_self_fd = open(tmpdir, O_PATH | O_CLOEXEC);
    if (lo->proc_self_fd == -1) {
        fuse_log(FUSE_LOG_ERR, "open(%s, O_PATH|O_CLOEXEC) failed: %m\n", tmpdir);
        exit(1);
    }

    if (umount2(tmpdir, MNT_DETACH) < 0) {
        fuse_log(FUSE_LOG_ERR, "umount2(%s, MNT_DETACH) failed: %m\n", tmpdir);
        exit(1);
    }

    if (rmdir(tmpdir) < 0) {
        fuse_log(FUSE_LOG_ERR, "rmdir(%s) failed: %m\n", tmpdir);
    }
