func (am *AuthUserMiddleware) AdminAuth() gin.HandlerFunc {
	return func(ctx *gin.Context) {
		// Extract and validate the token from the request header or query
		token := ExtractToken(ctx)
		if len(token) == 0 {
			handler.HandleResponse(ctx, errors.Unauthorized(reason.UnauthorizedError), nil)
			ctx.Abort()
			return
		}

		// Retrieve admin user info based on the token
		userInfo, err := am.authService.GetAdminUserCacheInfo(ctx, token)
		if err != nil || userInfo == nil {
			handler.HandleResponse(ctx, errors.Unauthorized(reason.UnauthorizedError), nil)
			ctx.Abort()
			return
		}

		// Optional: Verify that the user actually has admin privileges.
		// This check is important if GetAdminUserCacheInfo could return non-admin data.
		if !userInfo.IsAdmin {
			handler.HandleResponse(ctx, errors.Forbidden("Access denied: admin privileges required."), nil)
			ctx.Abort()
			return
		}

		// Store the user info in the context, so later handlers can access it
		ctx.Set("userInfo", userInfo)
		ctx.Next()
	}
}
