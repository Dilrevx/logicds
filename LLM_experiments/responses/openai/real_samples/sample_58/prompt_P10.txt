func (am *AuthUserMiddleware) AdminAuth() gin.HandlerFunc {
	return func(ctx *gin.Context) {
		// Extract the token from the request header or query parameters.
		token := ExtractToken(ctx)
		if len(token) == 0 {
			handler.HandleResponse(ctx, errors.Unauthorized(reason.UnauthorizedError), nil)
			ctx.Abort()
			return
		}

		// Retrieve admin user info from cache/service based on the token.
		userInfo, err := am.authService.GetAdminUserCacheInfo(ctx, token)
		if err != nil || userInfo == nil {
			handler.HandleResponse(ctx, errors.Forbidden(reason.UnauthorizedError), nil)
			ctx.Abort()
			return
		}

		// Ensure that the user is active and truly has admin privileges.
		// Here we assume that userInfo.Role is set and entity.AdminRole is the constant for admin.
		if userInfo.UserStatus == entity.UserStatusDeleted || userInfo.Role != entity.AdminRole {
			handler.HandleResponse(ctx, errors.Unauthorized(reason.UnauthorizedError), nil)
			ctx.Abort()
			return
		}

		// Save the user info in the context for downstream handlers.
		ctx.Set(ctxUUIDKey, userInfo)
		ctx.Next()
	}
}
