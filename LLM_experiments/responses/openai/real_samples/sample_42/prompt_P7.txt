if (sec_hdr_type == LIBLTE_MME_SECURITY_HDR_TYPE_PLAIN_NAS) {
    switch (msg_type) {
        // Allow these messages in plaintext unconditionally.
        case LIBLTE_MME_MSG_TYPE_EMM_INFORMATION:
        case LIBLTE_MME_MSG_TYPE_EMM_STATUS:
        case LIBLTE_MME_MSG_TYPE_DETACH_REQUEST:
            break;

        // For Attach Reject / Service Reject, only allow if emm_cause equals 25.
        case LIBLTE_MME_MSG_TYPE_ATTACH_REJECT:
        case LIBLTE_MME_MSG_TYPE_SERVICE_REJECT:
            if (emm_cause != 25) {
                logger.error("Rejecting unprotected NAS message MSG_TYPE=%02X with SEC_HDR_TYPE=%02X and invalid emm_cause=%d",
                             msg_type,
                             sec_hdr_type,
                             emm_cause);
                return;
            }
            break;

        // Reject any other NAS message type arriving without integrity protection.
        default:
            logger.error("Not handling NAS message MSG_TYPE=%02X with SEC_HDR_TYPE=%02X without integrity protection!",
                         msg_type,
                         sec_hdr_type);
            return;
    }
}
