  // drop messages if integrity protection isn't applied (see TS 24.301 Sec. 4.4.4.2)
  //
  // According to the standard, only a very limited set of NAS messages (such as
  // the service request) may be sent without an integrity protection. All other
  // message types must have integrity protection. Here we check that if the security
  // header indicates a plain NAS message then the message type is one of those allowed.
  //
  // Note: LIBLTE_NAS_MSG_TYPE_SERVICE_REQUEST is assumed to be the constant that indicates
  // the allowed message type which can be sent in the clear.
  if ((sec_hdr_type == LIBLTE_MME_SECURITY_HDR_TYPE_PLAIN_NAS) &&
      (msg_type != LIBLTE_NAS_MSG_TYPE_SERVICE_REQUEST)) {
    logger.error("Dropping NAS message: msg type %02X without integrity protection", msg_type);
    return;
  }
  
  // Normal processing continues here for messages which passed the integrity protection check.
  // [Any remaining code for further processing of the NAS PDU would follow here.]
}
