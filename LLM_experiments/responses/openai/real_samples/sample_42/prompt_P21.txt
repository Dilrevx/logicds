  // drop messages if integrity protection isn't applied (see TS 24.301 Sec. 4.4.4.2)
  if (sec_hdr_type == LIBLTE_MME_SECURITY_HDR_TYPE_PLAIN_NAS) {
    switch (msg_type) {
      // Only allow plaintext messages that are explicitly permitted.
      case LIBLTE_MME_MSG_TYPE_DETACH_REQUEST:
      case LIBLTE_MME_MSG_TYPE_EMM_INFORMATION:
      case LIBLTE_MME_MSG_TYPE_EMM_STATUS:
        // These message types are allowed in plaintext.
        break;
      case LIBLTE_MME_MSG_TYPE_ATTACH_REJECT:
      case LIBLTE_MME_MSG_TYPE_SERVICE_REJECT: {
          // For Attach/Service Rejects, allow only if emm_cause equals 25.
          // Assuming the NAS message structure contains an 'emm_cause' field.
          LIBLTE_BYTE_MSG_STRUCT* msg = (LIBLTE_BYTE_MSG_STRUCT*)pdu.get();
          if (msg->emm_cause != 25) {
            logger.error("Not handling NAS message MSG_TYPE=%02X with invalid emm_cause=%d in plaintext!",
                         msg_type, msg->emm_cause);
            return;
          }
          break;
      }
      default:
        logger.error("Not handling NAS message MSG_TYPE=%02X with SEC_HDR_TYPE=%02X without integrity protection!",
                     msg_type, sec_hdr_type);
        return;
    }
  }
