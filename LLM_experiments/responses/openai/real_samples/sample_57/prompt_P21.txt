def _read_from_sections(user, collection_url, permission):
    """Get regex sections with proper full matching to prevent bypass vulnerabilities."""
    filename = os.path.expanduser(config.get("rights", "file"))
    rights_type = config.get("rights", "type").lower()
    # Initialize config parser; note that 'user' and 'collection_url' are set as defaults 
    # but we will use fullmatch below to avoid partial regex matches.
    regex = ConfigParser({"login": user, "path": collection_url})
    if rights_type in DEFINED_RIGHTS:
        log.LOGGER.debug("Rights type '%s'" % rights_type)
        # Using StringIO to simulate a file-like object from the defined rights string.
        regex.readfp(StringIO(DEFINED_RIGHTS[rights_type]))
    elif rights_type == "from_file":
        log.LOGGER.debug("Reading rights from file %s" % filename)
        if not regex.read(filename):
            log.LOGGER.error("File '%s' not found for rights" % filename)
            return False
    else:
        log.LOGGER.error("Unknown rights type '%s'" % rights_type)
        return False

    for section in regex.sections():
        re_user = regex.get(section, "user")
        re_collection = regex.get(section, "collection")
        log.LOGGER.debug("Test if '%s:%s' matches against '%s:%s' from section '%s'" %
                         (user, collection_url, re_user, re_collection, section))
        # Use fullmatch to ensure the entire string is matched and prevent bypass via regex metacharacters.
        user_match = re.fullmatch(re_user, user)
        if user_match:
            # If there are groups in the user regex, insert them into the collection regex.
            try:
                re_collection_formatted = re_collection.format(*user_match.groups())
            except IndexError:
                log.LOGGER.error("Mismatch in formatting groups in section '%s'" % section)
                continue
            # Again, use fullmatch to force the collection URL to match entirely.
            if re.fullmatch(re_collection_formatted, collection_url):
                log.LOGGER.debug("Section '%s' matches" % section)
                if permission in regex.get(section, "permission"):
                    return True
            else:
                log.LOGGER.debug("Section '%s' does not match collection URL" % section)
    return False
