	if (msg->tsigstatus != dns_rcode_noerror || tsig.error != dns_rcode_noerror) {
		result = dns_rcode_notauth;
		return result;
	}

	/* 
	   IMPORTANT SECURITY ADDITION:
	   TSIG alone is not sufficient protection for zone transfers (AXFR) or NOTIFY messages.
	   An attacker with knowledge of a TSIG key name might craft a valid‐looking request.
	   Therefore, for sensitive operations such as AXFR and NOTIFY, enforce an ACL check.
	*/
	if (msg->opcode == dns_opcode_axfr || msg->opcode == dns_opcode_notify) {
		/* acl_check() is assumed to validate that the requestor is authorized to
		   perform the sensitive zone operation. */
		if (!acl_check(msg->client_address)) {
			result = dns_rcode_notauth;
			return result;
		}
	}

	result = ISC_R_SUCCESS;
	return result;
