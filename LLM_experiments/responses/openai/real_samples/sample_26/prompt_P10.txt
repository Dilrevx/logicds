isc_result_t
dns_message_signer(dns_message_t *msg, dns_name_t *signer) {
	isc_result_t result = ISC_R_SUCCESS;
	dns_rdata_t rdata = DNS_RDATA_INIT;

	REQUIRE(DNS_MESSAGE_VALID(msg));
	REQUIRE(signer != NULL);
	REQUIRE(msg->from_to_wire == DNS_MESSAGE_INTENTPARSE);

	/* If there’s no TSIG or SIG(0), nothing to sign */
	if (msg->tsig == NULL && msg->sig0 == NULL)
		return (ISC_R_NOTFOUND);

	/* Ensure that verification has been attempted. */
	if (msg->verify_attempted == 0)
		return (DNS_R_NOTVERIFIEDYET);

	/*
	 * Allocate a dynamic buffer for the signer name if necessary.
	 * This ensures that the signer has its own buffer, not one supplied
	 * by an untrusted request.
	 */
	if (!dns_name_hasbuffer(signer)) {
		isc_buffer_t *dynbuf = NULL;
		result = isc_buffer_allocate(msg->mctx, &dynbuf, 512);
		if (result != ISC_R_SUCCESS)
			return (result);
		dns_name_setbuffer(signer, dynbuf);
		dns_message_takebuffer(msg, &dynbuf);
	}

	/*
	 * Handle SIG(0) signatures first. (SIG(0) is typically used for dynamic
	 * update authentication, not for zone transfer or NOTIFY messages.)
	 */
	if (msg->sig0 != NULL) {
		dns_rdata_sig_t sig;

		result = dns_rdataset_first(msg->sig0);
		INSIST(result == ISC_R_SUCCESS);
		dns_rdataset_current(msg->sig0, &rdata);

		result = dns_rdata_tostruct(&rdata, &sig, NULL);
		if (result != ISC_R_SUCCESS)
			return (result);

		/* For SIG(0) we require that the signature verified properly */
		if (msg->verified_sig && msg->sig0status == dns_rcode_noerror)
			result = ISC_R_SUCCESS;
		else
			result = DNS_R_SIGINVALID;
		dns_name_clone(&sig.signer, signer);
		dns_rdata_freestruct(&sig);
	} else {
		/*
		 * Process TSIG signed messages.
		 * Vulnerability fix: For AXFR or NOTIFY messages, even if TSIG verification
		 * succeeded, further ACL-based authorization is required so that an attacker
		 * cannot bypass TSIG authentication simply by sending a properly constructed 
		 * TSIG-signed packet.
		 */
		dns_name_t *identity;
		dns_rdata_any_tsig_t tsig;

		result = dns_rdataset_first(msg->tsig);
		INSIST(result == ISC_R_SUCCESS);
		dns_rdataset_current(msg->tsig, &rdata);

		result = dns_rdata_tostruct(&rdata, &tsig, NULL);
		INSIST(result == ISC_R_SUCCESS);
		if (msg->tsigstatus != dns_rcode_noerror)
			result = DNS_R_TSIGVERIFYFAILURE;
		else if (tsig.error != dns_rcode_noerror)
			result = DNS_R_TSIGERRORSET;
		else
			result = ISC_R_SUCCESS;

		/*
		 * NEW: Additional ACL check for sensitive operations.
		 * If the message is an AXFR (zone transfer) or a NOTIFY packet,
		 * then ensure that the TSIG key used is explicitly permitted
		 * to authorize such operations. The function
		 * tsig_key_allows_transfer() represents this ACL check.
		 * (Implement tsig_key_allows_transfer() as appropriate.)
		 */
		if ((msg->is_axfr || msg->is_notify) && msg->tsigkey != NULL) {
			if (!tsig_key_allows_transfer(msg->tsigkey)) {
				dns_rdata_freestruct(&tsig);
				return (DNS_R_TRANSFER_NOT_ALLOWED);
			}
		}

		dns_rdata_freestruct(&tsig);

		/*
		 * TSIG verification must have produced a valid key.
		 * If msg->tsigkey is NULL, something is seriously wrong.
		 */
		if (msg->tsigkey == NULL) {
			INSIST(result != ISC_R_SUCCESS);
		} else {
			identity = dns_tsigkey_identity(msg->tsigkey);
			if (identity == NULL) {
				if (result == ISC_R_SUCCESS)
					result = DNS_R_NOIDENTITY;
				identity = &msg->tsigkey->name;
			}
			dns_name_clone(identity, signer);
		}
	}

	return (result);
}
