    /* Check that the signature algorithm in the certificate_verify message
     * matches the algorithm of the certificate. If it does not match any
     * of the supported and expected algorithms then fail.
     */
    if (args->sigAlgo == ed25519_sa_algo) {
    #ifdef HAVE_ED25519
        if (!ssl->peerEd25519KeyPresent) {
            WOLFSSL_MSG("Peer sent ED25519 sig but certificate does not contain an ED25519 key");
            ret = SIG_VERIFY_E;
            goto exit_dcv;
        }
    #else
        WOLFSSL_MSG("ED25519 not enabled but ED25519 sig received");
        ret = SIG_VERIFY_E;
        goto exit_dcv;
    #endif
    }
    else if (args->sigAlgo == ed448_sa_algo) {
    #ifdef HAVE_ED448
        if (!ssl->peerEd448KeyPresent) {
            WOLFSSL_MSG("Peer sent ED448 sig but certificate does not contain an ED448 key");
            ret = SIG_VERIFY_E;
            goto exit_dcv;
        }
    #else
        WOLFSSL_MSG("ED448 not enabled but ED448 sig received");
        ret = SIG_VERIFY_E;
        goto exit_dcv;
    #endif
    }
    else if (args->sigAlgo == ecc_dsa_sa_algo) {
    #ifdef HAVE_ECC
        if (!ssl->peerEccDsaKeyPresent) {
            WOLFSSL_MSG("Peer sent ECC sig but certificate does not contain an ECC key");
            ret = SIG_VERIFY_E;
            goto exit_dcv;
        }
    #else
        WOLFSSL_MSG("ECC not enabled but ECC sig received");
        ret = SIG_VERIFY_E;
        goto exit_dcv;
    #endif
    }
    else if (args->sigAlgo == falcon_level1_sa_algo ||
             args->sigAlgo == falcon_level5_sa_algo) {
    #ifdef HAVE_PQC
        if (!ssl->peerFalconKeyPresent) {
            WOLFSSL_MSG("Peer sent Falcon sig but certificate does not contain a Falcon key");
            ret = SIG_VERIFY_E;
            goto exit_dcv;
        }
    #else
        WOLFSSL_MSG("PQC not enabled but Falcon sig received");
        ret = SIG_VERIFY_E;
        goto exit_dcv;
    #endif
    }
    else if (args->sigAlgo == rsa_sa_algo) {
    #ifndef NO_RSA
        /* TLS 1.3 does not support PKCS#1.5 in certificate_verify. */
        WOLFSSL_MSG("Peer sent unsupported PKCS#1.5 RSA sig in TLS 1.3");
        ERROR_OUT(INVALID_PARAMETER, exit_dcv);
    #else
        WOLFSSL_MSG("RSA not enabled but RSA sig received");
        ret = SIG_VERIFY_E;
        goto exit_dcv;
    #endif
    }
    else if (args->sigAlgo == rsa_pss_sa_algo) {
    #ifndef NO_RSA
        if (ssl->peerRsaKey == NULL || !ssl->peerRsaKeyPresent) {
            WOLFSSL_MSG("Peer sent RSA-PSS sig but certificate does not contain an RSA key");
            ret = SIG_VERIFY_E;
            goto exit_dcv;
        }
    #else
        WOLFSSL_MSG("RSA not enabled but RSA-PSS sig received");
        ret = SIG_VERIFY_E;
        goto exit_dcv;
    #endif
    }
    else {
        /* Unknown signature algorithm: fail the verification */
        WOLFSSL_MSG("Peer sent unknown or mismatched signature algorithm in certificate_verify message");
        ret = SIG_VERIFY_E;
        goto exit_dcv;
    }
