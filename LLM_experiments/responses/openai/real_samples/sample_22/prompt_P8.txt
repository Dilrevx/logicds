#ifdef HAVE_ED25519
    /* Ensure that the ED25519 certificate is both present (pointer valid)
       and marked as present. This prevents an attacker from providing
       a mismatched signature algorithm without a valid cert. */
    if (args->sigAlgo == ed25519_sa_algo &&
       (ssl->peerEd25519Key == NULL || !ssl->peerEd25519KeyPresent)) {
        WOLFSSL_MSG("Peer sent ED25519 signature but no valid ED25519 certificate");
        ret = SIG_VERIFY_E;
        goto exit_dcv;
    }
#endif

#ifdef HAVE_ED448
    /* Check that the ED448 certificate is available before verifying
       an ED448 signature. */
    if (args->sigAlgo == ed448_sa_algo &&
       (ssl->peerEd448Key == NULL || !ssl->peerEd448KeyPresent)) {
        WOLFSSL_MSG("Peer sent ED448 signature but no valid ED448 certificate");
        ret = SIG_VERIFY_E;
        goto exit_dcv;
    }
#endif

#ifdef HAVE_ECC
    /* Verify that the ECC certificate is provided when an ECC signature is used.
       This additional pointer check avoids potential bypass if only a flag is set. */
    if (args->sigAlgo == ecc_dsa_sa_algo &&
       (ssl->peerEccDsaKey == NULL || !ssl->peerEccDsaKeyPresent)) {
        WOLFSSL_MSG("Peer sent ECC signature but no valid ECC certificate");
        ret = SIG_VERIFY_E;
        goto exit_dcv;
    }
#endif

#ifdef HAVE_PQC
    /* Combine Falcon level 1 and level 5 check.  The algorithm must match with a valid
       Falcon certificate. */
    if ((args->sigAlgo == falcon_level1_sa_algo ||
         args->sigAlgo == falcon_level5_sa_algo) &&
         !ssl->peerFalconKeyPresent) {
        WOLFSSL_MSG("Peer sent Falcon signature but no valid Falcon certificate");
        ret = SIG_VERIFY_E;
        goto exit_dcv;
    }
#endif

#ifndef NO_RSA
    /* For RSA we check that the peer's RSA certificate is present and valid.
       Previously, the PKCS#1.5 case only emitted an error message without verifying
       the pointer, potentially leading to undefined behaviour.
       Now, both rsa_sa_algo and rsa_pss_sa_algo require a valid pointer. */
    if (args->sigAlgo == rsa_sa_algo) {
        if (ssl->peerRsaKey == NULL || !ssl->peerRsaKeyPresent) {
            WOLFSSL_MSG("Peer sent PKCS#1.5 RSA signature but no valid RSA certificate");
            ret = SIG_VERIFY_E;
            goto exit_dcv;
        }
    }
    if (args->sigAlgo == rsa_pss_sa_algo &&
       (ssl->peerRsaKey == NULL || !ssl->peerRsaKeyPresent)) {
        WOLFSSL_MSG("Peer sent RSA-PSS signature but no valid RSA certificate");
        ret = SIG_VERIFY_E;
        goto exit_dcv;
    }
#endif
