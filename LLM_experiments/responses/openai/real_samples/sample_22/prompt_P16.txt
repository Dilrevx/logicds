    int valid_cert = 0;  /* Flag if the signature algorithm matches the provided certificate */

    #ifdef HAVE_ED25519
        if (args->sigAlgo == ed25519_sa_algo) {
            valid_cert = ssl->peerEd25519KeyPresent;
            if (!valid_cert) {
                WOLFSSL_MSG("Peer sent ED25519 signature but no ED25519 certificate");
            }
        }
    #endif

    #ifdef HAVE_ED448
        if (args->sigAlgo == ed448_sa_algo) {
            valid_cert = ssl->peerEd448KeyPresent;
            if (!valid_cert) {
                WOLFSSL_MSG("Peer sent ED448 signature but no ED448 certificate");
            }
        }
    #endif

    #ifdef HAVE_ECC
        if (args->sigAlgo == ecc_dsa_sa_algo) {
            valid_cert = ssl->peerEccDsaKeyPresent;
            if (!valid_cert) {
                WOLFSSL_MSG("Peer sent ECC signature but no ECC certificate");
            }
        }
    #endif

    #ifdef HAVE_PQC
        if (args->sigAlgo == falcon_level1_sa_algo) {
            valid_cert = ssl->peerFalconKeyPresent;
            if (!valid_cert) {
                WOLFSSL_MSG("Peer sent Falcon Level 1 signature but certificate key mismatch");
            }
        }
        if (args->sigAlgo == falcon_level5_sa_algo) {
            valid_cert = ssl->peerFalconKeyPresent;
            if (!valid_cert) {
                WOLFSSL_MSG("Peer sent Falcon Level 5 signature but certificate key mismatch");
            }
        }
    #endif

    #ifndef NO_RSA
        if (args->sigAlgo == rsa_sa_algo) {
            WOLFSSL_MSG("Peer sent PKCS#1.5 algorithm which is not valid in this context");
            ERROR_OUT(INVALID_PARAMETER, exit_dcv);
        }
        if (args->sigAlgo == rsa_pss_sa_algo) {
            valid_cert = (ssl->peerRsaKey != NULL && ssl->peerRsaKeyPresent);
            if (!valid_cert) {
                WOLFSSL_MSG("Peer sent RSA PSS signature but no valid RSA certificate");
            }
        }
    #endif

    /* Final validation: if none of the recognized algorithms had a matching certificate,
       then treat it as an invalid signature authentication attempt. */
    if (!valid_cert) {
        ret = SIG_VERIFY_E;
        goto exit_dcv;
    }
