            }
        }
    #endif

    #ifdef HAVE_ECC
        if (args->sigAlgo == ecc_dsa_sa_algo) {
            if (ssl->peerEccDsaKeyPresent) {
                cert_valid = 1;
            } else {
                WOLFSSL_MSG("Peer sent ECC sig but no ECC cert");
            }
        }
    #endif

    #ifdef HAVE_PQC
        if (args->sigAlgo == falcon_level1_sa_algo) {
            if (ssl->peerFalconKeyPresent) {
                cert_valid = 1;
            } else {
                WOLFSSL_MSG("Peer sent Falcon Level 1 sig but no Falcon cert");
            }
        }
        if (args->sigAlgo == falcon_level5_sa_algo) {
            if (ssl->peerFalconKeyPresent) {
                cert_valid = 1;
            } else {
                WOLFSSL_MSG("Peer sent Falcon Level 5 sig but no Falcon cert");
            }
        }
    #endif

    /* RSA signature handling */
    #ifndef NO_RSA
        if (args->sigAlgo == rsa_sa_algo) {
            WOLFSSL_MSG("Peer sent PKCS#1.5 algo, which is not permitted in TLS 1.3");
            ERROR_OUT(INVALID_PARAMETER, exit_dcv);
        }
        if (args->sigAlgo == rsa_pss_sa_algo) {
            if (ssl->peerRsaKey != NULL && ssl->peerRsaKeyPresent) {
                cert_valid = 1;
            } else {
                WOLFSSL_MSG("Peer sent RSA PSS sig but no RSA cert");
            }
        }
    #endif

    /* Final check: if none of the valid certificate-key combinations 
       has been detected, then the signature verification must fail. */
    if (!cert_valid) {
         ret = SIG_VERIFY_E;
         goto exit_dcv;
