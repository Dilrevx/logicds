#ifdef HAVE_ED25519
    if (ssl->peerEd25519KeyPresent) {
        if (args->sigAlgo != ed25519_sa_algo) {
            WOLFSSL_MSG("Mismatch: ED25519 certificate present but sigAlgo is not ed25519");
            ret = SIG_VERIFY_E;
            goto exit_dcv;
        }
    } else if (args->sigAlgo == ed25519_sa_algo) {
        WOLFSSL_MSG("Peer sent ED25519 sig but no ED25519 cert");
        ret = SIG_VERIFY_E;
        goto exit_dcv;
    }

#ifdef HAVE_ED448
    if (ssl->peerEd448KeyPresent) {
        if (args->sigAlgo != ed448_sa_algo) {
            WOLFSSL_MSG("Mismatch: ED448 certificate present but sigAlgo is not ed448");
            ret = SIG_VERIFY_E;
            goto exit_dcv;
        }
    } else if (args->sigAlgo == ed448_sa_algo) {
        WOLFSSL_MSG("Peer sent ED448 sig but no ED448 cert");
        ret = SIG_VERIFY_E;
        goto exit_dcv;
    }
#endif

#ifdef HAVE_ECC
    if (ssl->peerEccDsaKeyPresent) {
        if (args->sigAlgo != ecc_dsa_sa_algo) {
            WOLFSSL_MSG("Mismatch: ECC certificate present but sigAlgo is not ECC");
            ret = SIG_VERIFY_E;
            goto exit_dcv;
        }
    } else if (args->sigAlgo == ecc_dsa_sa_algo) {
        WOLFSSL_MSG("Peer sent ECC sig but no ECC cert");
        ret = SIG_VERIFY_E;
        goto exit_dcv;
    }
#endif

#ifdef HAVE_PQC
    if (ssl->peerFalconKeyPresent) {
        if (args->sigAlgo != falcon_level1_sa_algo &&
            args->sigAlgo != falcon_level5_sa_algo) {
            WOLFSSL_MSG("Mismatch: Falcon certificate present but unexpected Falcon sigAlgo");
            ret = SIG_VERIFY_E;
            goto exit_dcv;
        }
    } else if (args->sigAlgo == falcon_level1_sa_algo ||
               args->sigAlgo == falcon_level5_sa_algo) {
        WOLFSSL_MSG("Peer sent Falcon sig but no Falcon cert");
        ret = SIG_VERIFY_E;
        goto exit_dcv;
    }
#endif

#ifndef NO_RSA
    if (ssl->peerRsaKeyPresent) {
        /* Only RSA-PSS is allowed in TLSv1.3 */
        if (args->sigAlgo != rsa_pss_sa_algo) {
            WOLFSSL_MSG("Mismatch: RSA certificate present but sigAlgo is not RSA-PSS");
            ret = SIG_VERIFY_E;
            goto exit_dcv;
        }
    } else {
        if (args->sigAlgo == rsa_sa_algo) {
            WOLFSSL_MSG("Peer sent PKCS#1.5 rsa sig which is not allowed in certificate");
            ERROR_OUT(INVALID_PARAMETER, exit_dcv);
        }
        if (args->sigAlgo == rsa_pss_sa_algo) {
            WOLFSSL_MSG("Peer sent RSA-PSS sig but no RSA certificate");
            ret = SIG_VERIFY_E;
            goto exit_dcv;
        }
    }
#endif

/* Ensure that at least one certificate key type was present.
   Without a matching certificate, the handshake must be aborted. */
#if defined(HAVE_ED25519) || defined(HAVE_ED448) || defined(HAVE_ECC) || defined(HAVE_PQC) || !defined(NO_RSA)
    if (!ssl->peerEd25519KeyPresent && !ssl->peerEd448KeyPresent &&
        !ssl->peerEccDsaKeyPresent && !ssl->peerFalconKeyPresent &&
#ifndef NO_RSA
        !ssl->peerRsaKeyPresent
#else
        1
#endif
       ) {
        WOLFSSL_MSG("No valid certificate public key found matching the sigAlgo");
        ret = SIG_VERIFY_E;
        goto exit_dcv;
    }
#endif
