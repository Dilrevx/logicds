#if defined(HAVE_ED25519) || defined(HAVE_ED448) || defined(HAVE_ECC) || defined(HAVE_PQC) || !defined(NO_RSA)
    /* 
     * Unified check: The CertificateVerify message’s signature algorithm MUST match the type of key
     * that was provided in the peer’s Certificate message.  For TLS v1.3 the certificate’s key type
     * determines the only acceptable signature algorithm.  This prevents an attacker from “mixing and matching”
     * algorithms (for example, using an ED25519 signature algorithm with an RSA certificate), which could otherwise
     * bypass the certificate check.
     */
    if (ssl->peerEd25519KeyPresent) {
    #ifdef HAVE_ED25519
        if (args->sigAlgo != ed25519_sa_algo) {
            WOLFSSL_MSG("Mismatch: expected ED25519 signature algorithm for ED25519 cert");
            ret = SIG_VERIFY_E;
            goto exit_dcv;
        }
    #else
        WOLFSSL_MSG("ED25519 certificate present but ED25519 support is disabled");
        ret = SIG_VERIFY_E;
        goto exit_dcv;
    #endif
    }
    else if (ssl->peerEd448KeyPresent) {
    #ifdef HAVE_ED448
        if (args->sigAlgo != ed448_sa_algo) {
            WOLFSSL_MSG("Mismatch: expected ED448 signature algorithm for ED448 cert");
            ret = SIG_VERIFY_E;
            goto exit_dcv;
        }
    #else
        WOLFSSL_MSG("ED448 certificate present but ED448 support is disabled");
        ret = SIG_VERIFY_E;
        goto exit_dcv;
    #endif
    }
    else if (ssl->peerEccDsaKeyPresent) {
    #ifdef HAVE_ECC
        if (args->sigAlgo != ecc_dsa_sa_algo) {
            WOLFSSL_MSG("Mismatch: expected ECC DSA signature algorithm for ECC cert");
            ret = SIG_VERIFY_E;
            goto exit_dcv;
        }
    #else
        WOLFSSL_MSG("ECC certificate present but ECC support is disabled");
        ret = SIG_VERIFY_E;
        goto exit_dcv;
    #endif
    }
    else if (ssl->peerFalconKeyPresent) {
    #ifdef HAVE_PQC
        if (args->sigAlgo != falcon_level1_sa_algo &&
            args->sigAlgo != falcon_level5_sa_algo) {
            WOLFSSL_MSG("Mismatch: expected Falcon signature algorithm for Falcon cert");
            ret = SIG_VERIFY_E;
            goto exit_dcv;
        }
    #else
        WOLFSSL_MSG("Falcon certificate present but PQC support is disabled");
        ret = SIG_VERIFY_E;
        goto exit_dcv;
    #endif
    }
    else if (ssl->peerRsaKeyPresent) {
    #ifndef NO_RSA
        /* In TLS v1.3 only RSA_PSS is allowed for RSA; reject PKCS#1.5 (rsa_sa_algo) unconditionally.
         * This is in accord with the specification that RSA signatures must use RSASSA-PSS.
         */
        if (args->sigAlgo != rsa_pss_sa_algo) {
            WOLFSSL_MSG("Mismatch: expected RSA_PSS signature algorithm for RSA cert");
            ret = SIG_VERIFY_E;
            goto exit_dcv;
        }
    #else
        WOLFSSL_MSG("RSA certificate present but RSA support is disabled");
        ret = SIG_VERIFY_E;
        goto exit_dcv;
    #endif
    }
    else {
        WOLFSSL_MSG("No peer certificate key present; aborting verification");
        ret = SIG_VERIFY_E;
        goto exit_dcv;
    }
#endif
