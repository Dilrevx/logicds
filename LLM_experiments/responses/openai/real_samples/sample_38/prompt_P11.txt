  // Enforce integrity check on plain header NAS messages to prevent bypass.
  // Assuming that nas_msg.hdr.security_header_type indicates the security header type
  // and that the value srsran::nas_5g::security_header_type::plain_nas identifies plain NAS messages.
  // The function verify_nas_integrity() is assumed to perform the proper integrity verification
  // on the supplied PDU and return true if the integrity check passes.
  if (nas_msg.hdr.security_header_type == srsran::nas_5g::security_header_type::plain_nas) {
    if (!verify_nas_integrity(pdu)) {
      logger.error("Integrity check failed for plain header NAS message: %s (0x%02x)",
                   nas_msg.hdr.message_type.to_string(), nas_msg.hdr.message_type);
      return -1;
    }
  }

  // Process NAS message types.
  switch (nas_msg.hdr.message_type) {

      handle_registration_reject(nas_msg.registration_reject());
      break;
