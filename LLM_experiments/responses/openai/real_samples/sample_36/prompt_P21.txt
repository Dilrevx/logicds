int nas_5g::write_pdu(srsran::unique_byte_buffer_t pdu)
{
  logger.info(pdu->msg, pdu->N_bytes, "DL PDU (length %d)", pdu->N_bytes);

  nas_5gs_msg nas_msg;

  if (nas_msg.unpack_outer_hdr(pdu) != SRSRAN_SUCCESS) {
    logger.error("Unable to unpack outer NAS header");
    return SRSRAN_ERROR;
  }

  // The fix: Prevent accepting plain NAS messages after security activation.
  // It is assumed that nas_msg provides a method (or flag) to determine if
  // a security context is already active. If not, this should be implemented accordingly.
  switch (nas_msg.hdr.security_header_type)
  {
    case nas_5gs_hdr::security_header_type_opts::plain_5gs_nas_message:
      // If a security context is active, plain NAS messages MUST be rejected.
      if (nas_msg.is_security_activated()) {
        logger.error("Plain NAS messages are not allowed after NAS security activation");
        return SRSRAN_ERROR;
      }
      break;

    case nas_5gs_hdr::security_header_type_opts::integrity_protected:
      if (!integrity_check(pdu.get())) {
        logger.error("Not handling NAS message with integrity check error");
        return SRSRAN_ERROR;
      }
      break;

    case nas_5gs_hdr::security_header_type_opts::integrity_protected_and_ciphered:
      if (!integrity_check(pdu.get())) {
        logger.error("Not handling NAS message with integrity check error");
        return SRSRAN_ERROR;
      } else {
        cipher_decrypt(pdu.get());
      }
      break;

    case nas_5gs_hdr::security_header_type_opts::integrity_protected_with_new_5G_nas_context:
      // Depending on the implementation, additional checks may be needed.
      break;

    case nas_5gs_hdr::security_header_type_opts::integrity_protected_and_ciphered_with_new_5G_nas_context:
      logger.error("Not handling NAS message with new 5G NAS context and ciphering");
      return SRSRAN_ERROR;

    default:
      logger.error("Not handling NAS message with unknown security header");
      return SRSRAN_ERROR;
  }

  // Continue processing the NAS message...
  // (Additional processing code goes here)

  return SRSRAN_SUCCESS;
}
