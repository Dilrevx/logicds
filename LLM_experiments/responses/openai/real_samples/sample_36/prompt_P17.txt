int nas_5g::write_pdu(srsran::unique_byte_buffer_t pdu)
{
  // Validate the pointer and its internal buffer to prevent null pointer dereferences.
  if (!pdu || !pdu->msg) {
    logger.error("Invalid PDU pointer or missing message buffer");
    return SRSRAN_ERROR;
  }

  // Check that the PDU length is within acceptable bounds.
  // Adjust MAX_ALLOWED_PDU_LENGTH to match the system’s policy.
  const size_t MAX_ALLOWED_PDU_LENGTH = 1024;  // Example maximum limit.
  if (pdu->N_bytes == 0 || pdu->N_bytes > MAX_ALLOWED_PDU_LENGTH) {
    logger.error("PDU length %zu is not within allowed range (1-%zu)", pdu->N_bytes, MAX_ALLOWED_PDU_LENGTH);
    return SRSRAN_ERROR;
  }

  // Instead of passing untrusted data as a format string, use a constant format string and use
  // the %.*s specifier so that only the intended number of bytes are printed.
  logger.info("DL PDU (length %d) Data: %.*s", pdu->N_bytes, pdu->N_bytes, pdu->msg);

  nas_5gs_msg nas_msg;

  if (nas_msg.unpack_outer_hdr(pdu) != SRSRAN_SUCCESS) {
    logger.error("Unable to unpack outer NAS header");
    return SRSRAN_ERROR;
  }

  // ... Further processing of the NAS message would go here ...

  return SRSRAN_SUCCESS;
}
