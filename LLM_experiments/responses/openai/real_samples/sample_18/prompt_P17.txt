int
gnutls_ocsp_resp_check_crt(gnutls_ocsp_resp_t resp,
                           unsigned int indx, gnutls_x509_crt_t crt)
{
	int ret;
	gnutls_digest_algorithm_t digest;
	gnutls_datum_t rdn_hash = { NULL, 0 }, rserial = { NULL, 0 };
	gnutls_datum_t cserial = { NULL, 0 };
	gnutls_datum_t dn = { NULL, 0 };
	uint8_t cdn_hash[MAX_HASH_SIZE];
	size_t t, hash_len;

	/* Define a maximum acceptable size for certificate serial numbers.
	 * This prevents allocating (and later using) an overly large buffer
	 * that could otherwise be abused.
	 */
#define MAX_SERIAL_ALLOWED 64

	ret = gnutls_ocsp_resp_get_single(resp, indx, &digest, &rdn_hash,
					    NULL, &rserial, NULL, NULL, NULL,
					    NULL, NULL);
	if (ret < 0)
		return gnutls_assert_val(ret);

	if (rserial.size == 0 || digest == GNUTLS_DIG_UNKNOWN) {
		ret = gnutls_assert_val(GNUTLS_E_OCSP_RESPONSE_ERROR);
		goto cleanup;
	}

	hash_len = _gnutls_hash_get_algo_len(hash_to_entry(digest));
	if (hash_len != rdn_hash.size) {
		ret = gnutls_assert_val(GNUTLS_E_OCSP_RESPONSE_ERROR);
		goto cleanup;
	}

	/* Validate that the serial number size is within a safe limit.
	 * An attacker could craft an OCSP response containing an absurdly
	 * long serial number in rserial, so we reject such responses.
	 */
	if (rserial.size > MAX_SERIAL_ALLOWED) {
		ret = gnutls_assert_val(GNUTLS_E_PARSING_ERROR);
		goto cleanup;
	}

	cserial.size = rserial.size;
	cserial.data = gnutls_malloc(cserial.size);
	if (cserial.data == NULL) {
		ret = gnutls_assert_val(GNUTLS_E_MEMORY_ERROR);
		goto cleanup;
	}

	t = cserial.size;
	/* Copy the serial number from the response into our allocated memory.
	 * Since we now control the maximum size of rserial.size, this copy is safe.
	 */
	memcpy(cserial.data, rserial.data, t);

	/* --- Further processing of the certificate --- */
	/* For example, computing the hash of the certificate DN and comparing it
	 * with the one from the OCSP response.
	 *
	 * (Additional processing code would be here.)
	 */

	ret = 0; /* assume success if further verifications pass */

cleanup:
	/* Free allocated resources */
	gnutls_free(cserial.data);
	gnutls_free(dn.data);
	return ret;
}
