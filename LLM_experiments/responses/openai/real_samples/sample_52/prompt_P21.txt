ret_t
cherokee_validator_ldap_check (cherokee_validator_ldap_t *ldap,
                               cherokee_connection_t     *conn)
{
    int                              re;
    ret_t                            ret;
    size_t                           size;
    char                            *dn;
    LDAPMessage                     *message;
    LDAPMessage                     *first;
    char                            *attrs[] = { LDAP_NO_ATTRS, NULL };
    cherokee_validator_ldap_props_t *props   = VAL_LDAP_PROP(ldap);

    /* Sanity checks
     * Ensure that both username and password are provided.
     */
    if ((conn->validator == NULL) ||
        cherokee_buffer_is_empty (&conn->validator->user) ||
        cherokee_buffer_is_empty (&conn->validator->password))
        return ret_error;

    /* Proceed with LDAP search.
     * For example, perform an LDAP search for the given user.
     */
    re = ldap_search_s(props->handle,         /* LDAP connection handle */
                       props->bind_dn,        /* Base DN */
                       LDAP_SCOPE_SUBTREE,
                       /* Assuming the filter is correctly built to include the username.
                        * Adjust the filter if necessary to prevent injection.
                        */
                       "(&(objectClass=organizationalPerson)(uid=%s))",
                       attrs,
                       0,                     /* Only retrieve DN */
                       &message);
    if (re != LDAP_SUCCESS) {
        return ret_error;
    }

    /* Retrieve the first entry from the search results */
    first = ldap_first_entry(props->handle, message);
    if (first == NULL) {
        ldap_msgfree(message);
        return ret_error;
    }

    /* Extract the distinguished name (DN) of the entry */
    dn = ldap_get_dn(props->handle, first);
    if (dn == NULL) {
        ldap_msgfree(message);
        return ret_error;
    }

    /* Attempt to bind (i.e., authenticate) using the DN and the provided password */
    re = ldap_simple_bind_s(props->handle, dn, conn->validator->password.data);
    ldap_memfree(dn);
    ldap_msgfree(message);

    /* If the bind is not successful, authentication fails */
    if (re != LDAP_SUCCESS) {
        return ret_error;
    }

    return ret_ok;
}
