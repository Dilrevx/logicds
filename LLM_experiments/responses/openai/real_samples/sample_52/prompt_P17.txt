ret_t
cherokee_validator_ldap_check (cherokee_validator_ldap_t *ldap,
                               cherokee_connection_t     *conn)
{
    int                              re;
    ret_t                            ret = RET_ERROR;  /* Default return value on error */
    size_t                           size;
    char                            *dn = NULL;
    LDAPMessage                     *message = NULL;
    LDAPMessage                     *first = NULL;
    char                            *attrs[] = { LDAP_NO_ATTRS, NULL };
    cherokee_validator_ldap_props_t *props = VAL_LDAP_PROP(ldap);

    /* Sanity checks: ensure essential properties are present */
    if (props == NULL || props->search_base == NULL || props->filter == NULL) {
        /* Missing mandatory LDAP configuration */
        return RET_ERROR;
    }

    /*
     * SECURITY FIX:
     * If user-supplied data (e.g. username) is used to construct the LDAP filter,
     * escape any special characters to prevent LDAP injection.
     * Assume that conn->user contains the untrusted username.
     */
    char *escaped_user = ldap_escape_filter(conn->user, 0);
    if (escaped_user == NULL) {
        return RET_ERROR;
    }

    /* Build a safe filter string.
     * For example, if the filter should match the "uid" attribute,
     * construct it as "(uid=escaped_user)".
     */
    char *safe_filter = NULL;
    if (asprintf(&safe_filter, "(uid=%s)", escaped_user) == -1) {
        free(escaped_user);
        return RET_ERROR;
    }
    free(escaped_user);

    /* Optionally, update the filter in props or use safe_filter directly */
    /* For this repair we use safe_filter directly in the search */
    re = ldap_search_ext_s(props->ldap,            /* LDAP connection handle */
                           props->search_base,     /* Search base DN */
                           LDAP_SCOPE_SUBTREE,     /* Subtree search */
                           safe_filter,            /* Secure filter string */
                           attrs,                  /* No attributes returned */
                           0,                      /* attrsonly flag */
                           NULL,                   /* server controls */
                           NULL,                   /* client controls */
                           NULL,                   /* timeout */
                           0,                      /* sizelimit */
                           &message);              /* result pointer */
    free(safe_filter); /* no longer needed */

    if (re != LDAP_SUCCESS || message == NULL) {
        if (message) {
            ldap_msgfree(message);
        }
        return RET_ERROR;
    }

    first = ldap_first_entry(props->ldap, message);
    if (first == NULL) {
        ldap_msgfree(message);
        return RET_ERROR;
    }

    dn = ldap_get_dn(props->ldap, first);
    if (dn == NULL) {
        ldap_msgfree(message);
        return RET_ERROR;
    }

    /*
     * If you need to verify that the DN returned is exactly what you expect,
     * compare it with an expected value (which ideally is known and securely stored).
     * For example, if props->expected_dn is set, then check it.
     */
    if (props->expected_dn != NULL) {
        if (strcasecmp(dn, props->expected_dn) != 0) {
            ldap_memfree(dn);
            ldap_msgfree(message);
            return RET_ERROR;
        }
    }

    /* Successful LDAP check; safe cleanup */
    ldap_memfree(dn);
    ldap_msgfree(message);

    ret = RET_OK;
    return ret;
}
