def create_edit_shelf(shelf, page_title, page, shelf_id=False):
    sync_only_selected_shelves = current_user.kobo_only_shelves_sync

    # If editing an existing shelf, ensure the shelf belongs to the current user.
    if shelf_id:
        shelf_instance = calibre_db.session.query(ub.Shelf).filter(
            ub.Shelf.id == shelf_id,
            ub.Shelf.user_id == current_user.id
        ).first()
        if not shelf_instance:
            # Abort or return an appropriate error message if the shelf doesn't exist or isn't owned by the user.
            abort(403, description="Unauthorized access to shelf.")
        # Update the shelf with validated input.
        # (Assuming 'shelf' is a dict containing allowed fields; perform explicit field assignments)
        if 'name' in shelf:
            shelf_instance.name = shelf['name']
        # Add further fields as required, ensuring proper validation.
    else:
        # Creating a new shelf: build the shelf object with controlled data.
        try:
            # Only allow necessary and validated fields to be inserted.
            new_shelf = ub.Shelf(
                name=shelf.get('name'),
                user_id=current_user.id,
                kobo_sync=True if sync_only_selected_shelves else False
            )
            calibre_db.session.add(new_shelf)
        except Exception as e:
            # Roll back any changes on error and abort with a server error.
            calibre_db.session.rollback()
            abort(500, description="Failed to create shelf.")

    # Commit the changes if everything is valid.
    try:
        calibre_db.session.commit()
    except Exception as e:
        calibre_db.session.rollback()
        abort(500, description="Database commit failed.")

    # Finally, render the appropriate page with a success message or redirect.
    return render_template(page, title=page_title, shelf=shelf)
