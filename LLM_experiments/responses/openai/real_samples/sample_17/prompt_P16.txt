	/* Check if trivial authentication is disallowed and if the session used a trivial method.
	 * This ensures that only secure authentication methods are accepted when the client policy forbids trivial ones.
	 * It is assumed that:
	 *  - client_options.disallow_trivial_auth is a boolean flag set according to the client configuration.
	 *  - AUTH_TYPE_TRIVIAL is defined as the constant indicating a trivial authentication method.
	 *
	 * If both conditions are met, terminate the session with an error.
	 */
	if (client_options.disallow_trivial_auth && (cli_ses.lastauthtype == AUTH_TYPE_TRIVIAL)) {
		error("Trivial authentication method is disallowed by configuration");
		disconnect("Authentication refused: Trivial authentication method not permitted");
		return;
	}

	/* Note: in delayed-zlib mode, setting authdone here will enable compression in the transport layer */
	ses.authstate.authdone = 1;
