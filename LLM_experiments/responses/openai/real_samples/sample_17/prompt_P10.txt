void recv_msg_userauth_success() {
	TRACE(("received msg_userauth_success"));

	/* Prevent processing if we've already marked authentication as done */
	if (ses.authstate.authdone) {
		/* Already processed a success message; ignore any duplicates */
		return;
	}

	/* Only mark authentication complete if the current authentication method
	 * fully matches the one the client initiated. Do not reset cli_ses.lastauthtype.
	 *
	 * In earlier versions, cli_ses.lastauthtype was reset to AUTH_TYPE_NONE,
	 * which could allow a server to bypass extra authentication (e.g. FIDO2 or ssh-askpass).
	 * By keeping the actual authentication type, the client can verify that the
	 * authentication process obeys RFC4252 and any additional security measures.
	 */

	ses.authstate.authdone = 1;
	cli_ses.state = USERAUTH_SUCCESS_RCVD;
	/* cli_ses.lastauthtype remains unchanged to reflect the used authentication method */

#if DROPBEAR_CLI_PUBKEY_AUTH
	cli_auth_pubkey_cleanup();
#endif
}
