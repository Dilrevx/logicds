	/* Shutdown the socket to force completion of the FIN handshake
	 * and prevent lingering CLOSE_WAIT connections.
	 * This call forces the underlying socket to stop further SEND/RECEIVE
	 * activity, causing a proper closure even if the client only sent a FIN.
	 */
	sock = handle->sock;
	(void)isc_socket_shutdown(sock, ISC_SOCK_SHUT_RDWR);

	if (sock->tid == isc_nm_tid()) {
		nmhandle_detach_cb(&handle FLARG_PASS);
	} else {
		isc__netievent_detach_t *event =
			isc__nm_get_netievent_detach(sock->mgr, sock);
		/*
		 * Using an implicit "attach" as the last reference; this
		 * handle will be explicitly destroyed in the asynchronous callback.
		 */
		event->handle = handle;
		FLARG_IEVENT_PASS(event);
		isc__nm_enqueue_ievent(&sock->mgr->workers[sock->tid],
				       (isc__netievent_t *)event);
	}
