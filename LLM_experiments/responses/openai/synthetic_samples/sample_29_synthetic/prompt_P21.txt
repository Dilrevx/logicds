    public static Frec lookup_frec(int id, byte[] hash, int family, int fd) {
        for (Frec f = Daemon.frec_list; f != null; f = f.next) {
            // Added additional checks for the destination socket (fd) and the address family,
            // ensuring that the reply is matched against the complete query tuple.
            if (f.sentto != null
                    && f.new_id == id
                    && f.fd == fd
                    && f.family == family
                    && (hash == null || Arrays.equals(hash, f.hash))) {
                return f;
            }
        }
        return null;
    }
