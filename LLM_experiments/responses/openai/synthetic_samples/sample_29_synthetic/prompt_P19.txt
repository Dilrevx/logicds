    public static Frec lookup_frec(int id, byte[] hash, int family, int fd) {
        // Ensure the provided hash is not null to avoid NullPointerException
        if (hash == null) {
            throw new IllegalArgumentException("hash cannot be null");
        }

        // Traverse the linked list of Frec objects
        for (Frec f = Daemon.frec_list; f != null; f = f.next) {
            // First, perform basic integer checks to avoid doing hash comparisons unnecessarily
            if (f.id != id || f.family != family || f.fd != fd) {
                continue;
            }

            // Ensure that the stored hash is non-null and has the same length as the provided hash,
            // which is critical before calling MessageDigest.isEqual.
            if (f.hash == null || f.hash.length != hash.length) {
                continue;
            }

            // Use constant-time comparison to prevent timing attacks that could leak information
            if (MessageDigest.isEqual(f.hash, hash)) {
                return f;
            }
        }

        // If no matching Frec is found, return null.
        return null;
    }
