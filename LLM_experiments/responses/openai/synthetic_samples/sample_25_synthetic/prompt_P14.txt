    /* Verify that the peer sent the expected key for the chosen signature algorithm.
       Any failure in matching the expected key type is a fatal error,
       terminating the handshake with a "decrypt_error" alert. */
    if (args.sigAlgo == Commons.ed25519_sa_algo) {
        if (!ssl.peerEd25519KeyPresent) {
            Wslib.WOLFSSL_MSG("Certificate Verify failure: Expected ED25519 key not present");
            return Commons.INVALID_PARAMETER;
        }
    }
    else if (args.sigAlgo == Commons.ed448_sa_algo) {
        if (!ssl.peerEd448KeyPresent) {
            Wslib.WOLFSSL_MSG("Certificate Verify failure: Expected ED448 key not present");
            return Commons.INVALID_PARAMETER;
        }
    }
    else if (args.sigAlgo == Commons.ecc_dsa_sa_algo) {
        if (!ssl.peerEccDsaKeyPresent) {
            Wslib.WOLFSSL_MSG("Certificate Verify failure: Expected ECC key not present");
            return Commons.INVALID_PARAMETER;
        }
    }
    else if (args.sigAlgo == Commons.rsa_sa_algo) {
        Wslib.WOLFSSL_MSG("Certificate Verify failure: PKCS#1.5 signature is not permitted in TLS 1.3");
        return Commons.INVALID_PARAMETER;
    }
    else if (args.sigAlgo == Commons.rsa_pss_sa_algo) {
        if (!ssl.peerRsaKeyPresent) {
            Wslib.WOLFSSL_MSG("Certificate Verify failure: Expected RSA key not present");
            return Commons.INVALID_PARAMETER;
        }
    }
    else {
        Wslib.WOLFSSL_MSG("Certificate Verify failure: Unknown signature algorithm");
        return Commons.INVALID_PARAMETER;
    }
