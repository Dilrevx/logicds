public void adminAuth(GinContext ctx) {
    String token = extractToken(ctx);
    // Check for null or empty token to avoid processing an invalid token.
    if (token == null || token.isEmpty()) {
        handleResponse(ctx, AuthError.unauthorized(AuthError.UNAUTHORIZED_REASON));
        ctx.abort();
        return;
    }

    // Retrieve cached admin user info using the token.
    AuthService.CacheResult result = authService.getAdminUserCacheInfo(ctx, token);
    
    // Check if an error occurred or we didn't find the user info.
    if (result.err != null || result.userInfo == null) {
        handleResponse(ctx, AuthError.unauthorized(AuthError.UNAUTHORIZED_REASON));
        ctx.abort();
        return;
    }
    
    // Verify that the user actually has admin privileges.
    if (!result.userInfo.isAdmin()) {
        handleResponse(ctx, AuthError.unauthorized("Admin privileges required"));
        ctx.abort();
        return;
    }
    
    // If all checks pass, continue with processing as an admin.
    // ... rest of the function logic for authenticated admin users ...
}
