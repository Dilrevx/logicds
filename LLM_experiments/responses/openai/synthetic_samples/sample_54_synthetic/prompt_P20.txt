         * Proceed with the bind mount of /proc/self/fd onto the temporary directory.
         * The addition of O_CLOEXEC when opening the directory further reduces risk by
         * ensuring that the file descriptor is not inherited by unintended child processes.
         */
        if (OsApi.mount("/proc/self/fd", tmpdir, null, NamespaceFlags.MS_BIND, null) < 0) {
            FuseLog.log(NamespaceFlags.FUSE_LOG_ERR, "mount(/proc/self/fd, %s, MS_BIND): %m\n", tmpdir);
