    public int write_pdu(UniqueByteBuffer pdu) {
        logger.info(pdu.msg, pdu.N_bytes, "DL PDU (length %d)", pdu.N_bytes);
        Nas5gsMsg nas_msg = new Nas5gsMsg();

        // Unpack the outer header; if this fails, abort processing.
        if (nas_msg.unpack_outer_hdr(pdu) != SRSRAN_SUCCESS) {
            logger.error("Unable to unpack outer NAS header");
            return SRSRAN_ERROR;
        }

        // FIX: Reject nas_sm_cmd_replay if registration is complete.
        // It prevents processing of replay commands after successful registration.
        if (nas_msg.msg_type == Nas5gsMsgType.nas_sm_cmd_replay && isRegistrationComplete()) {
            logger.error("Received nas_sm_cmd_replay after registration complete; rejecting message");
            return SRSRAN_ERROR;
        }

        // Process the NAS PDU based on its security header type.
        switch (nas_msg.hdr.security_header_type) {
            case Nas5gsHdr.security_header_type_opts.plain_5gs_nas_message:
                // No extra processing needed for plain messages.
                break;

            case Nas5gsHdr.security_header_type_opts.integrity_protected:
                if (!integrity_check(pdu.get())) {
                    logger.error("Not handling NAS message with integrity check error");
                    return SRSRAN_ERROR;
                }
                break;

            case Nas5gsHdr.security_header_type_opts.integrity_protected_and_ciphered:
                if (!integrity_check(pdu.get())) {
                    logger.error("Not handling NAS message with integrity check error");
                    return SRSRAN_ERROR;
                } else {
                    cipher_decrypt(pdu.get());
                }
                break;

            case Nas5gsHdr.security_header_type_opts.integrity_protected_with_new_5G_nas_context:
                // Additional processing could be added here if needed.
                break;

            case Nas5gsHdr.security_header_type_opts.integrity_protected_and_ciphered_with_new_5G_nas_context:
                // Do not support messages that are both ciphered and use new context.
                return SRSRAN_ERROR;

            default:
                logger.error("Not handling NAS message with unknown security header");
                return SRSRAN_ERROR;
        }

        if (pcap != null) {
            // (Optional) pcap logging and further processing code.
        }

        // Continue with the rest of the function processing...
        // (Assuming additional processing and eventually a successful return.)
        return SRSRAN_SUCCESS;
    }
