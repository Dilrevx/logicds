    /* Determine the appropriate name to use for the CRL lookup.
       Per specification:
         (1) If the Distribution Point (DP) includes a cRLIssuer entry,
             then use that as the name (and later verify that the CRL's
             issuing distribution point extension is correct).
         (2) Otherwise, fall back to the certificate issuer.
       (Assume that “dp” is a pointer to the DP structure and “cert” is the current certificate.) */
    X509_NAME *crl_name = NULL;
    if (dp != NULL && dp->cRLIssuer != NULL) {
        crl_name = dp->cRLIssuer;
    } else {
        crl_name = cert->issuer;
    }

    /* Retrieve the CRLs using the appropriate lookup name */
    if (Commons.OPENSSL_VERSION_NUMBER >= Commons.OPENSSL_1_1 && !Commons.HAVE_LIBRESSL) {
        crls = storeCtx.get1_crls(crl_name);
    }
    else if (Commons.OPENSSL_VERSION_NUMBER >= Commons.OPENSSL_1_0 && !Commons.HAVE_LIBRESSL) {
        crls = Commons.get1_crls(storeCtx, crl_name);
    }
