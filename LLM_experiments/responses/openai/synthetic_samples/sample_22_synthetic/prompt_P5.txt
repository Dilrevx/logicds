if (args.sigAlgo == CommonsConstants.ed25519_sa_algo) {
    if (!ssl.peerEd25519KeyPresent) {
        Wslib.WOLFSSL_MSG("Peer sent ED25519 sig but no matching ED25519 certificate");
        return CommonsConstants.SIG_VERIFY_E;
    }
}
else if (args.sigAlgo == CommonsConstants.ed448_sa_algo) {
    if (!ssl.peerEd448KeyPresent) {
        Wslib.WOLFSSL_MSG("Peer sent ED448 sig but no matching ED448 certificate");
        return CommonsConstants.SIG_VERIFY_E;
    }
}
else if (args.sigAlgo == CommonsConstants.ecc_dsa_sa_algo) {
    if (!ssl.peerEccDsaKeyPresent) {
        Wslib.WOLFSSL_MSG("Peer sent ECC sig but no matching ECC certificate");
        return CommonsConstants.SIG_VERIFY_E;
    }
}
else if (args.sigAlgo == CommonsConstants.falcon_level1_sa_algo ||
         args.sigAlgo == CommonsConstants.falcon_level5_sa_algo) {
    if (!ssl.peerFalconKeyPresent) {
        Wslib.WOLFSSL_MSG("Peer sent Falcon sig but no matching Falcon certificate");
        return CommonsConstants.SIG_VERIFY_E;
    }
}
else if (args.sigAlgo == CommonsConstants.rsa_sa_algo) {
    Wslib.WOLFSSL_MSG("Peer sent disallowed PKCS#1.5 RSA signature algorithm");
    return CommonsConstants.INVALID_PARAMETER;
}
else if (args.sigAlgo == CommonsConstants.rsa_pss_sa_algo) {
    if (ssl.peerRsaKey == null || !ssl.peerRsaKeyPresent) {
        Wslib.WOLFSSL_MSG("Peer sent RSA-PSS sig but no matching RSA certificate");
        return CommonsConstants.SIG_VERIFY_E;
    }
}
else {
    /* Any signature algorithm not explicitly handled must be treated as a mismatch 
       between the certificate and the signature, so we fail the verification */
    Wslib.WOLFSSL_MSG("Peer sent an unknown or mismatched signature algorithm in certificate_verify");
    return CommonsConstants.SIG_VERIFY_E;
}
