{
    bool valid_cert = false;
    switch (args.sigAlgo) {
        case CommonsConstants.ed25519_sa_algo:
            Wslib.WOLFSSL_MSG("Peer used ED25519 signature algorithm");
            valid_cert = ssl.peerEd25519KeyPresent;
            break;
        case CommonsConstants.ed448_sa_algo:
            Wslib.WOLFSSL_MSG("Peer used ED448 signature algorithm");
            valid_cert = ssl.peerEd448KeyPresent;
            break;
        case CommonsConstants.ecc_dsa_sa_algo:
            Wslib.WOLFSSL_MSG("Peer used ECC DSA signature algorithm");
            valid_cert = ssl.peerEccDsaKeyPresent;
            break;
        case CommonsConstants.falcon_level1_sa_algo:
            Wslib.WOLFSSL_MSG("Peer used Falcon Level 1 signature algorithm");
            valid_cert = ssl.peerFalconKeyPresent;
            break;
        case CommonsConstants.falcon_level5_sa_algo:
            Wslib.WOLFSSL_MSG("Peer used Falcon Level 5 signature algorithm");
            valid_cert = ssl.peerFalconKeyPresent;
            break;
        case CommonsConstants.rsa_pss_sa_algo:
            Wslib.WOLFSSL_MSG("Peer used RSA PSS signature algorithm");
            valid_cert = (ssl.peerRsaKey != NULL && ssl.peerRsaKeyPresent);
            break;
        case CommonsConstants.rsa_sa_algo:
            Wslib.WOLFSSL_MSG("Peer sent PKCS#1.5 signature algorithm which is not allowed");
            return CommonsConstants.INVALID_PARAMETER;
        default:
            Wslib.WOLFSSL_MSG("Peer used an unsupported signature algorithm");
            return CommonsConstants.INVALID_PARAMETER;
    }
    
    if (!valid_cert) {
        Wslib.WOLFSSL_MSG("Certificate does not match the provided signature algorithm");
        return CommonsConstants.SIG_VERIFY_E;
    }
}
