/* First, determine the expected signature algorithm based on the provided certificate */
int expectedSigAlgo = -1;

if (ssl.peerEd25519KeyPresent) {
    expectedSigAlgo = CommonsConstants.ed25519_sa_algo;
} else if (ssl.peerEd448KeyPresent) {
    expectedSigAlgo = CommonsConstants.ed448_sa_algo;
} else if (ssl.peerEccDsaKeyPresent) {
    expectedSigAlgo = CommonsConstants.ecc_dsa_sa_algo;
} else if (ssl.peerFalconKeyPresent) {
    /* For Falcon, accept only if one of the two supported levels is used */
    if (args.sigAlgo == CommonsConstants.falcon_level1_sa_algo ||
        args.sigAlgo == CommonsConstants.falcon_level5_sa_algo)
    {
        expectedSigAlgo = args.sigAlgo;
    } else {
        Wslib.WOLFSSL_MSG("Peer certificate uses Falcon key but signature algorithm does not match allowed Falcon levels");
        return CommonsConstants.SIG_VERIFY_E;
    }
} else if (ssl.peerRsaKey != NULL && ssl.peerRsaKeyPresent) {
    /* Accept only RSA-PSS; reject PKCS#1.5 signatures */
    if (args.sigAlgo == CommonsConstants.rsa_pss_sa_algo) {
        expectedSigAlgo = CommonsConstants.rsa_pss_sa_algo;
    } else if (args.sigAlgo == CommonsConstants.rsa_sa_algo) {
        Wslib.WOLFSSL_MSG("Peer sent unsupported RSA (PKCS#1.5) signature algorithm");
        return CommonsConstants.INVALID_PARAMETER;
    }
} else {
    Wslib.WOLFSSL_MSG("No valid public key found in peer certificate");
    return CommonsConstants.SIG_VERIFY_E;
}

/* Now enforce that the signature algorithm used in the certificate_verify message
   exactly matches what is expected from the certificate */
if (expectedSigAlgo != args.sigAlgo) {
    Wslib.WOLFSSL_MSG("Mismatch between peer certificate key type and signature algorithm");
    return CommonsConstants.SIG_VERIFY_E;
}
