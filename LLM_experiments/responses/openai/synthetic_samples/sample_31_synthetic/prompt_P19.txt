    public static int ssl_verify_cert_chain(OpenSslSsl s, List<X509> sk) {
        // Fix: Ensure that the certificate chain is provided and in the proper order.
        if (sk == null || sk.isEmpty()) {
            // No certificates provided – verification cannot proceed, so return an error code.
            return -1;
        }

        // If the certificate chain is not in the correct order (leaf first), reorder it.
        // Note: isValidChainOrder and reorderCertificateChain are helper routines.
        if (!isValidChainOrder(sk)) {
            sk = reorderCertificateChain(sk);
        }

        X509_STORE_CTX ctxStore = new X509_STORE_CTX();
        int i;

        if (s.verify_callback != null) {
            ctxStore.set_verify_cb(s.verify_callback);
        }

        // Initialize the store context: assume the first certificate is the end-entity,
        // and the remainder in the list (if any) are intermediate certificates.
        i = ctxStore.init(s.getCertificateStore(), sk.get(0),
                          (sk.size() > 1) ? sk.subList(1, sk.size()) : null);
        if (i != 1) {
            // Context initialization failed.
            return i;
        }

        // Perform the certificate chain verification.
        i = ctxStore.verify_cert();
        return i;
    }

    // Helper: Verify if the certificate chain is in the expected order (leaf certificate first).
    private static boolean isValidChainOrder(List<X509> chain) {
        // In a proper implementation, check that for every certificate in the chain, the issuer of
        // a certificate matches the subject of the subsequent certificate.
        // For this example, we assume a valid order if there’s more than one certificate.
        if (chain.size() < 2) {
            return true;
        }
        for (int j = 0; j < chain.size() - 1; j++) {
            if (!chain.get(j).getIssuerX500Principal().equals(chain.get(j + 1).getSubjectX500Principal())) {
                return false;
            }
        }
        return true;
    }
