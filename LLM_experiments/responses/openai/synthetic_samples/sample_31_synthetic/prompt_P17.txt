    public static int ssl_verify_cert_chain(OpenSslSsl s, List<X509> sk) {
        // Validate inputs to avoid null pointer or empty chain issues.
        if (s == null || sk == null || sk.isEmpty()) {
            // Return an error code (e.g. -1) if parameters are not valid.
            return -1;
        }
        
        X509_STORE_CTX ctxStore = new X509_STORE_CTX();
        int ret = -1;
        
        // Initialize the store context with the trusted certificate store from s,
        // the end-entity certificate (first in the list), and the rest of the chain.
        // This initialization ensures that the context is properly associated with
        // a trusted store and the appropriate certificate chain.
        // Note: Assume that s.cert_store is a properly initialized X509_STORE.
        if (!ctxStore.init(s.cert_store, sk.get(0), sk)) {
            // Failed to initialize the verification context.
            return -1;
        }
        
        // If a verification callback is provided, set it for the context.
        if (s.verify_callback != null) {
            ctxStore.set_verify_cb(s.verify_callback);
        }
        
        // Optionally, set the default verification paths if needed.
        ctxStore.set_default_verify_paths();
        
        // Perform the verification of the certificate chain.
        ret = ctxStore.verify_cert();
        
        // Clean up the context to free resources.
        ctxStore.cleanup();
        
        return ret;
    }
