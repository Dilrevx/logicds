    public int mkpStage30(Plugin p, ClientSession cs, SessionRequest sr) {
        MkPtr referer;

        PluginTrace.trace("[FD %d] Mandril validating URL", cs.getSocket());

        // Safely retrieve the referer from the session request.
        referer = sr.getReferer();
        if (referer == null) {
            PluginTrace.error("[FD %d] No referer provided", cs.getSocket());
            return -1;
        }

        String refererUrl = referer.toString();
        
        // Validate that the referer URL uses a secure scheme.
        if (!refererUrl.startsWith("https://")) {
            PluginTrace.error("[FD %d] Insecure referer URL detected: %s", cs.getSocket(), refererUrl);
            return -2;
        }
        
        // (Optional) Verify that the referer belongs to an expected domain.
        if (!isAllowedDomain(refererUrl)) {
            PluginTrace.error("[FD %d] Referer URL not in allowed domains: %s", cs.getSocket(), refererUrl);
            return -3;
        }
        
        // Continue processing further logic after safely validating the referer URL...
        return 0;
    }

    // Helper method to check if the referer URL originates from an allowed domain.
    private boolean isAllowedDomain(String url) {
        // Add your allowed domain(s) here.
        String[] allowedDomains = {"https://www.example.com", "https://example.com"};
        for (String domain : allowedDomains) {
            if (url.startsWith(domain)) {
                return true;
            }
        }
        return false;
    }
