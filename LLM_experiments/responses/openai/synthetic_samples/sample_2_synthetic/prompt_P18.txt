    public static int checkPolicy(X509StoreCtx ctx) {
        int ret;

        // If this context has a parent, skip further processing.
        if (ctx.parent != null) {
            return 1;
        }

        // If bare-ta signing is enabled, attempt to push a null certificate onto the chain.
        if (ctx.bareTaSigned && !pushNullCert(ctx.chain)) {
            raiseError(X509Error.ERR_R_CRYPTO_LIB);
            return -1;
        }

        try {
            // Perform the policy check.
            ret = X509PolicyChecker.check(ctx);
        } finally {
            // Ensure proper cleanup: If a null certificate was pushed, always pop it.
            if (ctx.bareTaSigned) {
                popCert(ctx.chain);
            }
        }

        // If an internal error was detected during policy checking, report it.
        if (ret == X509PolicyChecker.X509_PCY_TREE_INTERNAL) {
            raiseError(X509Error.ERR_R_X509_LIB);
            return -1;
        }

        // Return the result of the policy check.
        return ret;
    }
