switch (sec_hdr_type) {
    case LIBLTE_MME_SECURITY_HDR_TYPE_PLAIN_NAS:
    case LIBLTE_MME_SECURITY_HDR_TYPE_SERVICE_REQUEST:
    case LIBLTE_MME_SECURITY_HDR_TYPE_INTEGRITY_WITH_NEW_EPS_SECURITY_CONTEXT:
        // Plain or secured with new EPS security context (if applicable) are left untouched.
        break;

    case LIBLTE_MME_SECURITY_HDR_TYPE_INTEGRITY:
        if (!integrity_check(pdu.get())) {
            logger.error("Not handling NAS message with integrity check error (INTEGRITY)");
            return;
        }
        break;

    case LIBLTE_MME_SECURITY_HDR_TYPE_INTEGRITY_AND_CIPHERED:
        if (!integrity_check(pdu.get())) {
            logger.error("Not handling NAS message with integrity check error (INTEGRITY_AND_CIPHERED)");
            return;
        }
        cipher_decrypt(pdu.get());
        break;

    case LIBLTE_MME_SECURITY_HDR_TYPE_INTEGRITY_AND_CIPHERED_WITH_NEW_EPS_SECURITY_CONTEXT:
        // Previously, this case fell through without validation.
        if (!integrity_check(pdu.get())) {
            logger.error("Not handling NAS message with integrity check error (INTEGRITY_AND_CIPHERED_WITH_NEW_EPS_SECURITY_CONTEXT)");
            return;
        }
        cipher_decrypt(pdu.get());
        break;

    default:
        logger.error("Not handling NAS message with SEC_HDR_TYPE=%02X", sec_hdr_type);
        return;
}
