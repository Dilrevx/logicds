public static int BuildMessage(TlsContext ssl,
                               byte[] output,
                               int outSz,
                               byte[] input,
                               int inSz,
                               int type,
                               int hashOutput,
                               int sizeOnly,
                               int asyncOkay,
                               int epochOrder) {
    BuildMsgArgs args;
    BuildMsgArgs lcl_args = new BuildMsgArgs();

    // Validate pointers and buffer sizes to avoid null dereference and overflows
    if (ssl == null || ssl.options == null || output == null || input == null) {
        return CommonsConstants.BAD_FUNC_ARG;
    }
    if (outSz < 0 || inSz < 0 || outSz > output.length || inSz > input.length) {
        return CommonsConstants.BAD_FUNC_ARG;
    }
    
    // Optionally, check that the message type is within an acceptable range.
    if (type < 0 || type > CommonsConstants.MAX_MESSAGE_TYPE) {
        return CommonsConstants.BAD_MSG_TYPE;
    }

    args = lcl_args;
    // Reset message building state before starting.
    ssl.options.buildMsgState = CommonsConstants.BUILD_MSG_BEGIN;
    
    // For TLS 1.1 mode configure the Initialization Vector (IV) size.
    if (ssl.options.tls1_1) {
        args.ivSz = CommonsConstants.BLOCK_SIZE;
    }
    
    // The remainder of BuildMessage should safely process input into output.
    // For example purposes, assume we need to copy the message (ensure proper bounds)
    int copySize = Math.min(inSz, outSz);
    try {
        System.arraycopy(input, 0, output, 0, copySize);
    } catch (ArrayIndexOutOfBoundsException e) {
        // Defensive coding: if something is wrong despite our checks, set state and return error.
        ssl.options.buildMsgState = CommonsConstants.BUILD_MSG_ERROR;
        return CommonsConstants.BUFFER_ERROR;
    }
    
    // Additional processing of message (hashing, encryption,
    // and any asynchronous handling) goes here as required...
    
    // Set the state to indicate that message build completed successfully.
    ssl.options.buildMsgState = CommonsConstants.BUILD_MSG_COMPLETE;
    return CommonsConstants.SUCCESS;
}
