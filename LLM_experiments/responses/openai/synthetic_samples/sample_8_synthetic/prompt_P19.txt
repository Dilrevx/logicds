public int selectMethods() {
    // Retrieve the supported methods from client and server
    String[] clientMethods = client.getMethods();
    String[] serverMethods = server.getMethods();
    
    // Validate that the method arrays are not null and contain entries
    if (clientMethods == null || serverMethods == null) {
        throw new IllegalStateException("Client or server methods array is null.");
    }
    
    // For safety, initialize extStart to null
    String extStart = null;

    // Loop through client methods to find one that is also offered by the server
    for (int i = 0; i < clientMethods.length; i++) {
        String clientMethod = clientMethods[i];
        if (clientMethod == null) {
            continue; // Skip potential null in client methods
        }
        
        // Cross-check with server provided methods
        for (String serverMethod : serverMethods) {
            if (serverMethod == null) {
                continue; // Skip potential null in server methods
            }
            
            if (clientMethod.equals(serverMethod)) {
                // Ensure that the matched method is secure (e.g., not a known vulnerable or insecure method)
                if (isMethodAllowed(clientMethod)) {
                    extStart = clientMethod;
                    // Return the index of the selected client method. This prevents fallback to weak methods.
                    return i;
                }
            }
        }
    }
    
    // If no valid and secure method was found, return error indication.
    return -1;
}

/**
 * Checks whether the provided method is allowed.
 * This function can be customized to blacklist insecure methods.
 */
private boolean isMethodAllowed(String method) {
    // Example: blacklist method names which are known to be insecure.
    if ("INSECURE".equalsIgnoreCase(method) || "NULL".equalsIgnoreCase(method)) {
        return false;
    }
    // Other method-specific validations can be added here.
    return true;
}
