public int selectMethods() {
    String[] clientMethods = client.getMethods();
    String[] serverMethods = server.getMethods();

    // Look for the extension markers in the key exchange (KEX) string of both sides.
    String clientExt = findSubstring(clientMethods[SSHConstants.SSH_KEX], "," + SSHConstants.KEX_EXTENSION_CLIENT);
    String serverExt = findSubstring(serverMethods[SSHConstants.SSH_KEX], "," + SSHConstants.KEX_EXTENSION_SERVER);

    // Ensure that both sides agree on whether the extension is present.
    // A mismatch can indicate an attempted Terrapin attack.
    if ((clientExt != null && serverExt == null) || (clientExt == null && serverExt != null)) {
        return SSHConstants.SSH_ERROR;
    }

    // If extensions were negotiated, remove the extension section from the KEX algorithm lists.
    if (clientExt != null) {
        clientMethods[SSHConstants.SSH_KEX] = truncateString(clientMethods[SSHConstants.SSH_KEX], clientExt);
    }
    if (serverExt != null) {
        serverMethods[SSHConstants.SSH_KEX] = truncateString(serverMethods[SSHConstants.SSH_KEX], serverExt);
    }

    // Process each of the supported methods ensuring that negotiated methods are valid.
    for (int i = 0; i < SSHConstants.SSH_KEX_METHODS; i++) {
        crypto.setKexMethod(i, findMatching(serverMethods[i], clientMethods[i]));

        // If we are determining a MAC and the preceding KEX method indicates an AEAD algorithm,
        // substitute the AEAD-specific MAC method.
        if ((i == SSHConstants.SSH_MAC_C_S || i == SSHConstants.SSH_MAC_S_C) && (i - 2 >= 0)) {
            String aeadHmac = findAeadHmac(crypto.getKexMethod(i - 2));
            if (aeadHmac != null) {
                crypto.setKexMethod(i, aeadHmac);
            }
        }
        // For mandatory method slots, any failure to negotiate is fatal.
        if (crypto.getKexMethod(i) == null && i < SSHConstants.SSH_LANG_C_S) {
            return SSHConstants.SSH_ERROR;
        }
        // For language options (if any) that are optional, substitute an empty string.
        else if (i >= SSHConstants.SSH_LANG_C_S && crypto.getKexMethod(i) == null) {
            crypto.setKexMethod(i, "");
        }
    }
    return SSHConstants.SSH_OK;
}
