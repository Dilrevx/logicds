String kexMethods = clientMethods[SSHConstants.SSH_KEX];
if (kexMethods.indexOf("," + SSHConstants.KEX_EXTENSION_CLIENT) != -1) {
    // Remove any occurrence of the insecure extension entry rather than truncating the entire list.
    String[] methods = kexMethods.split(",");
    List<String> filteredMethods = new ArrayList<String>(methods.length);
    for (int i = 0; i < methods.length; i++) {
        if (!methods[i].equals(SSHConstants.KEX_EXTENSION_CLIENT)) {
            filteredMethods.add(methods[i]);
        }
    }
    // Reassemble the list without the extension
    clientMethods[SSHConstants.SSH_KEX] = String.join(",", filteredMethods);
}
