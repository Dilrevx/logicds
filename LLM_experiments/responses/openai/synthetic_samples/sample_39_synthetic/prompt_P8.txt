if (security_mode_command.selected_nas_security_algorithms == null ||
    security_mode_command.selected_nas_security_algorithms.ciphering_algorithm == null ||
    security_mode_command.selected_nas_security_algorithms.integrity_protection_algorithm == null) {
    logger.error("Missing NAS security algorithms in security_mode_command");
    send_security_mode_reject(Cause5gmmType.Cause.ue_security_capabilities_mismatch);
    return SRSRAN_ERROR;
}

try {
    ctxt_base.cipher_algo = CipheringAlgorithmIdEnum.fromValue(
        security_mode_command.selected_nas_security_algorithms.ciphering_algorithm.value
    );
} catch (IllegalArgumentException e) {
    logger.error("Invalid ciphering algorithm value: " +
                 security_mode_command.selected_nas_security_algorithms.ciphering_algorithm.value);
    send_security_mode_reject(Cause5gmmType.Cause.ue_security_capabilities_mismatch);
    return SRSRAN_ERROR;
}

try {
    ctxt_base.integ_algo = IntegrityAlgorithmIdEnum.fromValue(
        security_mode_command.selected_nas_security_algorithms.integrity_protection_algorithm.value
    );
} catch (IllegalArgumentException e) {
    logger.error("Invalid integrity protection algorithm value: " +
                 security_mode_command.selected_nas_security_algorithms.integrity_protection_algorithm.value);
    send_security_mode_reject(Cause5gmmType.Cause.ue_security_capabilities_mismatch);
    return SRSRAN_ERROR;
}

if (security_mode_command.replayed_ue_security_capabilities == null ||
    !check_replayed_ue_security_capabilities(security_mode_command.replayed_ue_security_capabilities)) {
    logger.warning("Sending Security Mode Reject due to security capabilities mismatch or missing capabilities");
    send_security_mode_reject(Cause5gmmType.Cause.ue_security_capabilities_mismatch);
    return SRSRAN_ERROR;
}

if (initial_sec_command) {
    set_k_gnb_count(0);
    ctxt_base.tx_count = 0;
    initial_sec_command = false;
