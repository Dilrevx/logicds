                // First, attempt to perform the basic SPNEGO authentication.
                if (NgxConst.NGX_DECLINED == AuthSpnego.ngxHttpAuthSpnegoBasic(r, ctx, alcf)) {
                    SpnegoDebug.d0("Basic auth failed");

                    /* 
                     * Prevent basic-auth bypass by validating the username.
                     * If the username is malformed (e.g. contains unexpected characters),
                     * we consider the authentication attempt invalid and do not fall back to
                     * sending basic-auth headers.
                     */
                    if (!AuthSpnego.isValidUsername(ctx.user)) {
                        SpnegoDebug.d0("Malformed username detected, rejecting authentication");
                        ctx.ret = NgxConst.NGX_HTTP_UNAUTHORIZED;
                        return ctx.ret;
                    }

                    /* 
                     * If the username is valid, continue with setting only the basic auth headers.
                     * This ensures that even if SPNEGO fails, proper basic authentication error headers
                     * are set.
                     */
                    if (NgxConst.NGX_ERROR == AuthSpnego.ngxHttpAuthSpnegoHeadersBasicOnly(r, ctx, alcf)) {
                        SpnegoDebug.d0("Error setting headers");
                        ctx.ret = NgxConst.NGX_HTTP_INTERNAL_SERVER_ERROR;
                        return ctx.ret;
                    }
                    ctx.ret = NgxConst.NGX_HTTP_UNAUTHORIZED;
                    return ctx.ret;
                }
