            /* First verify that the certificate is allowed to act as a CA.
             * This must be done before any explicit elliptic curve parameter checks,
             * so that a non‐CA certificate that nevertheless has explicit EC parameters
             * does not cause a later check to override the CA validation error.
             */
            if ((x.ex_flags & X509.EXFLAG_CA) == 0 && x.ex_pathlen != -1 && strict) {
                ctx.error = X509StoreCtx.X509_V_ERR_INVALID_EXTENSION;
                ret = 0;

                    ret = 0;

                    ctx.error = X509StoreCtx.X509_V_ERR_EC_KEY_EXPLICIT_PARAMS;
