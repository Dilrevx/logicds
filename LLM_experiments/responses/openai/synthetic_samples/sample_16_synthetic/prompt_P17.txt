    public static int ssl3_get_cert_verify(SSL s) {
        EVP_PKEY pkey = null;
        byte[] p;
        int al, ret = 0;
        long n;
        int type = 0, i = 0, j = 0;
        X509 peer;
        EVP_MD_CTX mctx = new EVP_MD_CTX();
        Helper.EVP_MD_CTX_init(mctx);

        boolean[] okHolder = new boolean[1];
        n = s.method.ssl_get_message(s,
                Constants.SSL3_ST_SR_CERT_VRFY_A,
                Constants.SSL3_ST_SR_CERT_VRFY_B,
                -1,
                Constants.SSL3_RT_MAX_PLAIN_LENGTH,
                okHolder);
        if (!okHolder[0]) {
            Helper.EVP_MD_CTX_cleanup(mctx);
            return (int)n;
        }

        if (s.session.peer != null) {
            peer = s.session.peer;
            pkey = Helper.X509_get_pubkey(peer);
            type = Helper.X509_certificate_type(peer, pkey);
        } else {
            peer = null;
            pkey = null;
        }

        // FIX: Ensure we have a valid public key.
        if (pkey == null) {
            Helper.EVP_MD_CTX_cleanup(mctx);
            s.method.raise_alert(s, Constants.SSL3_AD_HANDSHAKE_FAILURE);
            return -1;
        }

        // Ensure the message length is within the expected bounds.
        if (n < 0 || n > Constants.SSL3_RT_MAX_PLAIN_LENGTH) {
            Helper.EVP_MD_CTX_cleanup(mctx);
            s.method.raise_alert(s, Constants.SSL3_AD_DECODE_ERROR);
            return -1;
        }

        // Assume the received message bytes (which include the signature data)
        // are stored in s.init_msg (or use the proper buffer in your code).
        p = s.init_msg;

        // Determine the digest type based on the peer's certificate type.
        int mdType = Helper.ssl_get_cert_vfy_md(type);
        if (!Helper.EVP_DigestVerifyInit(mctx, null, mdType, null, pkey)) {
            Helper.EVP_MD_CTX_cleanup(mctx);
            s.method.raise_alert(s, Constants.SSL3_AD_HANDSHAKE_FAILURE);
            return -1;
        }

        // Update digest with the handshake data.
        if (!Helper.EVP_DigestVerifyUpdate(mctx, s.handshakeBuffer, s.handshakeBufferLen)) {
            Helper.EVP_MD_CTX_cleanup(mctx);
            s.method.raise_alert(s, Constants.SSL3_AD_HANDSHAKE_FAILURE);
            return -1;
        }

        // Verify the signature using the digest context.
        ret = Helper.EVP_DigestVerifyFinal(mctx, p, (int) n);
        if (ret <= 0) {
            Helper.EVP_MD_CTX_cleanup(mctx);
            s.method.raise_alert(s, Constants.SSL3_AD_HANDSHAKE_FAILURE);
            return -1;
        }

        Helper.EVP_MD_CTX_cleanup(mctx);
        return 1;
    }
