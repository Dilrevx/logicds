    public int ssl3_connect() {
        int ret = -1;
        switch (state.handshakeStage) {
            case 0:
                System.out.println("Client: Sending ClientHello");
                state.handshakeStage = HandshakeStates.SSL3_ST_CR_FINISHED_A;
                ret = 1;
                break;
            case HandshakeStates.SSL3_ST_CR_FINISHED_A:
            case HandshakeStates.SSL3_ST_CR_FINISHED_B:
            case HandshakeStates.SSL_ST_OK:
                // Safeguard: Prevent processing ChangeCipherSpec messages
                // to mitigate the CCS injection vulnerability.
                System.out.println("Client: In Finished state or handshake complete. Skipping CCS processing for security reasons.");
                return ret;
            default:
                return ret;
        }
        return ret;
    }
