    public static String getDnsKey() {
        String key = "";
        String envKey = Env.getenv("DNS_KEY");
        if (envKey != null && !envKey.isEmpty()) {
            key = envKey;
        } else if (Config.Server.getDnsKey() != null && !Config.Server.getDnsKey().isEmpty()) {
            key = Config.Server.getDnsKey();
        }
        if (key.isEmpty()) {
            throw new IllegalStateException("DNS key is not configured.");
        }
        return key;
    }
