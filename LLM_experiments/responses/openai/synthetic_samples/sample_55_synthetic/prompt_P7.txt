public static String getDnsKey() {
    // Retrieve the DNS key from the environment variable.
    String envKey = Env.getenv("DNS_KEY");
    if (envKey != null && !envKey.isEmpty()) {
        return envKey;
    }
    
    // Retrieve the DNS key from the configuration.
    String configKey = Config.Server.getDnsKey();
    if (configKey != null && !configKey.isEmpty()) {
        return configKey;
    }
    
    // If no key is provided from either source, throw an exception.
    throw new IllegalStateException("DNS API key is not configured.");
}
