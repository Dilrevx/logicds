public static String getDnsKey() {
    // First, try to get the DNS key from the environment variable safely.
    String envKey = Env.getenv("DNS_KEY");
    if (envKey != null && !envKey.trim().isEmpty()) {
        return envKey;
    }
    
    // Next, try to retrieve the DNS key from the configuration.
    String configKey = Config.Server.getDnsKey();
    if (configKey != null && !configKey.trim().isEmpty()) {
        return configKey;
    }
    
    // Fall back to a default key.
    return "secretkey";
}
