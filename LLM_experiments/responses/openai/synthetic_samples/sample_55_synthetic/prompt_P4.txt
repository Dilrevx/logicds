    public static String getDnsKey() {
        String key = null;
        String envKey = Env.getenv("DNS_KEY");
        String configKey = Config.Server.getDnsKey();
        
        if (envKey != null && !envKey.isEmpty()) {
            key = envKey;
        } else if (configKey != null && !configKey.isEmpty()) {
            key = configKey;
        }
        
        if (key == null) {
            throw new IllegalStateException("DNS key is not provided. Please set DNS_KEY environment variable or configure it in the server settings.");
        }
        
        return key;
    }
