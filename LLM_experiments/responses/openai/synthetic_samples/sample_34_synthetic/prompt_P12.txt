        static pj_uint32_t last_restart_time = 0;

        const pj_uint32_t MIN_RESTART_INTERVAL_MS = 5000;  // 5 seconds

        if (current_time - last_restart_time >= MIN_RESTART_INTERVAL_MS) {
            last_restart_time = current_time;
