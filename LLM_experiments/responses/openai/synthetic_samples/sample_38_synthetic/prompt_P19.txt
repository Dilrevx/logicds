    public int write_pdu(UniqueByteBuffer pdu) {
        logger.info(pdu.msg, pdu.N_bytes, "DL PDU (length %d)", pdu.N_bytes);

        Nas5gsMsg nas_msg = new Nas5gsMsg();

        if (nas_msg.unpack_outer_hdr(pdu) != SRSRAN_SUCCESS) {
            logger.error("Unable to unpack outer NAS header");
            return SRSRAN_ERROR;
        }

        // Capture the pdu contents once to avoid multiple fetches that may lead to different data views.
        byte[] pduData = pdu.get();

        switch (nas_msg.hdr.security_header_type) {
            case Nas5gsHdr.security_header_type_opts.plain_5gs_nas_message:
                // No integrity or confidentiality protection, pduData remains unchanged.
                break;
            case Nas5gsHdr.security_header_type_opts.integrity_protected:
                if (!integrity_check(pduData)) {
                    logger.error("Not handling NAS message with integrity check error");
                    return SRSRAN_ERROR;
                }
                break;
            case Nas5gsHdr.security_header_type_opts.integrity_protected_and_ciphered:
                if (!integrity_check(pduData)) {
                    logger.error("Not handling NAS message with integrity check error");
                    return SRSRAN_ERROR;
                } else {
                    // Perform decryption and update pdu with the decrypted data.
                    byte[] decrypted = cipher_decrypt(pduData);
                    if (decrypted == null) {
                        logger.error("Cipher decryption failed");
                        return SRSRAN_ERROR;
                    }
                    // Update the UniqueByteBuffer with decrypted data.
                    pdu.set(decrypted);
                    // Update our local pduData reference for any subsequent validation, if needed.
                    pduData = decrypted;
                }
                break;
            case Nas5gsHdr.security_header_type_opts.integrity_protected_with_new_5G_nas_context:
                // Optionally, handle the new NAS context case or leave it unmodified.
                break;
            case Nas5gsHdr.security_header_type_opts.integrity_protected_and_ciphered_with_new_5G_nas_context:
                logger.error("Not handling NAS message with integrity protection & ciphering using new 5G NAS context");
                return SRSRAN_ERROR;
            default:
                logger.error("Not handling NAS message with unknown security header");
                return SRSRAN_ERROR;
        }

        if (pcap != null) {
            pcap.write_nas(pdu.msg, pdu.N_bytes);
        }

        logger.info(pdu.msg, pdu.N_bytes, "Decrypted DL PDU (length %d)", pdu.N_bytes);

        if (nas_msg.unpack(pdu) != SRSRAN_SUCCESS) {
            logger.error("Unable to unpack complete NAS pdu");
            return SRSRAN_ERROR;
        }

        return SRSRAN_SUCCESS;
    }
