public class RrcSecurityParameters
{
    public CIPHERING_ALGORITHM_ID_ENUM cipher_algo;
    public INTEGRITY_ALGORITHM_ID_ENUM integ_algo;

    public RrcSecurityParameters(CIPHERING_ALGORITHM_ID_ENUM cipher, INTEGRITY_ALGORITHM_ID_ENUM integrity)
    {
        // Reject use of plain NIA0 which bypasses RRC security activation.
        if (cipher == CIPHERING_ALGORITHM_ID_ENUM.NIA0_plain || integrity == INTEGRITY_ALGORITHM_ID_ENUM.NIA0_plain)
        {
            throw new SecurityException("Plain NIA0 algorithms are not permitted for secure RRC operation.");
        }

        cipher_algo = cipher;
        integ_algo = integrity;
    }
}
</repair>
