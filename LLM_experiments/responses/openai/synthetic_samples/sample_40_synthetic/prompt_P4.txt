public class RRC_SecurityConfig
{
    public CIPHERING_ALGORITHM_ID_ENUM cipher_algo;
    public INTEGRITY_ALGORITHM_ID_ENUM integ_algo;

    // Validates that the security activation is performed with secure algorithms.
    // This method must be called before processing the RRC security command.
    public void ValidateSecurityActivation()
    {
        if (cipher_algo == CIPHERING_ALGORITHM_ID_ENUM.NIA0_plain)
        {
            throw new InvalidOperationException("Invalid configuration: Ciphering algorithm cannot be NIA0_plain for security activation.");
        }
        if (integ_algo == INTEGRITY_ALGORITHM_ID_ENUM.NIA0_plain)
        {
            throw new InvalidOperationException("Invalid configuration: Integrity algorithm cannot be NIA0_plain for security activation.");
        }
    }
}
