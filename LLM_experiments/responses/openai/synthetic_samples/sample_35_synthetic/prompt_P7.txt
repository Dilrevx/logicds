    // Retrieve the provider value from the system property.
    String adminProviderValue = ADMIN_PROVIDER.getValue();

    // Define a whitelist of allowed provider values.
    Set<String> allowedProviders = new HashSet<>();
    allowedProviders.add("LOCAL");
    allowedProviders.add("CORP");
    allowedProviders.add("DEFAULT");

    // Validate the retrieved value against the allowed providers.
    if (!allowedProviders.contains(adminProviderValue)) {
        log.error("Unauthorized or invalid ADMIN_PROVIDER value detected: " + adminProviderValue);
        throw new SecurityException("Invalid admin provider configuration");
    }

    // The value is validated, so it's safe to initialize the provider.
    initProvider(adminProviderValue);
