// Retrieve the provider value from secure configuration
String provider = ADMIN_PROVIDER.getValue();

// Ensure that the value for the admin provider is one of the explicitly allowed values.
// This whitelist prevents an attacker from injecting an arbitrary provider value via
// the admin.authorizedJIDs system property.
if (!"InternalAdminProvider".equals(provider) && !"BackupAdminProvider".equals(provider)) {
    throw new SecurityException("Unauthorized admin provider value: " + provider);
}

initProvider(provider);
