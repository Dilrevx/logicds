    // Initialize admin provider preserving existing functionality
    initProvider(ADMIN_PROVIDER.getValue());

    // Register a listener for user deletion events using the event dispatcher mechanism.
    // This listener ensures that if an admin-related user account is deleted, any lingering
    // administrative privileges are revoked, thereby mitigating privilege escalation risks.
    EventDispatcher.registerListener(UserEvent.USER_DELETED, new UserDeletionListener() {
        @Override
        public void onUserDeleted(User user) {
            // Verify that the user is not null to prevent null pointer exceptions
            if (user != null) {
                // Revoke any admin privileges associated with this user account
                revokeAdminPrivileges(user);
            }
        }
    });
