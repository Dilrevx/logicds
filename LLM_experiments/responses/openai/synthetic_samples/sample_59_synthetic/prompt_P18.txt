public VerifyResult verifyCaptcha(Context ctx, String key, String captcha) {
    // Validate input parameters to prevent null dereferences.
    if (key == null || captcha == null) {
        return new VerifyResult(false, "Key or captcha must not be null.");
    }
    
    // Fetch the expected captcha value from a secure captcha repository/service.
    // (This could be a cache, database or in-memory store.)
    String expectedCaptcha = CaptchaService.getCaptchaByKey(key);
    if (expectedCaptcha == null) {
        return new VerifyResult(false, "Captcha not found or already expired.");
    }
    
    // To prevent replay attacks, remove the captcha immediately regardless of validation result.
    CaptchaService.removeCaptchaByKey(key);

    // Use constant time comparison to prevent timing attacks.
    if (constantTimeEquals(expectedCaptcha, captcha)) {
        return new VerifyResult(true, "Captcha verified successfully.");
    } else {
        return new VerifyResult(false, "Captcha does not match.");
    }
}

// Helper method for constant time string comparison.
private boolean constantTimeEquals(String a, String b) {
    if (a == null || b == null || a.length() != b.length()) {
        return false;
    }
    int result = 0;
    for (int i = 0; i < a.length(); i++) {
        result |= a.charAt(i) ^ b.charAt(i);
    }
    return result == 0;
}
