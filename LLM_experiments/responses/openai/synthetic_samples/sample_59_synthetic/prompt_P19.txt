public VerifyResult verifyCaptcha(Context ctx, String key, String captcha) {
    // Validate inputs to ensure they are not null.
    if (key == null || captcha == null) {
        return VerifyResult.failure("Invalid captcha input.");
    }
    
    // Retrieve the stored captcha value securely using the provided key.
    // Assuming that ctx.getCaptchaValue(key) is the safe method to fetch the captcha.
    String storedCaptcha = ctx.getCaptchaValue(key);
    
    // Invalidate or remove the captcha entry immediately to enforce one-time use.
    ctx.removeCaptchaValue(key);
    
    // If the stored captcha is missing (expired or never set), reject the verification.
    if (storedCaptcha == null) {
        return VerifyResult.failure("Captcha has expired or is invalid.");
    }
    
    // Compare the provided captcha with the stored value using a constant-time
    // comparison to mitigate timing attacks.
    if (!constantTimeEquals(storedCaptcha, captcha)) {
        return VerifyResult.failure("Captcha does not match.");
    }
    
    // If all checks pass, the captcha is verified.
    return VerifyResult.success("Captcha verified successfully.");
}

// Utility method to perform constant-time comparison of two strings.
private boolean constantTimeEquals(String a, String b) {
    // If lengths differ, they cannot be equal.
    if (a.length() != b.length()) {
        return false;
    }
    
    byte[] aBytes = a.getBytes(java.nio.charset.StandardCharsets.UTF_8);
    byte[] bBytes = b.getBytes(java.nio.charset.StandardCharsets.UTF_8);
    
    // Use MessageDigest.isEqual for constant-time comparison.
    return java.security.MessageDigest.isEqual(aBytes, bBytes);
}
