public VerifyResult verifyCaptcha(Context ctx, String key, String captcha) {
    CaptchaRepo.CaptchaResult fetched = captchaRepo.getCaptcha(ctx, key);
    if (fetched.err != null || fetched.value == null) {
        return new VerifyResult(false, null);
    }
    // Invalidate the captcha immediately to prevent replay attacks
    captchaRepo.removeCaptcha(ctx, key);
    boolean isValid = trimSpace(captcha).equals(fetched.value);
    return new VerifyResult(isValid, null);
}
