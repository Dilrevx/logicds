    public VerifyResult verifyCaptcha(Context ctx, String key, String captcha) {
        // Validate inputs to ensure they are not null.
        if (key == null || captcha == null) {
            return VerifyResult.FAILURE;
        }

        // Retrieve the expected captcha value from a secure store.
        // Assume getCaptchaFromStore returns an object or null if the captcha was not found
        // (e.g. captcha expired or already validated).
        Captcha expectedCaptcha = getCaptchaFromStore(ctx, key);
        if (expectedCaptcha == null || expectedCaptcha.getAnswer() == null) {
            return VerifyResult.EXPIRED;
        }

        // Convert both the expected captcha answer and the provided captcha into UTF-8 encoded byte arrays.
        // This normalization also promotes a constant-time comparison.
        byte[] expectedBytes = expectedCaptcha.getAnswer().trim().getBytes(StandardCharsets.UTF_8);
        byte[] providedBytes = captcha.trim().getBytes(StandardCharsets.UTF_8);

        // If lengths differ, reject immediately.
        if (expectedBytes.length != providedBytes.length) {
            return VerifyResult.FAILURE;
        }

        // Use a constant-time comparison method to mitigate timing-based side channel attacks.
        if (!MessageDigest.isEqual(expectedBytes, providedBytes)) {
            return VerifyResult.FAILURE;
        }

        // Invalidate the captcha once it has been successfully verified to prevent replay attacks.
        removeCaptchaFromStore(ctx, key);

        return VerifyResult.SUCCESS;
    }
