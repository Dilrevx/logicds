import java.nio.charset.StandardCharsets;
import java.security.MessageDigest;

// Retrieve the captcha data from the repository.
CaptchaRepo.CaptchaResult fetched = captchaRepo.getCaptcha(ctx, key);
if (fetched.err != null || fetched.value == null) {
    // Return failure if there was an error fetching the captcha or if the captcha value is missing.
    return new VerifyResult(false, null);
}
if (captcha == null) {
    // Return failure if the user-supplied captcha is null.
    return new VerifyResult(false, null);
}

String realCaptcha = fetched.value;
// Instead of using a plain equals() comparison, use a constant-time comparison to protect against timing attacks.
boolean valid = MessageDigest.isEqual(
        trimSpace(captcha).getBytes(StandardCharsets.UTF_8),
        realCaptcha.getBytes(StandardCharsets.UTF_8)
);

return new VerifyResult(valid, null);
