    public static void recv_msg_userauth_success() {
        Logger.trace("received msg_userauth_success");
        
        // Verify that the session is in the correct state to process a userauth success.
        // For example, we expect that a userauth request was sent.
        if (cli_ses.state != ClientSession.USERAUTH_REQUEST_SENT) {
            Logger.error("Unexpected msg_userauth_success received when session state is: " + cli_ses.state);
            return;
        }
        
        // Proceed only if the session’s authentication is indeed pending.
        if (ses == null || ses.authstate == null || ses.authstate.authdone) {
            Logger.error("Session or authstate is invalid, or authentication already marked done.");
            return;
        }
        
        // Update authentication state
        ses.authstate.authdone = true;
        cli_ses.state = ClientSession.USERAUTH_SUCCESS_RCVD;
        cli_ses.lastauthtype = ClientSession.AUTH_TYPE_NONE;
        
        // Clean up any resources related to public key authentication.
        ClientSession.cli_auth_pubkey_cleanup();
    }
