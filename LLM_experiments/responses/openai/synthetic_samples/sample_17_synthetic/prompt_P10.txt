    public static void recv_msg_userauth_success() {
        Logger.trace("received msg_userauth_success");

        // Ensure a valid and expected authentication method was used.
        if (cli_ses.lastauthtype == ClientSession.AUTH_TYPE_NONE) {
            Logger.error("Authentication reported as successful without a valid auth method. " +
                         "Possible server attack detected. Aborting connection.");
            throw new RuntimeException("Protocol error: no valid authentication method in use.");
        }

        // For multifactor scenarios, verify that all required methods (e.g. FIDO2 or SSH-Askpass) are complete.
        // This should match the client’s expectation before accepting the authentication success.
        if (!ses.authstate.hasCompletedRequiredMethods()) {
            Logger.error("Authentication success received but required multifactor steps were not completed. " +
                         "Aborting connection to prevent potential agent abuse.");
            throw new RuntimeException("Protocol error: incomplete multifactor authentication.");
        }

        // If all checks pass, mark the session as authenticated.
        ses.authstate.authdone = true;
        cli_ses.state = ClientSession.USERAUTH_SUCCESS_RCVD;
        // Retain the lastauthtype value instead of overwriting it, so subsequent security checks remain valid.
        // Clean up any temporary state related to public key authentication.
        ClientSession.cli_auth_pubkey_cleanup();
    }
