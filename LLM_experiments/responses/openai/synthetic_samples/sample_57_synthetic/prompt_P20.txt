    public boolean readFromSections(String user, String collectionUrl, String permission) {

        Map<String, String> defaults = new HashMap<String, String>();
        defaults.put("login", sanitize(user));
        defaults.put("path", sanitize(collectionUrl));
        // Optionally include permission if further processing is needed

        }
    }

    // Utility method to sanitize input strings, removing problematic characters
    private String sanitize(String input) {
        if (input == null) {
            return "";
        }
        // Remove any newline characters and trim whitespace
        // This basic sanitization can be expanded based on application needs
        return input.replaceAll("[\\r\\n]", "").trim();
    }
