    public void serveHttp(HttpResponse w, HttpRequest r) {
        ConfigLoader.LoadResult cLoad = ConfigLoader.getConfig(tlsConfigPath);
        WebConfig c = cLoad.config;
        Exception err = cLoad.err;
        if (err != null) {
            // Log the error without leaking sensitive config details
            logger.log("msg", "Unable to parse configuration", "err", err);
            httpError(w, HttpStatus.statusText(HttpStatus.INTERNAL_SERVER_ERROR),
                    HttpStatus.INTERNAL_SERVER_ERROR);
            return;
        }

        // First, set any custom headers from the configuration.
        for (Map.Entry<String, String> e : c.getHttpConfig().header.entrySet()) {
            w.getHeaders().put(e.getKey(), e.getValue());
        }

        // ----------------------------------------------------------------------------
        // SECURITY FIX:
        // Instead of silently allowing access when no users are configured,
        // require authentication. Allowing an unauthenticated request when the
        // configuration file contains user-related settings may expose the
        // exporter endpoint to unintended public access.
        // ----------------------------------------------------------------------------
        if (c.getUsers().isEmpty()) {
            logger.log("msg", "Access denied: No users configured for authentication.");
            w.getHeaders().put("WWW-Authenticate", "Basic");
            httpError(w, HttpStatus.statusText(HttpStatus.UNAUTHORIZED), HttpStatus.UNAUTHORIZED);
            return;
        }

        // Process basic authentication.
        HttpRequest.BasicAuthResult ba = r.basicAuth();
        String user = ba.user;
        String pass = ba.pass;
        boolean authProvided = ba.present;
        if (authProvided) {
            String hashedPassword = c.getUsers().get(user);
            boolean validUser = hashedPassword != null;

            // Use a fake hash if no user match to mitigate timing attacks.
            if (!validUser) {
                hashedPassword = FAKE_HASH;
            }

            // Cache key includes username, hashed password and provided password.
            String cacheKey = hexEncode(concat(user, hashedPassword, pass));
            AuthCache.GetResult cached = cache.get(cacheKey);
            boolean authOk = cached.value;
            boolean cacheHit = cached.present;

            if (!cacheHit) {
                Exception bcryptErr;
                // Synchronized block to prevent concurrent expensive bcrypt operations.
                synchronized (bcryptMtx) {
                    bcryptErr = Bcrypt.compareHashAndPassword(
                            hashedPassword.getBytes(), pass.getBytes());
                }

                authOk = bcryptErr == null;
                cache.set(cacheKey, authOk);
            }

            if (authOk && validUser) {
                handler.serveHttp(w, r);
                return;
            }
        }

        // If authentication failed or was not present then return 401.
        w.getHeaders().put("WWW-Authenticate", "Basic");
        httpError(w, HttpStatus.statusText(HttpStatus.UNAUTHORIZED), HttpStatus.UNAUTHORIZED);
    }
