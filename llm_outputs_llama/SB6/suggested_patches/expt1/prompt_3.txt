import commons.*;

public class X509Verifier {

            if (cur.tag == MbedtlsConstants.MBEDTLS_ASN1_DNS) {
                String dns = new String(cur.data, 0, cur.len);
                if (dns.equals(cn)) {
                    return;
                }
            } else if (cur.tag == MbedtlsConstants.MBEDTLS_ASN1_URI) {
                String uri = new String(cur.data, 0, cur.len);
                if (uri.startsWith("https://" + cn + "/") || uri.equals("https://" + cn)) {
                    return;
                }
            }
            
        }
    }
