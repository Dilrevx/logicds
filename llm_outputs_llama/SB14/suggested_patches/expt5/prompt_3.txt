package main;

import commons.*;

public class NmHandleDetach {
    public static void isc__nmhandle_detach(HandleRef handlep) {
        Require.check(handlep != null);
        Require.check(HandleRef.valid(handlep));
        NmHandle handle = handlep.handle;
        handlep.handle = null;
        NmSocket sock = handle.sock;

        if (sock.tid == NodeManager.nmTid()) {
            NodeManager.nmhandle_detach_cb(handle);
            // Add a check to ensure the connection is properly closed
            if (sock.state == TCP_FIN_WAIT_1) {
                sock.state = TCP_CLOSED;
                // Send an ACK for the received FIN
                sendAck(sock);
            }
        } else {
            NetiEventDetach event =
                    NodeManager.getNetiEventDetach(sock.mgr, sock);
            event.handle = handle;
            EventPasser.passEvent(event);
            NodeManager.enqueueEvent(
                    sock.mgr.workers[sock.tid], event);
            // Add a check to ensure the connection is properly closed
            if (sock.state == TCP_FIN_WAIT_1) {
                sock.state = TCP_CLOSED;
                // Send an ACK for the received FIN
                sendAck(sock);
            }
        }
        // Add a timeout to ensure the connection is closed after a certain period
        if (sock.state == TCP_CLOSED) {
            setTimeout(sock, CLOSE_TIMEOUT);
        }

    }
}
