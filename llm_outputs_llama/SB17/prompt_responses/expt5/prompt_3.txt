        private void verifyCRLScope(X509CRL crl, X500Principal subject) {
            // Verify the issuer and scope of the complete CRL as per the specification
            if (crl.getExtensionValue("2.5.29.31")!= null) { // Check for issuing distribution point extension
                // Verify the distribution point name and the indirectCRL boolean
                verifyDistributionPoint(crl, subject);
            } else {
                // Verify that the CRL issuer matches the certificate issuer
                if (!crl.getIssuer().equals(subject)) {
                    throw new CertificateException("CRL issuer does not match certificate issuer");
                }
            }

            // Verify the onlyContainsUserCerts, onlyContainsCACerts, and onlyContainsAttributeCerts booleans
            verifyOnlyContains(crl);
        }

        private void verifyDistributionPoint(X509CRL crl, X500Principal subject) {
            // Verify the distribution point name and the indirectCRL boolean
            if (crl.getExtensionValue("2.5.29.31.1")!= null) { // Check for distribution point name
                // Verify that one of the names in the IDP matches one of the names in the DP
                verifyDistributionPointName(crl, subject);
            }

            // Verify the onlyContainsUserCerts, onlyContainsCACerts, and onlyContainsAttributeCerts booleans
            verifyOnlyContains(crl);
        }

        private void verifyOnlyContains(X509CRL crl) {
            // Verify the onlyContainsUserCerts boolean
            if (crl.getExtensionValue("2.5.29.31.2")!= null) { // Check for onlyContainsUserCerts
                // Verify that the certificate does not include the basic constraints extension with the cA boolean asserted
                if (crl.getExtensionValue("2.5.29.19")!= null) { // Check for basic constraints extension
                    throw new CertificateException("Certificate includes basic constraints extension with cA boolean asserted");
                }
            }

            // Verify the onlyContainsCACerts boolean
            if (crl.getExtensionValue("2.5.29.31.3")!= null) { // Check for onlyContainsCACerts
                // Verify that the certificate includes the basic constraints extension with the cA boolean asserted
                if (crl.getExtensionValue("2.5.29.19") == null) { // Check for basic constraints extension
                    throw new CertificateException("Certificate does not include basic constraints extension with cA boolean asserted");
                }
            }

            // Verify that the onlyContainsAttributeCerts boolean is not asserted
            if (crl.getExtensionValue("2.5.29.31.4")!= null) { // Check for onlyContainsAttributeCerts
                throw new CertificateException("onlyContainsAttributeCerts boolean is asserted");
            }
