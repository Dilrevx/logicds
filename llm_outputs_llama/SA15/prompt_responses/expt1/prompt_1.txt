public static int eap_tlv_process(EapSm sm, EapPeapData data,
                                  EapMethodRet ret, WPABuf req, boolean force_failure) {
    int resp_status;
    int tlvStatus = WPA_GET_BE16(EAP_TLV_RESULT_SUCCESS);
    if (req == null || req.buf == null) {
        return EAP_FAIL;
    }
    if (req.used < 4) {
        return EAP_FAIL;
    }
    if (force_failure) {
        tlvStatus = WPA_GET_BE16(EAP_TLV_RESULT_FAILURE);
    }
    resp_status = eap_tlv_build(sm, data, ret, tlvStatus);
    return resp_status;
}
