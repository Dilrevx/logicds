// Unpack the outer NAS header
if (nas_msg.unpack_outer_hdr(pdu)!= SRSRAN_SUCCESS) {
    logger.error("Unable to unpack outer NAS header");
    return SRSRAN_ERROR;
}

// Validate the sequence number
if (nas_msg.hdr.seq_number!= expected_seq_number) {
    logger.error("Invalid sequence number");
    return SRSRAN_ERROR;
}

// Assign the current message header based on the security header type
Nas5gsHdr::security_header_type_opts current_message_header;

switch (nas_msg.hdr.security_header_type) {
    case Nas5gsHdr.security_header_type_opts.plain_5gs_nas_message:
        // Check if a security context has already been established
        if (security_context_exists) {
            logger.error("Plain NAS message received after security context establishment");
            return SRSRAN_ERROR;
        }
        current_message_header = nas_msg.hdr.security_header_type;
        break;
    case Nas5gsHdr.security_header_type_opts.integrity_protected:
        if (!integrity_check(pdu.get())) {
            logger.error("Integrity check failed");
            return SRSRAN_ERROR;
        }
        current_message_header = nas_msg.hdr.security_header_type;
        break;
    case Nas5gsHdr.security_header_type_opts.integrity_protected_and_ciphered:
        if (!integrity_check(pdu.get())) {
            logger.error("Integrity check failed");
            return SRSRAN_ERROR;
        } else {
            cipher_decrypt(pdu.get());
        }
        current_message_header = nas_msg.hdr.security_header_type;
        break;
    case Nas5gsHdr.security_header_type_opts.integrity_protected_with_new_5G_nas_context:
        current_message_header = nas_msg.hdr.security_header_type;
        break;
    case Nas5gsHdr.security_header_type_opts.integrity_protected_and_ciphered_with_new_5G_nas_context:
        return SRSRAN_ERROR;
    default:
        logger.error("Unknown security header type");
        return SRSRAN_ERROR;
}
