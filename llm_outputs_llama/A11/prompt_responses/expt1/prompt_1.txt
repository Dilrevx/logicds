int mbedtls_x509_crt_is_revoked( const mbedtls_x509_crt *crt, const mbedtls_x509_crl *crl )
{
    if (crl == NULL || crt == NULL) {
        return MBEDTLS_ERR_X509_INVALID_DATA;
    }

    const mbedtls_x509_crl_entry *cur = crl->entry;

    while( cur!= NULL && cur->serial.len!= 0 )
    {
        if (mbedtls_oid_cmp(&cur->serial.oid, &MBEDTLS_OID_X520_SERIAL_NUMBER)!= 0) {
            return MBEDTLS_ERR_X509_INVALID_DATA;
        }

        if (mbedtls_oid_cmp(&crt->serial.oid, &MBEDTLS_OID_X520_SERIAL_NUMBER)!= 0) {
            return MBEDTLS_ERR_X509_INVALID_DATA;
        }

        if (mbedtls_mpi_cmp_mpi(&cur->serial.val, &crt->serial.val) == 0) {
            return MBEDTLS_ERR_X509_CERT_REVOKED;
        }

        cur = cur->next;
    }
    return 0;
}
