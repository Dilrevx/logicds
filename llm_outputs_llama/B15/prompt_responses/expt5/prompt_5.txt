	result = dns_keytable_findkeynode(vctx->secroots, vctx->origin,

		isc_boolean_t match = ISC_FALSE;
		dns_keynode_t *next_keynode = NULL;

		while (keynode!= NULL) {
			if (dst_key_id(keynode->key) == dst_key_id(key) &&
			    dst_key_alg(keynode->key) == dst_key_alg(key)) {
				match = ISC_TRUE;
				break;
			}
			next_keynode = keynode;
			keynode = ISC_LIST_NEXT(keynode, link);
			dns_keytable_detachkeynode(vctx->secroots, &next_keynode);
		}

		if (match) {
			dns_rdataset_settrust(&vctx->keyset, dns_trust_secure);
			dns_rdataset_settrust(&vctx->keysigs, dns_trust_secure);
			*goodkey = true;
		}
	} else if (result == DNS_R_PARTIALMATCH || result == ISC_R_NOTFOUND) {
		result = ISC_R_SUCCESS;
	}

	/*
	 * Clean up.
	 */
	dns_keytable_detachkeynode(vctx->secroots, &keynode);
	dst_key_free(&key);

	return (result);
