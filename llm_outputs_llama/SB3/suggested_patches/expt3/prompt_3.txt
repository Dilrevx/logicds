package main;

import commons.Logger;
import commons.Session;
import commons.ClientSession;
import commons.CliOptions;

public class AuthHandler {
    public static Session ses = new Session();
    public static ClientSession cli_ses = new ClientSession();
    public static CliOptions cli_opts = new CliOptions();

    public static void recv_msg_userauth_success() {

Logger.trace("received msg_userauth_success");
ses.authstate.authdone = true;
cli_ses.state = ClientSession.USERAUTH_SUCCESS_RCVD;

// Parse the SSH_MSG_USERAUTH_FAILURE message
byte[] msg = ses.in_buffer.get_msg_userauth_failure();
String[] authMethods = msg.get_auth_methods();

// Store the supported authentication methods
cli_ses.supportedAuthMethods = authMethods;

// Get the last authentication method used
String lastAuthMethod = cli_ses.lastauthtype;

// Check if the last authentication method is supported by the server
if (Arrays.asList(cli_ses.supportedAuthMethods).contains(lastAuthMethod)) {
    // Set the last authentication type to NONE only if the method is supported
    cli_ses.lastauthtype = ClientSession.AUTH_TYPE_NONE;
} else {
    // Handle the case where the authentication method is not supported
    Logger.error("Unsupported authentication method: " + lastAuthMethod);
    // You can add additional error handling or logging here
}

ClientSession.cli_auth_pubkey_cleanup();

    }
}
