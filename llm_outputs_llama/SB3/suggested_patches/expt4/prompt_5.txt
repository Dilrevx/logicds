package main;

import commons.Logger;
import commons.Session;
import commons.ClientSession;
import commons.CliOptions;

public class AuthHandler {
    public static Session ses = new Session();
    public static ClientSession cli_ses = new ClientSession();
    public static CliOptions cli_opts = new CliOptions();


    public static void recv_msg_userauth_success() {
        Logger.trace("received msg_userauth_success");
        if (ses.authstate.authdone) {
            Logger.trace("Ignoring duplicate msg_userauth_success");
            return;
        }
        ses.authstate.authdone = true;
        cli_ses.state = ClientSession.USERAUTH_SUCCESS_RCVD;
        cli_ses.lastauthtype = ClientSession.AUTH_TYPE_NONE;
        ClientSession.cli_auth_pubkey_cleanup();
        // Verify the authentication method used
        if (!isRFCCompliantAuthMethod(cli_ses.lastauthtype)) {
            Logger.error("Non-RFC-compliant authentication method used");
            // Disconnect or take other security measures
            cli_ses.disconnect();
        }
    }

    private static boolean isRFCCompliantAuthMethod(int authType) {
        // Implement the logic to check if the authentication method is RFC-compliant
        // For example:
        return authType == ClientSession.AUTH_TYPE_PUBLICKEY || authType == ClientSession.AUTH_TYPE_PASSWORD;
    }

}
