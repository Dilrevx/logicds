int gnutls_ocsp_resp_check_crt(gnutls_ocsp_resp_t resp, gnutls_x509_crt_t crt)
{
    unsigned char *cserial;
    size_t cserial_size;
    int ret;

    /* Retrieve the serial number from the certificate */
    ret = gnutls_x509_crt_get_serial(crt, &cserial, &cserial_size);
    if (ret < 0) {
        return ret;
    }

    cserial_size = gnutls_x509_crt_get_serial_size(crt);

    /* Compare the serial number from the certificate with the serial number from the OCSP response */
    unsigned char *oserial;
    size_t oserial_size;
    ret = gnutls_ocsp_resp_get_serial(resp, &oserial, &oserial_size);
    if (ret < 0) {
        return ret;
    }

    if (cserial_size!= oserial_size || memcmp(cserial, oserial, cserial_size)!= 0) {
        /* Return an error indicating an OCSP response error */
        return GNUTLS_E_OCSP_RESPONSE_ERROR;
    }

    return 0;
}
