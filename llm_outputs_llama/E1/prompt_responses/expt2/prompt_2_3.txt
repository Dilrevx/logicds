    break;
  case nas_5gs_hdr::security_header_type_opts::integrity_protected:
    if (sequence_number_check(pdu.get()) == false) {
      logger.error("Not handling NAS message with sequence number error");
      return SRSRAN_ERROR;
    } else if (integrity_check(pdu.get()) == false) {
      logger.error("Not handling NAS message with integrity check error");
      return SRSRAN_ERROR;
    }
    current_message_header = nas_msg.hdr.security_header_type;
    break;
  case nas_5gs_hdr::security_header_type_opts::integrity_protected_and_ciphered:
    if (sequence_number_check(pdu.get()) == false) {
      logger.error("Not handling NAS message with sequence number error");
      return SRSRAN_ERROR;
    } else if (integrity_check(pdu.get()) == false) {
      logger.error("Not handling NAS message with integrity check error");
      return SRSRAN_ERROR;
    } else {

    }

    break;

    current_message_header = nas_msg.hdr.security_header_type;
    break;
