// Check if a security context has already been established
if (has_sec_ctxt) {
  // Reject plain NAS messages after security activation
  if (nas_msg.hdr.security_header_type == nas_5gs_hdr::security_header_type_opts::plain_5gs_nas_message) {
    logger.error("Rejecting plain NAS message after security activation");
    return SRSRAN_ERROR;
  }
}

switch (nas_msg.hdr.security_header_type) {
  case nas_5gs_hdr::security_header_type_opts::plain_5gs_nas_message:
    // This case should not be reached if has_sec_ctxt is true
    break;
  case nas_5gs_hdr::security_header_type_opts::integrity_protected:
    // Sequence number check
    if (seq_num_check(pdu.get()) == false) {
      logger.error("Sequence number check failed");
      return SRSRAN_ERROR;
    }
    // Integrity check
    if (integrity_check(pdu.get()) == false) {
      logger.error("Not handling NAS message with integrity check error");
      return SRSRAN_ERROR;
    }
    break;
  case nas_5gs_hdr::security_header_type_opts::integrity_protected_and_ciphered:
    // Sequence number check
    if (seq_num_check(pdu.get()) == false) {
      logger.error("Sequence number check failed");
      return SRSRAN_ERROR;
    }
    // Integrity check
    if (integrity_check(pdu.get()) == false) {
      logger.error("Not handling NAS message with integrity check error");
      return SRSRAN_ERROR;
    } else {
      // Decryption
      cipher_decrypt(pdu.get());
    }
    break;
  case nas_5gs_hdr::security_header_type_opts::integrity_protected_with_new_5G_nas_context:
    // Handle new 5G NAS context
    break;
  case nas_5gs_hdr::security_header_type_opts::integrity_protected_and_ciphered_with_new_5G_nas_context:
    return SRSRAN_ERROR;
  default:
    logger.error("Not handling NAS message with unknown security header");
    break;
}

// Assign current message header
current_message_header = nas_msg.hdr.security_header_type;
