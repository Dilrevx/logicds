public class X509CertVerifier {

        for (i = 0; i < num; i++) {
            xtmp = ctx.chain.get(i);
            if (xtmp == null) {
                Helper.raiseError("X509_R_CERT_UNTRUSTED");
                return -1;
            }
            if (Helper.cert_self_signed(xtmp)) {
                if (i == 0) {
                    ok = 1;
                    break;
                } else {
                    Helper.raiseError("X509_R_SELF_SIGNED_CERT_IN_CHAIN");
                    return -1;
                }
            }
            if (i == 0) continue;
            xtmp2 = ctx.chain.get(i - 1);
            if (xtmp2 == null) {
                Helper.raiseError("X509_R_CERT_UNTRUSTED");
                return -1;
            }
            if (!Helper.check_issued(ctx, xtmp, xtmp2)) {
                Helper.raiseError("X509_R_CERT_NOT_YET_VALID");
                return -1;
            }
            if (!Helper.check_revocation(ctx, xtmp)) {
                Helper.raiseError("X509_R_CERT_REVOKED");
                return -1;
            }
            if (!Helper.check_cert(ctx, xtmp)) {
                Helper.raiseError("X509_R_CERT_NOT_YET_VALID");
                return -1;
            }
        }
        if (ok == 0) {
            Helper.raiseError("X509_R_NO_VALID_SCTS");
            return -1;
        }
        return 1;
    }

}
