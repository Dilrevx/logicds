package main;

import commons.*;
public class X509Verifier {

        int ret = X509Utils.x509_verify_cert_valid(ctx, leaf, null);
        if (ret!= X509Constants.X509_V_OK) {
            ctx.error = ret;
            ctx.error_depth = depth;
            return 0;
        }
        if (!X509Utils.x509_verify_cert_hostname(ctx, leaf, name)) {
            ctx.error = X509Constants.X509_V_ERR_HOSTNAME_MISMATCH;
            ctx.error_depth = depth;
            return 0;
        }
        if ((ctx.chains_count < ctx.max_chains? (ctx.chains[ctx.chains_count] = X509Utils.x509_verify_chain_dup(chain)) : null) == null) {
            ctx.error = X509Constants.X509_V_ERR_OUT_OF_MEM;
            ctx.error_depth = depth;
            return 0;
        }
        ctx.chains_count++;
        ctx.error = X509Constants.X509_V_OK;
        ctx.error_depth = depth;
        return 1;

}
