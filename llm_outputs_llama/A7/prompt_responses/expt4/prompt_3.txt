static int check_ca_chain(X509_STORE_CTX *ctx, STACK_OF(X509) *chain)
{
    X509 *cert;
    int i;

    for (i = 0; i < sk_X509_num(chain); i++) {
        cert = sk_X509_value(chain, i);
        if (!X509_check_ca(cert)) {
            return 0; /* Not a CA chain */
        }
    }
    return 1; /* Valid CA chain */
}

static int X509_check_ca(X509 *cert)
{
    /* Check if the certificate has the basicConstraints extension with cA set to TRUE */
    X509_EXTENSION *ext;
    ASN1_BOOLEAN *ca;

    ext = X509_get_ext(d2i_X509(cert), NID_basic_constraints);
    if (ext == NULL) {
        return 0; /* No basicConstraints extension */
    }

    ca = (ASN1_BOOLEAN *)X509V3_EXT_d2i(ext);
    if (ca == NULL ||!ca->data[0]) {
        return 0; /* cA is not set to TRUE */
    }

    return 1; /* cA is set to TRUE */
}
