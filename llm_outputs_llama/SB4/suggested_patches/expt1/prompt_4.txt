package main;

import commons.*;

public class OcspRespChecker {

        if (ret < 0) {
            return ret;
        }

        if (rserial.size!= cserial.size
                ||!Helper.memcmp(cserial.data, rserial.data, rserial.size)) {
            return ErrorCodes.GNUTLS_E_OCSP_RESPONSE_ERROR;
        }

        ret = Helper.gnutls_x509_crt_get_dn(crt, dn);
        if (ret < 0) {
            return ret;
        }

        ret = Helper.gnutls_x509_dn_get_rdn_raw(dn, 0, rdn_hash);
        if (ret < 0) {
            return ret;
        }

        ret = Helper.gnutls_hash(digest, rdn_hash.data, rdn_hash.size, cdn_hash);
        if (ret < 0) {
            return ret;
        }

        if (hash_len!= rdn_hash.size ||!Helper.memcmp(cdn_hash, rdn_hash.data, hash_len)) {
            return ErrorCodes.GNUTLS_E_OCSP_RESPONSE_ERROR;
        }

        return 0;
    }

}
