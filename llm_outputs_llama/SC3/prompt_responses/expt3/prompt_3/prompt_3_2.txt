private boolean isEmptyAclFile(String aclFile) {
    try (BufferedReader br = new BufferedReader(new FileReader(aclFile))) {
        String line;
        while ((line = br.readLine())!= null) {
            line = line.trim();
            if (!line.isEmpty() &&!line.startsWith("#")) {
                return false; 
            }
        }
        return true; 
    } catch (IOException e) {
        return true; 
    }
}
