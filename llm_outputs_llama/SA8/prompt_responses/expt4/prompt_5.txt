    public int selectMethods() {
        String[] clientMethods = client.getMethods();
        String[] serverMethods = server.getMethods();
        String extStart;

        // Check if the client supports the extension
        if (clientMethods[SSHConstants.SSH_KEX].contains(SSHConstants.KEX_EXTENSION_CLIENT)) {
            // Extract the extension part
            extStart = findSubstring(clientMethods[SSHConstants.SSH_KEX], "," + SSHConstants.KEX_EXTENSION_CLIENT);
            if (extStart!= null) {
                // Remove the extension part from the client methods
                clientMethods[SSHConstants.SSH_KEX] = truncateString(clientMethods[SSHConstants.SSH_KEX], extStart);
            }
        }

        // Iterate over the methods and find a match
        for (int i = 0; i < SSHConstants.SSH_KEX_METHODS; i++) {
            String clientMethod = clientMethods[i];
            String serverMethod = serverMethods[i];

            // Check if the client method is an AEAD method
            if (isAeadMethod(clientMethod)) {
                // Find a matching AEAD method on the server
                String aeadMethod = findAeadMethod(serverMethod);
                if (aeadMethod!= null) {
                    crypto.setKexMethod(i, aeadMethod);
                } else {
                    // If no matching AEAD method is found, return an error
                    return SSHConstants.SSH_ERROR;
                }
            } else {
                // Find a matching method on the server
                crypto.setKexMethod(i, findMatching(serverMethod, clientMethod));
            }

            // Check if a method was selected
            if (crypto.getKexMethod(i) == null && i < SSHConstants.SSH_LANG_C_S) {
                return SSHConstants.SSH_ERROR;
            } else if (i >= SSHConstants.SSH_LANG_C_S && crypto.getKexMethod(i) == null) {
                crypto.setKexMethod(i, "");
            }
        }
        return SSHConstants.SSH_OK;
    }

    private boolean isAeadMethod(String method) {
        return method.equals("chacha20-poly1305@openssh.com") || method.equals("aes128-gcm@openssh.com");
    }

    private String findAeadMethod(String methods) {
        String[] methodArray = methods.split(",");
        for (String method : methodArray) {
            if (isAeadMethod(method)) {
                return method;
            }
        }
        return null;
    }
