public int selectMethods() {
    String[] clientMethods = client.getMethods();
    String[] serverMethods = server.getMethods();
    String extStart;
    if (clientMethods == null || serverMethods == null) {
        throw new NullPointerException("Methods cannot be null");
    }
    if (clientMethods.length == 0 || serverMethods.length == 0) {
        throw new IllegalArgumentException("Methods cannot be empty");
    }
    for (String method : clientMethods) {
        if (method == null || method.isEmpty()) {
            throw new IllegalArgumentException("Method cannot be null or empty");
        }
    }
    for (String method : serverMethods) {
        if (method == null || method.isEmpty()) {
            throw new IllegalArgumentException("Method cannot be null or empty");
        }
    }
    extStart = Arrays.stream(clientMethods)
           .filter(serverMethods::contains)
           .findFirst()
           .orElseThrow(() -> new NoSuchElementException("No common methods found"));
    // Rest of the function remains the same
}
