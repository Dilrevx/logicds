package main;

import commons.*;

public class Tls12Builder {

    if (input == null || input.length < inSz) {
        return CommonsConstants.BAD_FUNC_ARG;
    }
    if (output == null || output.length < outSz) {
        return CommonsConstants.BAD_FUNC_ARG;
    }
    if (type < 0 || type > 255) {
        return CommonsConstants.BAD_FUNC_ARG;
    }
    if (hashOutput < 0 || hashOutput > 1) {
        return CommonsConstants.BAD_FUNC_ARG;
    }
    if (sizeOnly < 0 || sizeOnly > 1) {
        return CommonsConstants.BAD_FUNC_ARG;
    }
    if (asyncOkay < 0 || asyncOkay > 1) {
        return CommonsConstants.BAD_FUNC_ARG;
    }
    if (epochOrder < 0 || epochOrder > 1) {
        return CommonsConstants.BAD_FUNC_ARG;
    }
    
}
