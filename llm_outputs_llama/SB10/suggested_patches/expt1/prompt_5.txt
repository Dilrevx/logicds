package main;

import commons.*;

public class Tls12Builder {

if (args.ivSz > 0) {
    args.iv = new byte[args.ivSz];
    int ret = ssl.rng.generateBlock(args.iv, args.ivSz);
    if (ret!= 0) {
        return ret;
    }
    ssl.lastIv = args.iv;
    // Removed the code that was setting the IV to zero
    args.sz = inSz + args.ivSz;
}

}
