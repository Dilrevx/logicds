<repair>
    private AdminManager() {
        // Load an admin provider.
        initProvider(ADMIN_PROVIDER.getValue());
        
        // Add an event listener to the UserEventDispatcher
        UserEventDispatcher.addListener(new UserEventListener() {
            @Override
            public void userDeleting(User user) {
                // Remove the deleted user from the admin list
                removeAdminAccount(user.getUsername());
            }

            @Override
            public void userCreated(User user) {
                // No action needed for user creation
            }

            @Override
            public void userModified(User user) {
                // No action needed for user modification
            }
        });
    }
</repair>
