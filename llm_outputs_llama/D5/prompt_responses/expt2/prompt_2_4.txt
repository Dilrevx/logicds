    private AdminManager() {
        // Load an admin provider.
        initProvider(ADMIN_PROVIDER.getValue());

        // Add a UserEventDispatcher listener to monitor user events.
        UserEventDispatcher.addListener(new UserEventListener() {
            @Override
            public void userDeleting(User user) {
                // Remove the admin account when a user is deleted.
                if (isAdmin(user)) {
                    removeAdmin(user.getUsername());
                }
            }

            @Override
            public void userCreated(User user) {
                // No action required when a user is created.
            }

            @Override
            public void userModified(User user) {
                // No action required when a user is modified.
            }
        });
    }
</repair>
