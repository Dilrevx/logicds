    private AdminManager() {
        // Load an admin provider.
        Class<?> adminProviderClass = ADMIN_PROVIDER.getValue();
        if (adminProviderClass!= null &&!adminProviderClass.getName().startsWith("org.jivesoftware.openfire")) {
            Log.warn("Ignoring admin provider class '{}' as it is not from a trusted package", adminProviderClass.getName());
            adminProviderClass = DefaultAdminProvider.class;
        }
        initProvider(adminProviderClass);
    }
