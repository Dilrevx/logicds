private AdminManager() {
    // Get the authorized JIDs from the system property
    String authorizedJIDs = System.getProperty("admin.authorizedJIDs");

    // Validate and sanitize the input JIDs
    if (authorizedJIDs!= null) {
        String[] jids = authorizedJIDs.split(",");
        for (String jid : jids) {
            if (!isValidJID(jid)) {
                throw new SecurityException("Invalid JID: " + jid);
            }
        }
    }

    // Load an admin provider securely
    loadAdminProviderSecurely(ADMIN_PROVIDER.getValue());
}

private boolean isValidJID(String jid) {
    // Implement a regular expression to validate the JID format
    String jidPattern = "^[a-zA-Z0-9_-]+@[a-zA-Z0-9_-]+\\.[a-zA-Z]{2,}$";
    return jid.matches(jidPattern);
}
