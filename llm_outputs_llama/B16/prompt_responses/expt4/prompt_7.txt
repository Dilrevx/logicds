static struct frec *lookup_frec(unsigned short id, void *hash, struct sockaddr *source_addr, unsigned short source_port)
{
  struct frec *f;

  for(f = daemon->frec_list; f; f = f->next)
    if (f->sentto && f->new_id == id && 
	(!hash || memcmp(hash, f->hash, HASH_SIZE) == 0) &&
	(f->source_addr.ss_family == source_addr->sa_family) &&
	(memcmp(&f->source_addr, source_addr, 
		(source_addr->sa_family == AF_INET)? sizeof(struct sockaddr_in) : sizeof(struct sockaddr_in6)) == 0) &&
	(f->source_port == source_port))
      return f;
      
  return NULL;
}
