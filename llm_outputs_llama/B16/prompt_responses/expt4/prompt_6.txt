static struct frec *lookup_frec(unsigned short id, void *hash, int fd, int family)
{
  struct frec *f;

  for(f = daemon->frec_list; f; f = f->next)
    if (f->sentto && f->new_id == id && 
	(!hash || memcmp(hash, f->hash, HASH_SIZE) == 0) &&
	((family == AF_INET && f->rfd4 == fd) || (family == AF_INET6 && f->rfd6 == fd)) &&
	(f->sfd == fd))
      return f;
      
  return NULL;
}
