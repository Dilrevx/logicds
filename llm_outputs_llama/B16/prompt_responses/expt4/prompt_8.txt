static struct frec *lookup_frec(unsigned short id, void *hash, int fd, struct sockaddr *source)
{
  struct frec *f;

  for(f = daemon->frec_list; f; f = f->next)
    if (f->sentto && f->new_id == id && 
	(!hash || memcmp(hash, f->hash, HASH_SIZE) == 0) &&
	((f->family == AF_INET && f->rfd4 == fd && 
	  memcmp(&f->sfd4.sin_addr, &source->sin_addr, sizeof(struct in_addr)) == 0 &&
	  f->sfd4.sin_port == source->sin_port) ||
	 (f->family == AF_INET6 && f->rfd6 == fd && 
	  memcmp(&f->sfd6.sin6_addr, &source->sin6_addr, sizeof(struct in6_addr)) == 0 &&
	  f->sfd6.sin6_port == source->sin6_port)))
      return f;
      
  return NULL;
}
