static struct frec *lookup_frec(unsigned short id, void *hash, struct in_addr srcip, unsigned short srcport)
{
  struct frec *f;

  for(f = daemon->frec_list; f; f = f->next)
    if (f->sentto && f->new_id == id && 
	(!hash || memcmp(hash, f->hash, HASH_SIZE) == 0) &&
	(f->srcip.s_addr == srcip.s_addr) && (f->srcport == srcport))
      return f;
      
  return NULL;
}
