static struct frec *lookup_frec(unsigned short id, void *hash, struct in_addr *sentto_addr, unsigned short sentto_port)
{
  struct frec *f;

  for(f = daemon->frec_list; f; f = f->next)
    if (f->sentto && f->new_id == id && 
	(!hash || memcmp(hash, f->hash, HASH_SIZE) == 0) &&
	(f->sentto_addr.s_addr == sentto_addr->s_addr) &&
	(f->sentto_port == sentto_port))
      return f;
  return NULL;
}
