public static int checkChainExtensions(X509StoreCtx ctx) {
    int num = ctx.chain.size();
    int plen = 0;
    int proxy_path_length = 0;
    int must_be_ca = -1;
    int ret = 1;

    for (int i = 0; i < num; i++) {
        X509 x = ctx.chain.get(i);
        ret = Util.verifyCA();
        boolean strict = (ctx.param.flags & X509Flags.X509_V_FLAG_X509_STRICT)!= 0;

        if (x.getBasicConstraints()!= null && x.getBasicConstraints().isCA()) {
            must_be_ca = i;
        }

        if (x.getBasicConstraints()!= null && x.getBasicConstraints().getPathLengthConstraint()!= -1) {
            plen = x.getBasicConstraints().getPathLengthConstraint();
        }

        if (plen!= -1 && i > plen) {
            return 0;
        }

        if (x.getProxyCertInfoExtension()!= null) {
            proxy_path_length = x.getProxyCertInfoExtension().getPathLengthConstraint();
        }

        if (proxy_path_length!= -1 && i > proxy_path_length) {
            return 0;
        }

        if (i == num - 1 && must_be_ca == -1) {
            return 0;
        }
    }
    return ret;
}
