public static int mbedtls_x509_crt_is_revoked(
            MbedtlsX509Crt crt, MbedtlsX509Crl crl) {
    MbedtlsX509CrlEntry cur = crl.entry;
    while (cur!= null && cur.serial.len!= 0) {
        if (cur.serial.len == crt.serial.len && 
            memcmp(cur.serial.p, crt.serial.p, crt.serial.len) == 0) {
            return 1; // Certificate is revoked
        }
        cur = cur.next;
    }
    return 0; // Certificate is not revoked
}
