UserEventDispatcher.addListener(new UserEventListener() {
    @Override
    public void userDeleted(User user, Map<String, Object> params) {
        if (user!= null) {
            String username = user.getUsername();
            if (username!= null) {
                // Remove admin account associated with the deleted user's username
                ADMIN_PROVIDER.getValue().removeAdminAccount(username);
            }
        }
    }

    @Override
    public void userCreated(User user, Map<String, Object> params) {
        // No action required for user creation
    }

    @Override
    public void userModified(User user, Map<String, Object> params) {
        // No action required for user modification
    }
});
initProvider(ADMIN_PROVIDER.getValue());
