public static DnsResult nsXfrStart(DnsClient client, DnsZone zone, DnsRdatatype reqtype) {
    DnsResult result;
    String mnemonic;
    String questionName = zone.zoneName;
    String questionClass = "IN";
    boolean is_dlz = false;

    switch (reqtype) {
        case AXFR:
            mnemonic = "AXFR";
            break;
        case IXFR:
            mnemonic = "IXFR";
            break;
        default:
            throw new IllegalArgumentException("Unsupported request type");
    }

    if (client.view.zonetable == null) {
        throw new NullPointerException("Zone table is null");
    }

    if (questionName == null || questionName.isEmpty()) {
        throw new IllegalArgumentException("Question name is empty");
    }

    result = ZoneTableUtil.dnsZtFind(client.view.zonetable, questionName, 0, null, zone);

    if (result == null) {
        throw new NullPointerException("DNS result is null");
    }
    return result;
}
