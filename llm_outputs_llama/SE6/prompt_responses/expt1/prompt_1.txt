    public void write_pdu(int lcid, UniqueByteBuffer pdu) {
        if (pdu == null || pdu.N_bytes == 0) {
            logger.error("Invalid PDU provided");
            return;
        }

        int pd = 0;
        int msg_type = 0;
        int sec_hdr_type = 0;

        logger.info(pdu.msg, pdu.N_bytes, "DL %s PDU", rrc.getRbName(lcid));

        RefHolder pdHolder = new RefHolder();
        RefHolder secHdrHolder = new RefHolder();
        if (LiblteMme.parseMsgSecHeader(pdu.get(), pdHolder, secHdrHolder)!= 0) {
            logger.error("Failed to parse message security header");
            return;
        }
        sec_hdr_type = secHdrHolder.value;
    }
