package main;

import commons.*;

public class Tls13FinishedHandler {

    public static int DoTls13Finished(WOLFSSL ssl, byte[] input, int[] inOutIdx,
                                      int size, int totalSz, int sniff) {
        int ret;
        int finishedSz = 0;
        byte[] secret;
        byte[] mac = new byte[32];

        Tls13Helper.WOLFSSL_START(Constants.WC_FUNC_FINISHED_DO);
        Tls13Helper.WOLFSSL_ENTER("DoTls13Finished");

        if (inOutIdx[0] + size + ssl.keys.padSz > totalSz)
            return Constants.BUFFER_E;

        if (ssl.options & Constants.TLS_CLIENT && ssl.clientState == Constants.CONNECTED_AS_CLIENT && 
            ssl.verifyFlags & Constants.VERIFY_PEER_CERT && ssl.peerCert == null) {
            return Constants.BAD_CERTIFICATE;
        }

        if (ssl.options & Constants.TLS_SERVER && ssl.serverState == Constants.CONNECTED_AS_SERVER && 
            ssl.verifyFlags & Constants.VERIFY_PEER_CERT && ssl.peerCert == null) {
            return Constants.BAD_CERTIFICATE;
        }

}
