package main;

import commons.*;

public class Tls13FinishedHandler {

    public static int DoTls13Finished(WOLFSSL ssl, byte[] input, int[] inOutIdx,
                                      int size, int totalSz, int sniff) {
        int ret;
        int finishedSz = 0;
        byte[] secret;
        byte[] mac = new byte[32];

        if (input == null || inOutIdx == null || size < 0 || totalSz < 0) {
            return BAD_FUNC_ARG;
        }

        if (inOutIdx[0] + size > input.Length) {
            return BUFFER_E;
        }

        Tls13Helper.WOLFSSL_START(Constants.WC_FUNC_FINISHED_DO);
        ret = Tls13Helper.WOLFSSL_GET_SESSION(ssl, &session);
        if (ret!= 0) {
            return ret;
        }
        if (session == null) {
            return SSL_FATAL_ERROR;
        }
        secret = session.masterSecret;
        if (secret == null) {
            return SSL_FATAL_ERROR;
        }
        if (secret.Length!= Constants.WOLFSSL_MAX_MASTER_SECRET_SIZE) {
            return SSL_FATAL_ERROR;
        }
        ret = Tls13Helper.WOLFSSL_HMAC_SHA256(secret, 0, secret.Length, input, inOutIdx[0], size, mac);
        if (ret!= 0) {
            return ret;
        }
        finishedSz = size;
        inOutIdx[0] += size;
        return 0;
    }

}
