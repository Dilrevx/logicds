int mosquitto_acl_check_default(struct mosquitto_db *db, struct mosquitto *context, const char *topic, int access)
{
	char *local_acl;
	struct mosquitto__acl *acl_root;
	bool result;
	int i;
	int len, tlen, clen, ulen;
	char *s;
	struct mosquitto__security_options *security_opts = NULL;

	if(!db ||!context ||!topic) return MOSQ_ERR_INVAL;
	if(context->bridge) return MOSQ_ERR_SUCCESS;

	if(db->config->per_listener_settings){
		if(!context->listener) return MOSQ_ERR_ACL_DENIED;
		security_opts = &context->listener->security_options;
	}else{
		security_opts = &db->config->security_options;
	}

	if(security_opts == NULL) return MOSQ_ERR_ACL_DENIED;

	if(security_opts->acl_file){
		acl_root = mosquitto_acl_load(security_opts->acl_file);
		if(!acl_root) return MOSQ_ERR_ACL_DENIED;
	}

	len = strlen(topic);
	tlen = strlen(security_opts->topic);
	clen = strlen(security_opts->clientid);
	ulen = strlen(context->id);

	if(security_opts->acl_file){
		result = mosquitto_acl_check(acl_root, topic, access, context->id, context->username);
		mosquitto_acl_free(acl_root);
	}else{
		result = mosquitto_acl_check_default(security_opts, topic, access, context->id, context->username);
	}

	if(result) return MOSQ_ERR_SUCCESS;
	else return MOSQ_ERR_ACL_DENIED;
}
