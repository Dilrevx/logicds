int DoTls13Finished(WOLFSSL* ssl, const byte* input, word32* inOutIdx,
                           word32 size, word32 totalSz, int sniff)
{
    int    ret;
    word32 finishedSz = 0;
    byte*  secret;
    byte   mac[WC_MAX_DIGEST_SIZE];

    if (totalSz < TLS_FINISHED_SZ) {
        return BAD_FUNC_ARG;
    }

    if (ssl->options.side == WOLFSSL_CLIENT_END) {
        secret = ssl->arrays->clientFinishedSecret;
    } else {
        secret = ssl->arrays->serverFinishedSecret;
    }

    if (ssl->options.tls1_3 && ssl->options.resuming && ssl->sessionID.size > 0) {
        ret = wc_HmacHash(secret, TLS_FINISHED_SZ, mac, WC_SHA384);
    } else {
        ret = wc_HmacHash(secret, TLS_FINISHED_SZ, mac, ssl->hashes->digestType);
    }

    if (ret!= 0) {
        return ret;
    }

    if (XMEMCMP(mac, input + *inOutIdx, TLS_FINISHED_SZ)!= 0) {
        return VERIFY_FAILED;
    }

    *inOutIdx += TLS_FINISHED_SZ;
    return 0;
}
