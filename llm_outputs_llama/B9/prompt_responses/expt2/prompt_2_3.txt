    int validSigAlgo = 0;

    #ifdef HAVE_ED25519
        if (args->sigAlgo == ed25519_sa_algo) {
            validSigAlgo = ssl->peerEd25519KeyPresent;
            if (!validSigAlgo) {
                WOLFSSL_MSG("Peer sent ED25519 sig but not ED25519 cert");
            }
        }
    #endif

    #ifdef HAVE_ED448
        if (args->sigAlgo == ed448_sa_algo) {
            validSigAlgo = ssl->peerEd448KeyPresent;
            if (!validSigAlgo) {
                WOLFSSL_MSG("Peer sent ED448 sig but not ED448 cert");
            }
        }
    #endif

    #ifdef HAVE_ECC
        if (args->sigAlgo == ecc_dsa_sa_algo) {
            validSigAlgo = ssl->peerEccDsaKeyPresent;
            if (!validSigAlgo) {
                WOLFSSL_MSG("Peer sent ECC sig but not ECC cert");
            }
        }
    #endif

    #ifdef HAVE_PQC
        if (args->sigAlgo == falcon_level1_sa_algo || args->sigAlgo == falcon_level5_sa_algo) {
            validSigAlgo = ssl->peerFalconKeyPresent;
            if (!validSigAlgo) {
                WOLFSSL_MSG("Peer sent Falcon sig but not Falcon cert");
            }
        }
    #endif

    #ifndef NO_RSA
        if (args->sigAlgo == rsa_sa_algo) {
            WOLFSSL_MSG("Peer sent PKCS#1.5 algo but not in certificate");
            ERROR_OUT(INVALID_PARAMETER, exit_dcv);
        }

        if (args->sigAlgo == rsa_pss_sa_algo) {
            validSigAlgo = ssl->peerRsaKey!= NULL && ssl->peerRsaKeyPresent;
            if (!validSigAlgo) {
                WOLFSSL_MSG("Peer sent RSA-PSS sig but not RSA cert");
            }
        }
    #endif

    if (!validSigAlgo) {
        ret = SIG_VERIFY_E;
        goto exit_dcv;
    }
