int ssl3_get_cert_verify(SSL *s)
	{
	EVP_PKEY *pkey=NULL;
	unsigned char *p;
	int al,ok,ret=0;
	long n;
	int type=0,i,j;
	X509 *peer;
	const EVP_MD *md = NULL;
	EVP_MD_CTX mctx;
	EVP_MD_CTX_init(&mctx);

	n=s->method->ssl_get_message(s,
		SSL3_ST_SR_CERT_VRFY_A,
		SSL3_ST_SR_CERT_VRFY_B,
		-1,
		SSL3_RT_MAX_PLAIN_LENGTH,
		&ok);

	if (!ok) return((int)n);

	if (s->session->peer!= NULL)
		{
		peer=s->session->peer;
		pkey=X509_get_pubkey(peer);
		if (pkey == NULL) {
			SSLerr(SSL_F_SSL3_GET_CERT_VERIFY, ERR_R_PEM_LIB);
			return 0;
		}
		type=X509_certificate_type(peer,pkey);
		}
	else
		{
		peer=NULL;
		pkey=NULL;
		}

	if (pkey!= NULL) {
		if (type == 0) {
			SSLerr(SSL_F_SSL3_GET_CERT_VERIFY, SSL_R_UNKNOWN_CERT_TYPE);
			return 0;
		}
	}
