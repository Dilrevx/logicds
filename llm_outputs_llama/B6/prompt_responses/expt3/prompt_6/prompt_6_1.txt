    int ret = 0;
    mbedtls_x509_name *subject = &crt->subject;
    mbedtls_x509_sequence *san = NULL;

    // Check if the certificate contains a Subject Alternative Name (SAN) extension
    ret = mbedtls_x509_crt_get_ext_key_usage(crt, MBEDTLS_OID_SAN, &san);

        }

    for (int i = 0; i < subject->len; i++) {

            memcmp(name->oid.p, MBEDTLS_OID_AT_COMMON_NAME, MBEDTLS_OID_SIZE(MBEDTLS_OID_AT_COMMON_NAME)) == 0) {

int check_san_extension(mbedtls_x509_sequence *san, const char *expected_name) {
