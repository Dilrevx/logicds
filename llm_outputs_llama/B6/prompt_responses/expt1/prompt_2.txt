static void x509_crt_verify_name( const mbedtls_x509_crt *crt,
                                  const char *cn,
                                  uint32_t *flags )
{
    const mbedtls_x509_name *name;
    const mbedtls_x509_sequence *cur;
    size_t cn_len = strlen( cn );

    if( cn == NULL || cn_len == 0 )
    {
        *flags |= MBEDTLS_X509_BADCERT_CN_MISMATCH;
        return;
    }

    name = &crt->subject;
    for( cur = mbedtls_x509_sequence_get( name, MBEDTLS_X509_OID_CN ); cur!= NULL; cur = mbedtls_x509_sequence_next( cur ) )
    {
        if( mbedtls_oid_get_attr( &cur->oid, MBEDTLS_OID_SAN_URI )!= NULL )
            continue;

        if( mbedtls_oid_get_attr( &cur->oid, MBEDTLS_OID_SAN_EMAIL )!= NULL )
            continue;

        if( mbedtls_oid_get_attr( &cur->oid, MBEDTLS_OID_SAN_DNS )!= NULL )
            continue;

        if( mbedtls_oid_get_attr( &cur->oid, MBEDTLS_OID_CN )!= NULL )
        {
            if( cn_len == cur->val.len && memcmp( cn, cur->val.p, cn_len ) == 0 )
            {
                *flags &= ~MBEDTLS_X509_BADCERT_CN_MISMATCH;
                return;
            }
        }
    }

    *flags |= MBEDTLS_X509_BADCERT_CN_MISMATCH;
}
