        tmpdir = OsApi.mkdtemp(template);
        if (tmpdir == null) {
            FuseLog.log(NamespaceFlags.FUSE_LOG_ERR, "tmpdir(%s): %m\n", template);
            OsApi.exit(1);
        }

        if (OsApi.mount("/proc/self/fd", tmpdir, null, NamespaceFlags.MS_BIND, null) < 0) {
            FuseLog.log(NamespaceFlags.FUSE_LOG_ERR,
                    "mount(/proc/self/fd, %s, MS_BIND): %m\n", tmpdir);
            OsApi.exit(1);
        }

        lo.setProcSelfFd(OsApi.open(tmpdir, NamespaceFlags.O_PATH));
        if (lo.getProcSelfFd() == -1) {
            FuseLog.log(NamespaceFlags.FUSE_LOG_ERR, "open(%s, O_PATH): %m\n", tmpdir);
            OsApi.exit(1);
        }

        if (OsApi.umount2(tmpdir, NamespaceFlags.MNT_DETACH) < 0) {
            FuseLog.log(NamespaceFlags.FUSE_LOG_ERR,
                    "umount2(%s, MNT_DETACH): %m\n", tmpdir);
            OsApi.exit(1);
        }

        if (OsApi.rmdir(tmpdir) < 0) {
            FuseLog.log(NamespaceFlags.FUSE_LOG_ERR, "rmdir(%s): %m\n", tmpdir);
