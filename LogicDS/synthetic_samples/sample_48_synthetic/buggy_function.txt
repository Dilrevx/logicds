    public HttpError handle(EchoContext c, Server server, String secret, Handler next) {
        String path = c.getPath();
        String method = c.getMethod();

        if (server.defaultAuthSkipper(c)) {
            return next.handle(c);
        }

        if (hasPrefixes(path, new String[]{"/api/ping", "/api/v1/idp", "/api/user/:id"}) && "GET".equals(method)) {
            return next.handle(c);
        }

        String token = findAccessToken(c);
        if (token.length() == 0) {
            if (hasPrefixes(path, new String[]{"/o"})) {
                return next.handle(c);
            }
            if (hasPrefixes(path, new String[]{"/api/status", "/api/memo"}) && "GET".equals(method)) {
                return next.handle(c);
            }
            return new HttpError(STATUS_UNAUTHORIZED, "Missing access token");
        }

        Claims claims = new Claims();
        JwtParser.ParseResult parsed = JwtParser.parseWithClaims(token, claims, new JwtParser.KeyFn() {
            public Object apply(JwtToken t) {
                if (!"HS256".equals(t.getMethodAlg())) {
                    throw new RuntimeException("unexpected access token signing method");
                }
                Object kid = t.getHeader().get("kid");
                if (kid instanceof String && "v1".equals(kid)) {
                    return secret.getBytes();
                }
                throw new RuntimeException("unexpected access token kid");
            }
        });
        JwtToken accessToken = parsed.token;
        RuntimeException err = parsed.err;

        if (!audienceContains(claims.getAudience(), ACCESS_TOKEN_AUDIENCE)) {
            return new HttpError(STATUS_UNAUTHORIZED,
                    "Invalid access token, audience mismatch, got " + claims.getAudience()
                            + ", expected " + ACCESS_TOKEN_AUDIENCE);
        }

        boolean generateToken =
                (claims.getExpiresAtMillis() - System.currentTimeMillis()) < REFRESH_THRESHOLD_MILLIS;
        if (err != null) {
            if (err instanceof ValidationError) {
                ValidationError ve = (ValidationError) err;
                if (ve.getErrors() == ValidationError.VALIDATION_ERROR_EXPIRED) {
                    generateToken = true;
                }
            } else {
                return new HttpError(STATUS_UNAUTHORIZED, "Invalid or expired access token: " + err.getMessage());
            }
        }

        int userId;
        try {
            userId = Integer.parseInt(claims.getSubject());
        } catch (NumberFormatException e) {
            return new HttpError(STATUS_UNAUTHORIZED, "Malformed ID in the token.");
        }

        Object user = server.getUser(userId);
        if (user == null) {
            return new HttpError(STATUS_UNAUTHORIZED, "Failed to find user ID: " + userId);
        }

        if (generateToken) {
            HttpError genErr = generateTokenFunc(c, server, secret, userId);
            if (genErr != null && !accessToken.isValid()) {
                return genErr;
            }
        }

        c.setAttr(getUserIdContextKey(), userId);
        return next.handle(c);
    }
